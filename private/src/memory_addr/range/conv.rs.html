12/// A conversion from a standard library range to an address range.
13///
14/// Both [`core::range`] and [`core::ops`] range types are supported. Endpoints
15/// are converted through [`Into<A>`](Into). Missing lower bounds become address
16/// zero.
17///
18/// | Source range | Output type |
19/// | --- | --- |
20/// | `Range`, `RangeTo` | [`AddrRange<A>`] |
21/// | `RangeFrom`, `RangeFull` | [`AddrRangeFrom<A>`] |
22/// | `RangeInclusive`, `RangeToInclusive` | [`GeneralAddrRange<A>`] |
23///
24/// Exclusive endpoints are preserved without validation, including reversed
25/// endpoints that produce an invalid `AddrRange`. The
26/// [`addr_range!`](crate::addr_range), [`va_range!`](crate::va_range), and
27/// [`pa_range!`](crate::pa_range) macros validate the converted range instead.
28/// Inclusive upper endpoints become exclusive by adding one. An
29/// inclusive upper endpoint of `usize::MAX` produces
30/// [`GeneralAddrRange::RangeFrom`], avoiding overflow.
31///
32/// Reversed inclusive ranges become empty bounded ranges at the converted
33/// start. An exhausted [`core::ops::RangeInclusive`] also becomes an empty
34/// bounded range, whose location is unspecified because its source endpoints
35/// are unspecified.
36///
37/// # Examples
38///
39/// ```
40/// use memory_addr::{AddrRange, AddrRangeBounds, GeneralAddrRange, IntoAddrRange};
41///
42/// let reversed = core::range::Range { start: 5usize, end: 2 };
43/// let bounded: AddrRange<usize> = reversed.into_addr_range();
44/// assert!(!bounded.is_valid());
45/// let reversed = core::range::RangeInclusive { start: 5usize, last: 2 };
46/// let inclusive: GeneralAddrRange<usize> = reversed.into_addr_range();
47/// assert!(inclusive.is_valid());
48/// assert!(inclusive.is_empty());
49///
50/// let mut exhausted = usize::MAX..=usize::MAX;
51/// assert_eq!(exhausted.next(), Some(usize::MAX));
52/// let empty: GeneralAddrRange<usize> = exhausted.into_addr_range();
53/// assert!(empty.is_empty());
54/// ```
55pub const trait IntoAddrRange<A: [const] MemoryAddr>: Sized {
56    /// The address range representation produced by this conversion.
57    ///
58    /// Its concrete type depends on the source range's upper bound.
59    type Into: [const] AddrRangeBounds<A>;
60
61    /// Converts the source into its corresponding address range representation.
62    ///
63    /// Inclusive ranges select a bounded or unbounded representation according
64    /// to whether their exclusive endpoint can be represented.
65    ///
66    /// # Examples
67    ///
68    /// ```
69    /// use memory_addr::{AddrRange, GeneralAddrRange, IntoAddrRange};
70    ///
71    /// let prefix: AddrRange<usize> = (..4usize).into_addr_range();
72    /// assert_eq!(prefix, AddrRange::new(0, 4));
73    /// let singleton: GeneralAddrRange<usize> = (4usize..=4).into_addr_range();
74    /// assert_eq!(singleton, GeneralAddrRange::Range(AddrRange::new(4, 5)));
75    /// ```
76    fn into_addr_range(self) -> Self::Into;
77
78    /// Converts the source into the general address range representation.
79    ///
80    /// This produces the same output type for every supported source range.
81    /// It applies the same endpoint and validity rules as
82    /// [`into_addr_range`](Self::into_addr_range).
83    ///
84    /// # Examples
85    ///
86    /// ```
87    /// use memory_addr::{AddrRangeBounds, GeneralAddrRange, IntoAddrRange};
88    ///
89    /// let prefix: GeneralAddrRange<usize> = (..4usize).into_general_addr_range();
90    /// assert_eq!(prefix.checked_end(), Some(4));
91    /// let full: GeneralAddrRange<usize> = (..=usize::MAX).into_general_addr_range();
92    /// assert_eq!(full.start(), 0);
93    /// assert_eq!(full.checked_end(), None);
94    /// assert!(full.contains(usize::MAX));
95    /// ```
96    fn into_general_addr_range(self) -> GeneralAddrRange<A> {
97        self.into_addr_range().into_general()
98    }
99}

101/// Converts inclusive endpoints into an empty, bounded, or unbounded range.
102///
103/// Reversed endpoints describe an empty range at `start`. Otherwise the last
104/// address determines whether the exclusive endpoint is representable.
105const fn from_inclusive<A: [const] MemoryAddr>(start: A, last: A) -> GeneralAddrRange<A> {
106    if start > last {
107        return GeneralAddrRange::Range(AddrRange { start, end: start });
108    }
109    match last.checked_add(1) {
110        Some(end) => GeneralAddrRange::Range(AddrRange { start, end }),
111        None => GeneralAddrRange::RangeFrom(AddrRangeFrom { start }),
112    }
113}

222/// Converts a legacy inclusive range while preserving its exhaustion state.
223///
224/// Exhaustion produces an empty range regardless of the remaining endpoint
225/// values.
226const impl<A: [const] MemoryAddr, T: [const] Into<A> + [const] Destruct> IntoAddrRange<A>
227    for core::ops::RangeInclusive<T>
228{
229    type Into = GeneralAddrRange<A>;
230
231    fn into_addr_range(self) -> Self::Into {
232        // The excluded bound exposes exhaustion without requiring `T: PartialOrd`.
233        let exhausted = matches!(self.end_bound(), Bound::Excluded(_));
234        let (start, last) = self.into_inner();
235        let start = start.into();
236        let last = last.into();
237        if exhausted {
238            GeneralAddrRange::Range(AddrRange { start, end: start })
239        } else {
240            from_inclusive(start, last)
241        }
242    }
243}

270/// Converts a standard library range into an address range.
271///
272/// The address type is inferred from context. The source's upper bound
273/// determines the output type as described by [`IntoAddrRange`].
274///
275/// # Panics
276///
277/// Panics if the converted range has reversed exclusive endpoints.
278///
279/// # Example
280///
281/// ```
282/// use memory_addr::{addr_range, AddrRange};
283///
284/// let range: AddrRange<usize> = addr_range!(0x1000usize..0x2000);
285/// assert_eq!(range.start, 0x1000usize);
286/// assert_eq!(range.end, 0x2000usize);
287/// ```
288///
289/// Inclusive ranges can include the maximum address:
290///
291/// ```
292/// use memory_addr::{addr_range, AddrRangeBounds, GeneralAddrRange};
293///
294/// let range: GeneralAddrRange<usize> = addr_range!(usize::MAX..=usize::MAX);
295/// assert_eq!(range.size(), 1);
296/// assert!(range.contains(usize::MAX));
297/// ```
298#[macro_export]
299macro_rules! addr_range {
300    ($range:expr) => {{
301        let range = $crate::IntoAddrRange::into_addr_range($range);
302        assert!(
303            $crate::AddrRangeBounds::is_valid(&range),
304            "invalid `AddrRange`: `start` must be less than or equal to `end`",
305        );
306        range
307    }};
308}

310/// Converts a standard library range into a virtual address range.
311///
312/// The output is [`AddrRange`], [`AddrRangeFrom`], or [`GeneralAddrRange`] with
313/// [`VirtAddr`](crate::VirtAddr) endpoints, following [`IntoAddrRange`].
314///
315/// # Panics
316///
317/// Panics if the converted range has reversed exclusive endpoints.
318///
319/// # Example
320///
321/// ```
322/// use memory_addr::{va_range, AddrRangeBounds};
323///
324/// let range = va_range!(0x1000..0x2000);
325/// assert_eq!(range.start, 0x1000.into());
326/// assert_eq!(range.end, 0x2000.into());
327/// assert!(va_range!(..=usize::MAX).contains(memory_addr::va!(usize::MAX)));
328/// ```
329#[macro_export]
330macro_rules! va_range {
331    ($range:expr) => {{
332        let range = $crate::IntoAddrRange::<$crate::VirtAddr>::into_addr_range($range);
333        assert!(
334            $crate::AddrRangeBounds::is_valid(&range),
335            "invalid `AddrRange`: `start` must be less than or equal to `end`",
336        );
337        range
338    }};
339}

341/// Converts a standard library range into a physical address range.
342///
343/// The output is [`AddrRange`], [`AddrRangeFrom`], or [`GeneralAddrRange`] with
344/// [`PhysAddr`](crate::PhysAddr) endpoints, following [`IntoAddrRange`].
345///
346/// # Panics
347///
348/// Panics if the converted range has reversed exclusive endpoints.
349///
350/// # Example
351///
352/// ```
353/// use memory_addr::{pa_range, AddrRangeBounds};
354///
355/// let range = pa_range!(0x1000..0x2000);
356/// assert_eq!(range.start, 0x1000.into());
357/// assert_eq!(range.end, 0x2000.into());
358/// assert_eq!(pa_range!(usize::MAX..).size(), 1);
359/// ```
360#[macro_export]
361macro_rules! pa_range {
362    ($range:expr) => {{
363        let range = $crate::IntoAddrRange::<$crate::PhysAddr>::into_addr_range($range);
364        assert!(
365            $crate::AddrRangeBounds::is_valid(&range),
366            "invalid `AddrRange`: `start` must be less than or equal to `end`",
367        );
368        range
369    }};
