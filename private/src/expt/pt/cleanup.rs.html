1//! Cleanup operations after page table hierarchy mutations.
2//!
3//! [`PendingCleanup`] performs accumulated TLB invalidations through the selected [`TlbFlusher`].
4//! [`TlbFlushes`] stores requests without automatically processing them. Table-frame deallocation
5//! happens during mutation and is not part of pending cleanup. See the
6//! [safety requirements for page table updates](crate#safely-updating-page-tables).
7
8use core::{mem::ManuallyDrop, ops::AddAssign, ptr};
9
10use crate::meta::VirtAddr;
11
12mod flush;
13
14pub use flush::{DynTlbFlusher, SMALL_FLUSH_THRESHOLD, TlbFlush, TlbFlusher, TlbFlushes};
15
16/// Pending TLB invalidations after page table hierarchy mutations.
17///
18/// Call [`Self::clean_up`] to perform TLB invalidation, or call [`Self::into_inner`] to take the
19/// requests and TLB flusher for manual processing.
20///
21/// Dropping this value also processes cleanup. Explicit cleanup makes the synchronization point
22/// visible to the caller.
23///
24/// The selected [`TlbFlusher`] determines the architecture, invalidation scope, and completion
25/// guarantees. See [`TlbFlusher`] for strategy selection and the
26/// [safety requirements for page table updates](crate#safely-updating-page-tables).
27#[must_use = "call `clean_up` or `into_inner` explicitly, rather than relying on Drop's cleanup"]
28pub struct PendingCleanup<A: VirtAddr> {
29    /// The pending invalidation requests.
30    tlb_flushes: TlbFlushes<A>,
31    /// The TLB flusher used to flush the TLB.
32    flusher: DynTlbFlusher<A>,
33}
34
35/// Constructs, combines, and processes cleanup batches.
36impl<A: VirtAddr> PendingCleanup<A> {
37    /// Creates an empty [`PendingCleanup`] with the given TLB flusher.
38    pub fn new(flusher: &DynTlbFlusher<A>) -> Self {
39        Self {
40            tlb_flushes: TlbFlushes::None,
41            flusher: flusher.clone(),
42        }
43    }
44
45    /// Creates an empty [`PendingCleanup`] with TLB flusher `F`.
46    pub fn new_with<F: TlbFlusher<A>>() -> Self {
47        Self::new(&DynTlbFlusher::new::<F>())
48    }
49
50    /// Performs the pending TLB invalidations.
51    ///
52    /// See the type-level contract for invalidation scope and synchronization requirements.
53    pub fn clean_up(self) {
54        // Just drop it and let the destructor handle it.
55        drop(self);
56    }
57
58    /// Consumes this batch and returns its requests and TLB flusher without processing them.
59    ///
60    /// Dropping the returned values does not perform TLB invalidation.
61    ///
62    /// # Safety
63    ///
64    /// The caller must process the returned requests through a suitable TLB flusher and complete
65    /// any further synchronization required by the
66    /// [safety requirements for page table updates](crate#safely-updating-page-tables).
67    pub unsafe fn into_inner(self) -> (TlbFlushes<A>, DynTlbFlusher<A>) {
68        let this = ManuallyDrop::new(self);
69
70        unsafe {
71            (
72                ptr::read(&raw const this.tlb_flushes),
73                ptr::read(&raw const this.flusher),
74            )
75        }
76    }
77
78    /// Appends another flush batch to this [`PendingCleanup`].
79    pub fn append_flushes(&mut self, other: TlbFlushes<A>) {
80        self.tlb_flushes.append(other);
81    }
82
83    /// Appends a single flush to this [`PendingCleanup`].
84    pub fn push_flush(&mut self, flush: TlbFlush<A>) {
85        self.tlb_flushes.add_assign(flush);
86    }
87
88    /// Transfers another [`PendingCleanup`] into this batch without processing either batch.
89    ///
90    /// # Safety
91    ///
92    /// This batch's TLB flusher must provide the invalidation scope and completion guarantees
93    /// required by both batches. The other batch's flusher is discarded.
94    pub unsafe fn append(&mut self, other: Self) {
95        // SAFETY: we take the responsibility for the cleanup.
96        let (flushes, _) = unsafe { other.into_inner() };
97        self.append_flushes(flushes);
98    }
99}
100
101/// Processes any requests still held by a cleanup batch.
102impl<A: VirtAddr> Drop for PendingCleanup<A> {
103    /// Performs the same cleanup as [`Self::clean_up`].
104    fn drop(&mut self) {
105        self.tlb_flushes.flush_with(&self.flusher);
106    }
107}
