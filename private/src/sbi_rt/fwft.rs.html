9/// Set the configuration value of a specific firmware feature.
10///
11/// # Parameters
12///
13/// - `feature`: The identifier of the feature to set.
14/// - `value`: The value to set for the feature.
15/// - `flags`: Flags to modify the behavior of the set operation.
16///
17/// # Return value
18///
19/// A successful return results in the requested firmware feature to be set according to the `value` and `flags` parameters. In case of failure, `feature` value is not modified and the possible error codes returned in `SbiRet.error` are shown in the table below:
20///
21/// | Error code                  | Description
22/// |:----------------------------|:---------------------------------
23/// | `SbiRet::success()`         | `feature` was set successfully.
24/// | `SbiRet::not_supported()`   | `feature` is not reserved and valid, but the platform does not support it due to one or more missing dependencies (Hardware or SBI implementation).
25/// | `SbiRet::invalid_param()`   | Provided `value` or `flags` parameter is invalid.
26/// | `SbiRet::denied()`          | `feature` set operation failed because either it was denied by the SBI implementation, or`feature` is reserved or is platform-specific and unimplemented.
27/// | `SbiRet::denied_locked()`   | `feature` set operation failed because the `feature` is locked.
28/// | `SbiRet::failed()`          | The set operation failed for unspecified or unknown other reasons.
29///
30/// This function is defined in RISC-V SBI Specification chapter 18.1.
31#[inline]
32#[doc(alias = "sbi_fwft_set")]
33pub fn fwft_set(feature: u32, value: usize, flags: usize) -> SbiRet {
34    sbi_call_3(EID_FWFT, SET, feature as _, value, flags)
35}

37/// Get the configuration value of a specific firmware feature.
38///
39/// # Parameters
40///
41/// - `feature`: The identifier of the feature to get.
42///
43/// # Return value
44///
45/// A successful return results in the firmware feature configuration value to be returned in `SbiRet.value`. In case of failure, the content of `SbiRet.value` is zero and the possible error codes returned in `SbiRet.error` are shown in the table below:
46///
47/// | Error code                  | Description
48/// |:----------------------------|:---------------------------------
49/// | `SbiRet::success()`         | Feature status was retrieved successfully.
50/// | `SbiRet::not_supported()`   | `feature` is not reserved and valid, but the platform does not support it due to one or more missing dependencies (Hardware or SBI implementation).
51/// | `SbiRet::denied()`          | `feature` is reserved or is platform-specific and unimplemented.
52/// | `SbiRet::failed()`          | The get operation failed for unspecified or unknown other reasons.
53///
54/// This function is defined in RISC-V SBI Specification chapter 18.2.
55#[inline]
56#[doc(alias = "sbi_fwft_get")]
57pub fn fwft_get(feature: u32) -> SbiRet {
58    sbi_call_1(EID_FWFT, GET, feature as _)
59}
