10/// Send an inter-processor interrupt to all harts defined in hart mask.
11///
12/// Inter-processor interrupts manifest at the receiving harts as the supervisor software interrupts.
13///
14/// # Return value
15///
16/// The possible return error codes returned in `SbiRet.error` are shown in the table below:
17///
18/// | Return code               | Description
19/// |:--------------------------|:----------------------------------------------
20/// | `SbiRet::success()`       | IPI was sent to all the targeted harts successfully.
21/// | `SbiRet::invalid_param()` | At least one hartid constructed from `hart_mask`, is not valid, i.e. either the hartid is not enabled by the platform or is not available to the supervisor.
22/// | `SbiRet::failed()`        | The request failed for unspecified or unknown other reasons.
23///
24/// This function is defined in RISC-V SBI Specification chapter 7.1.
25#[inline]
26#[doc(alias = "sbi_send_ipi")]
27pub fn send_ipi(hart_mask: HartMask) -> SbiRet {
28    let (hart_mask, hart_mask_base) = hart_mask.into_inner();
29    sbi_call_2(EID_SPI, SEND_IPI, hart_mask, hart_mask_base)
30}
