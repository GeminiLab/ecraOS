9/// Read a range of event attribute values from a software event.
10///
11/// The `event_id` parameter specifies the software event ID whereas `base_attr_id`
12/// and `attr_count` parameters specifies the range of event attribute IDs.
13///
14/// The event attribute values are written to a output shared memory which is specified
15/// by the `output` parameter where:
16///
17/// - The `output` parameter MUST be `XLEN / 8` bytes aligned
18/// - The size of output shared memory is assumed to be `(XLEN / 8) * attr_count`
19/// - The value of event attribute with ID `base_attr_id + i` should be read from offset `(XLEN / 8) * (base_attr_id + i)`
20///
21/// The possible error codes returned in sbiret.error are shown below.
22///
23/// # Return value
24///
25/// | Error code                  | Description
26/// |:----------------------------|:---------------------------------
27/// | `SbiRet::success()`         | Event attribute values read successfully.
28/// | `SbiRet::not_supported()`   | `event_id` is not reserved and valid, but the platform does not support it due to one or more missing dependencies (Hardware or SBI implementation).
29/// | `SbiRet::invalid_param()`   | `event_id` is invalid or `attr_count` is zero.
30/// | `SbiRet::bad_range()`       | One of the event attribute IDs in the range specified by `base_attr_id` and `attr_count` is reserved.
31/// | `SbiRet::invalid_address()` | The shared memory pointed to by the `output` parameter does not satisfy the requirements.
32/// | `SbiRet::failed()`          | The read failed for unspecified or unknown other reasons.
33#[doc(alias = "sbi_sse_read_attrs")]
34#[inline]
35pub fn sse_read_attrs(
36    event_id: u32,
37    base_attr_id: u32,
38    attr_count: u32,
39    output: SharedPtr<u8>,
40) -> SbiRet {
41    sbi_call_5(
42        EID_SSE,
43        READ_ATTRS,
44        event_id as _,
45        base_attr_id as _,
46        attr_count as _,
47        output.phys_addr_lo(),
48        output.phys_addr_hi(),
49    )
50}

52/// Write a range of event attribute values to a software event.
53///
54/// The event_id parameter specifies the software event ID whereas `base_attr_id` and
55/// `attr_count` parameters specifies the range of event attribute IDs.
56///
57/// The event attribute values are read from a input shared memory which is specified
58/// by the `input` parameter where:
59///
60/// - The `input` parameter MUST be `XLEN / 8` bytes aligned
61/// - The size of input shared memory is assumed to be `(XLEN / 8) * attr_count`
62/// - The value of event attribute with ID `base_attr_id + i` should be read from offset `(XLEN / 8) * (base_attr_id + i)`
63///
64/// For local events, the event attributes are updated only for the calling hart.
65/// For global events, the event attributes are updated for all the harts.
66/// The possible error codes returned in sbiret.error are shown below.
67///
68/// # Return value
69///
70/// | Error code                  | Description
71/// |:----------------------------|:---------------------------------

73/// | `SbiRet::not_supported()`   | `event_id` is not reserved and valid, but the platform does not support it due to one or more missing dependencies (Hardware or SBI implementation).
74/// | `SbiRet::invalid_param()`   | `event_id` is invalid or `attr_count` is zero.
75/// | `SbiRet::bad_range()`       | One of the event attribute IDs in the range specified by `base_attr_id` and `attr_count` is reserved or is read-only.
76/// | `SbiRet::invalid_address()` | The shared memory pointed to by the `input` parameter does not satisfy the requirements.
77/// | `SbiRet::failed()`          | The write failed for unspecified or unknown other reasons.
78#[doc(alias = "sbi_sse_write_attrs")]
79#[inline]
80pub fn sse_write_attrs(
81    event_id: u32,
82    base_attr_id: u32,
83    attr_count: u32,
84    input: SharedPtr<u8>,
85) -> SbiRet {
86    sbi_call_5(
87        EID_SSE,
88        WRITE_ATTRS,
89        event_id as _,
90        base_attr_id as _,
91        attr_count as _,
92        input.phys_addr_lo(),
93        input.phys_addr_hi(),
94    )
95}

97/// Register an event handler for the software event.
98///
99/// The `event_id` parameter specifies the event ID for which an event handler is being registered.
100/// The `handler_entry_pc` parameter MUST be 2-bytes aligned and specifies the `ENTRY_PC` event
101/// attribute of the software event whereas the `handler_entry_arg` parameter specifies the
102/// `ENTRY_ARG` event attribute of the software event.
103///
104/// For local events, the event is registered only for the calling hart.
105/// For global events, the event is registered for all the harts.
106///
107/// The event MUST be in `UNUSED` state otherwise this function will fail.
108///
109/// Upon success, the event state moves from `UNUSED` to `REGISTERED`. In case of an error, possible error codes are listed below.
110///
111/// # Return value
112///
113/// | Error code                  | Description
114/// |:----------------------------|:---------------------------------
115/// | `SbiRet::success()`         | Event handler is registered successfully.
116/// | `SbiRet::not_supported()`   | `event_id` is not reserved and valid, but the platform does not support it due to one or more missing dependencies (Hardware or SBI implementation).
117/// | `SbiRet::invalid_state()`   | `event_id` is valid but the event is not in `UNUSED` state.
118/// | `SbiRet::invalid_param()`   | `event_id` is invalid or `handler_entry_pc` is not 2-bytes aligned.    
119#[doc(alias = "sbi_sse_register")]
120#[inline]
121pub fn sse_register(event_id: u32, handler_entry_pc: usize, handler_entry_arg: usize) -> SbiRet {
122    sbi_call_3(
123        EID_SSE,
124        REGISTER,
125        event_id as _,
126        handler_entry_pc,
127        handler_entry_arg,
128    )
129}

200/// Complete the supervisor event handling for the highest priority event in `RUNNING` state on the calling hart.
201///
202/// If there were no events in `RUNNING` state on the calling hart then this function does nothing and returns `SBI_SUCCESS`
203/// otherwise it moves the highest priority event in `RUNNING` state to:
204///
205/// - `REGISTERED` if the event is configured as one-shot
206/// - `ENABLED` state otherwise
207///
208/// It then resumes the interrupted supervisor state.
209#[doc(alias = "sbi_sse_complete")]
210#[inline]
211pub fn sse_complete() -> SbiRet {
212    sbi_call_0(EID_SSE, COMPLETE)
213}

215/// The supervisor software can inject a software event with this function.
216///
217/// The `event_id` parameter refers to the ID of the event to be injected.
218///
219/// For local events, the `hart_id` parameter refers to the hart on which the event is to be injected.
220/// For global events, the `hart_id` parameter is ignored.
221///
222/// An event can only be injected if it is allowed by the event attribute.
223///
224/// If an event is injected from within an SSE event handler, if it is ready to be run,
225/// it will be handled according to the priority rules
226///
227/// - If it has a higher priority than the one currently running, then it will be handled immediately, effectively preempting the currently running one.
228/// - If it has a lower priority, it will be run after the one that is currently running completes.
229///
230/// In case of an error, possible error codes are listed below.
231///
232/// # Return value
233///
234/// | Error code                  | Description
235/// |:----------------------------|:---------------------------------
236/// | `SbiRet::success()`         | Event is successfully injected.
237/// | `SbiRet::not_supported()`   | `event_id` is not reserved and valid, but the platform does not support it due to one or more missing dependencies (Hardware or SBI implementation).
238/// | `SbiRet::invalid_param()`   | `event_id` is invalid or `hart_id` is invalid.
239/// | `SbiRet::failed()`          | The injection failed for unspecified or unknown other reasons.
240#[doc(alias = "sbi_sse_inject")]
241#[inline]
242pub fn sse_inject(event_id: u32, hart_id: usize) -> SbiRet {
243    sbi_call_2(EID_SSE, INJECT, event_id as _, hart_id)
244}

259/// | `SbiRet::already_started()` | Software events were already unmasked on the calling hart.

261#[doc(alias = "sbi_sse_hart_unmask")]
