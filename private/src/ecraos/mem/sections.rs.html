1//! Kernel image section information.
2//!
3//! # Sections and symbols
4//!
5//! This module provides information about sections in the kernel image that are used by the kernel
6//! itself during runtime. These sections do not exactly correspond to ELF sections. For example,
7//! the `rodata` section also covers the `.init_array` section. Currently there are 7 sections:
8//!
9//! | Section | Description |
10//! | ------- | ----------- |
11//! | text | The compiled kernel code. |
12//! | rodata | The read-only data. |
13//! | percpu | The initial data for per-CPU variables. |
14//! | data | The initialized writable global data. |
15//! | rela_dyn | The dynamic relocation entries. |
16//! | got | The global offset table. |
17//! | bss | The zero-initialized global data. |
18//!
19//! This module uses symbols defined in the linker script (`ecraos/link.ld`) to locate these
20//! sections. Each section has three linker symbols marking its start, true end, and aligned end.
21//! The alignment is specified by the linker script, currently 64 KiB, and does not depend on the
22//! runtime page size.
23//!
24//! # Relocation compatibility
25//!
26//! This module resolves symbol addresses with PC-relative addressing. Queries return the current
27//! runtime ranges, which will change after relocation to a new address. Previously returned ranges
28//! and [`Section`] values do not update automatically.
29//!
30//! # Usage inside the kernel crate
31//!
32//! Initialization code uses this module to map the kernel image properly and clear the BSS section.
33
34use exarch::mem::MemoryRegionFlags;
35use memory_addr::{AddrRangeBounds, VirtAddrRange};
36
37/// Defines sections from symbols and access flags.
38macro_rules! sections {
39    ($(
40        $name:ident ($name_aligned:ident) =>
41        $start_symbol:ident .. $end_symbol:ident .. $aligned_end_symbol:ident,
42        $flags_fn:ident: $flag0:ident $(| $flags:ident)*
43    ),* $(,)?) => {
44        unsafe extern "C" {
45            $(
46                #[doc = concat!("Marks the start of the ", stringify!($name), " section.")]
47                fn $start_symbol();
48                #[doc = concat!("Marks the content end of the ", stringify!($name), " section.")]
49                fn $end_symbol();
50                #[doc = concat!("Marks the padded end of the ", stringify!($name), " section.")]
51                fn $aligned_end_symbol();
52            )*
53        }
54
55        $(
56            #[doc = concat!(
57                "Returns the address range actually used by the ", stringify!($name), " section."
58            )]
59            #[doc = ""]
60            #[doc = concat!(
61                "Compared to [`", stringify!($name_aligned), "`], the returned range is usually ",
62                "smaller because it excludes trailing alignment padding. The ranges are equal ",
63                "when the content end is already aligned."
64            )]
65            pub fn $name() -> VirtAddrRange {
66                // SAFETY: The linker script guarantees `$end_symbol >= $start_symbol`.
67                // Relocation does not change their relative order.
68                unsafe {
69                    VirtAddrRange::new_unchecked(
70                        $crate::get_symbol_addr!($start_symbol).into(),
71                        $crate::get_symbol_addr!($end_symbol).into(),
72                    )
73                }
74            }
75
76            #[doc = concat!(
77                "Returns the address range occupied by the ", stringify!($name), " section."
78            )]
79            #[doc = ""]
80            #[doc = concat!(
81                "Compared to [`", stringify!($name), "`], the returned range is usually larger ",
82                "because it includes trailing alignment padding. The ranges are equal when the ",
83                "content end is already aligned."
84            )]
85            pub fn $name_aligned() -> VirtAddrRange {
86                // SAFETY: The linker script guarantees `$aligned_end_symbol >= $start_symbol`.
87                // Relocation does not change their relative order.
88                unsafe {
89                    VirtAddrRange::new_unchecked(
90                        $crate::get_symbol_addr!($start_symbol).into(),
91                        $crate::get_symbol_addr!($aligned_end_symbol).into(),
92                    )
93                }
94            }
95
96            #[doc = concat!(
97                "Returns the mapping flags for the ", stringify!($name), " section."
98            )]
99            #[doc = ""]
100            #[doc = concat!(
101                "The flags are constant and have the semantics described by ",
102                "[`Section::flags`]."
103            )]
104            pub const fn $flags_fn() -> MemoryRegionFlags {
105                MemoryRegionFlags::$flag0
106                $(
107                    .union(MemoryRegionFlags::$flags)
108                )*
109            }
110        )*
111
112        #[doc = "The number of sections returned by [`all_sections`]."]
113        pub const SECTION_COUNT: usize = {
114            [$(stringify!($name)),*].len()
115        };
116
117        /// A snapshot of one kernel image section.
118        #[derive(Debug, Clone, Copy)]
119        pub struct Section {
120            /// The name of the section.
121            pub name: &'static str,
122            /// The virtual address range excluding trailing alignment padding.
123            pub range: VirtAddrRange,
124            /// The virtual address range including trailing linker alignment padding.
125            pub aligned_range: VirtAddrRange,
126            /// The access flags used to map the section.
127            ///
128            /// It describes the intended attributes rather than querying current page-table
129            /// permissions. It is determined by the kernel itself as a constant. [`super::pmm`]
130            /// adds [`MemoryRegionFlags::KERNEL`] when processing the sections.
131            pub flags: MemoryRegionFlags,
132        }
133
134        #[doc = "Returns a snapshot of all declared sections in linker order."]
135        #[doc = ""]
136        #[doc = "The array includes each section even when it is empty."]
137        pub fn all_sections() -> [Section; SECTION_COUNT] {
138            [$(Section {
139                name: stringify!($name),
140                range: $name(),
141                aligned_range: $name_aligned(),
142                flags: $flags_fn(),
143            }),*]
144        }
145    };
146}

158/// Returns the entire kernel image range, including alignment padding.
159///
160/// Spans `_skernel.._ekernel`, covering all sections returned by [`all_sections`].
161pub fn kernel_range() -> VirtAddrRange {
162    unsafe extern "C" {
163        /// Marks the start of the linked kernel image.
164        fn _skernel();
165        /// Marks the padded end of the linked kernel image.
166        fn _ekernel();
167    }
168
169    // SAFETY: The linker script guarantees `_ekernel >= _skernel`.
170    // Relocation does not change their relative order.
171    unsafe {
172        VirtAddrRange::new_unchecked(
173            crate::get_symbol_addr!(_skernel).into(),
174            crate::get_symbol_addr!(_ekernel).into(),
175        )
176    }
177}

179/// Clears BSS contents, including the reserved BSP per-CPU early slot.
180///
181/// Zeros [`bss`] without clearing trailing alignment padding.
182///
183/// # Safety
184///
185/// The BSP must call this once after initial relocation, before initializing per-CPU storage,
186/// using BSS-backed globals, or starting APs. Interrupts must be disabled. The BSS range must be
187/// mapped writable, with no live references or concurrent accesses to it.
188pub(crate) unsafe fn clear_bss() {
189    let bss_range = bss();
190    // SAFETY: The caller guarantees writable memory and exclusive access. The linker script
191    // guarantees the BSS range is valid.
192    let bss_slice =
193        unsafe { core::slice::from_raw_parts_mut(bss_range.start.as_mut_ptr(), bss_range.size()) };
194
195    bss_slice.fill(0);
196}
