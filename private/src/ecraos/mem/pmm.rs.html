1//! Physical memory region management.
2//!
3//! # The table
4//!
5//! This module maintains the kernel's global table of physical memory regions,
6//! as a list of [`PhysMemoryRegion`]s. Each `PhysMemoryRegion` describes a
7//! physical address range, its flags, and a description. Regions are nonempty,
8//! sorted by physical address, and non-overlapping. Their boundaries are of
9//! byte precision.
10//!
11//! The table itself is not public, it should be accessed through the methods
12//! provided in this module. The table is protected by a read-write lock, so
13//! concurrent read-only access is possible.
14//!
15//! # Initialization
16//!
17//! The table is initialized during early boot with a fixed capacity, from the
18//! platform memory information and kernel and loader reservations. After heap
19//! initialization, its storage can be switched to support additional regions
20//! without the early capacity limit, and can be extended with discovered device
21//! regions.
22//!
23//! # Cross-module interaction
24//!
25//! Page allocators use the table to identify usable RAM, and [`super::vmm`]
26//! uses the table to map the direct-mapping areas.
27//!
28//! # Access
29//!
30//! Use [`lock_read_phys_mem_regions`] and [`try_lock_read_phys_mem_regions`] to
31//! inspect the table under a shared read lock.
32//!
33//! Use [`lock_write_phys_mem_regions`] and [`try_lock_write_phys_mem_regions`]
34//! to modify the table under an exclusive write lock, these functions are
35//! unsafe because they require the caller to preserve the region invariants.
36//! These functions **will NOT** update the direct-mapping areas or the page
37//! allocator.
38//!
39//! The table may be appended (and overwritten) with
40//! [`add_and_map_phys_regions`]. Adding new regions automatically maps them to
41//! the direct-mapping area (and identity-maps them if the identity mappings are
42//! still active). Therefore, adding new regions is a relatively expensive
43//! operation, so it is recommended to add regions in batches.
44//!
45//! Removing regions is not supported yet.
46

61/// The capacity of the kernel physical memory region table before the heap allocator is available.
62pub(crate) const MAX_EARLY_PHYS_MEMORY_REGIONS: usize = 64;

64/// Fixed-capacity storage for physical memory regions before the heap allocator is available.
65pub(crate) type EarlyRegionsStorage = HeaplessVec<PhysMemoryRegion, MAX_EARLY_PHYS_MEMORY_REGIONS>;
66
67/// Dynamically extensible storage for physical memory regions after the heap allocator is available.
68pub(crate) type FinalRegionsStorage = Vec<PhysMemoryRegion>;
69
70/// Fixed-capacity physical memory regions used before the heap allocator is available.
71pub(crate) type EarlyRegions = PhysMemoryRegions<EarlyRegionsStorage>;
72
73/// Dynamically extensible physical memory regions used after the heap allocator is available.
74pub(crate) type FinalRegions = PhysMemoryRegions<FinalRegionsStorage>;

76/// A [`PhysMemoryRegions`] that is backed by either a `HeaplessVec` or a `Vec`.
77///
78/// The enum allows the unification of both storage types and the in-place replacement of the
79/// storage type, and is used to store the physical memory regions in the kernel.
80#[expect(
81    clippy::large_enum_variant,
82    reason = "inline storage must work before allocation"
83)]
84enum RuntimeRegions {
85    /// The heap allocator is unavailable, so `HeaplessVec` is being used.
86    Early(EarlyRegions),
87    /// The heap allocator is ready and `Vec` is being used.
88    Final(FinalRegions),
89}
90
91/// Storage-independent access to the underlying storage.
92impl RuntimeRegions {
93    /// Borrows the region contents independently of their current storage.
94    fn as_slice(&self) -> &[PhysMemoryRegion] {
95        match self {
96            Self::Early(regions) => regions.as_slice(),
97            Self::Final(regions) => regions.as_slice(),
98        }
99    }
100
101    /// Borrows the region contents mutably independently of their current storage.
102    ///
103    /// # Safety
104    ///
105    /// The caller must preserve the region invariants: regions must remain sorted and
106    /// non-overlapping, every range must be nonempty, and all flags must remain valid.
107    /// Valid flags contain only defined bits and exactly one role.
108    unsafe fn as_mut_slice(&mut self) -> &mut [PhysMemoryRegion] {
109        // SAFETY: The caller promises to maintain the invariant.
110        unsafe {
111            match self {
112                Self::Early(regions) => regions.as_mut_slice(),
113                Self::Final(regions) => regions.as_mut_slice(),
114            }
115        }
116    }
117}

119/// [`RuntimeRegions`] wrapped in a [`RwLockNoIrq`] to allow concurrent reads and exclusive writes.
120type RegionTable = RwLockNoIrq<RuntimeRegions>;
121
122/// The global physical memory regions table.
123static PHYS_MEM_REGIONS: LazyInit<RegionTable> = LazyInit::new();

125/// Acquires a shared read lock on the physical memory regions table and runs a reader while holding
126/// it.
127///
128/// Local interrupts and preemption are disabled throughout the callback. The callback must not
129/// acquire a write lock on the same table as it will deadlock. Acquiring a read lock on the same
130/// table from the callback is generally safe, but should be avoided if possible.
131fn read_region_table<R>(table: &RegionTable, reader: impl FnOnce(&RuntimeRegions) -> R) -> R {
132    let regions = table.read();
133    reader(&regions)
134}
135
136/// Tries to acquire a shared read lock on the physical memory regions table and run a reader while
137/// holding it.
138///
139/// Returns `None` if the lock could not be acquired.
140///
141/// Local interrupts and preemption are disabled throughout the callback. The callback must not
142/// acquire a write lock on the same table as it will deadlock. Acquiring a read lock on the same
143/// table from the callback is generally safe, but should be avoided if possible.
144fn try_read_region_table<R>(
145    table: &RegionTable,
146    reader: impl FnOnce(&RuntimeRegions) -> R,
147) -> Option<R> {
148    let regions = table.try_read()?;
149    Some(reader(&regions))
150}
151
152/// Acquires an exclusive write lock on the physical memory regions table and runs a writer while
153/// holding it.
154///
155/// Local interrupts and preemption are disabled throughout the callback. The callback must not
156/// acquire another read or write lock on the same table as it will deadlock.
157fn write_region_table<R>(table: &RegionTable, writer: impl FnOnce(&mut RuntimeRegions) -> R) -> R {
158    let mut regions = table.write();
159    writer(&mut regions)
160}
161
162/// Tries to acquire an exclusive write lock and runs the writer if successful.
163///
164/// Returns `None` if the lock could not be acquired.
165///
166/// Local interrupts and preemption are disabled throughout the callback. The callback must not
167/// acquire another read or write lock on the same table as it will deadlock.
168fn try_write_region_table<R>(
169    table: &RegionTable,
170    writer: impl FnOnce(&mut RuntimeRegions) -> R,
171) -> Option<R> {
172    let mut regions = table.try_write()?;
173    Some(writer(&mut regions))
174}

176/// Errors that can occur when trying to lock the global physical memory regions table.
177#[derive(Debug, thiserror::Error)]
178pub enum PhysMemoryRegionsLockError {
179    /// The global physical memory regions table has not been initialized yet.
180    #[error("global physical memory regions table not initialized")]
181    NotInitialized,
182    /// The requested lock could not be acquired immediately.
183    #[error("global physical memory regions table already locked")]
184    AlreadyLocked,
185}

187/// Acquires a shared read lock on the global physical memory regions table and runs a reader while
188/// holding it.
189///
190/// Local interrupts and preemption are disabled throughout the callback. The callback must not
191/// acquire a write lock on the same table as it will deadlock. Acquiring a read lock on the same
192/// table from the callback is generally safe, but should be avoided if possible.
193///
194/// Some operations, such as console output or logging, may access the table implicitly and must be
195/// checked carefully.
196///
197/// # Panics
198///
199/// Panics before early memory discovery initializes the table.
200pub fn lock_read_phys_mem_regions<R>(reader: impl FnOnce(&[PhysMemoryRegion]) -> R) -> R {
201    read_region_table(&PHYS_MEM_REGIONS, |table| reader(table.as_slice()))
202}

204/// Tries to acquire a shared read lock on the global physical memory regions table and run a reader
205/// while holding it.
206///
207/// Local interrupts and preemption are disabled throughout the callback. The callback must not
208/// acquire a write lock on the same table as it will deadlock. Acquiring a read lock on the same
209/// table from the callback is generally safe, but should be avoided if possible.
210///
211/// Some operations, such as console output or logging, may access the table implicitly and must be
212/// checked carefully.
213///
214/// # Errors
215///
216/// Returns an [`PhysMemoryRegionsLockError`] if the global physical memory regions table has not
217/// been initialized yet or if the lock could not be acquired.
218pub fn try_lock_read_phys_mem_regions<R>(
219    reader: impl FnOnce(&[PhysMemoryRegion]) -> R,
220) -> Result<R, PhysMemoryRegionsLockError> {
221    let regions = PHYS_MEM_REGIONS
222        .get()
223        .ok_or(PhysMemoryRegionsLockError::NotInitialized)?;
224    try_read_region_table(regions, |table| reader(table.as_slice()))
225        .ok_or(PhysMemoryRegionsLockError::AlreadyLocked)
226}

285/// Switches the global physical memory regions table to dynamically extensible storage.
286///
287/// Call after heap initialization and direct-map activation, before removing identity
288/// mappings. The caller must exclude concurrent table updates and other storage switches
289/// until this call returns.
290///
291/// Does nothing if the table already uses final storage.
292///
293/// # Errors
294///
295/// Returns [`PhysMemoryRegionsLockError::NotInitialized`] if the table is unavailable,
296/// or [`PhysMemoryRegionsLockError::AlreadyLocked`] if either required lock cannot be
297/// acquired immediately.
298pub(super) fn switch_to_final_storage() -> Result<(), PhysMemoryRegionsLockError> {
299    let regions = PHYS_MEM_REGIONS
300        .get()
301        .ok_or(PhysMemoryRegionsLockError::NotInitialized)?;
302
303    let early = try_read_region_table(regions, |table| match table {
304        RuntimeRegions::Early(regions) => Some(regions.clone()),
305        RuntimeRegions::Final(_) => None,
306    })
307    .ok_or(PhysMemoryRegionsLockError::AlreadyLocked)?;
308
309    if let Some(early_regions) = early {
310        // The conversion allocates, so do it without holding the ownership-table lock.
311        let mut final_regions: FinalRegions = early_regions.into();
312        relocate_region_descriptions(&mut final_regions);
313        try_write_region_table(regions, |table| {
314            *table = RuntimeRegions::Final(final_regions);
315        })
316        .ok_or(PhysMemoryRegionsLockError::AlreadyLocked)?;
317    }
318
319    Ok(())
320}

322/// Refreshes early kernel string references before their low identity mappings are removed.
323///
324/// Region descriptions are kernel static strings captured before the high-half jump. Copying the
325/// region entries does not relocate these runtime pointers automatically.
326fn relocate_region_descriptions(regions: &mut FinalRegions) {
327    // SAFETY: We update the region descriptions in place, without changing the ranges or attributes.
328    for region in unsafe { regions.iter_mut() } {
329        let address = VirtAddr::from_ptr_of(region.desc.as_ptr());
330        let desc = if vmm::direct_mapping_virt_to_phys(address).is_some() {
331            region.desc
332        } else {
333            let high = vmm::direct_mapping_phys_to_virt(PhysAddr::from_usize(address.as_usize()))
334                .expect("early region description exceeds the direct map");
335            // SAFETY: Early descriptions are immutable kernel static strings. Both aliases
336            // currently map the same read-only kernel bytes, so length and UTF-8 are unchanged
337            // and the high alias remains valid after the identity alias is removed.
338            unsafe {
339                core::str::from_utf8_unchecked(core::slice::from_raw_parts(
340                    high.as_ptr(),
341                    region.desc.len(),
342                ))
343            }
344        };
345        region.desc = desc;
346    }
347}

349/// Clones the global physical memory regions table if it's using the final storage.
350///
351/// The returned copy can be modified outside the table lock and installed with
352/// [`replace_final_regions`].
353///
354/// # Panics
355///
356/// Panics if the global physical memory regions table is not initialized or is using the early
357/// storage.
358pub(super) fn clone_final_regions() -> FinalRegions {
359    read_region_table(&PHYS_MEM_REGIONS, |table| {
360        let RuntimeRegions::Final(regions) = table else {
361            panic!("early physical region storage is still in use");
362        };
363        regions.clone()
364    })
365}

367/// Replaces the global physical memory regions table with a new set of regions, returning the
368/// previous one.
369///
370/// Replaces the table under an exclusive lock. Callers preparing a replacement from
371/// [`clone_final_regions`] must coordinate the complete read-modify-replace operation
372/// with other writers to avoid overwriting concurrent changes.
373///
374/// # Panics
375///
376/// Panics if the global physical memory regions table is not initialized or is using the early
377/// storage.
378pub(super) fn replace_final_regions(regions: FinalRegions) -> FinalRegions {
379    let previous = write_region_table(&PHYS_MEM_REGIONS, |table| {
380        let RuntimeRegions::Final(current) = table else {
381            panic!("physical region storage is still fixed");
382        };
383        core::mem::replace(current, regions)
384    });
385    previous
386}

388/// Converts an identity-mapped kernel section range to its physical range.
389fn virt_range_to_id_phys_range(range: VirtAddrRange) -> PhysAddrRange {
390    PhysAddrRange::new(
391        PhysAddr::from_usize(range.start.as_usize()),
392        PhysAddr::from_usize(range.end.as_usize()),
393    )
394}

396/// Initializes the global physical memory regions table with the memory regions reported by the
397/// platform, the kernel sections, and the loader range. The table is initialized with the early
398/// storage.
399///
400/// Kernel and loader ranges are installed first. Platform reservations and boot data must not
401/// overlap them. Free RAM fills only gaps left by those occupied regions.
402///
403/// # Panics
404///
405/// Panics if the global physical memory regions table has already been initialized,
406/// kernel or loader regions are invalid or overlap, platform reservations conflict
407/// with existing regions, or the resulting table exceeds the early storage capacity.
408pub(super) fn init_early_phys_mem_regions(raw: &RawMemoryRegions, loader_range: PhysAddrRange) {
409    let mut regions = EarlyRegions::new();
410
411    // Insert the kernel sections first.
412    for section in sections::all_sections() {
413        if section.aligned_range.is_empty() {
414            continue;
415        }
416        regions
417            .insert_disjoint(PhysMemoryRegion {
418                range: virt_range_to_id_phys_range(section.aligned_range),
419                flags: section.flags | MemoryRegionFlags::KERNEL,
420                desc: section.name,
421            })
422            .expect("invalid or overlapping kernel memory section");
423    }
424
425    // Insert the loader range next.
426    regions
427        .insert_disjoint(PhysMemoryRegion {
428            range: loader_range,
429            flags: MemoryRegionFlags::BOOT_SERVICE
430                | MemoryRegionFlags::READ
431                | MemoryRegionFlags::WRITE,
432            desc: "loader",
433        })
434        .expect("invalid or overlapping loader memory region");
435
436    // Insert the non-free platform-reported memory regions next.
437    for region in raw {
438        if !region.flags.contains(MemoryRegionFlags::FREE) {
439            regions
440                .insert_disjoint(*region)
441                .expect("platform reservation conflicts with kernel or loader memory");
442        }
443    }
444
445    // Insert the free platform-reported memory regions last.
446    for region in raw {
447        if region.flags.contains(MemoryRegionFlags::FREE) {
448            regions
449                .insert_uncovered(*region)
450                .expect("too many physical memory regions");
451        }
452    }
453
454    // Initialize the physical memory regions table.
455    PHYS_MEM_REGIONS.init_once(RegionTable::new(RuntimeRegions::Early(regions)));
