11/// Reports a fatal kernel error and terminates execution.
12///
13/// Accepts a format string and optional arguments. Passes the formatted message and the macro
14/// call's module path, file, line, and column to [`terminate`] with [`TerminateReason::KPanic`].
15///
16/// This macro also marks the code path as cold.
17#[macro_export]
18macro_rules! kpanic {
19    ($($arg:tt)*) => {{
20        ::core::hint::cold_path();
21        $crate::terminate(
22            $crate::TerminateReason::KPanic,
23            $crate::TerminateLocation::from_raw_parts(
24                Some(module_path!()),
25                file!(),
26                line!(),
27                column!(),
28            ),
29            format_args!($($arg)*),
30        )
31    }};
32}

34/// The reason for terminating execution.
35///
36/// Identifies a Rust panic, a fatal kernel error, or an unexpected internal error.
37#[derive(Debug, Clone, Copy, PartialEq, Eq)]
38pub enum TerminateReason {
39    /// A Rust panic received by the consumer's panic handler.
40    ///
41    /// Used when a panic handler forwards a Rust panic to the application's termination function.
42    Panic,
43    /// A fatal kernel error reported with [`kpanic!`].
44    ///
45    /// Indicates that execution cannot continue because of a kernel error.
46    KPanic,
47    /// An unexpected internal error.
48    ///
49    /// Used by [`bug!`] and the methods in [`UnwrapOrBug`].
50    Bug,
51}
52
53impl TerminateReason {
54    /// Returns the short label for the termination reason.
55    ///
56    /// Returns `"Panic"`, `"Kernel panic"`, or `"Bug"` for display in an error message.
57    pub fn name(self) -> &'static str {
58        match self {
59            TerminateReason::Panic => "Panic",
60            TerminateReason::KPanic => "Kernel panic",
61            TerminateReason::Bug => "Bug",
62        }
63    }
64
65    /// Returns an explanation of the termination reason.
66    ///
67    /// Provides additional text for an error report, including a request to report internal bugs.
68    pub fn desc(self) -> &'static str {
69        match self {
70            TerminateReason::Panic => {
71                "\"Panic\" means some code called the `panic!` macro, usually from an external crate."
72            }
73            TerminateReason::KPanic => {
74                "\"Kernel panic\" means some code called the `kpanic!` macro, usually from the kernel itself or its components."
75            }
76            TerminateReason::Bug => {
77                "\"Bug\" means an unexpected internal error happened in the kernel or its components. Please report this issue to the maintainers."
78            }
79        }
80    }
81}

83/// The source location of a panic or bug.
84///
85/// Stores a file, line, column, and optional module path for an error report. Macros can supply
86/// all four fields. Rust's [`Location`] provides the file, line, and column, but no module path.
87pub struct TerminateLocation<'a> {
88    /// The caller's module path, when available.
89    ///
90    /// Contains `None` when constructed from a Rust panic location.
91    module_path: Option<&'a str>,
92    /// The caller's source file.
93    ///
94    /// Contains `<unknown>` when constructed without a Rust panic location.
95    file: &'a str,
96    /// The caller's source line.
97    ///
98    /// Contains zero when constructed without a Rust panic location.
99    line: u32,
100    /// The caller's source column.
101    ///
102    /// Contains zero when constructed without a Rust panic location.
103    column: u32,
104}
105
106impl<'a> TerminateLocation<'a> {
107    /// Creates a source location from an optional Rust panic location.
108    ///
109    /// Copies the file, line, and column from [`Location`] and leaves the module path as `None`.
110    /// If no location is supplied, uses `<unknown>` for the file and zero for the line and column.
111    pub fn from_panic_location(location: Option<&Location<'a>>) -> Self {
112        if let Some(location) = location {
113            Self::from_raw_parts(None, location.file(), location.line(), location.column())
114        } else {
115            Self::from_raw_parts(None, "<unknown>", 0, 0)
116        }
117    }
118
119    /// Creates a source location from its individual fields.
120    ///
121    /// Stores the supplied module path, file, line, and column without changing them.
122    pub fn from_raw_parts(
123        module_path: Option<&'a str>,
124        file: &'a str,
125        line: u32,
126        column: u32,
127    ) -> Self {
128        Self {
129            module_path,
130            file,
131            line,
132            column,
133        }
134    }
135
136    /// Returns the caller's module path when available.
137    ///
138    /// Returns `None` when constructed from a Rust panic location.
139    pub fn module_path(&self) -> Option<&'a str> {
140        self.module_path
141    }
142
143    /// Returns the caller's source file.
144    ///
145    /// Returns `<unknown>` when constructed without a Rust panic location.
146    pub fn file(&self) -> &'a str {
147        self.file
148    }
149
150    /// Returns the caller's source line.
151    ///
152    /// Returns zero when constructed without a Rust panic location.
153    pub fn line(&self) -> u32 {
154        self.line
155    }
156
157    /// Returns the caller's source column.
158    ///
159    /// Returns zero when constructed without a Rust panic location.
160    pub fn column(&self) -> u32 {
161        self.column
162    }
163}

165/// The actual execution termination function.
166///
167/// `expanic` uses this function to terminate the execution. [`terminate`] is the default
168/// implementation that calls `core::panic!` with the supplied information in its message. Actual
169/// kernels should provide their own implementation with the same signature as the default and mark
170/// it with the [`#[terminate_handler]`](terminate_handler) attribute.
171///
172/// With a replacement registered, a panic handler can forward ordinary Rust panics through this
173/// `terminate` interface. Without a replacement, the default implementation calls `core::panic!`,
174/// so invoking that fallback from a panic handler would start another panic.
175#[eii(terminate_handler)]
176pub fn terminate(
177    reason: TerminateReason,
178    location: TerminateLocation<'_>,
179    args: fmt::Arguments<'_>,
180) -> ! {
181    terminate_fallback(reason, location, args)
182}
183
184/// Keeps the default termination symbol available to optimized standalone consumers.
185///
186/// EII normally emits the default implementation only when the defining crate references its
187/// public symbol. This reference preserves that weak symbol for consumers that use the fallback.
188#[used]
189static DEFAULT_TERMINATE: fn(
190    TerminateReason,
191    TerminateLocation<'static>,
192    fmt::Arguments<'static>,
193) -> ! = terminate;
194
195/// Panics with the supplied reason, source location, and message.
196///
197/// Includes the original location in the panic message so it remains available to a panic handler.
198#[inline(never)]
199fn terminate_fallback(
200    reason: TerminateReason,
201    location: TerminateLocation<'_>,
202    args: fmt::Arguments<'_>,
203) -> ! {
204    core::panic!(
205        "[{}:{}] {} at {}:{}:{}:\n{}\n\n{}",
206        location.module_path().unwrap_or("<unknown module>"),
207        location.line(),
208        reason.name(),
209        location.file(),
210        location.line(),
211        location.column(),
212        args,
213        reason.desc(),
214    )
215}
