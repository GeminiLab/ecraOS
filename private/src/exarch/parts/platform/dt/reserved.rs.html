13/// The description attached to regions reserved by a device tree.
14const DEVICE_TREE_RESERVED_DESC: &str = "device-tree reserved memory";

16/// A statically located range declared by a device tree reservation mechanism.
17#[derive(Clone, Copy)]
18struct DeviceTreeReservedRegion {
19    /// The physical range that firmware reserves from general use.
20    range: PhysAddrRange,
21    /// Whether the standard direct mapping must omit this range.
22    no_map: bool,
23}
24
25/// A direct child of the device-tree reserved-memory node while it is being parsed.
26struct ReservedMemoryChild<'dt> {
27    /// The structure-token depth of this child node.
28    depth: usize,
29    /// The child reg property, if it describes a static physical range.
30    reg: Option<&'dt [u8]>,
31    /// Whether the child has a no-map property.
32    no_map: bool,
33    /// Whether the child has a reusable property.
34    reusable: bool,
35    /// Whether firmware marks the child enabled.
36    enabled: bool,
37}
38
39/// Records all static FDT memory reservations in the physical region table.
40///
41/// Reads the FDT reservation map and static reg properties below /reserved-memory. Complete
42/// ranges are recorded, including gaps in discovered RAM. Existing access and device attributes
43/// are retained. Gaps receive only a RESERVED or NO_MAP role. A /reserved-memory child with no-map
44/// becomes a NO_MAP region so both early aliases omit it. Dynamic size requests are not allocated
45/// here because this function runs before the kernel allocator is available.
46///
47/// # Errors
48///
49/// Returns an error when the reservation map, structure block, reserved-memory properties, or
50/// region capacity is invalid. All failures leave the input region table unchanged.
51pub fn apply_device_tree_reservations(
52    dev_tree: DevTree<'_>,
53    regions: &mut RawMemoryRegions,
54) -> Result<(), DevTreeError> {
55    let mut reserved = HeaplessVec::<DeviceTreeReservedRegion, MAX_RAW_MEM_REGIONS>::new();
56
57    // Parse both reservation sources before changing the discovered memory regions.
58    parse_reservation_map(dev_tree, &mut reserved)?;
59    parse_reserved_memory_nodes(dev_tree, &mut reserved)?;
60
61    // Keep all input regions intact if any reservation exceeds the fixed storage capacity.
62    let mut adjusted = regions.clone();
63    for reservation in reserved {
64        apply_reserved_region(&mut adjusted, reservation)?;
65    }
66    *regions = adjusted;
67
68    Ok(())
69}

71/// Reads entries from the FDT header's memory-reservation block.
72fn parse_reservation_map(
73    dev_tree: DevTree<'_>,
74    reserved: &mut HeaplessVec<DeviceTreeReservedRegion, MAX_RAW_MEM_REGIONS>,
75) -> Result<(), DevTreeError> {
76    let bytes = dev_tree.buf();
77    let mut offset = dev_tree.off_mem_rsvmap();
78    let end = dev_tree.off_dt_struct();
79
80    // The reservation map must be aligned and fit before the structure block begins.
81    if !offset.is_multiple_of(core::mem::align_of::<u64>()) || offset > end {
82        return Err(DevTreeError::InvalidOffset);
83    }
84
85    loop {
86        // Each entry contains two big-endian u64 values; (0, 0) terminates the map.
87        let entry_end = offset
88            .checked_add(2 * core::mem::size_of::<u64>())
89            .ok_or(DevTreeError::InvalidOffset)?;
90        let entry = bytes
91            .get(offset..entry_end)
92            .filter(|_| entry_end <= end)
93            .ok_or(DevTreeError::InvalidOffset)?;
94        let address = u64::from_be_bytes(
95            entry[..8]
96                .try_into()
97                .map_err(|_| DevTreeError::ParseError)?,
98        );
99        let size = u64::from_be_bytes(
100            entry[8..]
101                .try_into()
102                .map_err(|_| DevTreeError::ParseError)?,
103        );
104        if address == 0 && size == 0 {
105            return Ok(());
106        }
107        if size == 0 {
108            return Err(DevTreeError::ParseError);
109        }
110
111        reserved
112            .push(DeviceTreeReservedRegion {
113                range: physical_range(address, size)?,
114                no_map: false,
115            })
116            .map_err(|_| DevTreeError::NotEnoughMemory)?;
117        offset = entry_end;
118    }
119}

121/// Reads static reg ranges from direct children of the /reserved-memory node.
122fn parse_reserved_memory_nodes(
123    dev_tree: DevTree<'_>,
124    reserved: &mut HeaplessVec<DeviceTreeReservedRegion, MAX_RAW_MEM_REGIONS>,
125) -> Result<(), DevTreeError> {
126    let mut tokens = dev_tree.parse_iter();
127    let mut depth = 0usize;
128    let mut reserved_node_depth = None;
129    let mut address_cells = None;
130    let mut size_cells = None;
131    let mut ranges_empty = None;
132    let mut child: Option<ReservedMemoryChild<'_>> = None;
133
134    // Stream the structure once. Depth tracking identifies the reserved-memory node and its
135    // direct children without building an allocator-backed index.
136    while let Some(token) = tokens.next()? {
137        match token {
138            ParsedTok::BeginNode(node) => {
139                if depth == 1 && node.name == b"reserved-memory" {
140                    reserved_node_depth = Some(depth);
141                    address_cells = None;
142                    size_cells = None;
143                    ranges_empty = None;
144                } else if depth
145                    .checked_sub(1)
146                    .is_some_and(|parent_depth| reserved_node_depth == Some(parent_depth))
147                {
148                    child = Some(ReservedMemoryChild {
149                        depth,
150                        reg: None,
151                        no_map: false,
152                        reusable: false,
153                        enabled: true,
154                    });
155                }
156                depth = depth.checked_add(1).ok_or(DevTreeError::ParseError)?;
157            }
158            ParsedTok::Prop(property) => {
159                let owner_depth = depth.checked_sub(1).ok_or(DevTreeError::ParseError)?;
160                let name = property_name(dev_tree, property.name_offset)?;
161
162                // Cell counts and ranges belong to the parent; reg and policy properties belong
163                // only to its direct reservation children.
164                if reserved_node_depth == Some(owner_depth) {
165                    match name {
166                        "#address-cells" => {
167                            address_cells = Some(read_cell_count(property.prop_buf)?);
168                        }
169                        "#size-cells" => {
170                            size_cells = Some(read_cell_count(property.prop_buf)?);
171                        }
172                        "ranges" => ranges_empty = Some(property.prop_buf.is_empty()),
173                        _ => {}
174                    }
175                } else if let Some(child) = child.as_mut()
176                    && child.depth == owner_depth
177                {
178                    match name {
179                        "reg" => child.reg = Some(property.prop_buf),
180                        "no-map" if property.prop_buf.is_empty() => child.no_map = true,
181                        "no-map" => return Err(DevTreeError::ParseError),
182                        "reusable" if property.prop_buf.is_empty() => child.reusable = true,
183                        "reusable" => return Err(DevTreeError::ParseError),
184                        "status" => child.enabled = status_is_enabled(property.prop_buf)?,
185                        _ => {}
186                    }
187                }
188            }
189            ParsedTok::EndNode => {
190                let ended_depth = depth.checked_sub(1).ok_or(DevTreeError::ParseError)?;
191
192                // Defer decoding reg until the child closes so its properties can appear in any
193                // valid order.
194                if child
195                    .as_ref()
196                    .is_some_and(|child| child.depth == ended_depth)
197                {
198                    let child = child.take().ok_or(DevTreeError::ParseError)?;
199                    // The DT specification makes no-map and reusable mutually exclusive.
200                    if child.no_map && child.reusable {
201                        return Err(DevTreeError::ParseError);
202                    }
203                    if child.enabled
204                        && let Some(reg) = child.reg
205                    {
206                        let address_cells = address_cells.ok_or(DevTreeError::InvalidParameter(
207                            "reserved-memory is missing #address-cells",
208                        ))?;
209                        let size_cells = size_cells.ok_or(DevTreeError::InvalidParameter(
210                            "reserved-memory is missing #size-cells",
211                        ))?;
212                        append_reg_ranges(reg, address_cells, size_cells, child.no_map, reserved)?;
213                    }
214                }
215                if reserved_node_depth == Some(ended_depth) {
216                    if ranges_empty != Some(true) {
217                        return Err(DevTreeError::InvalidParameter(
218                            "reserved-memory must have an empty ranges property",
219                        ));
220                    }
221                    reserved_node_depth = None;
222                }
223                depth = ended_depth;
224            }
225            ParsedTok::Nop => {}
226        }
227    }
228
229    Ok(())
230}

232/// Checks whether a null-terminated status property enables its node.
233fn status_is_enabled(bytes: &[u8]) -> Result<bool, DevTreeError> {
234    let Some(value) = bytes.strip_suffix(&[0]) else {
235        return Err(DevTreeError::ParseError);
236    };
237    let value = core::str::from_utf8(value).map_err(DevTreeError::from)?;
238    Ok(matches!(value, "okay" | "ok"))
239}

241/// Returns a property name from the FDT strings block.
242fn property_name<'dt>(dev_tree: DevTree<'dt>, offset: usize) -> Result<&'dt str, DevTreeError> {
243    let strings_start = dev_tree.off_dt_strings();
244    let strings_end = strings_start
245        .checked_add(dev_tree.size_dt_strings() as usize)
246        .ok_or(DevTreeError::InvalidOffset)?;
247    let strings = dev_tree
248        .buf()
249        .get(strings_start..strings_end)
250        .ok_or(DevTreeError::InvalidOffset)?;
251    let suffix = strings.get(offset..).ok_or(DevTreeError::InvalidOffset)?;
252    let length = suffix
253        .iter()
254        .position(|byte| *byte == 0)
255        .ok_or(DevTreeError::ParseError)?;
256    core::str::from_utf8(&suffix[..length]).map_err(DevTreeError::from)
257}

259/// Decodes a one- or two-cell count from a reserved-memory property.
260fn read_cell_count(bytes: &[u8]) -> Result<usize, DevTreeError> {
261    let [a, b, c, d] = bytes else {
262        return Err(DevTreeError::ParseError);
263    };
264    let count = u32::from_be_bytes([*a, *b, *c, *d]) as usize;
265    if !(1..=2).contains(&count) {
266        return Err(DevTreeError::InvalidParameter(
267            "reserved-memory cell counts must be one or two",
268        ));
269    }
270    Ok(count)
271}

273/// Decodes static physical ranges from a reserved-memory reg property.
274fn append_reg_ranges(
275    reg: &[u8],
276    address_cells: usize,
277    size_cells: usize,
278    no_map: bool,
279    reserved: &mut HeaplessVec<DeviceTreeReservedRegion, MAX_RAW_MEM_REGIONS>,
280) -> Result<(), DevTreeError> {
281    // A reg value is a sequence of address/size tuples sized by the parent cell counts.
282    let tuple_size = (address_cells + size_cells)
283        .checked_mul(core::mem::size_of::<u32>())
284        .ok_or(DevTreeError::InvalidOffset)?;
285    if reg.is_empty() || !reg.len().is_multiple_of(tuple_size) {
286        return Err(DevTreeError::ParseError);
287    }
288
289    for tuple in reg.chunks_exact(tuple_size) {
290        let address_size = address_cells * core::mem::size_of::<u32>();
291        let address = decode_cells(&tuple[..address_size], address_cells)?;
292        let size = decode_cells(&tuple[address_size..], size_cells)?;
293        if size == 0 {
294            return Err(DevTreeError::ParseError);
295        }
296        reserved
297            .push(DeviceTreeReservedRegion {
298                range: physical_range(address, size)?,
299                no_map,
300            })
301            .map_err(|_| DevTreeError::NotEnoughMemory)?;
302    }
303
304    Ok(())
305}

307/// Decodes a one- or two-cell big-endian integer.
308fn decode_cells(bytes: &[u8], count: usize) -> Result<u64, DevTreeError> {
309    match count {
310        1 => Ok(u32::from_be_bytes(bytes.try_into().map_err(|_| DevTreeError::ParseError)?) as u64),
311        2 => Ok(u64::from_be_bytes(
312            bytes.try_into().map_err(|_| DevTreeError::ParseError)?,
313        )),
314        _ => Err(DevTreeError::InvalidParameter(
315            "reserved-memory cell counts must be one or two",
316        )),
317    }
318}

320/// Creates a semantic physical range and rejects address-space overflow.
321fn physical_range(address: u64, size: u64) -> Result<PhysAddrRange, DevTreeError> {
322    let end = address
323        .checked_add(size)
324        .ok_or(DevTreeError::InvalidOffset)?;
325    let start = usize::try_from(address).map_err(|_| DevTreeError::InvalidOffset)?;
326    let end = usize::try_from(end).map_err(|_| DevTreeError::InvalidOffset)?;
327    Ok(PhysAddrRange::new(pa!(start), pa!(end)))
328}

330/// Records a complete reservation while preserving existing access and device attributes.
331fn apply_reserved_region(
332    regions: &mut RawMemoryRegions,
333    reservation: DeviceTreeReservedRegion,
334) -> Result<(), DevTreeError> {
335    // Preserve evidence of required mappings before replacing any role with NO_MAP. Otherwise
336    // the page planner would only see a barrier and silently omit boot data or the console.
337    let required = MemoryRegionFlags::BOOT_SERVICE
338        | MemoryRegionFlags::KERNEL
339        | MemoryRegionFlags::DEVICE
340        | MemoryRegionFlags::EXECUTE;
341    if reservation.no_map
342        && regions.overlaps(reservation.range).any(|old| {
343            !old.flags.contains(MemoryRegionFlags::NO_MAP) && old.flags.intersects(required)
344        })
345    {
346        return Err(DevTreeError::InvalidParameter(
347            "no-map reservation overlaps a required early mapping",
348        ));
349    }
350    let role = if reservation.no_map {
351        MemoryRegionFlags::NO_MAP
352    } else {
353        MemoryRegionFlags::RESERVED
354    };
355    regions
356        .overlay_with(
357            PhysMemoryRegion {
358                range: reservation.range,
359                // Gaps have no known access or device attributes. Record ownership only.
360                flags: role,
361                desc: DEVICE_TREE_RESERVED_DESC,
362            },
363            |old| {
364                // An ordinary reservation cannot weaken an existing no-map declaration.
365                if old.flags.contains(MemoryRegionFlags::NO_MAP)
366                    || !reservation.no_map && !old.flags.contains(MemoryRegionFlags::FREE)
367                {
368                    return old;
369                }
370                let roles = MemoryRegionFlags::FREE
371                    | MemoryRegionFlags::RESERVED
372                    | MemoryRegionFlags::NO_MAP
373                    | MemoryRegionFlags::BOOT_SERVICE
374                    | MemoryRegionFlags::KERNEL;
375                PhysMemoryRegion {
376                    flags: (old.flags & !roles) | role,
377                    desc: DEVICE_TREE_RESERVED_DESC,
378                    ..old
379                }
380            },
381        )
382        .map_err(|_| DevTreeError::NotEnoughMemory)
