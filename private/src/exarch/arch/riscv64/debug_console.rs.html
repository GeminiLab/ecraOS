5/// The maximum number of bytes that can be read at once.
6const MAX_RW_SIZE: usize = 256;
7
8/// Tries to write bytes to the console from input u8 slice.
9/// Returns the number of bytes written.
10fn try_write_bytes(bytes: &[u8]) -> usize {
11    let ret = sbi_rt::console_write(sbi_rt::Physical::new(
12        bytes.len().min(MAX_RW_SIZE),
13        virt_to_phys(VirtAddr::from_ptr_of(bytes.as_ptr())).as_usize(),
14        0,
15    ));
16    if ret.error == 0 { ret.value } else { 0 }
17}
18
19/// Writes bytes to the console from input u8 slice.
20pub fn write_bytes(bytes: &[u8]) {
21    let mut write_len = 0;
22    let mut buf = [0; MAX_RW_SIZE];
23    while write_len < bytes.len() {
24        let n = buf.len().min(bytes.len() - write_len);
25        if n == 0 {
26            break;
27        }
28        // `bytes` can be from user space, copy it into a kernel buffer
29        // to correctly use `virt_to_phys`.
30        buf[..n].copy_from_slice(&bytes[write_len..write_len + n]);
31        let n = try_write_bytes(&buf[..n]);
32        if n == 0 {
33            break;
34        }
35        write_len += n;
36    }
37}
38
39/// Reads bytes from the console into the given mutable slice.
40/// Returns the number of bytes read.
41pub fn read_bytes(bytes: &mut [u8]) -> usize {
42    sbi_rt::console_read(sbi_rt::Physical::new(
43        bytes.len().min(MAX_RW_SIZE),
44        virt_to_phys(VirtAddr::from_mut_ptr_of(bytes.as_mut_ptr())).as_usize(),
45        0,
46    ))
47    .value
48}
