10/// Calibrates the LAPIC one-shot countdown against the PIT reference clock.

17static mut INIT_TICK: Ticks = Ticks(0);
18static mut TSC_FREQ_KHZ: u64 = 0;
19
20/// TSC frequency used when calibration fails. It's almost definitely wrong, but it should be enough
21/// for the system to boot and enumerate other time devices.
22const TSC_FREQ_KHZ_BLIND: u64 = 2_000_000;
23
24fn current_ticks() -> Ticks {
25    unsafe { Ticks(_rdtsc()) }
26}
27
28/// Converts nanoseconds to LAPIC countdown ticks.
29///
30/// The result is clamped to the nonzero range accepted by the 32-bit initial-count register.
31fn nanos_to_lapic_ticks(nanos: u64, frequency_khz: u64) -> u32 {
32    let ticks = nanos.saturating_mul(frequency_khz) / 1_000_000;
33    ticks.clamp(1, u32::MAX as u64) as u32
34}

36pub fn init_early() {
37    unsafe {
38        INIT_TICK = current_ticks();
39        TSC_FREQ_KHZ = calibration::calibrate_tsc_early().unwrap_or_else(|| {
40            dbcn_println!(
41                "TSC calibration failed, using blind value: {} kHz",
42                TSC_FREQ_KHZ_BLIND
43            );
44            TSC_FREQ_KHZ_BLIND
45        });
46    }
47}

49/// Returns monotonic TSC ticks since early initialization.
50pub fn monotonic_ticks() -> Ticks {
51    unsafe { Ticks(current_ticks().0 - INIT_TICK.0) }
52}
53
54/// Converts TSC ticks to nanoseconds.
55pub fn ticks_to_nanos(ticks: Ticks) -> Nanos {
56    unsafe { Nanos(ticks.0 * 1_000_000 / TSC_FREQ_KHZ) }
57}
58
59/// Converts nanoseconds to TSC ticks.
60pub fn nanos_to_ticks(nanos: Nanos) -> Ticks {
61    unsafe { Ticks(nanos.0 * TSC_FREQ_KHZ / 1_000_000) }
62}

64/// Programs an x86 one-shot timer for an absolute monotonic deadline.
65pub fn set_oneshot_timer(deadline: crate::time::TimeValue) {
66    let deadline_ns =
67        u64::try_from(deadline.as_nanos()).expect("x86 timer deadline exceeds u64 nanoseconds");
68    if super::imp::apic::tsc_deadline_supported() {
69        let deadline_ticks = unsafe { INIT_TICK.0 }
70            .checked_add(nanos_to_ticks(Nanos(deadline_ns)).0)
71            .expect("x86 timer deadline overflows TSC");
72        super::imp::apic::program_tsc_deadline(deadline_ticks.max(current_ticks().0 + 1));
73        return;
74    }
75    let now_ns = ticks_to_nanos(monotonic_ticks()).0;
76    let delta_ns = deadline_ns.saturating_sub(now_ns);
77    let count = nanos_to_lapic_ticks(delta_ns, super::imp::apic::timer_frequency_khz());
78    super::imp::apic::program_timer_initial(count);
79}

85    /// Verifies conversion in the LAPIC timer frequency domain.
86    #[test]
87    fn converts_nanoseconds_in_lapic_frequency_domain() {
88        assert_eq!(nanos_to_lapic_ticks(10_000_000, 1_000_000), 10_000_000);
89        assert_eq!(nanos_to_lapic_ticks(10_000_000, 3_500_000), 35_000_000);
90    }

92    /// Verifies clamping to the LAPIC initial-count register range.
93    #[test]
94    fn clamps_lapic_timer_count_to_hardware_range() {
95        assert_eq!(nanos_to_lapic_ticks(0, 1_000_000), 1);
96        assert_eq!(nanos_to_lapic_ticks(u64::MAX, u64::MAX), u32::MAX);
