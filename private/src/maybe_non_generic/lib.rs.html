25/// The kind of a replaced generic parameter.

27/// The kind decides how uses of the parameter are rewritten in the generated copy.

29enum ReplacementKind {
30    /// A const generic parameter.
31    ///
32    /// Its uses become the generated argument value.
33    Const,
34    /// A type generic parameter.
35    ///
36    /// Its associated items become fields of the generated argument.

40/// A generic parameter replaced by a trailing function parameter.

42/// It is parsed from a `const(...)` or `type(...)` macro argument.

44struct Replacement {

47    /// It must match the kind of the parameter in the function signature.

102/// The validated arguments of one macro invocation.
103///
104/// They describe the name, signature changes, body rewrites, and attributes of the generated copy.
105#[derive(Debug)]
106struct MacroArgs {
107    /// The name of the generated copy.
108    ///
109    /// It replaces the original function name in the copied signature.

152/// Generates a copy of a generic function with selected generic parameters replaced.
153///
154/// The attribute applies to a generic function with a body, which may be a free function, a
155/// function in an `impl` block, or a default method of a trait. The original function is emitted
156/// unchanged. The copy uses the name given by the first attribute argument, and each replaced
157/// generic parameter becomes a trailing function parameter. A const generic becomes a runtime
158/// value, and a type generic becomes a value that stands in for the type, such as a table of
159/// function pointers.
160///
161/// For example, the following copy takes the number of summed values as its last argument:
162///
163/// ```
164/// use maybe_non_generic::maybe_non_generic;
165///
166/// #[maybe_non_generic(sum_dyn, const(N => count: usize))]
167/// fn sum<const N: usize>(values: &[u32]) -> u32 {
168///     values.iter().take(N).sum()
169/// }
170///
171/// assert_eq!(sum::<2>(&[1, 2, 3]), 3);
172/// assert_eq!(sum_dyn(&[1, 2, 3], 2), 3);
173/// ```
174///
175/// # Syntax
176///
177/// The first argument names the copy:
178///
179/// ```text
180/// #[maybe_non_generic(copy_name, argument, ...)]
181/// ```
182///
183/// The name can be followed by any number of comma-separated arguments, and a trailing comma is
184/// allowed. Each argument has one of the following forms:
185///
186/// - `const(N => n: T)` replaces the const generic `N` with the trailing parameter `n: T`.
187/// - `type(H => h: T)` replaces the type generic `H` with the trailing parameter `h: T`.
188/// - `fn(path::f => path::g)` renames calls to the function path `path::f` in the copied body.
189/// - `fn(receiver.f => receiver.g)` renames method calls `receiver.f(...)` in the copied body.
190/// - `attr(...)` adds the attribute `#[...]` to the copy.
191/// - `dont_copy_attr` stops the attributes of the original function from being copied.
192///
193/// # Generic parameter replacement
194///
195/// Each `const(...)` or `type(...)` argument names one generic parameter of the function and the
196/// parameter that replaces it. The arguments must follow the order of the generic parameters in
197/// the function signature, and the new parameters are appended in that order. Generic parameters
198/// that are not replaced, including lifetimes, stay on the copy. Bounds and `where` predicates that
199/// mention a replaced parameter are removed.
200///
201/// ## Body rewrites
202///
203/// The body of the copy is rewritten as follows:
204///
205/// - A const parameter `N` used as an expression becomes `n`.
206/// - A call `H::f(args)` becomes `(h.f)(args)`, and an expression `H::ITEM` becomes `h.ITEM`. The
207///   function-pointer tables generated by `dyn_static_traits` have this shape.
208/// - In a call with a turbofish, such as `f::<N, H>(args)`, replaced generic arguments are removed
209///   from the last path segment and appended to the call arguments. A const argument passes `n`,
210///   and a type argument passes `::core::clone::Clone::clone(&h)`, so the type of `h` must
211///   implement `Clone`. References and `dyn_static_traits` tables do.
212///
213/// Nested items, such as functions declared inside the body, are left unchanged, because they
214/// cannot use the generic parameters of the outer function.
215///
216/// The original callee still expects its generic arguments, so a call that passes replaced
217/// arguments usually also needs a `fn(...)` rename to the callee's own copy. The following example
218/// replaces a clock type with a table generated by `dyn_static_traits` and renames the inner call:
219///
220/// ```
221/// use dyn_static_traits::dyn_static_traits;
222/// use maybe_non_generic::maybe_non_generic;
223///
224/// #[dyn_static_traits(DynClock)]
225/// trait Clock {
226///     fn now() -> u64;
227/// }
228///
229/// struct BootClock;
230///
231/// impl Clock for BootClock {
232///     fn now() -> u64 {
233///         100
234///     }
235/// }
236///
237/// #[maybe_non_generic(elapsed_dyn, type(C => clock: &DynClock))]
238/// fn elapsed<C: Clock>(start: u64) -> u64 {
239///     C::now() - start
240/// }
241///
242/// #[maybe_non_generic(
243///     scaled_elapsed_dyn,
244///     const(N => factor: u64),
245///     type(C => clock: &DynClock),
246///     fn(elapsed => elapsed_dyn),
247/// )]
248/// fn scaled_elapsed<const N: u64, C: Clock>(start: u64) -> u64 {
249///     elapsed::<C>(start) * N
250/// }
251///
252/// let clock = DynClock::new::<BootClock>();
253/// assert_eq!(scaled_elapsed::<2, BootClock>(40), 120);
254/// assert_eq!(scaled_elapsed_dyn(40, 2, &clock), 120);
255/// ```
256///
257/// In `scaled_elapsed_dyn`, the call `elapsed::<C>(start)` becomes
258/// `elapsed_dyn(start, ::core::clone::Clone::clone(&clock))`, and `N` becomes `factor`.
259///
260/// # Call renaming
261///
262/// A `fn(path => path)` argument renames calls whose function path has the same segment names as
263/// the source path. Generic arguments are ignored while matching, and arguments that are not
264/// replaced are kept, so `fn(Table::size => Table::size_dyn)` also matches
265/// `Table::<M>::size::<L>(...)`. Both paths must have the same number of segments.
266///
267/// A `fn(receiver.method => receiver.method)` argument renames method calls whose receiver has the
268/// same tokens as the given receiver, such as `self.clear(...)`. The source and target receivers
269/// must be the same. Replaced generic arguments in the method turbofish are appended to the call
270/// arguments.
271///
272/// The following example renames an associated function call and a method call in an `impl`
273/// block:
274///
275/// ```
276/// use maybe_non_generic::maybe_non_generic;
277///
278/// struct Table {
279///     entries: usize,
280/// }
281///
282/// impl Table {
283///     #[maybe_non_generic(entry_size_dyn, const(LEVEL => level: usize))]
284///     fn entry_size<const LEVEL: usize>() -> usize {
285///         8 << LEVEL
286///     }
287///
288///     #[maybe_non_generic(
289///         bytes_dyn,
290///         const(LEVEL => level: usize),
291///         fn(Self::entry_size => Self::entry_size_dyn),
292///     )]
293///     fn bytes<const LEVEL: usize>(&self) -> usize {
294///         Self::entry_size::<LEVEL>() * self.entries
295///     }
296///
297///     #[maybe_non_generic(
298///         double_bytes_dyn,
299///         const(LEVEL => level: usize),
300///         fn(self.bytes => self.bytes_dyn),
301///     )]
302///     fn double_bytes<const LEVEL: usize>(&self) -> usize {
303///         self.bytes::<LEVEL>() * 2
304///     }
305/// }
306///
307/// let table = Table { entries: 4 };
308/// assert_eq!(table.bytes::<1>(), 64);
309/// assert_eq!(table.bytes_dyn(1), 64);
310/// assert_eq!(table.double_bytes::<1>(), 128);
311/// assert_eq!(table.double_bytes_dyn(1), 128);
312/// ```
313///
314/// # Attributes
315///
316/// The copy receives the attributes of the original function, including doc comments, followed by
317/// the attributes from `attr(...)` arguments. `#[maybe_non_generic]` attributes are never copied,
318/// so stacking several of them on one function generates one copy per attribute. With
319/// `dont_copy_attr`, only the `attr(...)` attributes are applied.
320///
321/// The following example generates two copies of one function with different doc comments:
322///
323/// ```
324/// use maybe_non_generic::maybe_non_generic;
325///
326/// /// Sums the first `N` values.
327/// #[maybe_non_generic(sum_dyn, const(N => count: usize))]
328/// #[maybe_non_generic(
329///     sum_counted,
330///     const(N => count: usize),
331///     dont_copy_attr,
332///     attr(doc = "Sums the first `count` values."),
333/// )]
334/// fn sum<const N: usize>(values: &[u32]) -> u32 {
335///     values.iter().take(N).sum()
336/// }
337///
338/// assert_eq!(sum_dyn(&[1, 2, 3], 2), 3);
339/// assert_eq!(sum_counted(&[1, 2, 3], 2), 3);
340/// ```
341///
342/// `sum_dyn` receives the doc comment of `sum`, while `sum_counted` receives only the doc comment
343/// from `attr(...)`.
344///
345/// # Limitations
346///
347/// The macro reports a compile error instead of generating a copy that might still depend on a
348/// replaced parameter. It rejects:
349///
350/// - items other than functions with a body,
351/// - functions without generic parameters,
352/// - replacements that name a missing generic parameter, a parameter of the other kind, or the
353///   same parameter twice,
354/// - generated parameter names that are already bound by a function parameter or in the body,
355/// - function parameter types or a return type that still mention a replaced parameter,
356/// - generic arguments on an associated function of a replaced type, such as `H::f::<T>()`,
357/// - replaced parameters that remain in the body after rewriting, for example in a type
358///   annotation, in a generic argument expression such as `{ N - 1 }`, or in a qualified path such
359///   as `<H as Trait>::f()`.
360///
361/// The macro cannot see inside some syntax, so it treats that syntax as a remaining use of a
362/// replaced parameter. The body of the copy therefore cannot contain macro invocations, or
363/// patterns with literals, ranges, or paths, such as `0`, `1..=9`, or `Ordering::Less`. This
364/// applies even when the syntax does not mention a replaced parameter.
365#[proc_macro_attribute]
366pub fn maybe_non_generic(attrs: TokenStream1, input: TokenStream1) -> TokenStream1 {
367    maybe_non_generic_impl(attrs.into(), input.into())
368        .unwrap_or_else(|error| error.to_compile_error())
369        .into()
370}

372/// Expands the attribute arguments and annotated function into the original and its copy.
373///
374/// Free functions are tried before functions in `impl` blocks. Any other item is rejected.
375fn maybe_non_generic_impl(attrs: TokenStream, input: TokenStream) -> Result<TokenStream> {
376    let args = syn::parse2::<MacroArgs>(attrs)?;
377
378    if let Ok(item) = syn::parse2::<ItemFn>(input.clone()) {
379        return expand_item_fn(&args, item);
380    }
381
382    if let Ok(item) = syn::parse2::<ImplItemFn>(input) {
383        return expand_impl_item_fn(&args, item);
384    }
385
386    Err(syn::Error::new(
387        args.output.span(),
388        "maybe_non_generic can only be applied to functions with bodies",
389    ))
390}

392/// Returns whether an original attribute is copied to the generated copy.
393///
394/// Every attribute except `maybe_non_generic` is copied, so stacked invocations do not produce
395/// copies of copies.
396fn should_copy_attr(attr: &Attribute) -> bool {
397    attr.path()
398        .segments
399        .last()
400        .is_none_or(|segment| segment.ident != "maybe_non_generic")
401}

403/// Builds the attribute list of the generated copy.
404///
405/// The copied original attributes come first, followed by the attributes from `attr(...)`.
406fn attrs_for_copy(original: &[Attribute], args: &MacroArgs) -> Vec<Attribute> {
407    let mut attrs = Vec::new();
408
409    if args.copy_attrs {
410        attrs.extend(
411            original
412                .iter()
413                .filter(|attr| should_copy_attr(attr))
414                .cloned(),
415        );
416    }
417
418    attrs.extend(args.extra_attrs.iter().cloned());
419    attrs
420}

426    let mut copy = item.clone();
427    copy.attrs = attrs_for_copy(&item.attrs, args);
428    copy.sig = signature_for_copy(&item.sig, args)?;

431    Ok(quote! {

433        #copy

452/// The parsing implementation for the macro arguments.
453///
454/// It reads the copy name followed by comma-separated arguments.
455impl Parse for MacroArgs {
456    /// Parses and validates all macro arguments.
457    ///
458    /// A trailing comma after the last argument is accepted.
459    fn parse(input: ParseStream<'_>) -> Result<Self> {
460        let output = input.parse::<Ident>()?;
461        let mut args = Vec::new();
462
463        while input.parse::<Option<Token![,]>>()?.is_some() {
464            if input.is_empty() {
465                break;
466            }
467
468            args.push(input.parse::<MacroArg>()?);
469        }
470
471        validate_args(output, args)
472    }
473}

475/// The parsing implementation for one macro argument.
476///
477/// It selects the argument form from its leading keyword or identifier.
478impl Parse for MacroArg {
479    /// Parses one `const`, `type`, `fn`, `attr`, or `dont_copy_attr` argument.
480    ///
481    /// Unknown identifiers are rejected.
482    fn parse(input: ParseStream<'_>) -> Result<Self> {
483        if input.peek(Token![const]) {
484            input.parse::<Token![const]>()?;
485            parse_replacement(input, ReplacementKind::Const).map(MacroArg::Replacement)
486        } else if input.peek(Token![type]) {
487            input.parse::<Token![type]>()?;
488            parse_replacement(input, ReplacementKind::Type).map(MacroArg::Replacement)
489        } else if input.peek(Token![fn]) {
490            input.parse::<Token![fn]>()?;
491            parse_fn_replacement(input).map(MacroArg::FnReplacement)
492        } else {
493            let name = input.parse::<Ident>()?;
494            if name == "dont_copy_attr" {
495                Ok(MacroArg::DontCopyAttr)
496            } else if name == "attr" {
497                parse_attr(input).map(MacroArg::Attr)
498            } else {
499                Err(syn::Error::new(
500                    name.span(),
501                    "unknown maybe_non_generic argument",
502                ))
503            }
504        }
505    }
506}

508/// Parses the parenthesized body of a `const(...)` or `type(...)` argument.
509///
510/// The body must contain exactly one `Param => name: Type` replacement.
511fn parse_replacement(input: ParseStream<'_>, kind: ReplacementKind) -> Result<Replacement> {
512    let content;
513    parenthesized!(content in input);
514
515    let param = content.parse::<Ident>()?;
516    content.parse::<Token![=>]>()?;
517    let arg_name = content.parse::<Ident>()?;
518    content.parse::<Token![:]>()?;
519    let arg_type = content.parse::<Type>()?;
520
521    if !content.is_empty() {
522        return Err(content.error("const/type argument must contain exactly one replacement"));
523    }
524
525    Ok(Replacement {
526        kind,
527        param,
528        arg_name,
529        arg_type,
530    })
531}

533/// Parses the parenthesized body of a `fn(...)` argument.
534///
535/// The body must contain exactly one `source => target` replacement.
536fn parse_fn_replacement(input: ParseStream<'_>) -> Result<FnReplacement> {
537    let content;
538    parenthesized!(content in input);
539
540    let from = parse_fn_target(&content)?;
541    content.parse::<Token![=>]>()?;
542    let to = parse_fn_target(&content)?;
543
544    if !content.is_empty() {
545        return Err(content.error("fn argument must contain exactly one replacement"));
546    }
547
548    Ok(FnReplacement { from, to })
549}

551/// Parses one side of a function replacement.
552///
553/// Tokens are read up to the next `=>`. A field expression such as `self.clear` is a method
554/// target, and anything else must be a path.
555fn parse_fn_target(input: ParseStream<'_>) -> Result<FnTarget> {
556    let target = input.step(|cursor| {
557        let mut rest = *cursor;
558        let mut tokens = TokenStream::new();
559
560        while let Some((tt, next)) = rest.token_tree() {
561            if matches!(&tt, proc_macro2::TokenTree::Punct(punct) if punct.as_char() == '=')
562                && matches!(
563                    next.token_tree(),
564                    Some((proc_macro2::TokenTree::Punct(ref punct), _)) if punct.as_char() == '>'
565                )
566            {
567                break;
568            }
569
570            tokens.extend(std::iter::once(tt));
571            rest = next;
572        }
573
574        Ok((tokens, rest))
575    })?;
576
577    if let Ok(Expr::Field(ExprField {
578        base,
579        member: Member::Named(method),
580        ..
581    })) = syn::parse2::<Expr>(target.clone())
582    {
583        return Ok(FnTarget::Method {
584            receiver: *base,
585            method,
586        });
587    }
588
589    syn::parse2::<Path>(target).map(FnTarget::Path)
590}

592/// Parses the parenthesized body of an `attr(...)` argument.
593///
594/// The body is the content of an outer attribute, such as `allow(dead_code)`.
595fn parse_attr(input: ParseStream<'_>) -> Result<Attribute> {
596    let content;
597    parenthesized!(content in input);
598    let content_tokens = content.parse::<TokenStream>()?;
599    let attr_tokens = quote!(#[#content_tokens]);
600    let mut attrs = Attribute::parse_outer.parse2(attr_tokens)?;
601    attrs
602        .pop()
603        .ok_or_else(|| content.error("expected attribute contents"))
604}

606/// Collects parsed arguments into validated macro arguments.
607///
608/// Duplicate replaced parameters, duplicate generated names, and invalid function replacements
609/// are rejected.
610fn validate_args(output: Ident, args: Vec<MacroArg>) -> Result<MacroArgs> {
611    let mut replacements = Vec::new();
612    let mut fn_replacements = Vec::new();
613    let mut copy_attrs = true;
614    let mut extra_attrs = Vec::new();
615
616    for arg in args {
617        match arg {
618            MacroArg::Replacement(replacement) => {
619                if replacements
620                    .iter()
621                    .any(|existing: &Replacement| existing.param == replacement.param)
622                {
623                    return Err(syn::Error::new(
624                        replacement.param.span(),
625                        "duplicate replacement generic parameter",
626                    ));
627                }
628
629                if replacements
630                    .iter()
631                    .any(|existing| existing.arg_name == replacement.arg_name)
632                {
633                    return Err(syn::Error::new(
634                        replacement.arg_name.span(),
635                        "duplicate generated argument name",
636                    ));
637                }
638
639                replacements.push(replacement);
640            }
641            MacroArg::FnReplacement(replacement) => {
642                validate_fn_replacement(&replacement)?;
643                fn_replacements.push(replacement);
644            }
645            MacroArg::DontCopyAttr => copy_attrs = false,
646            MacroArg::Attr(attr) => extra_attrs.push(attr),
647        }
648    }
649
650    Ok(MacroArgs {
651        output,
652        replacements,
653        fn_replacements,
654        copy_attrs,
655        extra_attrs,
656    })
657}

659/// Checks that the source and target of a function replacement are compatible.
660///
661/// Both sides must be paths or methods, and method replacements must keep the same receiver.
662fn validate_fn_replacement(replacement: &FnReplacement) -> Result<()> {
663    match (&replacement.from, &replacement.to) {
664        (
665            FnTarget::Method {
666                receiver: from_receiver,
667                ..
668            },
669            FnTarget::Method {
670                receiver: to_receiver,
671                ..
672            },
673        ) => {
674            if method_receiver_matches(from_receiver, to_receiver) {
675                Ok(())
676            } else {
677                Err(syn::Error::new_spanned(
678                    to_receiver,
679                    "method replacement target receiver must match source receiver",
680                ))
681            }
682        }
683        (FnTarget::Path(_), FnTarget::Path(_)) => Ok(()),
684        (FnTarget::Path(_), FnTarget::Method { receiver, .. }) => Err(syn::Error::new_spanned(
685            receiver,
686            "function replacement source and target must have the same form",
687        )),
688        (FnTarget::Method { .. }, FnTarget::Path(path)) => Err(syn::Error::new_spanned(
689            path,
690            "function replacement source and target must have the same form",
691        )),
692    }
693}

727/// Checks the replacements against the generic parameters of a signature.
728///
729/// The signature must be generic, and each replacement must name an existing parameter of its kind
730/// in signature order.
731fn validate_replacements_against_signature(sig: &Signature, args: &MacroArgs) -> Result<()> {
732    if sig.generics.params.is_empty() {
733        return Err(syn::Error::new_spanned(
734            &sig.ident,
735            "maybe_non_generic requires a function with generic parameters",
736        ));
737    }
738
739    for replacement in &args.replacements {
740        if sig
741            .generics
742            .params
743            .iter()
744            .any(|param| generic_param_matches_replacement(param, replacement))
745        {
746            continue;
747        }
748
749        if sig
750            .generics
751            .params
752            .iter()
753            .any(|param| generic_param_ident(param) == &replacement.param)
754        {
755            match replacement.kind {
756                ReplacementKind::Const => {
757                    return Err(syn::Error::new(
758                        replacement.param.span(),
759                        "const(...) must name a const generic parameter",
760                    ));
761                }
762                ReplacementKind::Type => {
763                    return Err(syn::Error::new(
764                        replacement.param.span(),
765                        "type(...) must name a type generic parameter",
766                    ));
767                }
768            }
769        } else {
770            return Err(syn::Error::new(
771                replacement.param.span(),
772                "replacement generic parameter does not exist on the function",
773            ));
774        }
775    }
776
777    validate_replacement_order_matches_signature(sig, &args.replacements)
778}

780/// Checks that replacements are listed in signature order.
781///
782/// The order determines the order of the appended parameters.
783fn validate_replacement_order_matches_signature(
784    sig: &Signature,
785    replacements: &[Replacement],
786) -> Result<()> {
787    let signature_order: Vec<&Replacement> = sig
788        .generics
789        .params
790        .iter()
791        .filter_map(|param| replacement_for_generic_param(param, replacements))
792        .collect();
793
794    for (actual, expected) in replacements.iter().zip(signature_order) {
795        if actual.kind == expected.kind && actual.param == expected.param {
796            continue;
797        }
798
799        return Err(syn::Error::new(
800            actual.param.span(),
801            "replacement generic parameters must be listed in function generic parameter order",
802        ));
803    }
804
805    Ok(())
806}

808/// Returns whether a generic parameter is the target of a replacement.
809///
810/// Const replacements match const parameters, and type replacements match type parameters.
811fn generic_param_matches_replacement(param: &GenericParam, replacement: &Replacement) -> bool {
812    match (replacement.kind, param) {
813        (ReplacementKind::Const, GenericParam::Const(param)) => param.ident == replacement.param,
814        (ReplacementKind::Type, GenericParam::Type(param)) => param.ident == replacement.param,
815        _ => false,
816    }
817}

819/// Builds the signature of the generated copy.
820///
821/// The copy is renamed, replaced generic parameters and the predicates that mention them are
822/// removed, and the generated parameters are appended.
823fn signature_for_copy(original: &Signature, args: &MacroArgs) -> Result<Signature> {
824    validate_replacements_against_signature(original, args)?;
825    validate_generated_arg_names_against_inputs(original, &args.replacements)?;
826    validate_existing_signature_types_do_not_use_replacements(original, &args.replacements)?;
827
828    let mut sig = original.clone();
829    sig.ident = args.output.clone();
830    sig.generics = generics_for_copy(&sig.generics, &args.replacements);
831
832    for replacement in &args.replacements {
833        let arg_name = &replacement.arg_name;
834        let arg_type = &replacement.arg_type;
835        let pat: Pat = parse_quote!(#arg_name);
836        sig.inputs.push(FnArg::Typed(parse_quote!(#pat: #arg_type)));
837    }
838
839    Ok(sig)
840}

842/// Checks that the existing signature types do not mention replaced parameters.
843///
844/// Such types would refer to a parameter that the copy no longer declares.
845fn validate_existing_signature_types_do_not_use_replacements(
846    sig: &Signature,
847    replacements: &[Replacement],
848) -> Result<()> {
849    for input in &sig.inputs {
850        let FnArg::Typed(input) = input else {
851            continue;
852        };
853
854        if type_mentions_replaced_param(&input.ty, replacements) {
855            return Err(syn::Error::new_spanned(
856                &input.ty,
857                "function parameter type still references a replaced generic parameter",
858            ));
859        }
860    }
861
862    if return_type_mentions_replaced_param(&sig.output, replacements) {
863        return Err(syn::Error::new_spanned(
864            &sig.output,
865            "return type still references a replaced generic parameter",
866        ));
867    }
868
869    Ok(())
870}

933/// Removes replaced generic arguments from an argument list.
934///
935/// Returns the call arguments that replace them, in order. Replaced parameters nested inside
936/// other generic arguments are rejected.
937fn remove_replaced_generic_args(
938    args: &mut AngleBracketedGenericArguments,
939    replacements: &[Replacement],
940) -> Result<Vec<Expr>> {
941    let mut appended = Vec::new();
942    let mut kept = Punctuated::<GenericArgument, Comma>::new();
943
944    for arg in args.args.iter().cloned() {
945        let replacement = match &arg {
946            GenericArgument::Type(ty) => replacements.iter().find(|replacement| {
947                (replacement.kind == ReplacementKind::Type
948                    || replacement.kind == ReplacementKind::Const)
949                    && is_type_ident(ty, &replacement.param)
950            }),
951            GenericArgument::Const(expr) => replacements.iter().find(|replacement| {
952                replacement.kind == ReplacementKind::Const
953                    && expr_is_const_ident(expr, &replacement.param)
954            }),
955            _ => None,
956        };
957
958        if let Some(replacement) = replacement {
959            appended.push(replacement_expr(replacement));
960        } else if generic_argument_mentions_replaced_param(&arg, replacements) {
961            return Err(syn::Error::new_spanned(
962                arg,
963                "replaced generic parameters inside generic arguments are unsupported",
964            ));
965        } else {
966            kept.push(arg);
967        }
968    }
969
970    args.args = kept;
971    Ok(appended)
972}

974/// Removes replaced generic arguments from the last segment of a call path.
975///
976/// Returns the call arguments that replace them. Replaced parameters in the generic arguments of
977/// earlier segments are rejected.
978fn process_path_generic_args(path: &mut Path, replacements: &[Replacement]) -> Result<Vec<Expr>> {
979    let final_index = path.segments.len().saturating_sub(1);
980    let mut appended = Vec::new();
981
982    for (index, segment) in path.segments.iter_mut().enumerate() {
983        if let PathArguments::AngleBracketed(args) = &mut segment.arguments {
984            if index == final_index {
985                appended.extend(remove_replaced_generic_args(args, replacements)?);
986                if args.args.is_empty() {
987                    segment.arguments = PathArguments::None;
988                }
989            } else if angle_bracketed_args_mention_replaced_param(args, replacements) {
990                return Err(syn::Error::new_spanned(
991                    args.clone(),
992                    "replaced generic parameters inside generic arguments are unsupported",
993                ));
994            }
995        }
996    }
997
998    Ok(appended)
999}

1023/// Renames a call path according to the path replacements.
1024///
1025/// Segment names are replaced one by one, and generic arguments on the segments are kept.
1026fn rewrite_path_call(path: &mut Path, mappings: &[FnReplacement]) -> Result<()> {
1027    for mapping in mappings {
1028        let (FnTarget::Path(from), FnTarget::Path(to)) = (&mapping.from, &mapping.to) else {
1029            continue;
1030        };
1031
1032        if !paths_match_ignoring_turbofish(path, from) {
1033            continue;
1034        }
1035
1036        if path.segments.len() != to.segments.len() {
1037            return Err(syn::Error::new_spanned(
1038                path.clone(),
1039                "function path replacements must have the same number of segments",
1040            ));
1041        }
1042
1043        for (segment, replacement_segment) in path.segments.iter_mut().zip(to.segments.iter()) {
1044            segment.ident = replacement_segment.ident.clone();
1045        }
1046    }
1047
1048    Ok(())
1049}

1051/// A visitor that rewrites the body of the generated copy.
1052///
1053/// It collects errors while visiting, so several problems can be reported together.
1054struct BodyTransformer<'a> {
1055    /// The generic parameter replacements.
1056    ///
1057    /// They decide how parameter uses and generic arguments are rewritten.
1058    replacements: &'a [Replacement],
1059    /// The call renames.
1060    ///
1061    /// They are applied to function and method calls.
1062    fn_replacements: &'a [FnReplacement],
1063    /// The errors found while visiting.
1064    ///
1065    /// They are combined into one error after the visit.
1066    errors: Vec<syn::Error>,
1067}
1068
1069/// Rewrites copied bodies and the calls inside them.
1070///
1071/// Each transformer is consumed by one body rewrite.
1072impl<'a> BodyTransformer<'a> {
1073    /// Creates a body transformer for the macro arguments.
1074    ///
1075    /// The transformer starts without errors.
1076    fn new(args: &'a MacroArgs) -> Self {
1077        Self {
1078            replacements: &args.replacements,
1079            fn_replacements: &args.fn_replacements,
1080            errors: Vec::new(),
1081        }
1082    }
1083
1084    /// Rewrites a block and validates the result.
1085    ///
1086    /// Binding conflicts are checked before rewriting, and remaining uses of replaced parameters
1087    /// are checked after it.
1088    fn transform_block(mut self, block: &mut Block) -> Result<()> {
1089        if let Some(error) = body_binding_conflict_error(block, self.replacements) {
1090            self.errors.push(error);
1091        }
1092
1093        self.visit_block_mut(block);
1094        if let Some(error) = residual_replaced_param_error(block, self.replacements) {
1095            self.errors.push(error);
1096        }
1097
1098        if self.errors.is_empty() {
1099            Ok(())
1100        } else {
1101            let mut errors = self.errors.into_iter();
1102            let mut error = errors.next().expect("at least one transform error");
1103            for next in errors {
1104                error.combine(next);
1105            }
1106            Err(error)
1107        }
1108    }
1109
1110    /// Finds the type replacement for an associated item path such as `H::f`.
1111    ///
1112    /// Returns the replacement and the item name. Generic arguments on the item are rejected.
1113    fn type_replacement_for_path(&self, path: &Path) -> Result<Option<(&Replacement, Ident)>> {
1114        if path.leading_colon.is_some() || path.segments.len() != 2 {
1115            return Ok(None);
1116        }
1117
1118        let mut segments = path.segments.iter();
1119        let first = segments.next().expect("first path segment");
1120        let second = segments.next().expect("second path segment");
1121
1122        if !matches!(first.arguments, PathArguments::None) {
1123            return Ok(None);
1124        }
1125
1126        let Some(replacement) = self.replacements.iter().find(|replacement| {
1127            replacement.kind == ReplacementKind::Type && replacement.param == first.ident
1128        }) else {
1129            return Ok(None);
1130        };
1131
1132        if !matches!(second.arguments, PathArguments::None) {
1133            return Err(syn::Error::new_spanned(
1134                second.arguments.clone(),
1135                "generic associated type-parameter access/calls are unsupported",
1136            ));
1137        }
1138
1139        Ok(Some((replacement, second.ident.clone())))
1140    }
1141
1142    /// Renames a method call according to the method replacements.
1143    ///
1144    /// Replaced generic arguments in the method turbofish are moved to the call arguments.
1145    fn rewrite_method_call(&mut self, node: &mut ExprMethodCall) {
1146        for mapping in self.fn_replacements {
1147            let (
1148                FnTarget::Method { receiver, method },
1149                FnTarget::Method {
1150                    method: to_method, ..
1151                },
1152            ) = (&mapping.from, &mapping.to)
1153            else {
1154                continue;
1155            };
1156
1157            if node.method == *method && method_receiver_matches(&node.receiver, receiver) {
1158                node.method = to_method.clone();
1159
1160                if let Some(turbofish) = &mut node.turbofish {
1161                    let appended = match remove_replaced_generic_args(turbofish, self.replacements)
1162                    {
1163                        Ok(appended) => appended,
1164                        Err(error) => {
1165                            self.errors.push(error);
1166                            return;
1167                        }
1168                    };
1169                    if turbofish.args.is_empty() {
1170                        node.turbofish = None;
1171                    }
1172                    node.args.extend(appended);
1173                }
1174
1175                break;
1176            }
1177        }
1178    }
1179}

1181/// The mutable visiting implementation for body rewrites.
1182///
1183/// It rewrites calls, method calls, and paths, and visits other expressions normally.
1184impl VisitMut for BodyTransformer<'_> {
1185    /// Rewrites a function call expression.
1186    ///
1187    /// Arguments are visited first. A call through a replaced type parameter becomes a call of a
1188    /// field, and other path calls are renamed before their replaced generic arguments are moved.
1189    fn visit_expr_call_mut(&mut self, node: &mut ExprCall) {
1190        for arg in &mut node.args {
1191            self.visit_expr_mut(arg);
1192        }
1193
1194        if let Expr::Path(expr_path) = &mut *node.func {
1195            if let Some((replacement, item)) = match self.type_replacement_for_path(&expr_path.path)
1196            {
1197                Ok(replacement) => replacement,
1198                Err(error) => {
1199                    self.errors.push(error);
1200                    return;
1201                }
1202            } {
1203                let arg_name = &replacement.arg_name;
1204                node.func = parse_quote!((#arg_name.#item));
1205                return;
1206            }
1207
1208            if let Err(error) = rewrite_path_call(&mut expr_path.path, self.fn_replacements) {
1209                self.errors.push(error);
1210                return;
1211            }
1212
1213            let appended = match process_path_generic_args(&mut expr_path.path, self.replacements) {
1214                Ok(appended) => appended,
1215                Err(error) => {
1216                    self.errors.push(error);
1217                    return;
1218                }
1219            };
1220            node.args.extend(appended);
1221        } else {
1222            self.visit_expr_mut(&mut node.func);
1223        }
1224    }
1225
1226    /// Rewrites a method call expression.
1227    ///
1228    /// The receiver and arguments are visited before the method is renamed.
1229    fn visit_expr_method_call_mut(&mut self, node: &mut ExprMethodCall) {
1230        self.visit_expr_mut(&mut node.receiver);
1231        for arg in &mut node.args {
1232            self.visit_expr_mut(arg);
1233        }
1234        self.rewrite_method_call(node);
1235    }
1236
1237    /// Rewrites an expression.
1238    ///
1239    /// Const parameters become their generated arguments, and associated items of replaced type
1240    /// parameters become fields of their generated arguments.
1241    fn visit_expr_mut(&mut self, node: &mut Expr) {
1242        match node {
1243            Expr::Call(call) => self.visit_expr_call_mut(call),
1244            Expr::MethodCall(call) => self.visit_expr_method_call_mut(call),
1245            Expr::Path(expr_path) => {
1246                if expr_path.qself.is_none() && expr_path.path.segments.len() == 1 {
1247                    let ident = &expr_path.path.segments[0].ident;
1248                    if let Some(replacement) = self.replacements.iter().find(|replacement| {
1249                        replacement.kind == ReplacementKind::Const && replacement.param == *ident
1250                    }) {
1251                        let arg_name = &replacement.arg_name;
1252                        *node = parse_quote!(#arg_name);
1253                        return;
1254                    }
1255                }
1256
1257                if let Some((replacement, item)) =
1258                    match self.type_replacement_for_path(&expr_path.path) {
1259                        Ok(replacement) => replacement,
1260                        Err(error) => {
1261                            self.errors.push(error);
1262                            return;
1263                        }
1264                    }
1265                {
1266                    let arg_name = &replacement.arg_name;
1267                    *node = parse_quote!(#arg_name.#item);
1268                    return;
1269                }
1270
1271                visit_mut::visit_expr_path_mut(self, expr_path);
1272            }
1273            _ => visit_mut::visit_expr_mut(self, node),
1274        }
1275    }
1276
1277    /// Leaves a nested item unchanged.
1278    ///
1279    /// Nested items cannot use the generic parameters of the outer function, so names inside
1280    /// them refer to their own parameters or to other items.
1281    fn visit_item_mut(&mut self, _node: &mut Item) {}
1282}

1285///
1286/// Errors from the rewrite and its validation are combined into one error.
1287fn transform_body(block: &mut Block, args: &MacroArgs) -> Result<()> {
1288    BodyTransformer::new(args).transform_block(block)

1291/// A visitor that finds body bindings with the same name as a generated parameter.
1292///
1293/// Such a binding would shadow the generated parameter in rewritten expressions.
1294struct BindingConflictVisitor<'a> {
1295    /// The generic parameter replacements.
1296    ///
1297    /// Their generated parameter names are compared with body bindings.
1298    replacements: &'a [Replacement],
1299    /// The first conflict found.
1300    ///
1301    /// Later conflicts are not recorded.
1302    error: Option<syn::Error>,
1303}
1304
1305/// Checks blocks and patterns for conflicting bindings.
1306///
1307/// Only the first conflict is reported.
1308impl<'a> BindingConflictVisitor<'a> {
1309    /// Returns the first binding conflict in a block.
1310    ///
1311    /// Returns `None` when no pattern binds a generated parameter name.
1312    fn check(block: &'a Block, replacements: &'a [Replacement]) -> Option<syn::Error> {
1313        let mut visitor = Self {
1314            replacements,
1315            error: None,
1316        };
1317        visitor.visit_block(block);
1318        visitor.error
1319    }
1320
1321    /// Records a conflict for a pattern that binds a generated parameter name.
1322    ///
1323    /// Nothing is recorded after the first conflict.
1324    fn check_pat(&mut self, pat: &'a Pat) {
1325        if self.error.is_some() {
1326            return;
1327        }
1328
1329        if let Some(ident) = conflicting_pat_binding(pat, self.replacements) {
1330            self.error = Some(syn::Error::new(
1331                ident.span(),
1332                "body binding conflicts with a generated argument name",
1333            ));
1334        }
1335    }
1336}

1338/// The visiting implementation for binding conflict checks.
1339///
1340/// It checks every pattern in the body, including nested patterns.
1341impl<'ast> Visit<'ast> for BindingConflictVisitor<'ast> {
1342    /// Checks a pattern and then visits its subpatterns.
1343    ///
1344    /// Each nested identifier pattern is checked when it is visited.
1345    fn visit_pat(&mut self, node: &'ast Pat) {
1346        self.check_pat(node);
1347        visit::visit_pat(self, node);
1348    }
1349
1350    /// Skips a nested item.
1351    ///
1352    /// Bindings inside nested items cannot shadow the parameters of the outer function.
1353    fn visit_item(&mut self, _node: &'ast Item) {}
1354}

1356/// Reports a body binding with the same name as a generated parameter.
1357///
1358/// Returns `None` when there is no conflict.
1359fn body_binding_conflict_error(block: &Block, replacements: &[Replacement]) -> Option<syn::Error> {

1363/// Returns the binding of an identifier pattern that conflicts with a generated parameter.
1364///
1365/// Other patterns return `None`, because the visitor checks their subpatterns separately.
1366fn conflicting_pat_binding<'a>(pat: &'a Pat, replacements: &[Replacement]) -> Option<&'a Ident> {
1367    match pat {
1368        Pat::Ident(pat) => replacements
1369            .iter()
1370            .any(|replacement| replacement.arg_name == pat.ident)
1371            .then_some(&pat.ident),
1372        _ => None,
1373    }
1374}

1376/// Checks that generated parameter names are not already bound by function parameters.
1377///
1378/// Nested bindings in parameter patterns are included.
1379fn validate_generated_arg_names_against_inputs(
1380    sig: &Signature,
1381    replacements: &[Replacement],
1382) -> Result<()> {
1383    let mut existing_bindings = Vec::new();
1384
1385    for input in &sig.inputs {
1386        let FnArg::Typed(input) = input else {
1387            continue;
1388        };
1389        collect_pat_bindings(&input.pat, &mut existing_bindings);
1390    }
1391
1392    for existing in existing_bindings {
1393        if let Some(replacement) = replacements
1394            .iter()
1395            .find(|replacement| replacement.arg_name == *existing)
1396        {
1397            return Err(syn::Error::new(
1398                replacement.arg_name.span(),
1399                "generated argument name conflicts with an existing function parameter",
1400            ));
1401        }
1402    }
1403
1404    Ok(())
1405}

1407/// Collects the identifiers bound by a pattern.
1408///
1409/// Nested patterns are searched recursively.
1410fn collect_pat_bindings<'a>(pat: &'a Pat, bindings: &mut Vec<&'a Ident>) {
1411    match pat {
1412        Pat::Ident(pat) => {
1413            bindings.push(&pat.ident);
1414            if let Some((_at, subpat)) = &pat.subpat {
1415                collect_pat_bindings(subpat, bindings);
1416            }
1417        }
1418        Pat::Or(pat) => {
1419            for case in &pat.cases {
1420                collect_pat_bindings(case, bindings);
1421            }
1422        }
1423        Pat::Paren(pat) => collect_pat_bindings(&pat.pat, bindings),
1424        Pat::Reference(pat) => collect_pat_bindings(&pat.pat, bindings),
1425        Pat::Slice(pat) => {
1426            for elem in &pat.elems {
1427                collect_pat_bindings(elem, bindings);
1428            }
1429        }
1430        Pat::Struct(pat) => {
1431            for field in &pat.fields {
1432                collect_pat_bindings(&field.pat, bindings);
1433            }
1434        }
1435        Pat::Tuple(pat) => {
1436            for elem in &pat.elems {
1437                collect_pat_bindings(elem, bindings);
1438            }
1439        }
1440        Pat::TupleStruct(pat) => {
1441            for elem in &pat.elems {
1442                collect_pat_bindings(elem, bindings);
1443            }
1444        }
1445        Pat::Type(pat) => collect_pat_bindings(&pat.pat, bindings),
1446        _ => {}
1447    }
1448}

1450/// Builds the generics of the generated copy.
1451///
1452/// Replaced parameters are removed, together with bounds and `where` predicates that mention
1453/// them.
1454fn generics_for_copy(generics: &Generics, replacements: &[Replacement]) -> Generics {
1455    let mut result = generics.clone();
1456    result.params = generics
1457        .params
1458        .iter()
1459        .filter(|param| replacement_for_generic_param(param, replacements).is_none())
1460        .cloned()
1461        .map(|mut param| {
1462            if let GenericParam::Type(param) = &mut param {
1463                param.bounds = retained_type_param_bounds(&param.bounds, replacements);
1464            }
1465            param
1466        })
1467        .collect();
1468
1469    if let Some(where_clause) = &mut result.where_clause {
1470        where_clause.predicates = where_clause
1471            .predicates
1472            .iter()
1473            .filter(|predicate| !predicate_mentions_replaced_param(predicate, replacements))
1474            .cloned()
1475            .collect();
1476    }
1477
1478    result
1479}

1545/// Returns whether a type mentions a replaced parameter.
1546///
1547/// Types that the macro cannot analyze, such as macro invocations, are treated as mentions.
1548fn type_mentions_replaced_param(ty: &Type, replacements: &[Replacement]) -> bool {
1549    match ty {
1550        Type::Array(ty) => {
1551            type_mentions_replaced_param(&ty.elem, replacements)
1552                || expr_mentions_replaced_param(&ty.len, replacements)
1553        }
1554        Type::BareFn(ty) => {
1555            ty.inputs
1556                .iter()
1557                .any(|arg| type_mentions_replaced_param(&arg.ty, replacements))
1558                || return_type_mentions_replaced_param(&ty.output, replacements)
1559        }
1560        Type::Group(ty) => type_mentions_replaced_param(&ty.elem, replacements),
1561        Type::ImplTrait(ty) => ty
1562            .bounds
1563            .iter()
1564            .any(|bound| type_param_bound_mentions_replaced_param(bound, replacements)),
1565        Type::Infer(_) => false,
1566        Type::Macro(_) => true,
1567        Type::Never(_) => false,
1568        Type::Paren(ty) => type_mentions_replaced_param(&ty.elem, replacements),
1569        Type::Path(ty) => {
1570            ty.qself
1571                .as_ref()
1572                .is_some_and(|qself| type_mentions_replaced_param(&qself.ty, replacements))
1573                || type_path_mentions_replaced_param(&ty.path, replacements)
1574        }
1575        Type::Ptr(ty) => type_mentions_replaced_param(&ty.elem, replacements),
1576        Type::Reference(ty) => type_mentions_replaced_param(&ty.elem, replacements),
1577        Type::Slice(ty) => type_mentions_replaced_param(&ty.elem, replacements),
1578        Type::TraitObject(ty) => ty
1579            .bounds
1580            .iter()
1581            .any(|bound| type_param_bound_mentions_replaced_param(bound, replacements)),
1582        Type::Tuple(ty) => ty
1583            .elems
1584            .iter()
1585            .any(|elem| type_mentions_replaced_param(elem, replacements)),
1586        _ => true,
1587    }
1588}

1603/// Returns whether an expression mentions a replaced parameter.
1604///
1605/// Expressions that the macro cannot analyze, such as macro invocations, are treated as mentions.
1606fn expr_mentions_replaced_param(expr: &Expr, replacements: &[Replacement]) -> bool {
1607    match expr {
1608        Expr::Array(expr) => expr
1609            .elems
1610            .iter()
1611            .any(|elem| expr_mentions_replaced_param(elem, replacements)),
1612        Expr::Assign(expr) => {
1613            expr_mentions_replaced_param(&expr.left, replacements)
1614                || expr_mentions_replaced_param(&expr.right, replacements)
1615        }
1616        Expr::Async(expr) => block_mentions_replaced_param(&expr.block, replacements),
1617        Expr::Await(expr) => expr_mentions_replaced_param(&expr.base, replacements),
1618        Expr::Binary(expr) => {
1619            expr_mentions_replaced_param(&expr.left, replacements)
1620                || expr_mentions_replaced_param(&expr.right, replacements)
1621        }
1622        Expr::Block(expr) => block_mentions_replaced_param(&expr.block, replacements),
1623        Expr::Break(expr) => expr
1624            .expr
1625            .as_ref()
1626            .is_some_and(|expr| expr_mentions_replaced_param(expr, replacements)),
1627        Expr::Call(expr) => {
1628            expr_mentions_replaced_param(&expr.func, replacements)
1629                || expr
1630                    .args
1631                    .iter()
1632                    .any(|arg| expr_mentions_replaced_param(arg, replacements))
1633        }
1634        Expr::Cast(expr) => {
1635            expr_mentions_replaced_param(&expr.expr, replacements)
1636                || type_mentions_replaced_param(&expr.ty, replacements)
1637        }
1638        Expr::Closure(expr) => {
1639            expr.inputs
1640                .iter()
1641                .any(|pat| pat_mentions_replaced_param(pat, replacements))
1642                || return_type_mentions_replaced_param(&expr.output, replacements)
1643                || expr_mentions_replaced_param(&expr.body, replacements)
1644        }
1645        Expr::Const(expr) => block_mentions_replaced_param(&expr.block, replacements),
1646        Expr::Continue(_) => false,
1647        Expr::Field(expr) => expr_mentions_replaced_param(&expr.base, replacements),
1648        Expr::ForLoop(expr) => {
1649            pat_mentions_replaced_param(&expr.pat, replacements)
1650                || expr_mentions_replaced_param(&expr.expr, replacements)
1651                || block_mentions_replaced_param(&expr.body, replacements)
1652        }
1653        Expr::Group(expr) => expr_mentions_replaced_param(&expr.expr, replacements),
1654        Expr::Index(expr) => {
1655            expr_mentions_replaced_param(&expr.expr, replacements)
1656                || expr_mentions_replaced_param(&expr.index, replacements)
1657        }
1658        Expr::If(expr) => {
1659            expr_mentions_replaced_param(&expr.cond, replacements)
1660                || expr
1661                    .then_branch
1662                    .stmts
1663                    .iter()
1664                    .any(|stmt| stmt_mentions_replaced_param(stmt, replacements))
1665                || expr.else_branch.as_ref().is_some_and(|(_, else_expr)| {
1666                    expr_mentions_replaced_param(else_expr, replacements)
1667                })
1668        }
1669        Expr::Infer(_) => false,
1670        Expr::Let(expr) => {
1671            pat_mentions_replaced_param(&expr.pat, replacements)
1672                || expr_mentions_replaced_param(&expr.expr, replacements)
1673        }
1674        Expr::Lit(_) => false,
1675        Expr::Loop(expr) => block_mentions_replaced_param(&expr.body, replacements),
1676        Expr::Macro(_) => true,
1677        Expr::Match(expr) => {
1678            expr_mentions_replaced_param(&expr.expr, replacements)
1679                || expr.arms.iter().any(|arm| {
1680                    pat_mentions_replaced_param(&arm.pat, replacements)
1681                        || arm.guard.as_ref().is_some_and(|(_if, guard)| {
1682                            expr_mentions_replaced_param(guard, replacements)
1683                        })
1684                        || expr_mentions_replaced_param(&arm.body, replacements)
1685                })
1686        }
1687        Expr::MethodCall(expr) => {
1688            expr_mentions_replaced_param(&expr.receiver, replacements)
1689                || expr.turbofish.as_ref().is_some_and(|args| {
1690                    angle_bracketed_args_mention_replaced_param(args, replacements)
1691                })
1692                || expr
1693                    .args
1694                    .iter()
1695                    .any(|arg| expr_mentions_replaced_param(arg, replacements))
1696        }
1697        Expr::Paren(expr) => expr_mentions_replaced_param(&expr.expr, replacements),
1698        Expr::Path(expr) => {
1699            expr.qself
1700                .as_ref()
1701                .is_some_and(|qself| type_mentions_replaced_param(&qself.ty, replacements))
1702                || expr_path_mentions_replaced_param(&expr.path, replacements)
1703        }
1704        Expr::Range(expr) => {
1705            expr.start
1706                .as_ref()
1707                .is_some_and(|expr| expr_mentions_replaced_param(expr, replacements))
1708                || expr
1709                    .end
1710                    .as_ref()
1711                    .is_some_and(|expr| expr_mentions_replaced_param(expr, replacements))
1712        }
1713        Expr::Reference(expr) => expr_mentions_replaced_param(&expr.expr, replacements),
1714        Expr::Repeat(expr) => {
1715            expr_mentions_replaced_param(&expr.expr, replacements)
1716                || expr_mentions_replaced_param(&expr.len, replacements)
1717        }
1718        Expr::Return(expr) => expr
1719            .expr
1720            .as_ref()
1721            .is_some_and(|expr| expr_mentions_replaced_param(expr, replacements)),
1722        Expr::Struct(expr) => {
1723            expr_path_mentions_replaced_param(&expr.path, replacements)
1724                || expr
1725                    .fields
1726                    .iter()
1727                    .any(|field| expr_mentions_replaced_param(&field.expr, replacements))
1728                || expr
1729                    .rest
1730                    .as_ref()
1731                    .is_some_and(|rest| expr_mentions_replaced_param(rest, replacements))
1732        }
1733        Expr::Tuple(expr) => expr
1734            .elems
1735            .iter()
1736            .any(|elem| expr_mentions_replaced_param(elem, replacements)),
1737        Expr::Try(expr) => expr_mentions_replaced_param(&expr.expr, replacements),
1738        Expr::TryBlock(expr) => block_mentions_replaced_param(&expr.block, replacements),
1739        Expr::Unary(expr) => expr_mentions_replaced_param(&expr.expr, replacements),
1740        Expr::Unsafe(expr) => block_mentions_replaced_param(&expr.block, replacements),
1741        Expr::While(expr) => {
1742            expr_mentions_replaced_param(&expr.cond, replacements)
1743                || block_mentions_replaced_param(&expr.body, replacements)
1744        }
1745        Expr::Yield(expr) => expr
1746            .expr
1747            .as_ref()
1748            .is_some_and(|expr| expr_mentions_replaced_param(expr, replacements)),
1749        _ => true,
1750    }
1751}
