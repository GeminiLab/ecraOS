1//! The slab allocator.
2//!
3//! [`SlabAllocator`] owns one cache per size class and selects the class from an allocation layout.
4//! It allocates no memory itself. When a class needs a slab, the allocator reports how many pages
5//! to supply, and the caller registers that memory and retries. The caller also serializes local
6//! operations and decides how to route a free, using [`SlabAllocator::owner_cpu`] to find the
7//! owning CPU. Remote frees reach a slab header without borrowing the allocator.
8//!
9//! [`SlabCache`] offers the same machinery for a single caller-selected class.
10
11use core::{alloc::Layout, ptr::NonNull};
12
13use memory_addr::VirtAddr;
14
15use crate::{
16    SlabLayout,
17    cache::SlabCache,
18    error::{AllocError, AllocResult},
19    size_class::SizeClass,
20    slab_page::{SlabPage, SlabPageHeader},
21};
22
23/// Result of a slab allocation attempt.
24pub enum SlabAllocResult {
25    /// A successfully allocated object.
26    Allocated(NonNull<u8>),
27    /// Slab memory that the caller must supply before retrying the allocation.
28    ///
29    /// The caller provides exactly this many contiguous mapped writable pages, aligned to the whole
30    /// span, and registers them with [`SlabAllocator::add_slab`].
31    NeedsSlab {
32        /// Size class whose cache needs a slab.
33        size_class: SizeClass,
34        /// Exact number of hardware pages in the required slab.
35        ///
36        /// The caller must supply exactly this many contiguous mapped writable pages, aligned to
37        /// the total size of the pages.
38        pages: usize,
39    },
40}
41
42/// Result of a local slab deallocation.
43pub enum SlabDeallocResult {
44    /// The object was released and its slab stays cached.
45    Done,
46    /// The slab became empty and exceeded the cache's retained empty slab, so the caller must
47    /// release its memory.
48    FreeSlab {
49        /// Virtual base address of the complete slab memory region.
50        base: VirtAddr,
51        /// Exact number of hardware pages to release.
52        pages: usize,
53    },
54}
55
56/// A slab allocator over caller-supplied slab memory.
57///
58/// The hardware page size is fixed at construction to 4, 16, or 64 KiB. Slab sizes follow
59/// `max(min(65536, slot_size * 512), page_size)` and must be aligned to their complete span, where
60/// `slot_size` comes from the selected size class. See the [crate documentation](crate) for the
61/// page counts this produces.
62///
63/// The caller serializes operations through the allocator and keeps slab memory alive until it is
64/// returned or every user of it has finished. Dropping the allocator does not release slab memory.
65pub struct SlabAllocator {
66    /// Independent class states, each permanently bound to a static layout.
67    caches: [SlabCache; SizeClass::SIZE_CLASS_COUNT],
68    /// Hardware page size used to report slab memory requests and returns.
69    page_size: usize,
70}
71
72/// Allocation, slab registration, and ownership operations for [`SlabAllocator`].
73impl SlabAllocator {
74    /// Creates an empty slab allocator.
75    ///
76    /// `page_size` must be 4, 16, or 64 KiB, and other values return [`AllocError::InvalidParam`].
77    /// No slab memory is obtained here. Each class begins empty and asks for a slab on first use.
78    pub fn new(page_size: usize) -> AllocResult<Self> {
79        let page_log2 = Self::page_log2(page_size)?;
80        Ok(Self {
81            caches: core::array::from_fn(|index| {
82                SlabCache::with_layout(SlabLayout::for_class(SizeClass::ALL[index], page_log2))
83            }),
84            page_size,
85        })
86    }
87
88    /// Returns the base-2 logarithm of a supported hardware page size in bytes.
89    fn page_log2(page_size: usize) -> AllocResult<usize> {
90        match page_size {
91            4096 => Ok(12),
92            16384 => Ok(14),
93            65536 => Ok(16),
94            _ => Err(AllocError::InvalidParam),
95        }
96    }
97
98    /// Allocates an object matching the requested allocation layout.
99    ///
100    /// Selects the size class from the layout and takes a slot from that class. Returns
101    /// [`SlabAllocResult::NeedsSlab`] when the class has no slot available. A size or alignment
102    /// above 2048 bytes returns [`AllocError::InvalidParam`].
103    pub fn alloc(&mut self, layout: Layout) -> AllocResult<SlabAllocResult> {
104        let class = SizeClass::from_layout(layout).ok_or(AllocError::InvalidParam)?;
105        let cache = &mut self.caches[class.index()];
106        match cache.alloc_object() {
107            Some(address) => {
108                // SAFETY: A successfully allocated cache slot has a non-null live address.
109                let object = unsafe { NonNull::new_unchecked(address.as_mut_ptr()) };
110                Ok(SlabAllocResult::Allocated(object))
111            }
112            None => Ok(SlabAllocResult::NeedsSlab {
113                size_class: class,
114                pages: cache.layout().slab_size / self.page_size,
115            }),
116        }
117    }
118
119    /// Registers slab memory for the selected size class.
120    ///
121    /// The memory must be the size that [`Self::alloc`] reported for `class`, and the allocator
122    /// takes no separate size argument because the class fixes it. A zero or misaligned base
123    /// returns [`AllocError::InvalidParam`] before memory is accessed.
124    ///
125    /// Registration is unsafe, so safe code cannot register an arbitrary address as slab memory.
126    ///
127    /// ```compile_fail
128    /// use exslab::{SizeClass, SlabAllocator};
129    /// use memory_addr::va;
130    ///
131    /// let mut allocator = SlabAllocator::new(4096).unwrap();
132    /// let _ = allocator.add_slab(SizeClass::Bytes8, va!(0x1000), 0);
133    /// ```
134    ///
135    /// # Safety
136    ///
137    /// An accepted base must identify exclusive mapped writable slab memory of the exact requested
138    /// span, aligned to that span. Keep this memory valid until it is returned, or until all cache
139    /// operations, slots, and remote producers end.
140    pub unsafe fn add_slab(
141        &mut self,
142        class: SizeClass,
143        base: VirtAddr,
144        owner_cpu: u16,
145    ) -> AllocResult {
146        // SAFETY: The caller supplies complete exclusive slab memory for this cache's bound layout.
147        unsafe { self.caches[class.index()].add_slab(base, owner_cpu) }
148    }
149
150    /// Releases an object through its owning allocator.
151    ///
152    /// Each size class retains at most one empty slab. When another slab of that class becomes
153    /// empty, it is removed and returned through [`SlabDeallocResult::FreeSlab`] with its exact
154    /// page count.
155    ///
156    /// # Safety
157    ///
158    /// The pointer must be a live allocation from this allocator, with its original allocation
159    /// layout. Release it exactly once and serialize this call with the allocator's other
160    /// operations. Remote frees use [`Self::remote_free_object`].
161    pub unsafe fn dealloc(&mut self, object: NonNull<u8>, layout: Layout) -> SlabDeallocResult {
162        let class = SizeClass::from_layout(layout).expect("unsupported allocation layout");
163        let cache = &mut self.caches[class.index()];
164        // SAFETY: The caller identifies a live slot of the cache selected by its allocation layout.
165        match unsafe { cache.dealloc_object(VirtAddr::from_mut_ptr_of(object.as_ptr())) } {
166            Some(base) => SlabDeallocResult::FreeSlab {
167                base,
168                pages: cache.layout().slab_size / self.page_size,
169            },
170            None => SlabDeallocResult::Done,
171        }
172    }
173
174    /// Returns the logical CPU identifier that owns an allocation.
175    ///
176    /// Requires only the object address, its original allocation layout, and the hardware page
177    /// size, so it works from any CPU and borrows no allocator state. A kernel typically
178    /// compares the result with the current CPU to choose between a local and a remote free.
179    /// The slab size for the layout determines which address bits to mask off, so no header is
180    /// scanned to find the owner.
181    ///
182    /// An unsupported layout or page size returns [`AllocError::InvalidParam`] before memory is
183    /// accessed.
184    ///
185    /// # Safety
186    ///
187    /// The pointer must identify a live allocation with its original allocation layout and hardware
188    /// page size. Slab memory must remain valid throughout this call. No incompatible reference to
189    /// the header may overlap the read of its immutable owner field.
190    pub unsafe fn owner_cpu(
191        object: NonNull<u8>,
192        layout: Layout,
193        page_size: usize,
194    ) -> AllocResult<u16> {
195        let page_log2 = Self::page_log2(page_size)?;
196        let class = SizeClass::from_layout(layout).ok_or(AllocError::InvalidParam)?;
197        let slab_layout = SlabLayout::for_class(class, page_log2);
198        let address = VirtAddr::from_mut_ptr_of(object.as_ptr());
199        // SAFETY: The caller supplies a live allocation with a matching slab layout.
200        let slab = unsafe { SlabPage::from_object(address, slab_layout) };
201        // SAFETY: Only the immutable owner field is read while the slab memory remains valid.
202        Ok(unsafe { (*slab.as_ptr().cast::<SlabPageHeader>()).owner_cpu })
203    }
204
205    /// Releases an object that belongs to another CPU's allocator.
206    ///
207    /// Publishes the slot to its slab through [`SlabPageHeader::remote_free`], which needs no
208    /// access to the owning allocator. The owner drains published slots when a later allocation or
209    /// deallocation uses that slab. This call does not return slab memory itself.
210    ///
211    /// An unsupported layout or page size returns [`AllocError::InvalidParam`] before memory is
212    /// accessed.
213    ///
214    /// # Safety
215    ///
216    /// The pointer must be a live slot with its original allocation layout and hardware page size.
217    /// It must be freed once. Slab memory must remain valid until successful publication. The owner
218    /// may then reuse the slot or reclaim the slab memory immediately. No whole-header reference
219    /// may overlap publication.
220    pub unsafe fn remote_free_object(
221        object: NonNull<u8>,
222        layout: Layout,
223        page_size: usize,
224    ) -> AllocResult {
225        let page_log2 = Self::page_log2(page_size)?;
226        let class = SizeClass::from_layout(layout).ok_or(AllocError::InvalidParam)?;
227        let slab_layout = SlabLayout::for_class(class, page_log2);
228        let address = VirtAddr::from_mut_ptr_of(object.as_ptr());
229        // SAFETY: The caller guarantees a live slot with a matching slab layout.
230        let slab = unsafe { SlabPage::from_object(address, slab_layout) };
231        // SAFETY: The caller transfers one live slot to its sole owner through publication.
232        unsafe { SlabPageHeader::remote_free(slab.cast(), address) };
233        Ok(())
234    }
235}

237/// Default hardware page configuration for [`SlabAllocator`].
238impl Default for SlabAllocator {
239    /// Creates an empty allocator configured for 4 KiB hardware pages.
240    fn default() -> Self {
241        Self::new(4096).expect("4 KiB hardware pages are supported")
242    }
