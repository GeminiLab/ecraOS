24/// Represents the MADT - this contains the MADT header fields. You can then iterate over a `Madt`
25/// to read each entry from it.
26///
27/// In modern versions of ACPI, the MADT can detail one of four interrupt models:
28/// - The ancient dual-i8259 legacy PIC model
29/// - The Advanced Programmable Interrupt Controller (APIC) model
30/// - The Streamlined Advanced Programmable Interrupt Controller (SAPIC) model (for Itanium systems)
31/// - The Generic Interrupt Controller (GIC) model (for ARM systems)
32///
33/// The MADT is a variable-sized structure consisting of a static header and then a variable number of entries.
34/// This type only contains the static portion, and then uses pointer arithmetic to parse the following entries.
35/// To make this sound, this type is `!Unpin` - this prevents `Madt` being moved, which would leave
36/// the entries behind.
37#[repr(C, packed)]
38#[derive(Debug)]
39pub struct Madt {
40    pub header: SdtHeader,
41    pub local_apic_address: u32,
42    pub flags: u32,
43    _pinned: PhantomPinned,
44}
45
46unsafe impl AcpiTable for Madt {
47    const SIGNATURE: Signature = Signature::MADT;
48
49    fn header(&self) -> &SdtHeader {
50        &self.header
51    }
52}
53
54impl Madt {
55    pub fn entries(self: Pin<&Self>) -> MadtEntryIter<'_> {
56        let ptr = unsafe { Pin::into_inner_unchecked(self) as *const Madt as *const u8 };
57        MadtEntryIter {
58            pointer: unsafe { ptr.add(mem::size_of::<Madt>()) },
59            remaining_length: self.header.length - mem::size_of::<Madt>() as u32,
60            _phantom: PhantomData,
61        }
62    }
63
64    pub fn supports_8259(&self) -> bool {
65        { self.flags }.get_bit(0)
66    }
67
68    pub fn get_mpwk_mailbox_addr(self: Pin<&Self>) -> Result<u64, AcpiError> {
69        for entry in self.entries() {
70            if let MadtEntry::MultiprocessorWakeup(entry) = entry {
71                return Ok(entry.mailbox_address);
72            }
73        }
74        Err(AcpiError::InvalidMadt(MadtError::UnexpectedEntry))
75    }
76}

78#[derive(Debug)]
79pub struct MadtEntryIter<'a> {
80    pointer: *const u8,
81    /*
82     * The iterator can only have at most `u32::MAX` remaining bytes, because the length of the
83     * whole SDT can only be at most `u32::MAX`.
84     */
85    remaining_length: u32,
86    _phantom: PhantomData<&'a ()>,
87}

110impl<'a> Iterator for MadtEntryIter<'a> {
111    type Item = MadtEntry<'a>;
112
113    fn next(&mut self) -> Option<Self::Item> {
114        while self.remaining_length > 0 {
115            let entry_pointer = self.pointer;
116            let header = unsafe { *(self.pointer as *const EntryHeader) };
117
118            if header.length as u32 > self.remaining_length {
119                warn!(
120                    "Invalid entry of type {} in MADT - extending past length of table. Ignoring",
121                    header.entry_type
122                );
123                return None;
124            }
125
126            self.pointer = unsafe { self.pointer.byte_offset(header.length as isize) };
127            self.remaining_length = self.remaining_length.saturating_sub(header.length as u32);
128
129            macro_rules! construct_entry {
130                ($entry_type:expr,
131                 $entry_pointer:expr,
132                 $(($value:expr => $variant:path as $type:ty)),*
133                ) => {
134                    match $entry_type {
135                        $(
136                            $value => {
137                                return Some($variant(unsafe {
138                                    &*($entry_pointer as *const $type)
139                                }))
140                            }
141                         )*
142
143                        /*
144                         * These entry types are reserved by the ACPI standard. We should skip them
145                         * if they appear in a real MADT.
146                         */
147                        0x11..=0x7f => {}
148
149                        /*
150                         * These entry types are reserved for OEM use. Atm, we just skip them too.
151                         * TODO: work out if we should ever do anything else here
152                         */
153                        0x80..=0xff => {}
154                    }
155                }
156            }
157
158            #[rustfmt::skip]
159            construct_entry!(
160                header.entry_type,
161                entry_pointer,
162                (0x0 => MadtEntry::LocalApic as LocalApicEntry),
163                (0x1 => MadtEntry::IoApic as IoApicEntry),
164                (0x2 => MadtEntry::InterruptSourceOverride as InterruptSourceOverrideEntry),
165                (0x3 => MadtEntry::NmiSource as NmiSourceEntry),
166                (0x4 => MadtEntry::LocalApicNmi as LocalApicNmiEntry),
167                (0x5 => MadtEntry::LocalApicAddressOverride as LocalApicAddressOverrideEntry),
168                (0x6 => MadtEntry::IoSapic as IoSapicEntry),
169                (0x7 => MadtEntry::LocalSapic as LocalSapicEntry),
170                (0x8 => MadtEntry::PlatformInterruptSource as PlatformInterruptSourceEntry),
171                (0x9 => MadtEntry::LocalX2Apic as LocalX2ApicEntry),
172                (0xa => MadtEntry::X2ApicNmi as X2ApicNmiEntry),
173                (0xb => MadtEntry::Gicc as GiccEntry),
174                (0xc => MadtEntry::Gicd as GicdEntry),
175                (0xd => MadtEntry::GicMsiFrame as GicMsiFrameEntry),
176                (0xe => MadtEntry::GicRedistributor as GicRedistributorEntry),
177                (0xf => MadtEntry::GicInterruptTranslationService as GicInterruptTranslationServiceEntry),
178                (0x10 => MadtEntry::MultiprocessorWakeup as MultiprocessorWakeupEntry)
179            );
180        }
181
182        None
183    }
184}

430/// Polarity indicates what signal mode the interrupt line needs to be in to be considered 'active'.
431#[derive(Debug, Clone, Copy, PartialEq, Eq)]
432pub enum Polarity {
433    SameAsBus,
434    ActiveHigh,
435    ActiveLow,
436}
437
438/// Trigger mode of an interrupt, describing how the interrupt is triggered.
439///
440/// When an interrupt is `Edge` triggered, it is triggered exactly once, when the interrupt
441/// signal goes from its opposite polarity to its active polarity.
442///
443/// For `Level` triggered interrupts, a continuous signal is emitted so long as the interrupt
444/// is in its active polarity.
445///
446/// `SameAsBus`-triggered interrupts will utilize the same interrupt triggering as the system bus
447/// they communicate across.
448#[derive(Debug, Clone, Copy, PartialEq, Eq)]
449pub enum TriggerMode {
450    SameAsBus,
451    Edge,
452    Level,
453}
454
455pub fn parse_mps_inti_flags(flags: u16) -> Result<(Polarity, TriggerMode), AcpiError> {
456    let polarity = match flags.get_bits(0..2) {
457        0b00 => Polarity::SameAsBus,
458        0b01 => Polarity::ActiveHigh,
459        0b11 => Polarity::ActiveLow,
460        _ => return Err(AcpiError::InvalidMadt(MadtError::MpsIntiInvalidPolarity)),
461    };
462
463    let trigger_mode = match flags.get_bits(2..4) {
464        0b00 => TriggerMode::SameAsBus,
465        0b01 => TriggerMode::Edge,
466        0b11 => TriggerMode::Level,
467        _ => return Err(AcpiError::InvalidMadt(MadtError::MpsIntiInvalidTriggerMode)),
468    };
469
470    Ok((polarity, trigger_mode))
