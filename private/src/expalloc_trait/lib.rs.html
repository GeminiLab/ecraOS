4use memory_addr::{PhysAddr, VirtAddr};

6/// A physical page allocator with a static interface.
7///
8/// Implementations allocate physically contiguous, page-aligned frames and translate their
9/// physical addresses into virtual addresses through which the caller can access them. All frame
10/// counts use the page size reported by [`Self::page_size_shift`]. The associated functions have
11/// no receiver, so generic code selects an implementation through its type parameter.
12/// [`DynPageAllocator`] stores the same functions for runtime selection.
13///
14/// # Allocation and access requirements
15///
16/// Allocations must remain reserved until released and must not overlap other live allocations.
17/// The interface does not guarantee that newly allocated memory is initialized. Callers must end
18/// all accesses before releasing an allocation and must not release it more than once.
19///
20/// Initialization, mapping availability, and concurrency requirements depend on the implementation.
21/// The default methods do not initialize the allocator, establish mappings, or synchronize access.
22#[dyn_static_traits::dyn_static_traits(DynPageAllocator)]
23pub trait PageAllocator {
24    /// Returns the base-2 logarithm of the allocator's page size.
25    ///
26    /// A page contains `1usize << page_size_shift()` bytes. The result must be less than
27    /// `usize::BITS` and must remain constant for the lifetime of every allocation made through
28    /// this type.
29    fn page_size_shift() -> usize;
30
31    /// Allocates one physical frame.
32    ///
33    /// By default, this delegates to [`Self::alloc_frames`] with a page count of one and returns
34    /// [`None`] when the allocation cannot be satisfied.
35    fn alloc_frame() -> Option<PhysAddr> {
36        Self::alloc_frames(1)
37    }
38
39    /// Allocates a contiguous run of physical frames.
40    ///
41    /// For a nonzero `page_count`, returns the physical address of the first frame, or [`None`]
42    /// when the requested run cannot be allocated. The returned address must be aligned to the
43    /// allocator's page size, and the allocation must cover `page_count` consecutive pages.
44    /// Zero-count behavior is defined by the implementation.
45    fn alloc_frames(page_count: usize) -> Option<PhysAddr>;
46
47    /// Allocates enough contiguous frames to cover a byte size.
48    ///
49    /// By default, this rounds `size_in_bytes` up to a whole number of pages and delegates to
50    /// [`Self::alloc_frames`]. A zero byte size delegates with a count of zero, so its behavior
51    /// depends on the implementation. Implementations may override this method to track the
52    /// original byte size. Callers should release the allocation through
53    /// [`Self::dealloc_frames_of_size`] with the same byte size.
54    fn alloc_frames_of_size(size_in_bytes: usize) -> Option<PhysAddr> {
55        Self::alloc_frames(size_in_bytes.div_ceil(1usize << Self::page_size_shift()))
56    }
57
58    /// Deallocates one physical frame.
59    ///
60    /// `addr` must identify a live one-frame allocation managed by this allocator. By default,
61    /// this delegates to [`Self::dealloc_frames`] with a page count of one.
62    fn dealloc_frame(addr: PhysAddr) {
63        Self::dealloc_frames(addr, 1)
64    }
65
66    /// Deallocates a contiguous run of physical frames.
67    ///
68    /// `addr` and `page_count` must exactly match the starting address and frame count of a live
69    /// allocation managed by this allocator. All accesses must have ended before this call.
70    /// Zero-count behavior is defined by the implementation.
71    fn dealloc_frames(addr: PhysAddr, page_count: usize);
72
73    /// Deallocates the frames covering a byte size.
74    ///
75    /// By default, this rounds `size_in_bytes` up to a whole number of pages and delegates to
76    /// [`Self::dealloc_frames`]. A zero byte size delegates with a count of zero.
77    ///
78    /// `addr` must identify a live allocation managed by this allocator, and `size_in_bytes` should
79    /// be the original byte size passed to [`Self::alloc_frames_of_size`]. Different byte sizes
80    /// that round to the same page count work with the default implementation, but an override may
81    /// require the exact original byte size. All accesses must have ended before this call.
82    fn dealloc_frames_of_size(addr: PhysAddr, size_in_bytes: usize) {
83        Self::dealloc_frames(
84            addr,
85            size_in_bytes.div_ceil(1usize << Self::page_size_shift()),
86        )
87    }
88
89    /// Translates a physical address into an accessible virtual address.
90    ///
91    /// `addr` must be supported by the implementation's physical-memory mapping. The returned
92    /// address must access the same physical memory, with a mapping that covers the bytes the
93    /// caller needs to access. Callers must keep the mapping and any allocation being accessed
94    /// valid for the duration of those accesses. Translation does not transfer ownership.
95    fn phys_to_virt(addr: PhysAddr) -> VirtAddr;
96}
