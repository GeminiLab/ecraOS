18unsafe fn aligned_ptr_in<T>(buf: &mut [u8], offset: usize) -> Result<*mut T, DevTreeError> {
19    // Get the aligned offset
20    let ptr = buf.as_ptr().add(offset);
21    let aligned_offset = offset + ptr.align_offset(align_of::<T>());
22
23    let t_slice_ref = buf
24        .get_mut(aligned_offset..aligned_offset + size_of::<T>())
25        .ok_or(DevTreeError::NotEnoughMemory)?;
26    Ok(t_slice_ref.as_mut_ptr() as *mut T)
27}

40struct DTIBuilder<'i, 'dt: 'i> {

51pub(super) struct DTINode<'i, 'dt: 'i> {
52    parent: *const Self,
53    first_child: *const Self,
54    // `next` is either
55    // 1. the next sibling node
56    // 2. the next node in DFS (some higher up node)
57    // It is 1 if (*next).parent == self.parent, otherwise it is 2.
58    next: *const Self,
59    pub(super) name: &'dt [u8],
60
61    // NOTE: We store props like C arrays. Props are a packed array after each node.
62    // This is the number of props after this node in memory.
63    pub(super) num_props: usize,
64    _index: PhantomData<&'i u8>,
65}

98impl<'i, 'dt: 'i> DTIBuilder<'i, 'dt> {
99    fn allocate_aligned_ptr<T>(&mut self) -> Result<*mut T, DevTreeError> {
100        unsafe {
101            let ptr = aligned_ptr_in::<T>(self.buf, self.front_off)?;
102            self.front_off = ptr.add(1) as usize - self.buf.as_ptr() as usize;
103            Ok(ptr)
104        }
105    }
106
107    pub fn parsed_node(&mut self, node: &ParsedBeginNode<'dt>) -> Result<(), DevTreeError> {
108        unsafe {
109            self.in_node_header = true;
110
111            let new_ptr = self.allocate_aligned_ptr::<DTINode>()?;
112            let parent = self.cur_node;
113
114            // Write the data
115            *new_ptr = DTINode {
116                parent,
117
118                // set by the next node we create
119                first_child: null_mut(),
120                // set by the next node we create
121                next: null_mut(),
122
123                name: node.name,
124                num_props: 0,
125                _index: PhantomData,
126            };
127
128            if !parent.is_null() {
129                debug_assert!(
130                    !self.prev_new_node.is_null(),
131                    "cur_node should not have been initialized without also intializing \
132                    prev_new_node"
133                );
134
135                // While we're streaming through new nodes, parent.next will always point to our
136                // last sibling. Use that fact to change our previous sibling to point towards this
137                // node node.
138                if !(*parent).next.is_null() {
139                    let prev_sibling = (*parent).next as *mut DTINode;
140                    (*prev_sibling).next = new_ptr;
141                }
142                (*parent).next = new_ptr;
143
144                // Update the last new node's next pointer to point towards us.
145                // This is needed to get DFS to work.
146                // In particular, this step is essential for nephew to point to us.
147                //
148                // / {
149                //   sib {
150                //      nephew {
151                //      };
152                //   };
153                //   us {
154                //   };
155                // };
156                (*self.prev_new_node).next = new_ptr;
157
158                // If this new node is the first node that follows the current one, it is the current's
159                // first child.
160                if (*parent).first_child.is_null() {
161                    (*parent).first_child = new_ptr;
162                }
163            }
164
165            // Update the current parent node to be us.
166            // Outside this function, the cur_node may change (if we see a EndNode token).
167            self.cur_node = new_ptr;
168            // Save a pointer to the last created node.
169            // See the note in the assignment to its `next` field above.
170            self.prev_new_node = new_ptr;
171        }
172
173        Ok(())
174    }

176    pub fn parsed_prop(&mut self, prop: &ParsedProp<'dt>) -> Result<(), DevTreeError> {
177        if !self.in_node_header {
178            return Err(DevTreeError::ParseError);
179        }
180
181        unsafe {
182            let new_ptr = self.allocate_aligned_ptr::<DTIProp>()?;
183            (*self.cur_node).num_props += 1;
184            *new_ptr = DTIProp::from(prop);
185        }
186
187        Ok(())
188    }
189
190    pub fn parsed_end_node(&mut self) -> Result<(), DevTreeError> {
191        // There were more EndNode tokens than BeginNode ones.
192        if self.cur_node.is_null() {
193            return Err(DevTreeError::ParseError);
194        }
195        // Unsafe is Ok.
196        // Lifetime : self.cur_node is a pointer into a buffer with the same lifetime as self
197        // Alignment: parsed_node verifies alignment when creating self.cur_node
198        // NonNull  : We check that self.cur_node is non-null above
199        // Mutability: We cast from a *const to a *mut.
200        //             We're the only thread which has access to the buffer at this time, so this
201        //             is thread-safe.
202        unsafe {
203            // Change the current node back to the parent.
204            self.cur_node = (*self.cur_node).parent as *mut DTINode;
205        }
206
207        // We are no longer in a node header.
208        // We are either going to see a new node next or parse another end_node.
209        self.in_node_header = false;
210
211        Ok(())
212    }
213}

215impl<'i, 'dt: 'i> DevTreeIndex<'i, 'dt> {
216    // Note: Our parsing method is unsafe - particularly due to its use of pointer arithmetic.
217    //
218    // We decide this is worth it for the following reasons:
219    // - It requires no allocator.
220    // - It has incredibly low overhead.
221    //   - This parsing method only requires a single allocation. (The buffer given as buf)
222    //   - This parsing method only requires a single iteration over the FDT.
223    // - It is very easy to test in isolation; parsing is entirely enclosed to this module.
224    unsafe fn init_builder<'a>(
225        buf: &'i mut [u8],
226        iter: &mut DevTreeParseIter<'a, 'dt>,
227    ) -> Result<DTIBuilder<'i, 'dt>, DevTreeError> {
228        let mut builder = DTIBuilder {
229            front_off: 0,
230            buf,
231            cur_node: null_mut(),
232            prev_new_node: null_mut(),
233            in_node_header: false,
234        };
235
236        while let Some(tok) = iter.next()? {
237            match tok {
238                ParsedTok::BeginNode(node) => {
239                    builder.parsed_node(&node)?;
240                    return Ok(builder);
241                }
242                ParsedTok::Nop => continue,
243                _ => break,
244            }
245        }
246        Err(DevTreeError::ParseError)
247    }

249    pub fn get_layout(fdt: &'i DevTree<'dt>) -> Result<Layout, DevTreeError> {
250        // Size may require alignment of DTINode.
251        let mut size = 0usize;
252
253        // We assert this because it makes size calculations easier.
254        // We don't have to worry about re-aligning between props and nodes.
255        // If they didn't have the same alignment, we would have to keep track
256        // of the last node and re-align depending on the last seen type.
257        //
258        // E.g. If we saw one node, two props, and then two nodes:
259        //
260        // size = \
261        // align_of::<DTINode> + size_of::<DTINode>
262        // + align_of::<DTIProp> + size_of::<DTIProp>
263        // + size_of::<DTIProp>
264        // + size_of::<DTIProp>
265        // + align_of::<DTINode> + size_of::<DTINode>
266        // + size_of::<DTINode>
267        const_assert_eq!(align_of::<DTINode>(), align_of::<DTIProp>());
268
269        let mut iter = DevTreeIter::new(fdt);
270        while let Some(item) = iter.next()? {
271            match item {
272                DevTreeItem::Node(_) => size += size_of::<DTINode>(),
273                DevTreeItem::Prop(_) => size += size_of::<DTIProp>(),
274            }
275        }
276
277        // Unsafe okay.
278        // - Size is not likely to be usize::MAX. (There's no way we find that many nodes.)
279        // - Align is a result of align_of, so it will be a non-zero power of two
280        unsafe {
281            Ok(Layout::from_size_align_unchecked(
282                size,
283                align_of::<DTINode>(),
284            ))
285        }
286    }

288    pub fn new(fdt: DevTree<'dt>, buf: &'i mut [u8]) -> Result<Self, DevTreeError> {
289        let mut iter = DevTreeParseIter::new(&fdt);
290
291        let mut builder = unsafe { Self::init_builder(buf, &mut iter) }?;
292
293        let this = Self {
294            fdt,
295            root: builder.cur_node,
296        };
297
298        // The builder should have setup a root node or returned an Err.
299        debug_assert!(!this.root.is_null());
300
301        // The buffer will be split into two parts, front and back:
302        //
303        // Front will be used as a temporary work section to  build the nodes as we parse them.
304        // The back will be used to save completely parsed nodes.
305        while let Some(item) = iter.next()? {
306            match item {
307                ParsedTok::BeginNode(node) => {
308                    builder.parsed_node(&node)?;
309                }
310                ParsedTok::Prop(prop) => {
311                    builder.parsed_prop(&prop)?;
312                }
313                ParsedTok::EndNode => {
314                    builder.parsed_end_node()?;
315                }
316                ParsedTok::Nop => continue,
317            }
318        }
319        Ok(this)
320    }
