8//! Virtualization Translation Control Register - EL2
9//!
10//! Provides control of stage2 translation of EL0/1
11
12use tock_registers::{
13    interfaces::{Readable, Writeable},
14    register_bitfields,
15};
16
17register_bitfields! {u64,
18    pub VTCR_EL2 [
19        /// Reserved, RES1.
20        RES1 OFFSET(31) NUMBITS(1) [],
21        /// When FEAT_SEL2 is implemented:
22        /// Non-secure stage 2 translation output address space for the Secure EL1&0 translation regime.
23        NSA OFFSET(30) NUMBITS(1) [
24            SecurePASpace = 0,
25            NonSecurePASpace = 1,
26        ],
27        /// Hardware dirty flag update in stage2 translations when EL2 is enabled
28        HD OFFSET(22) NUMBITS(1) [
29            /// Stage2 hardware management of dirty state disabled
30            Disabled = 0,
31            /// Stage2 hardware management of dirty state enabled
32            Enabled = 1,
33        ],
34        /// Hardware access flag update in stage2 translations when EL2 is enabled
35        HA OFFSET(21) NUMBITS(1) [
36            /// Stage2 hardware management of access state disabled
37            Disabled = 0,
38            /// Stage2 hardware management of access state enabled
39            Enabled = 1,
40        ],
41        /// VMID Size
42        VS OFFSET(19) NUMBITS(1) [
43            /// 8-bit VMID
44            Bits8 = 0,
45            /// 16-bit VMID
46            Bits16 = 1,
47        ],
48        /// Physical Address size of the second stage of translation
49        PS OFFSET(16) NUMBITS(3) [
50            /// 32 bits, 4GB
51            PA_32B_4GB = 0b000,
52            /// 36 bits, 64GB
53            PA_36B_64GB = 0b001,
54            /// 40 bits, 1TB
55            PA_40B_1TB = 0b010,
56            /// 42 bits, 4TB
57            PA_42B_4TB = 0b011,
58            /// 44 bits, 16TB
59            PA_44B_16TB = 0b100,
60            /// 48 bits, 256TB
61            PA_48B_256TB = 0b101,
62            /// 52 bits, 4PB
63            PA_52B_4PB = 0b110,
64        ],
65        /// Granule size used for `VTTBR_EL2`
66        TG0 OFFSET(14) NUMBITS(2) [
67            /// Granule size of 4KB
68            Granule4KB = 0b00,
69            /// Granule size of 16KB
70            Granule16KB = 0b10,
71            /// Granule size of 64KB
72            Granule64KB = 0b01,
73        ],
74        /// Shareability attribute for memory associated with translation table
75        /// walks using `VTTBR_EL2` and `VSTTBR_EL2`
76        SH0 OFFSET(12) NUMBITS(2) [
77            /// Non-shareable
78            Non = 0b00,
79            /// Outer sharable
80            Outer = 0b10,
81            /// Inner sharable
82            Inner = 0b11,
83        ],
84        /// Outer cacheability attribute for memory associated with translation table
85        /// walks using `VTTBR_EL2` and `VSTTBR_EL2`
86        ORGN0 OFFSET(10) NUMBITS(2) [
87            /// Normal non-cacheable memory
88            NormalNC = 0b00,
89            /// Normal Write-back, Read-allocate, Write-allocate
90            NormalWBRAWA = 0b01,
91            /// Normal Write-through, Read-allocate, no Write-allocate
92            NormalWTRAnWA = 0b10,
93            /// Normal Write-back, Read-allocate, no Write-allocate
94            NormalWBRAnWA = 0b11,
95        ],
96        /// Inner cacheability attribute for memory associated with translation table
97        /// walks using `VTTBR_EL2` and `VSTTBR_EL2`
98        IRGN0 OFFSET(8) NUMBITS(2) [
99            /// Normal non-cacheable memory
100            NormalNC = 0b00,
101            /// Normal Write-back, Read-allocate, Write-allocate
102            NormalWBRAWA = 0b01,
103            /// Normal Write-through, Read-allocate, no Write-allocate
104            NormalWTRAnWA = 0b10,
105            /// Normal Write-back, Read-allocate, no Write-allocate
106            NormalWBRAnWA = 0b11,
107        ],
108        /// Starting level of the stage2 translation lookup
109        SL0 OFFSET(6) NUMBITS(2) [
110            Granule4KBLevel2 = 0b00,
111            Granule4KBLevel1 = 0b01,
112            Granule4KBLevel0 = 0b10,
113            Granule4KBLevel3 = 0b11,
114        ],
115        /// The size of the offest of the memory region addressed by the `VTTBR_EL2`
116        T0SZ OFFSET(0) NUMBITS(6) [],
117
118    ]
119}
120pub struct Reg;
121

130    type T = u64;

132

134}

136pub const VTCR_EL2: Reg = Reg {};
