13/// Execute `FENCE.I` instruction on remote harts.
14///
15/// # Return value
16///
17/// The possible return error codes returned in `SbiRet.error` are shown in the table below:
18///
19/// | Return code               | Description
20/// |:--------------------------|:----------------------------------------------
21/// | `SbiRet::success()`       | A remote fence was sent to all the targeted harts successfully.
22/// | `SbiRet::invalid_param()` | At least one hartid constructed from `hart_mask`, is not valid, i.e. either the hartid is not enabled by the platform or is not available to the supervisor.
23/// | `SbiRet::failed()`        | The request failed for unspecified or unknown other reasons.
24///
25/// This function is defined in RISC-V SBI Specification chapter 8.1.
26#[inline]
27#[doc(alias = "sbi_remote_fence_i")]
28pub fn remote_fence_i(hart_mask: HartMask) -> SbiRet {
29    let (hart_mask, hart_mask_base) = hart_mask.into_inner();
30    sbi_call_2(EID_RFNC, REMOTE_FENCE_I, hart_mask, hart_mask_base)
31}

33/// Execute `SFENCE.VMA` instructions for all address spaces on remote harts.
34///
35/// This function instructs the remote harts to execute one or more `SFENCE.VMA` instructions,
36/// covering the range of virtual addresses between `start_addr` and `size`.
37///
38/// # Return value
39///
40/// The possible return error codes returned in `SbiRet.error` are shown in the table below:
41///
42/// | Return code                 | Description
43/// |:----------------------------|:----------------------------------------------
44/// | `SbiRet::success()`         | A remote fence was sent to all the targeted harts successfully.
45/// | `SbiRet::invalid_address()` | `start_addr` or `size` is not valid.
46/// | `SbiRet::invalid_param()`   | At least one hartid constructed from `hart_mask`, is not valid, i.e. either the hartid is not enabled by the platform or is not available to the supervisor.
47/// | `SbiRet::failed()`          | The request failed for unspecified or unknown other reasons.
48///
49/// This function is defined in RISC-V SBI Specification chapter 8.2.
50#[inline]
51#[doc(alias = "sbi_remote_sfence_vma")]
52pub fn remote_sfence_vma(hart_mask: HartMask, start_addr: usize, size: usize) -> SbiRet {
53    let (hart_mask, hart_mask_base) = hart_mask.into_inner();
54    sbi_call_4(
55        EID_RFNC,
56        REMOTE_SFENCE_VMA,
57        hart_mask,
58        hart_mask_base,
59        start_addr,
60        size,
61    )
62}

64/// Execute address-space-based `SFENCE.VMA` instructions on remote harts.

77/// | `SbiRet::invalid_address()` | `start_addr` or `size` is not valid.
78/// | `SbiRet::invalid_param()`   | Either `asid`, or at least one hartid constructed from `hart_mask`, is not valid, i.e. either the hartid is not enabled by the platform or is not available to the supervisor.
79/// | `SbiRet::failed()`          | The request failed for unspecified or unknown other reasons.
80///
81/// This function is defined in RISC-V SBI Specification chapter 8.3.
82#[inline]
83#[doc(alias = "sbi_remote_sfence_vma_asid")]
84pub fn remote_sfence_vma_asid(
85    hart_mask: HartMask,
86    start_addr: usize,
87    size: usize,
88    asid: usize,
89) -> SbiRet {
90    let (hart_mask, hart_mask_base) = hart_mask.into_inner();
91    sbi_call_5(
92        EID_RFNC,
93        REMOTE_SFENCE_VMA_ASID,
94        hart_mask,
95        hart_mask_base,
96        start_addr,
97        size,
98        asid,
99    )
100}
