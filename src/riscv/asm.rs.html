3macro_rules! instruction {
4    ($(#[$attr:meta])*, unsafe $fnname:ident, $asm:expr, $($options:tt)*) => (
5        $(#[$attr])*
6        #[inline(always)]
7        pub unsafe fn $fnname() {
8            #[cfg(any(target_arch = "riscv32", target_arch = "riscv64"))]
9            core::arch::asm!($asm, $($options)*);
10            #[cfg(not(any(target_arch = "riscv32", target_arch = "riscv64")))]
11            unimplemented!();
12        }
13    );
14    ($(#[$attr:meta])*, $fnname:ident, $asm:expr, $($options:tt)*) => (
15        $(#[$attr])*
16        #[inline(always)]
17        pub fn $fnname() {
18            #[cfg(any(target_arch = "riscv32", target_arch = "riscv64"))]
19            unsafe { core::arch::asm!($asm, $($options)*) };
20            #[cfg(not(any(target_arch = "riscv32", target_arch = "riscv64")))]
21            unimplemented!();
22        }
23    )
24}

26instruction!(
27    /// `NOP` instruction wrapper
28    ///
29    /// The `NOP` instruction does not change any architecturally visible state, except for
30    /// advancing the PC and incrementing any applicable performance counters.
31    ///
32    /// This function generates a no-operation; it's useful to prevent delay loops from being
33    /// optimized away.
34    , nop, "nop", options(nomem, nostack, preserves_flags));

36instruction!(
37    /// `WFI` instruction wrapper
38    ///
39    /// Provides a hint to the implementation that the current hart can be stalled until an
40    /// interrupt might need servicing. The WFI instruction is just a hint, and a legal
41    /// implementation is to implement WFI as a NOP.
42    ///
43    /// # Behavior
44    ///
45    /// - May cause the hart to enter a low-power state
46    /// - Will be interrupted by any enabled interrupt
47    /// - No guarantee of actual power savings (implementation-dependent)
48    ,wfi, "wfi", options(nomem, nostack, preserves_flags));

50instruction!(
51    /// `EBREAK` instruction wrapper
52    ///
53    /// Generates a breakpoint exception for use by debuggers.
54    ///
55    /// # Behavior
56    ///
57    /// When executed, this instruction causes a breakpoint exception to be raised,
58    /// which will typically be handled by a debugger or exception handler.
59    ///
60    /// # Safety
61    ///

63    /// which can disrupt normal program flow. Only call this when you intend to
64    /// trigger a breakpoint.

67instruction!(
68    /// `ECALL` instruction wrapper
69    ///
70    /// Generates an environment call exception for system services.
71    ///
72    /// # Behavior
73    ///
74    /// When executed in different privilege modes:
75    /// - U-mode: Generates environment-call-from-U-mode exception
76    /// - S-mode: Generates environment-call-from-S-mode exception
77    /// - M-mode: Generates environment-call-from-M-mode exception
78    ///
79    /// # Safety
80    ///
81    /// This function is unsafe because:
82    /// - It unconditionally generates an exception
83    /// - The stack pointer is **NOT** automatically saved/restored
84    /// - The exception handler is responsible for proper context management
85    /// - Improper use can crash the system
86    , unsafe ecall, "ecall", options(nomem, nostack, preserves_flags));

88instruction!(
89    /// `SFENCE.VMA` instruction wrapper (all address spaces and page table levels)
90    ///
91    /// Synchronizes updates to in-memory memory-management data structures with current execution.
92    /// Instruction execution causes implicit reads and writes to these data structures; however, these implicit references
93    /// are ordinarily not ordered with respect to loads and stores in the instruction stream.
94    /// Executing an `SFENCE.VMA` instruction guarantees that any stores in the instruction stream prior to the
95    /// `SFENCE.VMA` are ordered before all implicit references subsequent to the `SFENCE.VMA`.
96    , sfence_vma_all, "sfence.vma", options(nostack));

98instruction!(
99    /// `FENCE` instruction wrapper
100    ///
101    /// The FENCE instruction is used to order device I/O and memory accesses as viewed by other RISC-V
102    /// harts and external devices or coprocessors. Any combination of device input (I), device output
103    /// (O), memory reads (R), and memory writes (W) may be ordered with respect to any combination
104    /// of the same. Informally, no other RISC-V hart or external device can observe any operation in the
105    /// successor set following a FENCE before any operation in the predecessor set preceding the FENCE.
106    ///
107    /// The FENCE instruction also orders memory reads and writes made by the hart as observed by
108    /// memory reads and writes made by an external device. However, FENCE does not order observations
109    /// of events made by an external device using any other signaling mechanism.
110    , fence, "fence", options(nostack));

112instruction!(
113    /// `FENCE.I` instruction wrapper
114    ///
115    /// Used to synchronize the instruction and data streams. RISC-V does not guarantee that
116    /// stores to instruction memory will be made visible to instruction fetches on a
117    /// RISC-V hart until that hart executes a FENCE.I instruction.
118    ///
119    /// A FENCE.I instruction ensures that a subsequent instruction fetch on a RISC-V hart
120    /// will see any previous data stores already visible to the same RISC-V hart.
121    /// FENCE.I does not ensure that other RISC-V harts’ instruction fetches will observe the
122    /// local hart’s stores in a multiprocessor system. To make a store to instruction memory
123    /// visible to all RISC-V harts, the writing hart also has to execute a data FENCE before
124    /// requesting that all remote RISC-V harts execute a FENCE.I.
125    ///
126    /// The unused fields in the FENCE.I instruction, imm\[11:0\], rs1, and rd, are reserved for
127    /// finer-grain fences in future extensions. For forward compatibility, base
128    /// implementations shall ignore these fields, and standard software shall zero these fields.
129    , fence_i, "fence.i", options(nostack));

131/// `SFENCE.VMA` instruction wrapper
132///
133/// Synchronizes updates to in-memory memory-management data structures with current execution.
134/// Instruction execution causes implicit reads and writes to these data structures; however, these implicit references
135/// are ordinarily not ordered with respect to loads and stores in the instruction stream.
136/// Executing an `SFENCE.VMA` instruction guarantees that any stores in the instruction stream prior to the
137/// `SFENCE.VMA` are ordered before all implicit references subsequent to the `SFENCE.VMA`.
138#[inline(always)]
139#[cfg_attr(
140    not(any(target_arch = "riscv32", target_arch = "riscv64")),
141    allow(unused_variables)
142)]
143pub fn sfence_vma(asid: usize, addr: usize) {
144    #[cfg(any(target_arch = "riscv32", target_arch = "riscv64"))]
145    unsafe {
146        core::arch::asm!("sfence.vma {}, {}", in(reg) addr, in(reg) asid, options(nostack));
147    }
148    #[cfg(not(any(target_arch = "riscv32", target_arch = "riscv64")))]
149    unimplemented!();
150}

152/// Blocks the program for *at least* `cycles` CPU cycles.
153///
154/// This is implemented in assembly so its execution time is independent of the optimization
155/// level, however it is dependent on the specific architecture and core configuration.
156///
157/// NOTE that the delay can take much longer if interrupts are serviced during its execution
158/// and the execution time may vary with other factors. This delay is mainly useful for simple
159/// timer-less initialization of peripherals if and only if accurate timing is not essential. In
160/// any other case please use a more accurate method to produce a delay.
161#[inline]
162#[cfg_attr(
163    not(any(target_arch = "riscv32", target_arch = "riscv64")),
164    allow(unused_variables)
165)]
166pub fn delay(cycles: u32) {
167    #[cfg(any(target_arch = "riscv32", target_arch = "riscv64"))]
168    {
169        let real_cyc = 1 + cycles / 2;
170        unsafe {
171            core::arch::asm!(
172                "2:",
173                "addi {0}, {0}, -1",
174                "bne {0}, zero, 2b",
175                inout(reg) real_cyc => _,
176                options(nomem, nostack),
177            );
178        }
179    }
180    #[cfg(not(any(target_arch = "riscv32", target_arch = "riscv64")))]
181    unimplemented!();
182}
