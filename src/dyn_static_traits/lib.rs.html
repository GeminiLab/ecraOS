17/// Creates a compile error attached to a syntax node.
18///
19/// The returned token stream highlights `span` and reports `msg` when the macro expansion is
20/// compiled.
21fn error_at<S: Spanned, M: AsRef<str> + ?Sized, T>(span: &S, msg: &M) -> Result<T, TokenStream> {
22    let span = span.span();
23    let msg = msg.as_ref();
24    Err(quote_spanned!(span => compile_error!(#msg);))
25}

27/// Collects the static associated-method signatures from a trait.
28///
29/// A static trait contains only associated functions. Each function must have no receiver, which
30/// means its declaration starts with `fn name(...)` instead of `fn name(&self, ...)`,
31/// `fn name(&mut self, ...)`, or another receiver form.
32///
33/// Functions with receivers, generic parameters, asynchronous or variadic signatures, and
34/// non-function trait items are rejected because they cannot be represented by the generated
35/// function-pointer fields.
36fn collect_static_trait_methods(trait_item: &ItemTrait) -> Result<Vec<&Signature>, TokenStream> {
37    let mut result = Vec::with_capacity(trait_item.items.len());
38
39    for item in trait_item.items.iter() {
40        match item {
41            TraitItem::Fn(fn_item) => {
42                if !fn_item.sig.generics.params.is_empty() {
43                    return error_at(
44                        &fn_item.sig.generics,
45                        "Associated functions must not have generic parameters",
46                    );
47                }
48
49                if fn_item.sig.asyncness.is_some() {
50                    return error_at(
51                        &fn_item.sig.asyncness,
52                        "Associated functions must not be async",
53                    );
54                }
55
56                if fn_item.sig.variadic.is_some() {
57                    return error_at(
58                        &fn_item.sig.variadic,
59                        "Associated functions must not be variadic",
60                    );
61                }
62
63                if let Some(receiver) = fn_item.sig.receiver() {
64                    return error_at(receiver, "Associated functions must not have a receiver");
65                }
66
67                result.push(&fn_item.sig);
68            }
69            _ => return error_at(item, "Static traits must only contain functions"),
70        }
71    }
72
73    Ok(result)
74}

76/// Chooses the implementation type parameter name for the generated constructor.
77///
78/// `T` is preferred. When the trait name or one of its type or const parameters already uses a
79/// candidate, `T` followed by uppercase letters is tried next, from shorter to longer suffixes and
80/// alphabetically among suffixes of one length, as in `TA`, `TB`, ..., `TZ`, `TAA`. Lifetime
81/// parameters never conflict with the chosen name.
82fn implementation_type_ident(trait_item: &ItemTrait) -> Ident {
83    let occupied = trait_item
84        .generics
85        .params
86        .iter()
87        .filter_map(|param| match param {
88            GenericParam::Type(param) => Some(&param.ident),
89            GenericParam::Const(param) => Some(&param.ident),
90            GenericParam::Lifetime(_) => None,
91        })
92        .chain([&trait_item.ident])
93        .map(|ident| ident.unraw().to_string())
94        .collect::<Vec<_>>();
95
96    let mut suffix = Vec::new();
97    loop {
98        let candidate = core::iter::once('T')
99            .chain(suffix.iter().map(|&letter| char::from(letter)))
100            .collect::<String>();
101        if !occupied.contains(&candidate) {
102            return format_ident!("{}", candidate);
103        }
104        match suffix.iter().rposition(|&letter| letter != b'Z') {
105            Some(index) => {
106                suffix[index] += 1;
107                suffix[index + 1..].fill(b'A');
108            }
109            None => {
110                suffix.fill(b'A');
111                suffix.push(b'A');
112            }
113        }
114    }
115}

117/// Expands a static trait and its function-pointer table.
118///
119/// `input_trait` must be a static trait, meaning that every associated method is an associated
120/// function without a receiver. `output_ident` names the generated table. Each table field
121/// corresponds to one static associated method, and the generated constructor binds those fields to
122/// a selected trait implementation.
123fn dyn_static_traits_impl(
124    input_trait: ItemTrait,
125    output_ident: Ident,
126) -> Result<TokenStream, TokenStream> {
127    let vis = &input_trait.vis;
128    let trait_ident = &input_trait.ident;
129    let trait_generics = &input_trait.generics;
130    let (impl_generics, type_generics, where_clause) = trait_generics.split_for_impl();
131    let signatures = collect_static_trait_methods(&input_trait)?;
132    let impl_ident = implementation_type_ident(&input_trait);
133
134    let mut fields = TokenStream::new();
135    let mut init = TokenStream::new();
136    let mut clone_fields = TokenStream::new();
137
138    for signature in signatures {
139        let fn_ident = signature.ident.clone();
140        let safety = signature.unsafety;
141        let api = &signature.abi;
142        let inputs = &signature.inputs;
143        let output = &signature.output;
144
145        let field_desc = format!(
146            concat!(
147                "The function pointer for [`{}::{}`].\n\n",
148                "The pointer invokes this static associated function on the implementation\n",
149                "selected when the table was constructed."
150            ),
151            trait_ident, fn_ident
152        );
153
154        fields.append_all(quote! {
155            #[doc = #field_desc]
156            pub #fn_ident: #safety #api fn(#inputs) #output,
157        });
158
159        init.append_all(quote! {
160            #fn_ident: #impl_ident::#fn_ident,
161        });
162
163        clone_fields.append_all(quote! {
164            #fn_ident: self.#fn_ident,
165        });
166    }
167
168    let output_desc = format!(
169        concat!(
170            "A function-pointer table for [`{}`].\n\n",
171            "Each field stores one static associated method from the implementation selected by\n",
172            "[`Self::new`]. Each `new::<{}>` call fixes its implementation statically, while\n",
173            "tables built from different implementations share this type and can be selected at\n",
174            "runtime.",
175        ),
176        trait_ident, impl_ident
177    );
178    let new_desc = format!(
179        concat!(
180            "Creates a function-pointer table for a static-trait implementation.\n\n",
181            "`{0}` supplies the static associated methods stored in the returned fields. The\n",
182            "table can then be passed around independently of `{0}`.",
183        ),
184        impl_ident
185    );
186
187    Ok(quote! {
188        #input_trait
189
190        #[doc = #output_desc]
191        #vis struct #output_ident #trait_generics {
192            #fields
193        }
194
195        /// Constructs function-pointer tables from static-trait implementations.
196        ///
197        /// Each constructed table is independent of the implementation type that built it.
198        impl #impl_generics #output_ident #type_generics #where_clause {
199            #[doc = #new_desc]
200            pub fn new<#impl_ident: #trait_ident #type_generics>() -> Self {
201                Self {
202                    #init
203                }
204            }
205        }
206
207        /// Clones function-pointer tables by copying their fields.
208        ///
209        /// Function pointers are always copyable, so the trait's type parameters need not
210        /// implement `Clone`.
211        impl #impl_generics ::core::clone::Clone for #output_ident #type_generics #where_clause {
212            /// Returns a table whose fields point to the same functions.
213            ///
214            /// Each function pointer is copied from `self`.
215            fn clone(&self) -> Self {
216                Self {
217                    #clone_fields
218                }
219            }
220        }
221    })
222}

224/// Generates a function-pointer table for a static trait.
225///
226/// The attribute is applied to a static trait and takes the name of the table to generate as its
227/// only argument, as in `#[dyn_static_traits(DynClock)]`. The trait is emitted unchanged, so it
228/// still works as a generic parameter, and the generated table adds runtime selection between its
229/// implementations. The generated code needs no runtime support and works in `no_std` crates.
230///
231/// # Static traits
232///
233/// In this crate, a static trait means a trait whose items are exclusively static associated
234/// methods, which Rust calls associated functions with no receiver in their signature. For example,
235/// `fn page_size() -> usize` is static, while `fn page_size(&self) -> usize` is not.
236///
237/// Static traits are useful as generic parameters for static dependency injection. A generic
238/// function such as `fn read<T: Storage>()` can call `T::read()` with the implementation selected
239/// at compile time. This pattern applies to many scenarios like allocators, logging, clocks, device
240/// access, synchronization policies, platform services, and other static interfaces.
241///
242/// Because their functions have no receiver, static traits are not dyn compatible, so a
243/// `dyn Trait` object cannot select their implementation at runtime. The generated table provides
244/// dynamic behavior for the same static interface instead. It is a concrete struct of function
245/// pointers, not a `dyn Trait` object. Each `new::<T>` call still names its implementation type
246/// statically, but the resulting table no longer carries `T` in its own type. Tables built from
247/// different implementations can therefore be stored behind one common type, passed across an API
248/// boundary, or chosen conditionally at runtime.
249///
250/// # Generated items
251///
252/// For `#[dyn_static_traits(DynClock)]` on a trait `Clock`, the macro emits the trait unchanged,
253/// followed by these items:
254///
255/// - A struct `DynClock` with the visibility and generic parameters of `Clock`. It has one public
256///   field per associated function, named after the function.
257/// - An inherent `pub fn new<T: Clock>() -> Self` constructor that stores the functions of `T` in
258///   the fields. A function that `T` does not override uses the default body from the trait.
259/// - A `Clone` implementation that copies every function pointer. It places no `Clone` bound on
260///   the parameters of the trait.
261///
262/// The table, its fields, its constructor, and its `Clone` implementation receive generated doc
263/// comments. For a trait with one function, the expansion is roughly equivalent to the following
264/// code:
265///
266/// ```ignore
267/// trait Clock {
268///     fn now() -> u64;
269/// }
270///
271/// struct DynClock {
272///     pub now: fn() -> u64,
273/// }
274///
275/// impl DynClock {
276///     pub fn new<T: Clock>() -> Self {
277///         Self { now: T::now }
278///     }
279/// }
280///
281/// impl Clone for DynClock {
282///     fn clone(&self) -> Self {
283///         Self { now: self.now }
284///     }
285/// }
286/// ```
287///
288/// ## Function-pointer fields
289///
290/// Each field type keeps the parameters, return type, ABI, and `unsafe` qualifier of its function,
291/// so `unsafe extern "C" fn read(pointer: *const u32) -> u32;` produces a field of type
292/// `unsafe extern "C" fn(pointer: *const u32) -> u32`. Elided lifetimes in a signature become
293/// higher-ranked lifetimes of the function pointer, so a function such as
294/// `fn first(text: &str) -> &str` keeps its borrowing behavior. A field for an `unsafe fn` still
295/// requires an unsafe call and retains the safety contract of the original associated function.
296///
297/// A field is called with parentheses around the field access, such as `(table.now)()`, because
298/// `table.now()` would look for a method.
299///
300/// ## Generic traits
301///
302/// Type, const, and lifetime parameters of the trait become parameters of the table, and `new`
303/// requires its implementation parameter to implement the trait with the same arguments, as in
304/// `DynStore::<u8, 4>::new::<Zeros>()`. A generic table is `Clone` regardless of its type
305/// arguments. Every type and lifetime parameter of the trait must appear in at least one function
306/// signature, because the generated struct would otherwise leave it unused.
307///
308/// The implementation parameter of `new` is named `T` when that name is free. When the trait
309/// itself or one of its type or const parameters is named `T`, the macro uses the first free name
310/// among `TA` to `TZ`, then `TAA`, `TAB`, and so on, so trait parameters can use any name.
311///
312/// # Requirements
313///
314/// The annotated trait must contain only static associated methods. The macro reports a compile
315/// error for:
316///
317/// - functions with a `self` receiver,
318/// - functions with generic parameters, including lifetime parameters,
319/// - `async` or variadic functions,
320/// - associated constants, associated types, macro invocations, and other non-function items.
321///
322/// Functions may have default bodies. Some signatures pass these checks, but a function-pointer
323/// field cannot represent them, so the generated table fails to compile:
324///
325/// - signatures that mention `Self`, which refers to the table inside the generated struct,
326/// - `impl Trait` in a parameter or return type,
327/// - parameter patterns other than plain names and `_`, such as `mut value` or `(left, right)`,
328///   even in functions with default bodies.
329///
330/// Attributes on trait functions stay on the trait and are not copied to the fields. In
331/// particular, a function under a disabled `#[cfg]` still receives a field, so the constructor
332/// fails to compile.
333///
334/// # Examples
335///
336/// ```
337/// use dyn_static_traits::dyn_static_traits;
338///
339/// #[dyn_static_traits(DynClock)]
340/// trait Clock {
341///     fn now() -> u64;
342/// }
343///
344/// struct MonotonicClock;
345///
346/// impl Clock for MonotonicClock {
347///     fn now() -> u64 {
348///         42
349///     }
350/// }
351///
352/// // Static dependency injection uses the trait as a generic parameter.
353/// fn read_time<T: Clock>() -> u64 {
354///     T::now()
355/// }
356/// assert_eq!(read_time::<MonotonicClock>(), 42);
357///
358/// // Dynamic selection stores the generated function-pointer table.
359/// let clock = DynClock::new::<MonotonicClock>();
360/// assert_eq!((clock.now)(), 42);
361/// ```
362///
363/// Tables built from different implementations share one type, so a program can choose between
364/// them at runtime:
365///
366/// ```
367/// use dyn_static_traits::dyn_static_traits;
368///
369/// #[dyn_static_traits(DynConsole)]
370/// trait Console {
371///     fn name() -> &'static str;
372/// }
373///
374/// struct Serial;
375///
376/// impl Console for Serial {
377///     fn name() -> &'static str {
378///         "serial"
379///     }
380/// }
381///
382/// struct Framebuffer;
383///
384/// impl Console for Framebuffer {
385///     fn name() -> &'static str {
386///         "framebuffer"
387///     }
388/// }
389///
390/// fn select(graphics: bool) -> DynConsole {
391///     if graphics {
392///         DynConsole::new::<Framebuffer>()
393///     } else {
394///         DynConsole::new::<Serial>()
395///     }
396/// }
397///
398/// assert_eq!((select(false).name)(), "serial");
399/// assert_eq!((select(true).name)(), "framebuffer");
400/// ```
401///
402/// A generic trait produces a generic table, and default bodies fill the fields that an
403/// implementation does not override:
404///
405/// ```
406/// use dyn_static_traits::dyn_static_traits;
407///
408/// #[dyn_static_traits(DynStore)]
409/// trait Store<V: Copy, const N: usize> {
410///     fn read(index: usize) -> V;
411///
412///     fn capacity() -> usize {
413///         N
414///     }
415/// }
416///
417/// struct Zeros;
418///
419/// impl Store<u8, 4> for Zeros {
420///     fn read(_index: usize) -> u8 {
421///         0
422///     }
423/// }
424///
425/// let store = DynStore::<u8, 4>::new::<Zeros>();
426/// assert_eq!((store.read)(2), 0);
427/// assert_eq!((store.capacity)(), 4);
428/// ```
429///
430/// An `unsafe` associated function keeps its qualifier in the table:
431///
432/// ```
433/// use dyn_static_traits::dyn_static_traits;
434///
435/// #[dyn_static_traits(DynReader)]
436/// trait Reader {
437///     /// Reads a value through a raw pointer.
438///     ///
439///     /// `pointer` must be valid for reads.
440///     unsafe fn read(pointer: *const u32) -> u32;
441/// }
442///
443/// struct DirectReader;
444///
445/// impl Reader for DirectReader {
446///     unsafe fn read(pointer: *const u32) -> u32 {
447///         // SAFETY: The caller guarantees that `pointer` is valid for reads.
448///         unsafe { *pointer }
449///     }
450/// }
451///
452/// let reader = DynReader::new::<DirectReader>();
453/// let value = 7;
454/// // SAFETY: `value` is live for the duration of the call.
455/// assert_eq!(unsafe { (reader.read)(&raw const value) }, 7);
456/// ```
457#[proc_macro_attribute]
458pub fn dyn_static_traits(attrs: TokenStream1, input: TokenStream1) -> TokenStream1 {
459    let output_ident = parse_macro_input!(attrs as Ident);
460    let input_trait = parse_macro_input!(input as ItemTrait);
461
462    dyn_static_traits_impl(input_trait, output_ident)
463        .unwrap_or_else(|e| e)
464        .into()
465}
