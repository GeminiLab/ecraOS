1//! Abstract page allocator interface.
2//!
3//! This crate defines [`PageAllocator`], a static interface used by low-level
4//! components that need physically addressed pages and virtual addresses to
5//! them, such as page-table implementations. The accompanying
6//! [`DynPageAllocator`] function-pointer table permits choosing
7//! an implementation at runtime.
8
9#![no_std]
10
11use memory_addr::{PhysAddr, VirtAddr};
12
13/// A physical page allocator.
14///
15/// Implementations allocate physically contiguous, page-aligned frames and
16/// translate their physical addresses into virtual addresses through which the
17/// caller can access them. All frame counts use the page size reported by
18/// [`Self::page_size_shift`].
19#[dyn_static_traits::dyn_static_traits(DynPageAllocator)]
20pub trait PageAllocator {
21    /// Returns the base-2 logarithm of the allocator's page size.
22    ///
23    /// A page contains `1 << page_size_shift()` bytes. The value must remain
24    /// constant for the lifetime of every allocation made through this type.
25    fn page_size_shift() -> usize;
26
27    /// Allocates one frame.
28    ///
29    /// By default, this delegates to [`Self::alloc_frames`] with a page count
30    /// of one.
31    fn alloc_frame() -> Option<PhysAddr> {
32        Self::alloc_frames(1)
33    }
34
35    /// Allocates a contiguous run of frames.
36    ///
37    /// Returns the physical address of the first frame, or [`None`] when the
38    /// requested run cannot be allocated. The returned address must be aligned
39    /// to the allocator's page size.
40    fn alloc_frames(page_count: usize) -> Option<PhysAddr>;
41
42    /// Allocates enough contiguous frames to cover a byte size.
43    ///
44    /// By default, this delegates to [`Self::alloc_frames`] with a page count
45    /// computed from the byte size.
46    fn alloc_frames_of_size(size_in_bytes: usize) -> Option<PhysAddr> {
47        Self::alloc_frames(size_in_bytes.div_ceil(1usize << Self::page_size_shift()))
48    }
49
50    /// Deallocates one frame.
51    ///
52    /// By default, this delegates to [`Self::dealloc_frames`] with a page count
53    /// of one.
54    fn dealloc_frame(addr: PhysAddr) {
55        Self::dealloc_frames(addr, 1)
56    }
57
58    /// Deallocates a contiguous run of frames.
59    ///
60    /// `addr` and `page_count` must exactly match a previous allocation
61    /// returned by this allocator.
62    fn dealloc_frames(addr: PhysAddr, page_count: usize);
63
64    /// Deallocates the frames covering a byte size.
65    ///
66    /// By default, this delegates to [`Self::dealloc_frames`] with a page count
67    /// computed from the byte size. Therefore, `size_in_bytes` could actually
68    /// be different from the actual number used when allocating frames. However,
69    /// this is not promised to be the case for every allocator.
70    fn dealloc_frames_of_size(addr: PhysAddr, size_in_bytes: usize) {
71        Self::dealloc_frames(
72            addr,
73            size_in_bytes.div_ceil(1usize << Self::page_size_shift()),
74        )
75    }
76
77    /// Translates a physical address into an accessible virtual address.
78    fn phys_to_virt(addr: PhysAddr) -> VirtAddr;
79}
