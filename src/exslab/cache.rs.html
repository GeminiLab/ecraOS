1//! Size-class slab caches.
2//!
3//! A cache holds the slabs of one size class. It keeps slabs with free slots separate from full
4//! ones so that allocation rarely has to search, and retains one empty slab instead of returning it
5//! immediately.
6//!
7//! A remote free touches only the atomic list in a slab header, so it needs no access to the cache.
8//! The cache drains pending remote frees when a later allocation or deallocation uses that slab.
9
10use core::ptr::NonNull;
11
12use memory_addr::{MemoryAddr, VirtAddr};
13
14use crate::{
15    SlabLayout,
16    error::{AllocError, AllocResult},
17    slab_page::{SlabPage, SlabPageHeader},
18};

20/// A size-class slab cache.
21///
22/// The cache owns a partial list, a full list, and one retained empty slab. All three hold slabs
23/// registered with [`SlabCache::add_slab`]. The bound layout is fixed at construction, so every
24/// slab in the cache has the same size, slot count, and slot size.
25pub struct SlabCache {
26    /// Slabs with at least one locally available slot.
27    partial: Option<NonNull<SlabPageHeader>>,
28    /// Slabs with no locally available slot, though some may have pending remote frees.
29    full: Option<NonNull<SlabPageHeader>>,
30    /// Retained empty slab, kept out of both lists.
31    empty: Option<NonNull<SlabPageHeader>>,
32    /// Shared compile-time layout selected when the cache is constructed.
33    layout: &'static SlabLayout,
34}
35
36/// Slot allocation and slab memory management for [`SlabCache`].
37impl SlabCache {
38    /// Creates an empty size-class cache.
39    ///
40    /// `object_log2` and `slab_log2` are the base-2 logarithms of the slot and slab sizes in bytes.
41    /// They must select a power-of-two slot size from 8 to 2048 bytes and a power-of-two slab size
42    /// from 4 to 256 KiB. Other values return [`AllocError::InvalidParam`]. The resulting cache
43    /// holds no slab until the caller registers one with [`Self::add_slab`].
44    pub const fn new(object_log2: usize, slab_log2: usize) -> AllocResult<Self> {
45        match SlabLayout::select(object_log2, slab_log2) {
46            Some(layout) => Ok(Self::with_layout(layout)),
47            None => Err(AllocError::InvalidParam),
48        }
49    }
50
51    /// Binds an empty cache to one validated static slab layout record.
52    ///
53    /// Used by [`crate::SlabAllocator`], which selects the layout from its page-size policy.
54    pub(crate) const fn with_layout(layout: &'static SlabLayout) -> Self {
55        Self {
56            partial: None,
57            full: None,
58            empty: None,
59            layout,
60        }
61    }
62
63    /// Returns the immutable slab layout bound to this cache.
64    pub const fn layout(&self) -> &'static SlabLayout {
65        self.layout
66    }
67
68    /// Registers slab memory with this cache.
69    ///
70    /// Initializes the slab so that all of its slots are available, then adds it to the partial
71    /// list. Returns [`AllocError::InvalidParam`] before memory access if the base is zero or is
72    /// not aligned to [`SlabLayout::slab_size`].
73    ///
74    /// # Safety
75    ///
76    /// An accepted base must identify exclusive writable slab memory of the bound slab size. Keep
77    /// this memory valid until it is returned or all cache operations, slots, and remote producers
78    /// end.
79    pub unsafe fn add_slab(&mut self, base: VirtAddr, owner_cpu: u16) -> AllocResult {
80        if base.as_usize() == 0 || !base.is_aligned(self.layout.slab_size) {
81            return Err(AllocError::InvalidParam);
82        }
83        let slab = NonNull::new(base.as_mut_ptr_of::<SlabPage>()).unwrap();
84        // SAFETY: The caller supplies exclusive slab memory of the checked size and alignment.
85        unsafe {
86            SlabPage::init_at(slab, self.layout, owner_cpu);
87            Self::push_front(&mut self.partial, slab.cast());
88        }
89        Ok(())
90    }
91
92    /// Allocates one object slot.
93    ///
94    /// Takes a slot from the first partial slab. If there is none, a full slab with pending remote
95    /// frees is moved back to the partial list, and failing that the retained empty slab is reused.
96    /// Returns `None` when the cache holds no slab that can supply a slot, which is the caller's
97    /// signal to register a new slab.
98    ///
99    /// The caller serializes all operations through this cache.
100    pub fn alloc_object(&mut self) -> Option<VirtAddr> {
101        if let Some(slab) = self.partial {
102            // SAFETY: Private state fixes the slab layout and the mutable borrow serializes access.
103            return Some(unsafe { self.alloc_from_partial(slab) });
104        }
105        let mut current = self.full;
106        while let Some(slab) = current {
107            // SAFETY: Full-list headers remain live and links have exclusive owner access.
108            current = unsafe { (*slab.as_ptr()).list_next };
109            // SAFETY: Only the atomic remote head is accessed concurrently.
110            if unsafe { SlabPageHeader::has_remote_frees(slab) } {
111                // SAFETY: The cache owns the slab and serializes bitmap and list updates.
112                unsafe {
113                    Self::remove(&mut self.full, slab);
114                    Self::push_front(&mut self.partial, slab);
115                    return Some(self.alloc_from_partial(slab));
116                }
117            }
118        }
119        let slab = self.empty.take()?;
120        // SAFETY: A retained empty slab is live and has no list neighbors.
121        unsafe {
122            Self::push_front(&mut self.partial, slab);
123            Some(self.alloc_from_partial(slab))
124        }
125    }
126
127    /// Releases an object slot to this cache.
128    ///
129    /// Collects pending remote frees before judging whether the slab is empty, so that a slab
130    /// emptied by remote frees is not mistaken for a partial one. Returns `Some` with the slab
131    /// base when the slab became empty and the cache already retained another empty slab,
132    /// leaving that slab for the caller to release. Its memory spans [`SlabLayout::slab_size`]
133    /// bytes. Otherwise returns `None`, possibly after retaining this slab as the empty one.
134    ///
135    /// # Safety
136    ///
137    /// The address must identify one live slot belonging to this cache, released exactly once.
138    /// The caller must serialize this call with all other operations through the cache.
139    pub unsafe fn dealloc_object(&mut self, obj_addr: VirtAddr) -> Option<VirtAddr> {
140        // SAFETY: A live slot belongs to slab memory with this cache's exact slab layout.
141        let slab = unsafe { SlabPage::from_object(obj_addr, self.layout) };
142        let header = slab.cast::<SlabPageHeader>();
143        // SAFETY: Local fields have serialized owner access, without whole-header references.
144        unsafe {
145            debug_assert_eq!((*header.as_ptr()).size_class, self.layout.size_class);
146            let was_full = (*header.as_ptr()).local_free_count == 0;
147            SlabPage::local_free(slab, self.layout, obj_addr);
148            if was_full {
149                Self::remove(&mut self.full, header);
150                Self::push_front(&mut self.partial, header);
151            }
152            SlabPage::drain_remote_frees(slab, self.layout);
153            if (*header.as_ptr()).local_free_count != self.layout.slot_count as u16 {
154                return None;
155            }
156            Self::remove(&mut self.partial, header);
157            if self.empty.is_none() {
158                self.empty = Some(header);
159                None
160            } else {
161                Some(VirtAddr::from_mut_ptr_of(slab.as_ptr()))
162            }
163        }
164    }
165
166    /// Allocates a slot from a partial slab.
167    ///
168    /// Moves the slab to the full list if the final locally available slot is taken.
169    ///
170    /// # Safety
171    ///
172    /// The slab must belong to this cache's partial list and have its bound slab layout.
173    /// All owner accesses must be serialized and the slab memory must remain valid.
174    unsafe fn alloc_from_partial(&mut self, header: NonNull<SlabPageHeader>) -> VirtAddr {
175        let slab = header.cast::<SlabPage>();
176        // SAFETY: The caller guarantees a matching slab layout and serialized owner accesses.
177        unsafe {
178            SlabPage::drain_remote_frees(slab, self.layout);
179            let object =
180                SlabPage::local_alloc(slab, self.layout).expect("a partial slab has a free slot");
181            if (*header.as_ptr()).local_free_count == 0 {
182                Self::remove(&mut self.partial, header);
183                Self::push_front(&mut self.full, header);
184            }
185            object
186        }
187    }

189    /// Inserts a live unlinked slab at the front of an owner-managed list.
190    ///
191    /// # Safety
192    ///
193    /// All nodes must have live headers and valid links. The owner must have exclusive access
194    /// to the links, and `slab` must not already belong to a list.
195    unsafe fn push_front(
196        head: &mut Option<NonNull<SlabPageHeader>>,
197        slab: NonNull<SlabPageHeader>,
198    ) {
199        // SAFETY: Individual link fields are live and exclusively owner-managed.
200        unsafe {
201            (*slab.as_ptr()).list_prev = None;
202            (*slab.as_ptr()).list_next = *head;
203            if let Some(first) = *head {
204                (*first.as_ptr()).list_prev = Some(slab);
205            }
206        }
207        *head = Some(slab);
208    }
209
210    /// Removes one slab from an owner-managed doubly linked list.
211    ///
212    /// # Safety
213    ///
214    /// `slab` must belong to `head`, all links must identify live headers, and the owner must
215    /// serialize all link updates.
216    unsafe fn remove(head: &mut Option<NonNull<SlabPageHeader>>, slab: NonNull<SlabPageHeader>) {
217        // SAFETY: The caller guarantees live, exclusively owner-managed link fields.
218        unsafe {
219            let previous = (*slab.as_ptr()).list_prev;
220            let next = (*slab.as_ptr()).list_next;
221            if let Some(previous) = previous {
222                (*previous.as_ptr()).list_next = next;
223            } else {
224                *head = next;
225            }
226            if let Some(next) = next {
227                (*next.as_ptr()).list_prev = previous;
228            }
229            (*slab.as_ptr()).list_prev = None;
230            (*slab.as_ptr()).list_next = None;
231        }
232    }
