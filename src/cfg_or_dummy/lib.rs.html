9/// Collects identifiers bound by a function parameter pattern.
10///
11/// The supported pattern forms cover named parameters and the destructuring forms accepted by
12/// `cfg_or_dummy`. Patterns without collectable identifiers contribute no bindings to the
13/// fallback body.
14fn collect_ident_from_pat<'a>(pat: &'a Pat) -> Box<dyn Iterator<Item = &'a Ident> + 'a> {
15    match pat {
16        Pat::Ident(pat_ident) => Box::new(iter::once(&pat_ident.ident)),
17        Pat::Paren(pat_paren) => collect_ident_from_pat(&pat_paren.pat),
18        Pat::Reference(pat_reference) => collect_ident_from_pat(&pat_reference.pat),
19        Pat::Slice(pat_slice) => Box::new(pat_slice.elems.iter().flat_map(collect_ident_from_pat)),
20        Pat::Struct(pat_struct) => Box::new(
21            pat_struct
22                .fields
23                .iter()
24                .flat_map(|f| collect_ident_from_pat(&f.pat)),
25        ),
26        Pat::Tuple(pat_tuple) => Box::new(pat_tuple.elems.iter().flat_map(collect_ident_from_pat)),
27        Pat::TupleStruct(pat_tuple_struct) => Box::new(
28            pat_tuple_struct
29                .elems
30                .iter()
31                .flat_map(collect_ident_from_pat),
32        ),
33        _ => Box::new(iter::empty()),
34    }
35}

37/// Builds the conditional function expansion.
38///
39/// Emits the original function under `cond` and an unreachable dummy function under `not(cond)`.
40fn cfg_or_dummy_impl(cond: Meta, func: ItemFn) -> TokenStream {
41    let params = func
42        .sig
43        .inputs
44        .iter()
45        .filter_map(|f| match f {
46            FnArg::Typed(pat) => Some(pat),
47            _ => None,
48        })
49        .flat_map(|p| collect_ident_from_pat(&p.pat));
50
51    let mut func_unreachable = func.clone();
52    func_unreachable.block = parse_quote!({
53        #(
54            let _ = #params;
55        )*
56
57        unreachable!()
58    });
59
60    quote::quote!(
61        #[cfg(#cond)]
62        #[cfg_attr(doc, doc(cfg(#cond)))]
63        #func
64        #[cfg(not(#cond))]
65        #[cfg_attr(doc, doc(cfg(#cond)))]
66        #func_unreachable
67    )
68}

70/// Adds a condition-specific implementation and a dummy implementation for a free function.
71///
72/// The attribute accepts one `cfg` metadata item, such as `target_arch = "aarch64"` or
73/// `any(target_arch = "aarch64", target_arch = "riscv64")`. For a function annotated with
74/// `#[cfg_or_dummy(condition)]`, the macro emits two conditional definitions:
75///
76/// - `#[cfg(condition)]` emits the original function and its implementation.
77/// - `#[cfg(not(condition))]` emits a dummy function with the same interface.
78///
79/// Only the definition whose condition matches the current configuration is compiled. The dummy
80/// preserves the function's signature, visibility, attributes, generics, ABI, safety, and return
81/// type. Its body is unreachable, so it does not implement the target-specific operation.
82///
83/// # Documentation builds
84///
85/// This attribute is intended for functions inside modules enabled with
86/// `#[cfg(any(condition, doc))]`. During a documentation build for another target, `doc` keeps the
87/// module available. A function marked only with `#[cfg(condition)]` would disappear from the
88/// generated documentation, while an unconditionally compiled function could fail because its
89/// body refers to target-specific code. The dummy keeps the function interface available while
90/// avoiding that body.
91///
92/// The generated definitions receive `doc(cfg(condition))`, so rustdoc records the condition under
93/// which the real implementation is available.
94///
95/// # Examples
96///
97/// Keeps an x86-64 inline-assembly implementation available in rustdoc:
98///
99/// ```
100/// #[cfg(any(target_arch = "x86_64", doc))]
101/// pub mod cpu {
102///     use cfg_or_dummy::cfg_or_dummy;
103///
104///     #[cfg_or_dummy(target_arch = "x86_64")]
105///     pub fn pause() {
106///         // SAFETY: `pause` has no memory or stack effects exposed to Rust.
107///         unsafe {
108///             core::arch::asm!("pause", options(nomem, nostack, preserves_flags));
109///         }
110///     }
111/// }
112///
113/// #[cfg(target_arch = "x86_64")]
114/// cpu::pause();
115/// ```
116#[proc_macro_attribute]
117pub fn cfg_or_dummy(attr: TokenStream1, item: TokenStream1) -> TokenStream1 {
118    let attr = parse_macro_input!(attr as Meta);
119    let func = parse_macro_input!(item as ItemFn);
120
121    cfg_or_dummy_impl(attr, func).into()
122}
