47pub trait PerCPUAreaIf {
48    /// Returns the per-CPU area base virtual address for `cpu_id`.
49    fn percpu_area_base_for(cpu_id: usize) -> VirtAddr;

52const fn align_up_64(val: usize) -> usize {
53    (val + SIZE_64BIT - 1) & !(SIZE_64BIT - 1)
54}
55
56/// Returns the per-CPU data area size for one CPU.
57pub fn percpu_area_size() -> usize {
58    expercpu_macros::percpu_symbol_vma!(_percpu_end)
59        - expercpu_macros::percpu_symbol_vma!(_percpu_start)
60}
61
62/// Returns the allocation layout for one per-CPU data area.
63pub fn percpu_area_layout() -> Layout {
64    Layout::from_size_align(align_up_64(percpu_area_size()), PERCPU_AREA_ALIGN).unwrap()
65}
66
67fn validate_percpu_base(base: VirtAddr) {
68    assert!(
69        base.as_usize() != 0,
70        "per-CPU area base address must be non-null"
71    );
72    assert_eq!(
73        base.as_usize() % PERCPU_AREA_ALIGN,
74        0,
75        "per-CPU area base address must be 64-byte aligned"
76    );
77}

79/// Initializes the current CPU's per-CPU data area.
80///
81/// The caller supplies the base virtual address of the current CPU's per-CPU
82/// area. The function copies the `.percpu` initial image to that area and
83/// writes the architecture-specific per-CPU register for the current CPU.
84///
85/// # Safety
86///
87/// The caller must ensure that:
88///
89/// - `base` is the current CPU's writable per-CPU area base;
90/// - the area is at least `percpu_area_layout().size()` bytes and 64-byte
91///   aligned;
92/// - the destination does not overlap the `.percpu` initial image source; and
93/// - setting the current CPU's per-CPU register to this base is valid for the
94///   current execution context.
95pub unsafe fn init(base: VirtAddr) {
96    let percpu_start = VirtAddr::from_usize(expercpu_macros::percpu_symbol_vma!(_percpu_start));
97    unsafe {
98        init_from(base, percpu_start);
99    }
100}

102/// Initializes the current CPU's per-CPU data area using the early slot.
103///
104/// Early slot is a reserved area in the binary for initializing the per-CPU data area without
105/// relying on dynamic memory allocation. Early slot is bounded by the symbols
106/// `_percpu_early_slot_start` and `_percpu_early_slot_end`.
107///
108/// # Panics
109///
110/// Panics at runtime if the linker-provided early slot is smaller than the per-CPU initial image.
111///
112/// # Safety
113///
114/// The caller must ensure that the early slot is included in the binary, and is not used by other
115/// cores.
116#[cfg(feature = "early-slot")]
117pub unsafe fn init_in_early_slot() {
118    let early_slot_start = expercpu_macros::percpu_symbol_vma!(_percpu_early_slot_start);
119    let early_slot_end = expercpu_macros::percpu_symbol_vma!(_percpu_early_slot_end);
120
121    assert!(
122        early_slot_end - early_slot_start >= percpu_area_size(),
123        "early slot is too small"
124    );
125
126    unsafe {
127        init(VirtAddr::from_usize(early_slot_start));
128    }
129}
130
131/// Initializes the current CPU's per-CPU data area, copying from the early slot.
132///
133/// # Safety
134///
135/// See the safety requirements of [`init`] for more details.
136#[cfg(feature = "early-slot")]
137pub unsafe fn init_from_early_slot(base: VirtAddr) {
138    unsafe { init_from(base, early_slot_start()) };
139}

141/// Returns the start address of the early slot.
142#[cfg(feature = "early-slot")]
143#[inline]
144pub fn early_slot_start() -> VirtAddr {
145    VirtAddr::from_usize(expercpu_macros::percpu_symbol_vma!(
146        _percpu_early_slot_start
147    ))
148}

150unsafe fn init_from(base: VirtAddr, from: VirtAddr) {
151    validate_percpu_base(base);
152    let size = percpu_area_size();
153
154    unsafe {
155        core::ptr::copy(from.as_mut_ptr(), base.as_mut_ptr(), size);
156        write_percpu_reg(base);
157    }
158}

160/// Reads the architecture-specific per-CPU data register.
161#[cfg(not(feature = "host-test"))]
162pub fn read_percpu_reg() -> VirtAddr {
163    let tp: usize;
164    unsafe {
165        cfg_if::cfg_if! {
166            if #[cfg(target_arch = "x86_64")] {
167                core::arch::asm!("rdgsbase {}", out(reg) tp);
168            } else if #[cfg(any(target_arch = "riscv32", target_arch = "riscv64"))] {
169                // RISC-V `tp` may be a better long-term per-CPU base, but keep `gp`
170                // for compatibility with the current design.
171                core::arch::asm!("mv {}, gp", out(reg) tp)
172            } else if #[cfg(all(target_arch = "aarch64", not(feature = "arm-el2")))] {
173                core::arch::asm!("mrs {}, TPIDR_EL1", out(reg) tp)
174            } else if #[cfg(all(target_arch = "aarch64", feature = "arm-el2"))] {
175                core::arch::asm!("mrs {}, TPIDR_EL2", out(reg) tp)
176            } else if #[cfg(target_arch = "loongarch64")] {
177                core::arch::asm!("move {}, $r21", out(reg) tp)
178            } else if #[cfg(target_arch = "arm")] {
179                core::arch::asm!("mrc p15, 0, {}, c13, c0, 4", out(reg) tp)
180            } else {
181                compile_error!("unsupported architecture for expercpu");
182            }
183        }
184    }
185    VirtAddr::from_usize(tp + expercpu_macros::percpu_symbol_vma!(_percpu_start))
186}

193/// Writes the architecture-specific per-CPU data register.
194///
195/// # Safety
196///
197/// `tp` must be the valid per-CPU area base for the current CPU, and setting
198/// the architecture-specific per-CPU register to that base must be valid for
199/// the current execution context.
200#[cfg(not(feature = "host-test"))]
201pub unsafe fn write_percpu_reg(tp: VirtAddr) {
202    let tp = tp.as_usize() - expercpu_macros::percpu_symbol_vma!(_percpu_start);
203
204    unsafe {
205        cfg_if::cfg_if! {
206            if #[cfg(target_arch = "x86_64")] {
207                core::arch::asm!("wrgsbase {}", in(reg) tp);
208            } else if #[cfg(any(target_arch = "riscv32", target_arch = "riscv64"))] {
209                core::arch::asm!("mv gp, {}", in(reg) tp)
210            } else if #[cfg(all(target_arch = "aarch64", not(feature = "arm-el2")))] {
211                core::arch::asm!("msr TPIDR_EL1, {}", in(reg) tp)
212            } else if #[cfg(all(target_arch = "aarch64", feature = "arm-el2"))] {
213                core::arch::asm!("msr TPIDR_EL2, {}", in(reg) tp)
214            } else if #[cfg(target_arch = "loongarch64")] {
215                core::arch::asm!("move $r21, {}", in(reg) tp)
216            } else if #[cfg(target_arch = "arm")] {
217                core::arch::asm!("mcr p15, 0, {}, c13, c0, 4", in(reg) tp)
218            } else {
219                compile_error!("unsupported architecture for expercpu");
220            }
221        }
222    }
223}
