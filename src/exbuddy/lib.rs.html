12/// Maximum buddy order.
13///
14/// It determines the maximum size of buddy blocks - with 4 KiB pages this gives
15/// 2^20 x 4 KiB = 4 GiB blocks. No single allocation can exceed this size.
16/// Note that the buddy allocator can handle sections larger than this.
17pub const MAX_ORDER: usize = 20;

47/// Converts a page count to the smallest order that can contain it.
48///
49/// Returns `Err(BuddyError::InvalidPageCount)` if the page count is too
50/// large that even the largest order cannot contain it.
51///
52/// # Examples
53///
54/// ```
55/// use exbuddy::{page_count_to_order_ceiling, MAX_ORDER, BuddyError};
56///
57/// assert_eq!(page_count_to_order_ceiling(0), Ok(0));
58/// assert_eq!(page_count_to_order_ceiling(1), Ok(0));
59/// assert_eq!(page_count_to_order_ceiling(2), Ok(1));
60/// assert_eq!(page_count_to_order_ceiling(3), Ok(2));
61/// assert_eq!(page_count_to_order_ceiling(4), Ok(2));
62///
63/// assert_eq!(page_count_to_order_ceiling(1usize << MAX_ORDER), Ok(MAX_ORDER));
64/// assert_eq!(page_count_to_order_ceiling(1usize << MAX_ORDER + 1), Err(BuddyError::InvalidPageCount));
65/// ```
66#[inline]
67pub const fn page_count_to_order_ceiling(page_count: usize) -> BuddyResult<usize> {
68    if let Some(p) = page_count.checked_next_power_of_two() {
69        let order = p.trailing_zeros() as usize;
70        if order <= MAX_ORDER {
71            return Ok(order);
72        }
73    }
74
75    Err(BuddyError::InvalidPageCount)
76}

78/// Converts a page count to the largest order that can fit in it.
79///
80/// Returns `Err(BuddyError::InvalidPageCount)` if the page count is too
81/// small that even the smallest order cannot fit in it (i.e. 0).
82///
83/// # Examples
84///
85/// ```
86/// use exbuddy::{page_count_to_order_floor, MAX_ORDER, BuddyError, BuddyResult};
87///
88/// assert_eq!(page_count_to_order_floor(0), Err(BuddyError::InvalidPageCount));
89/// assert_eq!(page_count_to_order_floor(1), Ok(0));
90/// assert_eq!(page_count_to_order_floor(2), Ok(1));
91/// assert_eq!(page_count_to_order_floor(3), Ok(1));
92/// assert_eq!(page_count_to_order_floor(4), Ok(2));
93///
94/// assert_eq!(page_count_to_order_floor(1usize << MAX_ORDER), Ok(MAX_ORDER));
95/// assert_eq!(page_count_to_order_floor(1usize << MAX_ORDER + 1), Ok(MAX_ORDER));
96/// ```
97#[inline]
98pub const fn page_count_to_order_floor(page_count: usize) -> BuddyResult<usize> {
99    if let Some(p) = page_count.highest_one() {
100        let p = p as _;
101
102        return Ok(if p > MAX_ORDER { MAX_ORDER } else { p });
103    }
104
105    Err(BuddyError::InvalidPageCount)
106}
