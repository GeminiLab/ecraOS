1//! Periodic kernel timer events.
2//!
3//! The kernel timer uses per-CPU absolute one-shot deadlines to provide periodic timer events.
4
5use core::time::Duration;
6
7use expercpu::def_percpu;
8
9mod deadline;
10
11use deadline::select_next_deadline;
12
13/// The number of periodic timer events requested each second.
14///
15/// This is the selected kernel timer-event cadence in hertz.
16pub const TIMER_EVENT_HZ: u64 = 100;
17
18/// The interval between periodic timer-event deadlines in nanoseconds.
19///
20/// This is exact because one second is divisible by [`TIMER_EVENT_HZ`].
21const PERIODIC_INTERVAL_NANOS: u64 = 1_000_000_000 / TIMER_EVENT_HZ;
22
23/// The next deadline to program on the current hart.
24///
25/// Zero means this hart has not yet seeded its periodic schedule.
26#[def_percpu]
27static NEXT_DEADLINE_NANOS: u64 = 0;
28
29/// Initializes the periodic timer on the bootstrap hart.
30///
31/// This registers the shared handler before arming and enabling local interrupts.
32pub fn init_bsp() {
33    program_next_timer();
34    exarch::trap::enable_local();
35}
36
37/// Initializes the periodic timer on an application hart.
38///
39/// The bootstrap hart has already installed the shared timer handler.
40pub fn init_ap() {
41    program_next_timer();
42    exarch::trap::enable_local();
43}
44
45/// Handles a supervisor timer interrupt on the current hart.
46///
47/// The handler advances the periodic deadline and programs the next SBI event.
48pub(crate) fn timer_tick() {
49    crate::task::wake_sleepers(exarch::time::monotonic_time());
50    program_next_timer();
51}
52
53/// Programs the next periodic deadline for the current hart.
54///
55/// Expired schedules restart one interval after the current monotonic time.
56fn program_next_timer() {
57    let now = exarch::time::monotonic_time();
58    let now_ns = u64::try_from(now.as_nanos()).expect("monotonic time exceeds u64 nanoseconds");
59    let scheduled_ns = NEXT_DEADLINE_NANOS.read_current();
60    let (deadline_ns, following_ns) =
61        select_next_deadline(now_ns, scheduled_ns, PERIODIC_INTERVAL_NANOS);
62    NEXT_DEADLINE_NANOS.write_current(following_ns);
63    exarch::time::set_oneshot_timer(Duration::from_nanos(deadline_ns));
64}
