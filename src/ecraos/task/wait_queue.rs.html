1//! Cooperative completion wait queues.

3//! The pure model defines the publication and block-commit protocol shared by host tests and the
4//! kernel completion queue. Architecture-specific switching is layered on top by the scheduler.

8/// Errors returned before a join waiter is published.
9///
10/// Self-join is rejected because the current task cannot complete while waiting on itself.
11#[derive(Clone, Copy, Debug, Eq, PartialEq)]
12pub enum JoinValidationError {
13    /// The target task is the caller itself.
14    SelfJoin,
15}
16
17/// Validates that a task may join a target task.
18///
19/// A caller without a scheduler identity may join any target.
20pub fn validate_join(
21    current_task_id: Option<u64>,
22    target_task_id: u64,
23) -> Result<(), JoinValidationError> {
24    if current_task_id == Some(target_task_id) {
25        Err(JoinValidationError::SelfJoin)
26    } else {
27        Ok(())
28    }
29}

36/// Errors returned by the kernel completion waiter queue.
37///
38/// Duplicate task membership would permit one task to be woken more than once for one generation.
39#[derive(Clone, Copy, Debug, Eq, PartialEq)]
40pub(super) enum TaskWaitQueueError {
41    /// The task already occupies a completion waiter position.
42    Duplicate,
43}
44
45/// The state of one kernel completion waiter.
46///
47/// Published entries close the wake-before-block-commit race.
48#[derive(Debug)]
49enum TaskWaiterState {
50    /// Published but not yet committed as blocked.
51    Published,
52    /// Committed as blocked and awaiting completion wakeup.
53    Blocked,
54}
55
56/// One kernel completion waiter and its task ownership token.
57///
58/// The queue owns the token until wakeup or cancellation removes the entry.
59struct TaskWaiter {
60    /// The task waiting for completion.
61    task: Arc<Task>,
62    /// The publication state used by the block protocol.
63    state: TaskWaiterState,
64}
65
66/// The internal completion wait queue.
67///
68/// This queue is intentionally limited to task completion and `join`. Its lock is acquired before
69/// the scheduler run-queue lock and is released before any architecture context switch.
70pub(super) struct TaskWaitQueue {
71    /// Waiter entries protected from CPUs and local interrupt handlers.
72    inner: SpinNoIrq<VecDeque<TaskWaiter>>,
73}
74
75impl TaskWaitQueue {
76    /// Creates an empty completion wait queue.
77    ///
78    /// No task initially waits for the completion event.
79    pub(super) const fn new() -> Self {
80        Self {
81            inner: SpinNoIrq::new(VecDeque::new()),
82        }
83    }
84
85    /// Locks the completion wait queue.
86    ///
87    /// The guard disables local interrupts and must be dropped before switching contexts.
88    pub(super) fn lock(&self) -> TaskWaitQueueGuard<'_> {
89        // Completion locks follow NoPreemptIrqSave -> domain metadata -> Queue -> task lifecycle
90        // -> completion locks. A future lock-free wait queue could replace this lock after its
91        // memory-ordering proof.
92        TaskWaitQueueGuard {
93            _preempt: kernel_guard::NoPreemptIrqSave::new(),
94            guard: self.inner.lock(),
95        }
96    }
97}
98
99/// A locked completion wait queue.
100///
101/// The guard owns the lock across waiter publication and block commit to close the wakeup race.
102pub(super) struct TaskWaitQueueGuard<'a> {
103    /// Scheduler guard held for the lifetime of the queue lock.
104    _preempt: kernel_guard::NoPreemptIrqSave,
105    /// The IRQ-safe queue lock guard.
106    guard: SpinNoIrqGuard<'a, VecDeque<TaskWaiter>>,
107}
108
109impl TaskWaitQueueGuard<'_> {
110    /// Publishes one task as a completion waiter.
111    ///
112    /// The caller must hold this guard while checking completion and committing the block.
113    pub(super) fn register(&mut self, task: Arc<Task>) -> Result<(), TaskWaitQueueError> {
114        if self
115            .guard
116            .iter()
117            .any(|waiter| waiter.task.id() == task.id())
118        {
119            return Err(TaskWaitQueueError::Duplicate);
120        }
121        self.guard.push_back(TaskWaiter {
122            task,
123            state: TaskWaiterState::Published,
124        });
125        Ok(())
126    }
127
128    /// Commits one published task as blocked unless completion already woke it.
129    ///
130    /// `false` means the waiter was consumed by a wake-before-commit race and the caller must not
131    /// block.
132    pub(super) fn commit_block(&mut self, task_id: TaskId) -> bool {
133        let Some(waiter) = self
134            .guard
135            .iter_mut()
136            .find(|waiter| waiter.task.id() == task_id)
137        else {
138            return false;
139        };
140        waiter.state = TaskWaiterState::Blocked;
141        true
142    }
143
144    /// Removes and returns all published and blocked completion waiters.
145    ///
146    /// The caller releases this guard before enqueuing the returned tasks.
147    pub(super) fn wake_all(&mut self) -> Vec<Arc<Task>> {
148        self.guard.drain(..).map(|waiter| waiter.task).collect()
149    }
150}
