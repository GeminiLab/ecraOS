32/// Kernel task lifecycle states.
33///
34/// Runnable represents ownership by the future run queue, while Running records the sole CPU that
35/// currently owns execution of the task.
36#[derive(Clone, Copy, Debug, Eq, PartialEq)]
37pub enum TaskState {
38    /// A task that has not yet been published as runnable.
39    Created,
40    /// A task that is eligible to run and may be enqueued once.
41    Runnable,
42    /// A task executing on exactly one CPU.
43    Running {
44        /// The logical identifier of the CPU executing the task.
45        cpu_id: usize,
46    },
47    /// A task waiting for its current wait generation to be woken.
48    Blocked,
49    /// A task sleeping until its local timer deadline.
50    Sleeping,
51    /// A task registered on one wait queue and one timer deadline.
52    TimedWaiting,
53    /// A task that has completed and can never run again.
54    Exited,
55}
56
57/// The published outcome of a task that reached `Exited`.
58#[derive(Clone, Copy, Debug, Eq, PartialEq)]
59pub enum TaskExitStatus {
60    /// The task body returned normally.
61    Completed,
62}
63
64/// The outcome of one timed wait.
65#[derive(Clone, Copy, Debug, Eq, PartialEq)]
66pub enum WaitResult {
67    /// The wait queue notified the task before its deadline.
68    Woken,
69    /// The wait deadline expired before notification.
70    TimedOut,
71}

73/// Errors returned by rejected task lifecycle transitions.
74///
75/// Each error names the invariant that prevented the requested transition.
76#[derive(Clone, Copy, Debug, Eq, PartialEq)]
77pub enum TaskStateError {
78    /// The task is already runnable and publishing it again would double enqueue it.
79    AlreadyRunnable,
80    /// The task is already executing and cannot be started on another CPU.
81    AlreadyRunning,
82    /// The requested CPU does not own the running task.
83    WrongCpu,
84    /// The task has exited and cannot transition again.
85    Exited,
86    /// The requested transition is invalid from the current lifecycle state.
87    InvalidTransition,
88}

90/// A scheduler placement policy for one task.
91///
92/// Tasks with any-CPU placement may be published on the current CPU's queue, while pinned tasks
93/// must remain on their designated logical CPU.
94#[derive(Clone, Copy, Debug, Eq, PartialEq)]
95pub(super) enum TaskPlacement {
96    /// Allows publication on the scheduler-selected CPU.
97    AnyCpu,
98    /// Restricts publication and execution to one logical CPU.
99    Pinned(usize),
100}

102impl TaskState {
103    /// Publishes a created or blocked task as runnable.
104    ///
105    /// A runnable task rejects another publication so future queue ownership remains unique.
106    pub fn make_runnable(&mut self) -> Result<(), TaskStateError> {
107        match self {
108            Self::Created | Self::Blocked => {
109                *self = Self::Runnable;
110                Ok(())
111            }
112            Self::Runnable => Err(TaskStateError::AlreadyRunnable),
113            Self::Exited => Err(TaskStateError::Exited),
114            Self::Running { .. } | Self::Sleeping | Self::TimedWaiting => {
115                Err(TaskStateError::InvalidTransition)
116            }
117        }
118    }
119
120    /// Starts a runnable task on one CPU.
121    ///
122    /// A running task rejects all additional starts so it cannot execute on two CPUs.
123    pub fn start_running(&mut self, cpu_id: usize) -> Result<(), TaskStateError> {
124        match self {
125            Self::Runnable => {
126                *self = Self::Running { cpu_id };
127                Ok(())
128            }
129            Self::Running { .. } => Err(TaskStateError::AlreadyRunning),
130            Self::Exited => Err(TaskStateError::Exited),
131            Self::Created | Self::Blocked | Self::Sleeping | Self::TimedWaiting => {
132                Err(TaskStateError::InvalidTransition)
133            }
134        }
135    }
136
137    /// Blocks a task on its owning CPU.
138    ///
139    /// Only the CPU recorded by the running state may relinquish execution ownership.
140    pub fn block(&mut self, cpu_id: usize) -> Result<(), TaskStateError> {
141        match self {
142            Self::Running { cpu_id: owner } if *owner == cpu_id => {
143                *self = Self::Blocked;
144                Ok(())
145            }
146            Self::Running { .. } => Err(TaskStateError::WrongCpu),
147            Self::Exited => Err(TaskStateError::Exited),
148            Self::Created
149            | Self::Runnable
150            | Self::Blocked
151            | Self::Sleeping
152            | Self::TimedWaiting => Err(TaskStateError::InvalidTransition),
153        }
154    }
155
156    /// Exits a task on its owning CPU.
157    ///
158    /// Exited is terminal and only the CPU currently executing the task may enter it.
159    pub fn exit(&mut self, cpu_id: usize) -> Result<(), TaskStateError> {
160        match self {
161            Self::Running { cpu_id: owner } if *owner == cpu_id => {
162                *self = Self::Exited;
163                Ok(())
164            }
165            Self::Running { .. } => Err(TaskStateError::WrongCpu),
166            Self::Exited => Err(TaskStateError::Exited),
167            Self::Created
168            | Self::Runnable
169            | Self::Blocked
170            | Self::Sleeping
171            | Self::TimedWaiting => Err(TaskStateError::InvalidTransition),
172        }
173    }

176/// Converts one owned [`Arc`] strong reference into a context argument.
177///
178/// The returned raw argument exclusively represents the consumed strong reference until
179/// [`execution_arg_into_arc`] reconstructs it exactly once.
180pub(crate) fn execution_arg_from_arc<T>(owner: Arc<T>) -> usize {
181    Arc::into_raw(owner) as usize
182}

184/// Reconstructs the owned [`Arc`] strong reference represented by a context argument.
185///
186/// This consumes the raw token without changing its strong count.
187///
188/// # Safety
189///
190/// `argument` must have been returned by one unmatched [`execution_arg_from_arc`] call for `T`.
191/// It must be reconstructed exactly once.
192pub(crate) unsafe fn execution_arg_into_arc<T>(argument: usize) -> Arc<T> {
193    // SAFETY: The caller guarantees that this pointer is one unmatched Arc::into_raw token for T.
194    unsafe { Arc::from_raw(argument as *const T) }
195}

197/// Narrow task completion publication state.
198pub(super) struct Completion {
199    /// Whether the task closure has returned.
200    completed: AtomicBool,
201    /// Stable encoded exit outcome.
202    status: AtomicU8,
203    /// The internal wait queue used by joiners.
204    waiters: TaskWaitQueue,
205}
206
207impl Completion {
208    /// Creates unpublished completion state.
209    ///
210    /// Neither completion nor waiter notification is visible initially.
211    const fn new() -> Self {
212        Self {
213            completed: AtomicBool::new(false),
214            status: AtomicU8::new(0),
215            waiters: TaskWaitQueue::new(),
216        }
217    }
218
219    /// Publishes task completion.
220    ///
221    /// Release ordering makes all closure effects visible before completion is observed.
222    fn publish_completed(&self) {
223        self.status.store(1, Ordering::Relaxed);
224        self.completed.store(true, Ordering::Release);
225    }
226
227    /// Returns whether the task completion has been published.
228    ///
229    /// The acquire load observes closure effects before a joiner returns.
230    pub(super) fn is_completed(&self) -> bool {
231        self.completed.load(Ordering::Acquire)
232    }
233
234    /// Returns the published exit outcome, if the task has completed.
235    pub(super) fn status(&self) -> Option<TaskExitStatus> {
236        if self.is_completed() {
237            Some(TaskExitStatus::Completed)
238        } else {
239            None
240        }
241    }
242
243    /// Locks the completion waiter queue.
244    ///
245    /// The caller must release this guard before any scheduler context switch.
246    pub(super) fn lock_waiters(&self) -> TaskWaitQueueGuard<'_> {
247        self.waiters.lock()
248    }
249
250    /// Wakes all completion waiters and returns their task tokens.
251    ///
252    /// The caller enqueues returned tasks after releasing the completion queue lock.
253    pub(super) fn take_waiters(&self) -> alloc::vec::Vec<Arc<Task>> {
254        self.waiters.lock().wake_all()
255    }
256}

258/// An owned kernel task.
259///
260/// The task owns its stable ID, protected lifecycle, architecture context, guarded stack, closure,
261/// completion publication and current-stack lifetime marker.
262pub struct Task {
263    /// The stable identifier assigned at creation.
264    id: TaskId,
265    /// The lifecycle and exclusive running-CPU ownership.
266    state: SpinNoIrq<TaskState>,
267    /// The architecture context prepared before publication and written by switch code only at
268    /// exclusive incoming or outgoing context boundaries.
269    context: UnsafeCell<exarch::context::TaskContext>,
270    /// The owned guarded kernel stack.
271    stack: KernelStack,
272    /// The optional one-shot task body for a fresh task's initial trampoline.
273    ///
274    /// Adopted root tasks already execute their continuation and therefore leave this field empty.
275    body: UnsafeCell<Option<Box<dyn FnOnce() + Send>>>,
276    /// The completion state that outlives execution until final task reclamation.
277    completion: Arc<Completion>,
278    /// Whether execution is still using this task's stack.
279    current: AtomicBool,
280    /// A wakeup awaiting current-stack handoff before runnable publication.
281    ///
282    /// The lifecycle lock serializes this flag with the current marker and wake routing.
283    handoff_wakeup: AtomicBool,
284    /// Whether this task is registered on a synchronization wait queue.
285    waiting: AtomicBool,
286    /// Raw pointer to the one synchronization wait queue owning this registration.
287    wait_queue: AtomicU64,
288    /// Result published by the winner of a timed wait wake/timeout race.
289    wait_result: AtomicU8,
290    /// Whether the context is ready for publication or saving at a switch boundary.
291    context_initialized: AtomicBool,
292    /// The scheduler placement policy for this task.
293    placement: TaskPlacement,
294    /// The monotonically increasing generation of the current sleep request.
295    sleep_generation: AtomicU64,
296    /// The CPU owning the current timer entry, or `usize::MAX` when none is registered.
297    timer_cpu: AtomicUsize,
298    /// Remaining timer ticks for the current preemptive slice.
299    remaining_slice: AtomicU64,
300    /// Weak ownership link to the scheduling domain.
301    domain: SpinNoIrq<Weak<SchedulingDomain>>,
302    /// Stable domain identity retained after the domain is destroyed.
303    domain_id: AtomicU64,
304}

306// SAFETY: Task construction requires a Send closure. All shared mutable state is atomic or
307// protected by SpinNoIrq, except the context and body UnsafeCells. The context is accessed uniquely
308// at switch boundaries while the task is not running. The body is consumed exactly once by its own
309// trampoline.
310unsafe impl Send for Task {}
311
312// SAFETY: The same invariants as Send permit shared references. Lifecycle ownership prevents two
313// CPUs from running the task, and current-stack reclamation prevents access after stack release.
314unsafe impl Sync for Task {}

316/// Routes one committed wakeup around the current-stack handoff window.
317///
318/// The caller holds the lifecycle lock. A current Runnable task retains its publication token
319/// until the incoming stack clears the marker, while an already resumed task needs no publication.
320fn route_handoff_wakeup(state: TaskState, current: bool) -> (bool, bool) {
321    if state != TaskState::Runnable {
322        return (false, false);
323    }
324    if current {
325        (false, true)
326    } else {
327        (true, false)
328    }
329}
330
331/// Claims the deferred wake token on an incoming stack.
332///
333/// The lifecycle lock excludes routing and an in-place resume while the current marker is cleared.
334fn finish_handoff_wakeup(state: TaskState, pending: bool) -> bool {
335    pending && state == TaskState::Runnable
336}

342    fn route_locked_wakeup(&self, state: TaskState) -> bool {
343        let (publish, pending) = route_handoff_wakeup(state, self.current.load(Ordering::Acquire));
344        self.handoff_wakeup.store(pending, Ordering::Relaxed);

443    /// Associates a newly created task with one scheduling domain.
444    pub(super) fn set_domain(&self, domain: &DomainRef) {
445        let _guard = kernel_guard::NoPreempt::new();
446        *self.domain.lock() = Arc::downgrade(domain);
447        self.domain_id.store(domain.id(), Ordering::Release);
448        let ticks = domain
449            .policy()
450            .slice_ticks(core::time::Duration::from_millis(10))
451            .unwrap_or(0);
452        self.remaining_slice.store(ticks, Ordering::Release);
453    }

455    /// Accounts one timer tick and reports whether the slice expired.
456    pub(super) fn account_tick(&self) -> bool {
457        let mut current = self.remaining_slice.load(Ordering::Acquire);
458        while current != 0 {
459            match self.remaining_slice.compare_exchange_weak(
460                current,
461                current - 1,
462                Ordering::AcqRel,
463                Ordering::Acquire,
464            ) {
465                Ok(_) => return current == 1,
466                Err(observed) => current = observed,
467            }
468        }
469        false
470    }

503    /// Prepares the task's initial architecture context exactly once.
504    ///
505    /// Callers invoke this while the task is still private and Created, before publishing it as
506    /// runnable or making scheduler state active. The context owns one strong reference that the
507    /// initial trampoline reconstructs exactly once.
508    pub(crate) fn prepare_context(self: &Arc<Self>) {
509        let _guard = kernel_guard::NoPreempt::new();
510        let state = self.state.lock();
511        assert_eq!(
512            *state,
513            TaskState::Created,
514            "task context must be prepared before publication"
515        );
516        assert!(
517            !self.context_initialized.load(Ordering::Acquire),
518            "task context initialized twice"
519        );
520        let argument = execution_arg_from_arc(Arc::clone(self));
521        // SAFETY: The lifecycle lock and Created state keep the task private during initialization.
522        // The argument owns one strong reference and the initialized context passes it to
523        // task_trampoline exactly once. All fallible allocation completed in new_detached before
524        // the token was created.
525        unsafe {
526            exarch::context::init_task_context(
527                &mut *self.context.get(),
528                self.stack.top(),
529                task_trampoline,
530                argument,
531            )
532        };
533        self.context_initialized.store(true, Ordering::Release);
534    }

536    /// Marks the task running on one CPU.
537    ///
538    /// The prepared-context assertion, state transition, and current-stack marker are published
539    /// before switching to the task.
540    pub fn start_running(self: &Arc<Self>, cpu_id: usize) -> Result<(), TaskStateError> {
541        assert!(
542            self.context_initialized.load(Ordering::Acquire),
543            "task context was not prepared before starting"
544        );
545        let mut state = self.state.lock();
546        state.start_running(cpu_id)?;
547        self.handoff_wakeup.store(false, Ordering::Relaxed);
548        self.current.store(true, Ordering::Release);

558            TaskState::Running { cpu_id: owner } if owner == cpu_id => {

565            | TaskState::Runnable
566            | TaskState::Blocked
567            | TaskState::Sleeping
568            | TaskState::TimedWaiting => Err(TaskStateError::InvalidTransition),

602    /// Publishes completion and transitions the task to Exited on its owning CPU.
603    ///
604    /// The returned waiters must be routed to runnable queues after this task's completion lock is
605    /// no longer held.
606    pub(super) fn complete(&self, cpu_id: usize) -> alloc::vec::Vec<Arc<Task>> {
607        self.state
608            .lock()
609            .exit(cpu_id)
610            .expect("task completed outside its owning CPU");
611        if let Some(domain) = self.domain() {
612            domain.retire_task();
613        }
614        self.completion.publish_completed();
615        self.completion.take_waiters()
616    }

618    /// Marks a blocked task runnable after one completion wakeup.
619    ///
620    /// Exactly one waiter generation may wake this task. A true result grants queue publication
621    /// to the caller, while false transfers it to the incoming stack's handoff hook.
622    pub(crate) fn wake_from_completion(&self) -> Result<bool, TaskStateError> {
623        let _guard = kernel_guard::NoPreempt::new();
624        let mut state = self.state.lock();
625        state.make_runnable()?;
626        Ok(self.route_locked_wakeup(*state))
627    }

629    /// Wakes one synchronization waiter and publishes the notification result.
630    ///
631    /// Transition and queue-publication routing share one lifecycle lock acquisition. A true
632    /// result grants queue publication and false retains it until current-stack handoff.
633    pub(crate) fn wake_from_wait(&self) -> Result<bool, TaskStateError> {
634        let _guard = kernel_guard::NoPreempt::new();
635        let mut state = self.state.lock();
636        match *state {
637            TaskState::Blocked => state.make_runnable(),
638            TaskState::TimedWaiting => {
639                *state = TaskState::Runnable;
640                self.wait_result.store(1, Ordering::Release);
641                self.timer_cpu.store(usize::MAX, Ordering::Release);
642                Ok(())
643            }
644            _ => Err(TaskStateError::InvalidTransition),
645        }?;
646        Ok(self.route_locked_wakeup(*state))
647    }

649    /// Times out one synchronization waiter if its generation is still current.
650    ///
651    /// A true result grants queue publication. A current task retains its wake token until handoff.
652    pub(crate) fn timeout_wait(&self, generation: u64) -> bool {
653        if self.sleep_generation.load(Ordering::Acquire) != generation {
654            return false;
655        }
656        let _guard = kernel_guard::NoPreempt::new();
657        let mut state = self.state.lock();
658        if *state != TaskState::TimedWaiting {
659            return false;
660        }
661        *state = TaskState::Runnable;
662        self.wait_result.store(2, Ordering::Release);
663        self.timer_cpu.store(usize::MAX, Ordering::Release);
664        self.route_locked_wakeup(*state)
665    }

667    /// Rolls back a timed wait that could not acquire timer ownership.
668    pub(crate) fn cancel_timed_wait(&self, generation: u64, cpu_id: usize) -> bool {
669        if self.sleep_generation.load(Ordering::Acquire) != generation {
670            return false;
671        }
672        let _guard = kernel_guard::NoPreemptIrqSave::new();
673        let mut state = self.state.lock();
674        if *state != TaskState::TimedWaiting {
675            return false;
676        }
677        *state = TaskState::Running { cpu_id };
678        self.timer_cpu.store(usize::MAX, Ordering::Release);
679        self.wait_result.store(0, Ordering::Release);
680        true
681    }
682
683    /// Blocks this task on its owning CPU.
684    ///
685    /// The scheduler calls this only while the task is Running and the completion queue is locked.
686    pub(crate) fn block(&self, cpu_id: usize) -> Result<(), TaskStateError> {
687        let _guard = kernel_guard::NoPreempt::new();
688        self.state.lock().block(cpu_id)
689    }
690
691    /// Moves a running task into timer-owned sleeping state.
692    pub(crate) fn sleep(&self, cpu_id: usize) -> Result<(), TaskStateError> {
693        let _guard = kernel_guard::NoPreempt::new();
694        let mut state = self.state.lock();
695        match *state {
696            TaskState::Running { cpu_id: owner } if owner == cpu_id => {
697                *state = TaskState::Sleeping;
698                self.timer_cpu.store(cpu_id, Ordering::Release);
699                Ok(())
700            }
701            TaskState::Running { .. } => Err(TaskStateError::WrongCpu),
702            TaskState::Exited => Err(TaskStateError::Exited),
703            _ => Err(TaskStateError::InvalidTransition),
704        }
705    }
706
707    /// Starts one uniquely identified sleep request.
708    ///
709    /// A new generation invalidates every timer entry left by an earlier request.
710    pub(crate) fn begin_sleep(&self) -> u64 {
711        self.sleep_generation
712            .fetch_add(1, Ordering::AcqRel)
713            .checked_add(1)
714            .expect("task sleep generation space exhausted")
715    }
716
717    /// Returns the generation of the current timer operation.
718    pub(crate) fn sleep_generation(&self) -> u64 {
719        self.sleep_generation.load(Ordering::Acquire)
720    }
721
722    /// Wakes a task if the timer entry still names its current sleep request.
723    ///
724    /// Stale entries are discarded without changing a newer sleep or another blocked state.
725    pub(crate) fn wake_from_sleep(&self, generation: u64) -> bool {
726        if self.sleep_generation.load(Ordering::Acquire) != generation {
727            return false;
728        }
729        let _guard = kernel_guard::NoPreempt::new();
730        let result = {
731            let mut state = self.state.lock();
732            match *state {
733                TaskState::Sleeping => {
734                    *state = TaskState::Runnable;
735                    self.timer_cpu.store(usize::MAX, Ordering::Release);
736                    Ok(())
737                }
738                _ => state.make_runnable(),
739            }
740        };
741        result.is_ok()
742    }

744    /// Returns the architecture context pointer used by switch assembly.
745    ///
746    /// The caller must uphold the architecture context's unique-access and running-state rules.
747    pub(crate) const fn context_ptr(&self) -> *mut exarch::context::TaskContext {
748        self.context.get()
749    }

751    /// Returns whether any CPU may still be executing on this task's stack.
752    ///
753    /// Scheduler selection uses this during the unlocked context-switch handoff window.
754    pub(crate) fn is_current(&self) -> bool {
755        self.current.load(Ordering::Acquire)
756    }
757
758    /// Marks the stack non-current after execution has moved elsewhere.
759    ///
760    /// Deferred reaping calls this only from another active stack. A true result transfers a wake
761    /// publication retained during the switch window, with the lifecycle lock already released.
762    pub(crate) fn mark_not_current(&self) -> bool {
763        let state = self.state.lock();
764        self.current.store(false, Ordering::Release);
765        finish_handoff_wakeup(*state, self.handoff_wakeup.swap(false, Ordering::Relaxed))
766    }

768    /// Claims this task's single wait-queue registration slot.
769    pub(crate) fn claim_waiting(&self) -> Result<(), TaskError> {
770        self.waiting
771            .compare_exchange(false, true, Ordering::AcqRel, Ordering::Acquire)
772            .map(|_| ())
773            .map_err(|_| TaskError::AlreadyWaiting)
774    }
775
776    /// Releases this task's wait-queue registration slot.
777    pub(crate) fn release_waiting(&self) {
778        self.waiting.store(false, Ordering::Release);
779    }

781    /// Associates this task with its synchronization wait queue.
782    pub(crate) fn set_wait_queue(&self, queue: *const ()) {
783        self.wait_queue
784            .store(queue as usize as u64, Ordering::Release);
785    }
786
787    /// Removes and returns the synchronization wait queue pointer.
788    pub(crate) fn take_wait_queue(&self) -> Option<*const ()> {
789        let value = self.wait_queue.swap(0, Ordering::AcqRel);
790        (value != 0).then_some(value as usize as *const ())
791    }
792
793    /// Begins one timed wait and returns its generation token.
794    pub(crate) fn begin_timed_wait(&self, cpu_id: usize) -> Result<u64, TaskStateError> {
795        let generation = self.begin_sleep();
796        let _guard = kernel_guard::NoPreempt::new();
797        let mut state = self.state.lock();
798        match *state {
799            TaskState::Running { cpu_id: owner } if owner == cpu_id => {
800                *state = TaskState::TimedWaiting;
801                self.wait_result.store(0, Ordering::Release);
802                self.timer_cpu.store(cpu_id, Ordering::Release);
803                Ok(generation)
804            }
805            TaskState::Running { .. } => Err(TaskStateError::WrongCpu),
806            _ => Err(TaskStateError::InvalidTransition),
807        }
808    }
809
810    /// Returns the timer CPU for the current timed wait, if one is registered.
811    pub(crate) fn timer_cpu(&self) -> Option<usize> {
812        let cpu = self.timer_cpu.load(Ordering::Acquire);
813        (cpu != usize::MAX).then_some(cpu)
814    }
815
816    /// Returns and clears the result of the most recent timed wait.
817    pub(crate) fn take_wait_result(&self) -> Option<WaitResult> {
818        match self.wait_result.swap(0, Ordering::AcqRel) {
819            1 => Some(WaitResult::Woken),
820            2 => Some(WaitResult::TimedOut),
821            _ => None,
822        }
823    }

828    ///
829    /// KernelStack drops only after this assertion proves execution moved to another stack.
830    fn drop(&mut self) {
831        assert!(
832            !self.current.load(Ordering::Acquire),
833            "attempted to reclaim the current task stack"
834        );
835    }

838/// Runs a task's owned closure and transfers control to the narrow exit callback.
839///
840/// The raw argument owns one Arc strong reference. Completion is published before terminal state
841/// and waiter notification are observed, and this function never returns.
842unsafe extern "C" fn task_trampoline(task: usize) -> ! {
843    // SAFETY: prepare_context supplies exactly one unmatched Arc::into_raw token to the initial
844    // context, and architecture entry invokes this trampoline exactly once with that token.
845    let task: Arc<Task> = unsafe { execution_arg_into_arc(task) };
846    super::scheduler::finish_initial_switch();
847    let body = unsafe { (&mut *task.body.get()).take() }.expect("task body already consumed");
848    body();
849    let waiters = task.complete(crate::mp::current_cpu_id());
850    super::scheduler::wake_tasks(waiters);
851    super::exit_current(task)
852}

854/// A handle that joins one cooperative task completion.
855///
856/// The handle owns completion state but does not keep an exited task's stack alive.
857pub struct JoinHandle {
858    /// Stable identity used to reject self-join.
859    task_id: TaskId,
860    /// Completion state shared with the task and all observers.
861    completion: Arc<Completion>,
862    /// Weak target used for cross-domain wake placement without retaining the task.
863    task: Weak<Task>,
864    /// Whether this handle has consumed its join operation.
865    joined: bool,
866}
867
868impl JoinHandle {
869    /// Waits cooperatively until the target task exits.
870    ///
871    /// Joining the current task is rejected because it cannot make progress while blocked.
872    pub fn join(&mut self) -> Result<(), TaskError> {
873        if self.joined {
874            return Err(TaskError::AlreadyJoined);
875        }
876        let current_task_id = super::scheduler::current_task_id();
877        if validate_join(current_task_id, self.task_id).is_err() {
878            return Err(TaskError::WouldDeadlock);
879        }
880        while !self.completion.is_completed() {
881            if current_task_id.is_some() {
882                super::scheduler::block_current_on(&self.completion);
883            } else {
884                super::scheduler::yield_now();
885            }
886        }
887        self.joined = true;
888        Ok(())
889    }
890
891    /// Returns whether the target task has exited.
892    ///
893    /// This query never blocks or changes scheduler state.
894    pub fn is_finished(&self) -> bool {
895        self.completion.is_completed()
896    }
897
898    /// Returns the completed task outcome without consuming the join handle.
899    pub fn status(&self) -> Option<TaskExitStatus> {
900        self.completion.status()
901    }
902}

904/// Current-stack wakeup routing tests.
905///
906/// The pure transitions run under the same lifecycle lock in the production Task methods.
907#[cfg(test)]
908mod handoff_wakeup_tests {
909    use super::{TaskState, finish_handoff_wakeup, route_handoff_wakeup};
910
911    /// Retains exactly one publication when a wake wins before the stack handoff.
912    ///
913    /// Marker clearance transfers that publication to the incoming scheduler context.
914    #[test]
915    fn wake_before_handoff_is_not_lost() {
916        let (publish, pending) = route_handoff_wakeup(TaskState::Runnable, true);
917        assert!(!publish);
918        assert!(finish_handoff_wakeup(TaskState::Runnable, pending));
919        assert!(!finish_handoff_wakeup(TaskState::Runnable, false));
920        assert_eq!(
921            route_handoff_wakeup(TaskState::Runnable, false),
922            (true, false)
923        );
924    }
925
926    /// Clears pending publication when an already-awakened task resumes without switching.
927    ///
928    /// A late waker cannot publish an active Running task a second time.
929    #[test]
930    fn resumed_task_does_not_publish_again() {
931        assert_eq!(
932            route_handoff_wakeup(TaskState::Running { cpu_id: 1 }, true),
933            (false, false)
934        );
935        let (_, pending) = route_handoff_wakeup(TaskState::Runnable, true);
936        assert!(pending);
937        // Resuming current clears its old token under the same lock before another block.
938        let (_, pending) = route_handoff_wakeup(TaskState::Running { cpu_id: 1 }, true);
939        assert!(!finish_handoff_wakeup(TaskState::Runnable, pending));
940        // The new wake occurs after handoff, so it owns exactly one direct publication.
941        let (publish, pending) = route_handoff_wakeup(TaskState::Runnable, false);
942        assert!(publish);
943        assert!(!finish_handoff_wakeup(TaskState::Runnable, pending));
944    }
