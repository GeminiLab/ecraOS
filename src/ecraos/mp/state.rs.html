1//! Secondary CPU lifecycle states.
2//!
3//! The state machine is intentionally dependency-free so host tests can validate invalid
4//! transitions without building the no-std kernel.
5
6/// The lifecycle state of a CPU.
7#[derive(Debug, Clone, Copy, PartialEq, Eq)]
8pub enum CpuState {
9    /// The CPU has not received a startup request.
10    Offline,
11    /// The architecture accepted a startup request and initialization is pending.
12    Starting,
13    /// The CPU completed the required per-CPU initialization sequence.
14    Online,
15    /// Startup failed or exceeded its deadline.
16    Failed,
17}
18
19impl CpuState {
20    /// Checks whether a lifecycle state transition is valid.
21    pub const fn can_transition_to(self, to: CpuState) -> bool {
22        matches!(
23            (self, to),
24            (CpuState::Offline, CpuState::Starting)
25                | (CpuState::Starting, CpuState::Online)
26                | (CpuState::Starting, CpuState::Failed)
27        )
28    }
29
30    /// Converts the lifecycle state to its atomic storage encoding.
31    ///
32    /// The returned value is stable for values stored in the per-CPU state array.
33    pub const fn to_u8(self) -> u8 {
34        match self {
35            CpuState::Offline => 0,
36            CpuState::Starting => 1,
37            CpuState::Online => 2,
38            CpuState::Failed => 3,
39        }
40    }
41
42    /// Converts an atomic storage encoding to its lifecycle state.
43    ///
44    /// This function panics when `value` is not an encoding produced by [`CpuState::to_u8`].
45    pub const fn from_u8(value: u8) -> CpuState {
46        match value {
47            0 => CpuState::Offline,
48            1 => CpuState::Starting,
49            2 => CpuState::Online,
50            3 => CpuState::Failed,
51            _ => panic!("Invalid CPU state value"),
52        }
53    }
54}

56/// CPU lifecycle state-machine tests.
57///
58/// These tests cover valid transitions, rejected transitions, and atomic storage encoding.
59#[cfg(test)]
60mod tests {
61    use super::CpuState;
62
63    /// Verifies that an offline CPU can enter the startup sequence.
64    ///
65    /// Starting is the only valid state immediately following offline.
66    #[test]
67    fn offline_starts_as_starting() {
68        assert!(CpuState::Offline.can_transition_to(CpuState::Starting),);
69    }
70
71    /// Verifies that an offline CPU cannot publish itself as online.
72    ///
73    /// Every secondary CPU must pass through the starting state first.
74    #[test]
75    fn offline_cannot_skip_starting() {
76        assert!(!CpuState::Offline.can_transition_to(CpuState::Online),);
77    }
78
79    /// Verifies that a starting CPU can publish online exactly once.
80    ///
81    /// An online CPU cannot repeat the publication transition.
82    #[test]
83    fn starting_can_be_online_once() {
84        assert!(CpuState::Starting.can_transition_to(CpuState::Online),);
85        assert!(!CpuState::Online.can_transition_to(CpuState::Online),);
86    }
87
88    /// Verifies that a failed CPU cannot later publish itself as online.
89    ///
90    /// Failed is a terminal lifecycle state for the current startup attempt.
91    #[test]
92    fn failed_cannot_become_online() {
93        assert!(CpuState::Starting.can_transition_to(CpuState::Failed),);
94        assert!(!CpuState::Failed.can_transition_to(CpuState::Online),);
95    }
96
97    /// Verifies that every lifecycle state survives atomic encoding and decoding.
98    ///
99    /// This guards the representation shared by initialization, loads, and transitions.
100    #[test]
101    fn states_round_trip_through_atomic_encoding() {
102        for state in [
103            CpuState::Offline,
104            CpuState::Starting,
105            CpuState::Online,
106            CpuState::Failed,
107        ] {
108            assert_eq!(CpuState::from_u8(state.to_u8()), state);
109        }
110    }
111}
