13/// A [`FormalPhase`] that is stable (not transitioning).
14#[derive(Clone, Copy, Debug, Eq, PartialEq)]
15pub enum StableFormalPhase {
16    /// Same as [`FormalPhase::IdentityActive`].
17    IdentityActive,
18    /// Same as [`FormalPhase::Active`].
19    Active,
20}
21
22/// A phase of [`VmmState::Formal`].
23#[derive(Clone, Copy, Debug, Eq, PartialEq)]
24pub enum FormalPhase {
25    /// The formal page table is active on the hardware and contains identity mappings.
26    IdentityActive,
27    /// A mapping transaction is updating the formal page table or completing its invalidations.
28    Transitioning {
29        /// ID of the transaction currently changing the page table.
30        transaction_id: u64,
31    },
32    /// The formal page table is active on the hardware and contains no identity mappings.
33    Active,
34}
35
36/// State updates at the beginning and end of a formal page-table update.
37impl FormalPhase {
38    /// Begins an exclusive formal page-table update and returns the previous stable phase.
39    ///
40    /// Returns [`None`] without changing the phase if another transaction is in progress.
41    pub fn begin_transition(&mut self, transaction_id: u64) -> Option<StableFormalPhase> {
42        let stable = match *self {
43            Self::IdentityActive => StableFormalPhase::IdentityActive,
44            Self::Active => StableFormalPhase::Active,
45            Self::Transitioning { .. } => return None,
46        };
47        *self = Self::Transitioning { transaction_id };
48        Some(stable)
49    }
50
51    /// Ends an exclusive formal page-table update owned by `transaction_id`.
52    ///
53    /// Sets the phase to `next` and returns `true` on success. Returns `false` without changing
54    /// the phase if the transaction does not match.
55    pub fn finish_transition(&mut self, transaction_id: u64, next: StableFormalPhase) -> bool {
56        if *self != (Self::Transitioning { transaction_id }) {
57            return false;
58        }
59        *self = match next {
60            StableFormalPhase::IdentityActive => Self::IdentityActive,
61            StableFormalPhase::Active => Self::Active,
62        };
63        true
64    }
65}

67/// A phase of [`VmmState::Early`].
68#[derive(Clone, Copy, Debug, Eq, PartialEq)]
69pub enum EarlyPhase {
70    /// The early page table is ready, but is not active on the hardware.
71    TableReady,
72    /// The early page table is active on the hardware and contains identity mappings.
73    Active,
74}
75
76/// State updates at the beginning of early page-table activation.
77impl EarlyPhase {
78    /// Marks the early page table as active and returns `true` on success.
79    ///
80    /// Returns `false` without changing the phase if the early page table was already marked
81    /// active.
82    pub fn activate(&mut self) -> bool {
83        if *self != Self::TableReady {
84            return false;
85        }
86        *self = Self::Active;
87        true
88    }
89}

91/// The current VMM lifecycle state and its page-table data.
92pub enum VmmState {
93    /// The address layout is ready, but the first page table has not been created.
94    LayoutReady {
95        /// The early page-table type, using the early page allocator.
96        page_table_type: OpaquePageTableType<VirtAddr>,
97    },
98    /// The early page table has been created and may or may not be active on the hardware.
99    Early {
100        /// The early page-table, using the early page allocator.
101        page_table: OpaquePageTable<VirtAddr>,
102        /// The phase of the early page table, indicating whether it is active on the hardware.
103        phase: EarlyPhase,
104    },
105    /// The formal page table has been created and is active on the hardware.
106    Formal {
107        /// The formal page table, using the buddy allocator for new table frames.
108        page_table: OpaquePageTable<VirtAddr>,
109        /// Page-aligned direct-mapping records, including any retained `NO_MAP` ranges.
110        mappings: FinalRegions,
111        /// The phase of the formal page table, indicating whether identity mappings remain or a
112        /// page-table change is in progress.
113        phase: FormalPhase,
114    },
115}
116
117/// Transfer of the early page table to the formal allocator.
118impl VmmState {
119    /// Switches to the formal page table allocator without changing the page table hierarchy.
120    ///
121    /// Preserves the existing hierarchy and starts the formal phase with identity mappings active.
122    /// `mappings` must describe the direct mappings installed in that hierarchy.
123    ///
124    /// # Safety
125    ///
126    /// The descriptor must match the existing page-table hierarchy. Its allocator must be able to
127    /// access and deallocate every frame in that hierarchy.
128    ///
129    /// # Panics
130    ///
131    /// Panics if the VMM does not own an active early page table.
132    pub unsafe fn switch_to_formal_page_table(
133        &mut self,
134        page_table_type: OpaquePageTableType<VirtAddr>,
135        mappings: FinalRegions,
136    ) {
137        let Self::Early {
138            page_table,
139            phase: EarlyPhase::Active,
140        } = self
141        else {
142            panic!("formal handoff requires an active early page table");
143        };
144        let root = mem::replace(page_table, OpaquePageTable::dummy()).leak();
145        // SAFETY: Leaking the only owner preserves the hierarchy without calling its old methods.
146        // The caller guarantees that the descriptor matches and can manage every table frame.
147        let page_table = unsafe { page_table_type.new_pagetable_at(root) };
148        *self = Self::Formal {
149            page_table,
150            mappings,
151            phase: FormalPhase::IdentityActive,
152        };
153    }
154
155    /// Returns a short description of the current state.
156    pub const fn desc(&self) -> &'static str {
157        match self {
158            Self::LayoutReady { .. } => "layout ready",
159            Self::Early { phase, .. } => match phase {
160                EarlyPhase::TableReady => "early page table ready",
161                EarlyPhase::Active => "early page table active",
162            },
163            Self::Formal { phase, .. } => match phase {
164                FormalPhase::IdentityActive => "formal and identity mappings active",
165                FormalPhase::Transitioning { .. } => "formal mapping in transition",
166                FormalPhase::Active => "formal mappings active",
167            },
168        }
169    }
170}
