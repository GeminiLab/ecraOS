34/// Rounds an MMIO description to the base pages required by the direct map.
35///
36/// Device registers may be smaller than, or unaligned to, the selected page size. Mapping the
37/// outward-aligned range keeps the physical and virtual direct-map offsets identical.
38fn aligned_mmio_region(region: MemoryRegion, page_size: usize) -> MemoryRegion {
39    assert!(
40        page_size.is_power_of_two(),
41        "base page size must be a power of two"
42    );
43    assert!(!region.range.is_empty(), "MMIO range must not be empty");
44    MemoryRegion {
45        range: region
46            .range
47            .align_outwards(page_size)
48            .expect("MMIO range alignment overflow"),
49        ..region
50    }
51}

53/// Reports whether two physical ranges require any of the same base pages.
54///
55/// Adjacent byte ranges can still conflict when page rounding makes them share a mapping.
56fn ranges_share_mapping_page(left: PhysAddrRange, right: PhysAddrRange, page_size: usize) -> bool {
57    let left = left
58        .align_outwards(page_size)
59        .expect("physical range alignment overflow");
60    let right = right
61        .align_outwards(page_size)
62        .expect("physical range alignment overflow");
63    left.overlaps(right)
64}

66/// Reports whether an overlapping mapping changes page attributes.
67///
68/// Callers pass page-aligned ranges so duplicate early-console and platform mappings can be
69/// reused while incompatible aliases are rejected.
70fn mapping_flags_conflict(
71    range: PhysAddrRange,
72    flags: MappingFlags,
73    mapped: &[(PhysAddrRange, MappingFlags)],
74) -> bool {
75    mapped.iter().any(|(previous_range, previous_flags)| {
76        previous_range.overlaps(range) && *previous_flags != flags
77    })
78}

80/// The serialization lock for kernel mapping changes and delayed reuse.
81///
82/// The request payload has one writer throughout mutation, remote completion and finalization.
83static MAPPING_TRANSACTION: SpinNoPreempt<()> = SpinNoPreempt::new(());
84
85// Lock order:
86// Allocate the MP online/IPI-ready snapshot before acquiring the mapping transaction.
87// mapping transaction (preemption disabled, interrupts enabled)
88//   -> vmalloc metadata lock, only for prepare/finalize phases
89//   -> page-table SpinNoIrq lock, only for PTE mutation and error-path local cleanup
90//   -> release every IRQ-disabling and allocator lock
91//   -> finish pending local cleanup, then publish/send/wait for remote acknowledgements
92//   -> reacquire vmalloc metadata briefly to finalize or quarantine
93//   -> release physical frames outside vmalloc and page-table locks.
94// No scheduler queue lock crosses this boundary. The IPI handler takes none of these locks.
95// A future lock-free mapping/request structure may replace this serialized design after a complete
96// memory-ordering proof. Early not-ready CPUs may add fresh mappings locally, but cannot remove or
97// replace mappings without completing the normal remote protocol.
98
99/// An owned kernel mapping transaction and its stable remote target set.
100///
101/// Only this preemption guard survives the remote completion phase.
102pub(crate) struct MappingTransaction {
103    /// The MP-owned snapshot excluding the pinned caller.
104    ///
105    /// Allocation precedes acquisition of the serialization lock.
106    targets: Vec<LogicalCpuId>,
107    /// The serialization and CPU-pinning guard.
108    ///
109    /// It leaves interrupts enabled throughout remote completion.
110    _guard: SpinNoPreemptGuard<'static, ()>,
111}
112
113impl MappingTransaction {
114    /// Captures ready CPUs and serializes mutation without disabling interrupts.
115    ///
116    /// A ready CPU cannot spin on this lock with interrupts disabled while another owner awaits it.
117    pub(crate) fn begin() -> Result<Self, TlbShootdownError> {
118        let snapshot = mp::ipi_ready_cpu_ids();
119        let pin = kernel_guard::NoPreempt::new();
120        if mp::is_ipi_ready(mp::current_cpu_id()) && !exarch::trap::local_enabled() {
121            return Err(TlbShootdownError::Ipi(
122                crate::ipi::IpiError::InterruptsDisabled,
123            ));
124        }
125        let targets = tlb::remote_targets(snapshot, mp::current_cpu_id());
126        let guard = MAPPING_TRANSACTION.lock();
127        drop(pin);
128        Ok(Self {
129            targets,
130            _guard: guard,
131        })
132    }
133
134    /// Completes remote invalidation after the local page-table guard has been released.
135    ///
136    /// The same snapshot is used for every mutation in this transaction.
137    pub(crate) fn complete(&self, request: TlbInvalidation) -> Result<(), TlbShootdownError> {
138        tlb::shootdown(&self.targets, request)
139    }
140
141    /// Returns the captured remote logical destinations for diagnostics.
142    ///
143    /// The returned slice does not create a second CPU inventory.
144    pub(crate) fn targets(&self) -> &[LogicalCpuId] {
145        &self.targets
146    }
147}

250/// Computes the physical address corresponding to a direct-map virtual address.
251///
252/// Checks only whether the address lies in the configured direct-map window. It does not verify
253/// that a page-table mapping exists for the address, so callers must only pass mapped addresses.
254///
255/// Returns [`None`] when the layout is unavailable or the address lies outside the direct map.
256pub fn direct_mapping_virt_to_phys(addr: VirtAddr) -> Option<PhysAddr> {
257    let addr_space = VIRTUAL_ADDRESS_SPACE.get()?;
258    if !addr_space.layout.direct_mapping_range.contains(addr) {
259        return None;
260    }
261
262    Some(PhysAddr::from_usize(
263        addr.as_usize() - addr_space.layout.direct_mapping_range.start.as_usize(),
264    ))
265}

267/// Computes the direct-map virtual address corresponding to a physical address.
268///
269/// Checks that the final roots are active and that the address fits in the reserved direct-map
270/// window. It does not check whether that physical address is mapped there. NO_MAP regions
271/// intentionally have no direct-map mapping.
272///
273/// Returns [`None`] before the final roots are active or when the offset exceeds the reservation.
274pub fn direct_mapping_phys_to_virt(addr: PhysAddr) -> Option<VirtAddr> {
275    if !PAGE_TABLE_ACTIVE.load(Ordering::Acquire) {
276        return None;
277    }
278
279    let addr_space = VIRTUAL_ADDRESS_SPACE.get()?;
280    let offset = addr.as_usize();
281    let start = addr_space.layout.direct_mapping_range.start.as_usize();
282    let virt = VirtAddr::from_usize(start.checked_add(offset)?);
283    addr_space
284        .layout
285        .direct_mapping_range
286        .contains(virt)
287        .then_some(virt)
288}

290/// Runs a closure with exclusive access to the kernel page-table owner.
291///
292/// Operations use the allocator captured by the owner's descriptor. The page-table slot must
293/// already be initialized and cannot be observed while allocator handoff holds the same lock.
294#[inline]
295pub fn with_page_table<F, T>(f: F) -> T
296where
297    F: FnOnce(&mut OpaquePageTable<VirtAddr>) -> T,
298{
299    let mut pt = VIRTUAL_ADDRESS_SPACE.page_table_mutex.lock();
300    f(&mut pt)
301}

303/// Maps discovered device regions through the physical-memory direct map.
304///
305/// The bootstrap CPU calls this after the formal allocator is ready and before starting APs.
306/// Ranges are rounded to base pages, checked against existing direct-map aliases and the
307/// direct-map capacity, and mapped with their declared device attributes. Identical overlaps are
308/// reused so an early console range can also appear in platform discovery.
309pub fn map_mmio_regions(regions: &[MemoryRegion]) {
310    let page_size = page_size();
311    let physical_memory = crate::mem::pmm::phys_mem_regions();
312    let mut mapped = Vec::with_capacity(regions.len());
313
314    for region in regions {
315        assert!(
316            region.flags.contains(MemoryRegionFlags::DEVICE),
317            "platform MMIO range is missing the device-memory flag: {region:?}"
318        );
319        assert!(
320            region.flags.role_sanity_check() && region.flags.contains(MemoryRegionFlags::RESERVED),
321            "platform MMIO range must have only the reserved ownership role: {region:?}"
322        );
323
324        let aligned = aligned_mmio_region(*region, page_size);
325        let mapping_flags = region_flags_to_mapping(aligned.flags);
326        assert!(
327            !mapping_flags.is_empty(),
328            "platform MMIO range has no mapping permissions: {region:?}"
329        );
330
331        assert!(
332            !mapping_flags_conflict(aligned.range, mapping_flags, &mapped),
333            "overlapping platform MMIO ranges have different page attributes: {region:?}"
334        );
335        let mut is_already_mapped = false;
336        for memory_region in physical_memory {
337            if !ranges_share_mapping_page(memory_region.range, aligned.range, page_size) {
338                continue;
339            }
340            let existing_range = memory_region
341                .range
342                .align_outwards(page_size)
343                .expect("physical memory range alignment overflow");
344            let existing_flags = region_flags_to_mapping(memory_region.flags);
345            assert_eq!(
346                existing_flags, mapping_flags,
347                "platform MMIO range conflicts with an existing direct-map page: mmio={region:?}, existing={memory_region:?}"
348            );
349            if existing_range.start <= aligned.range.start
350                && existing_range.end >= aligned.range.end
351            {
352                is_already_mapped = true;
353            }
354        }
355        mapped.push((aligned.range, mapping_flags));
356
357        let virtual_start = direct_mapping_phys_to_virt(aligned.range.start)
358            .expect("platform MMIO start exceeds direct-map capacity");
359        let last_physical = PhysAddr::from_usize(
360            aligned
361                .range
362                .end
363                .as_usize()
364                .checked_sub(1)
365                .expect("aligned MMIO range must not be empty"),
366        );
367        direct_mapping_phys_to_virt(last_physical)
368            .expect("platform MMIO end exceeds direct-map capacity");
369
370        if is_already_mapped {
371            info!(
372                "Reusing direct-mapped platform MMIO: {} {:#x}",
373                region.desc, aligned.range
374            );
375            continue;
376        }
377
378        info!(
379            "Mapping platform MMIO: {} {:#x} flags={}",
380            region.desc, aligned.range, region.flags
381        );
382        with_page_table(|page_table| {
383            page_table
384                .map(
385                    virtual_start,
386                    aligned.range.start,
387                    aligned.range.size(),
388                    mapping_flags,
389                )
390                .expect("failed to map platform MMIO range");
391        });
392    }
393}

395/// Completes local cleanup and preserves its required remote invalidation scope.
396///
397/// The caller releases the page-table lock before calling this helper and supplies a request
398/// covering every changed base page. A full cleanup request also widens remote invalidation,
399/// including changes to huge mappings or intermediate descriptors. The returned request must
400/// complete on all captured remote CPUs before mappings are published or data frames are reused.
401pub(crate) fn finish_local_cleanup(
402    cleanup: PendingCleanup<VirtAddr>,
403    request: TlbInvalidation,
404) -> TlbInvalidation {
405    // SAFETY: Process every request locally through its captured strategy, then return the
406    // required remote scope to the mapping transaction. No pending invalidation is discarded.
407    let (mut flushes, flusher) = unsafe { cleanup.into_inner() };
408    let request = if matches!(flushes, TlbFlushes::Full) {
409        TlbInvalidation::All
410    } else {
411        request
412    };
413    flushes.flush_with(&flusher);
414    request
415}

432/// Initializes the virtual address space.
433///
434/// Detects the translation mode and calculates the kernel layout without allocating table pages.
435/// The initial opaque page table type captures the early allocator `AL` and [`LocalTlbFlusher`]
436/// for cleanup on the calling processor.
437pub(super) fn init_vmm_layout<AL: PageAllocator>() {
438    kprintln!("Virtual address space:");
439
440    let mode = select_va_mode(exarch::mem::virt_addr_space_modes());
441    exarch::mem::set_virt_addr_space_mode(mode);
442    let layout = calculate_vmm_layout(mode);
443
444    VIRTUAL_ADDRESS_SPACE.init_once(VirtualAddressSpace {
445        layout,
446        page_table_type_mutex: SpinNoIrq::new(exarch::mem::get_page_table_type::<
447            AL,
448            LocalTlbFlusher,
449        >(mode)),
450        page_table_mutex: SpinNoIrq::new(OpaquePageTable::dummy()),
451    });
452}

454/// Selects the virtual address space mode to use.
455///
456/// Currently, we require the virtual address space mode to cover both halves
457/// symmetrically, and we prefer those with the largest upper virtual address
458/// bits, and among which we prefer those with the largest page size.
459fn select_va_mode(va_modes: VirtAddrSpaceModes) -> VirtAddrSpaceMode {
460    kprintln!("  Platform virtual address space modes:");
461    for mode in &va_modes.modes {
462        kprintln!("    {}", *mode);
463    }
464
465    kprintln!(
466        "  Current virtual address space mode:\n    {}",
467        va_modes.modes[va_modes.current_index]
468    );
469
470    // Prefer an address-space mode that covers both halves. AArch64 exposes
471    // these as independent TTBR0 and TTBR1 regimes, while x86-64 and RISC-V
472    // expose a unified root.
473    let mut chosen_index = None::<usize>;
474    for (index, mode) in va_modes.modes.iter().enumerate() {
475        if matches!(
476            mode,
477            VirtAddrSpaceMode::Unified(..) | VirtAddrSpaceMode::Independent { .. }
478        ) && mode.upper_va_bits() == mode.lower_va_bits()
479            && mode.upper_page_shift() == mode.lower_page_shift()
480            && chosen_index
481                .map(|index| {
482                    let old_mode = va_modes.modes[index];
483
484                    old_mode.upper_va_bits() < mode.upper_va_bits()
485                        || (old_mode.upper_va_bits() == mode.upper_va_bits()
486                            && old_mode.upper_page_shift() < mode.upper_page_shift())
487                })
488                .unwrap_or(true)
489        {
490            chosen_index = Some(index);
491        }
492    }
493
494    let chosen_index = chosen_index.expect("No good virtual address space mode found");
495    let chosen_mode = va_modes.modes[chosen_index];
496
497    kprintln!(
498        "  Selected virtual address space mode:\n    {}",
499        chosen_mode
500    );
501
502    chosen_mode
503}

505/// Calculates the layout of the virtual address space based on the given mode.
506///
507/// Reserves the first half of the upper address range for the direct map and the next quarter for
508/// vmalloc, leaving the final quarter unused.
509fn calculate_vmm_layout(mode: VirtAddrSpaceMode) -> VirtualAddressSpaceLayout {
510    let upper_bits = mode.upper_va_bits().unwrap().get();
511    let page_shift = mode.upper_page_shift().unwrap().get();
512    let upper_start = va!((1usize << upper_bits).wrapping_neg());
513
514    kprintln!("  Upper half start    : {:#x}", upper_start);
515
516    // Top level memory areas:
517    // - 1st half: direct mapping area
518    // - 2nd half:
519    //   - 3rd quarter: vmalloc area
520    //   - 4th quarter: not used
521    // This scheme gives us at least the same size of these areas as Linux does.
522    let quarter_size = 1usize << (upper_bits - 2);
523    let direct_mapping_range = VirtAddrRange::from_start_size(upper_start, quarter_size * 2);
524    let vmalloc_range =
525        VirtAddrRange::from_start_size(upper_start + quarter_size * 2, quarter_size);
526    kprintln!(
527        "  Direct mapping area : {:x}, {}",
528        direct_mapping_range,
529        direct_mapping_range.size().size_display_wide()
530    );
531    kprintln!(
532        "  Vmalloc area        : {:x}, {}",
533        vmalloc_range,
534        vmalloc_range.size().size_display_wide()
535    );
536
537    VirtualAddressSpaceLayout {
538        mode,
539        page_shift,
540        direct_mapping_range,
541        vmalloc_range,
542    }
543}

545/// Initializes the page table with identity mappings and the physical-memory direct map.
546pub(super) fn init_vmm_mapping_early(phys_mem_regions: &MemoryRegions) {
547    let virt_phys_offset = direct_mapping_offset();
548    let mut pt = VIRTUAL_ADDRESS_SPACE
549        .page_table_type_mutex
550        .lock()
551        .new_pagetable_alloc()
552        .unwrap();
553
554    // TODO: the physical memory regions are not guaranteed to be page-aligned,
555    // that may cause issues when mapping them. Luckily, it's not a serious or
556    // urgent issue now, because free memory regions and kernel sections are
557    // almost guaranteed to be page-aligned. We should fix this in the future.
558    for region in phys_mem_regions {
559        if region.flags.contains(MemoryRegionFlags::NO_MAP) {
560            continue;
561        }
562        let mapping_flags = region_flags_to_mapping(region.flags);
563        if mapping_flags.is_empty() {
564            continue;
565        }
566
567        let paddr = region.range.start;
568        let vaddr_low = VirtAddr::from_usize(paddr.as_usize());
569        let vaddr_high = vaddr_low + virt_phys_offset;
570        let size = region.range.size();
571
572        // The high direct map is needed for all physical memory. The low
573        // identity map only needs to cover code/data used before the jump and
574        // pages allocated for the bootstrap page tables.
575        let early_range = early::phys_addr_range();
576        let needs_full_identity = region.flags.intersects(
577            MemoryRegionFlags::KERNEL
578                | MemoryRegionFlags::BOOT_SERVICE
579                | MemoryRegionFlags::EXECUTE,
580        );
581        if needs_full_identity {
582            pt.map(vaddr_low, paddr, size, mapping_flags).unwrap();
583        } else if region.range.start < early_range.end && early_range.start < region.range.end {
584            let identity_start = region.range.start.max(early_range.start);
585            let identity_end = region.range.end.min(early_range.end);
586            let identity_size = identity_end.as_usize() - identity_start.as_usize();
587            pt.map(
588                VirtAddr::from_usize(identity_start.as_usize()),
589                identity_start,
590                identity_size,
591                mapping_flags,
592            )
593            .unwrap();
594        }
595        pt.map(vaddr_high, paddr, size, mapping_flags).unwrap();
596    }
597
598    *VIRTUAL_ADDRESS_SPACE.page_table_mutex.lock() = pt;
599}

601/// Transfers the active page-table owner to the formal allocator.
602///
603/// Refreshes relocated function pointers and transfers the early owner's roots to the formal
604/// allocator without freeing them or calling old method pointers. Before this handoff, the formal
605/// allocator must have registered all early page-table allocations so it can release them.
606pub(super) fn use_formal_page_table_allocator() {
607    // Fix function addresses in the opaque page table type first.
608    let page_table_type = exarch::mem::get_page_table_type::<FormalPageAllocator, LocalTlbFlusher>(
609        VIRTUAL_ADDRESS_SPACE.layout.mode,
610    );
611    *VIRTUAL_ADDRESS_SPACE.page_table_type_mutex.lock() = page_table_type.clone();
612
613    let mut page_table_guard = VIRTUAL_ADDRESS_SPACE.page_table_mutex.lock();
614
615    let root = mem::replace(page_table_guard.deref_mut(), OpaquePageTable::dummy()).leak();
616
617    // SAFETY: leak transfers the existing roots and all their table pages out of the only owner.
618    // The refreshed descriptor has the same geometry and can access and release those pages.
619    *page_table_guard = unsafe { page_table_type.new_pagetable_at(root) };
620}

622/// Removes temporary low-half identity mappings from every ready CPU's active view.
623///
624/// Failure is fatal after local mutation and never restores an inconsistent local mapping.
625pub(super) fn remove_identical_mapping(phys_mem_regions: &MemoryRegions) {
626    info!("Removing identical mappings...");
627    let transaction = MappingTransaction::begin().expect("identity mapping transaction rejected");
628
629    let cleanup = with_page_table(|pt| {
630        pt.apply_actions_with_cleanup(phys_mem_regions.iter().map(|region| {
631            let vaddr_low = va!(region.range.start.as_usize());
632            let size = region.range.size();
633            info!(
634                "identity-map clearing range={:#x}",
635                VirtAddrRange::from_start_size(vaddr_low, size)
636            );
637            PageTableAction::Unmap {
638                vaddr: vaddr_low,
639                size,
640            }
641        }))
642        .expect("identity mapping mutation failed")
643    });
644    // The page-table guard is released before cleanup and remote completion.
645    let request = finish_local_cleanup(cleanup, TlbInvalidation::All);
646    transaction.complete(request).unwrap_or_else(|error| {
647        panic!(
648            "identity-map completion failed ranges={phys_mem_regions:?} targets={:?}: {error:?}",
649            transaction.targets()
650        );
651    });
652    info!(
653        "identity-map completion targets={:?} operation=All",
654        transaction.targets()
655    );
656}

658/// The kernel's formal page allocator for page-table storage.
659///
660/// Uses the initialized buddy allocator and the high physical-memory direct map. The buddy
661/// allocator adopts early allocations before this type becomes the active descriptor's allocator.
662pub struct FormalPageAllocator;
663
664impl PageAllocator for FormalPageAllocator {
665    /// Returns the logarithm of the kernel base-page size in bytes.
666    ///
667    /// Page-table allocations use the same granularity as the physical allocator.
668    fn page_size_shift() -> usize {
669        page_size_shift()
670    }
671
672    /// Allocates one physical frame for page-table storage.
673    ///
674    /// Returns [`None`] when the buddy allocator cannot satisfy the request.
675    fn alloc_frame() -> Option<PhysAddr> {
676        allocs::palloc::alloc_frame().ok()
677    }
678
679    /// Allocates contiguous physical frames for page-table storage.
680    ///
681    /// The allocation is aligned to at least one kernel base page.
682    fn alloc_frames(page_count: usize) -> Option<PhysAddr> {
683        allocs::palloc::alloc_frames(page_count, 1 << page_size_shift()).ok()
684    }
685
686    /// Releases one physical page-table frame to the buddy allocator.
687    ///
688    /// The frame must belong to this allocator, including registered early allocations.
689    fn dealloc_frame(addr: PhysAddr) {
690        allocs::palloc::dealloc_frame(addr).unwrap();
691    }
692
693    /// Releases contiguous physical page-table frames to the buddy allocator.
694    ///
695    /// The address and count must match an allocation adopted or created by this allocator.
696    fn dealloc_frames(addr: PhysAddr, page_count: usize) {
697        allocs::palloc::dealloc_frames(addr, page_count).unwrap();
698    }
699
700    /// Translates a physical page-table address through the kernel direct map.
701    ///
702    /// The final page-table roots must be active before this allocator accesses table storage.
703    fn phys_to_virt(addr: PhysAddr) -> VirtAddr {
704        VirtAddr::from_usize(addr.as_usize() + direct_mapping_offset())
705    }
706}

708/// Deferred page-table cleanup and remote invalidation tests.
709///
710/// Recording local invalidations avoids privileged instructions on the host.
711#[cfg(test)]
712mod tests {
713    use core::cell::RefCell;
714
715    use expt::pt::cleanup::{PendingCleanup, TlbFlush, TlbFlusher};
716    use expt::pte::MappingFlags;
717    use memory_addr::{PhysAddrRange, VirtAddr, VirtAddrRange, pa, va};
718
719    use crate::tlb::TlbInvalidation;
720
721    use super::{
722        MemoryRegion, MemoryRegionFlags, aligned_mmio_region, finish_local_cleanup,
723        mapping_flags_conflict, ranges_share_mapping_page,
724    };
725
726    std::thread_local! {
727        /// The local invalidations completed by the calling test thread.
728        ///
729        /// Each test starts with an independent empty recording.
730        static FLUSHES: RefCell<Vec<Option<VirtAddr>>> = const { RefCell::new(Vec::new()) };
731    }
732
733    /// A local invalidation recorder.
734    ///
735    /// It models the captured strategy without executing architecture instructions.
736    struct RecordingFlusher;
737
738    /// The host recording strategy implementation.
739    ///
740    /// Cleanup completes by storing observations in thread-local test state.
741    impl TlbFlusher<VirtAddr> for RecordingFlusher {
742        /// Records one completed local invalidation.
743        ///
744        /// The returned remote request must be inspected only after this record exists.
745        fn flush(address: Option<VirtAddr>) {
746            FLUSHES.with_borrow_mut(|flushes| flushes.push(address));
747        }
748    }
749
750    /// Completes local pages before returning the requested remote range.
751    ///
752    /// Batching must retain every pending address without widening a sufficient range.
753    #[test]
754    fn finishes_local_pages_before_remote_completion() {
755        let mut cleanup = PendingCleanup::new_with::<RecordingFlusher>();
756        cleanup.push_flush(TlbFlush::Page(va!(0x1000)));
757        cleanup.push_flush(TlbFlush::Page(va!(0x2000)));
758        let request = TlbInvalidation::Range(VirtAddrRange::new(va!(0x1000), va!(0x3000)));
759        let remote = finish_local_cleanup(cleanup, request);
760        FLUSHES.with_borrow(|flushes| {
761            assert_eq!(flushes.as_slice(), &[Some(va!(0x1000)), Some(va!(0x2000))]);
762        });
763        assert_eq!(remote, request);
764    }
765
766    /// Widens remote invalidation when a page-table mutation requires a full flush.
767    ///
768    /// Huge-leaf splits and accumulated flush overflow cannot be completed by one remote page.
769    #[test]
770    fn full_cleanup_requires_full_remote_invalidation() {
771        let mut cleanup = PendingCleanup::new_with::<RecordingFlusher>();
772        cleanup.push_flush(TlbFlush::Page(va!(0x1000)));
773        cleanup.push_flush(TlbFlush::Full);
774        let remote = finish_local_cleanup(cleanup, TlbInvalidation::Page(va!(0x1000)));
775        FLUSHES.with_borrow(|flushes| assert_eq!(flushes.as_slice(), &[None]));
776        assert_eq!(remote, TlbInvalidation::All);
777    }
778
779    /// Rounds an unaligned device range outward without changing its mapping attributes.
780    #[test]
781    fn aligns_mmio_range_outwards_to_base_pages() {
782        let region = MemoryRegion {
783            range: PhysAddrRange::new(pa!(0x1234), pa!(0x2abc)),
784            flags: MemoryRegionFlags::READ
785                | MemoryRegionFlags::WRITE
786                | MemoryRegionFlags::DEVICE
787                | MemoryRegionFlags::RESERVED,
788            desc: "UART registers",
789        };
790
791        let aligned = aligned_mmio_region(region, 0x1000);
792
793        assert_eq!(aligned.range, PhysAddrRange::new(pa!(0x1000), pa!(0x3000)));
794        assert_eq!(aligned.flags, region.flags);
795        assert_eq!(aligned.desc, region.desc);
796    }
797
798    /// Treats adjacent byte ranges in one page as a mapping conflict.
799    #[test]
800    fn detects_mmio_sharing_a_page_with_ram() {
801        let mmio = PhysAddrRange::new(pa!(0x1800), pa!(0x2000));
802        let ram = PhysAddrRange::new(pa!(0x1000), pa!(0x1800));
803
804        assert!(ranges_share_mapping_page(mmio, ram, 0x1000));
805        assert!(!ranges_share_mapping_page(
806            mmio,
807            PhysAddrRange::new(pa!(0x2000), pa!(0x2800)),
808            0x1000,
809        ));
810    }

812    /// Allows duplicate MMIO mappings only when their page attributes agree.
813    #[test]
814    fn rejects_overlapping_mmio_mappings_with_different_attributes() {
815        let range = PhysAddrRange::new(pa!(0x4000), pa!(0x5000));
816        let mapped = [(range, MappingFlags::READ | MappingFlags::DEVICE)];

818        assert!(!mapping_flags_conflict(
819            range,
