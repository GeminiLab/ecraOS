31/// Deferred page-table cleanup and remote invalidation tests.
32///
33/// Recording local invalidations avoids privileged instructions on the host.
34#[cfg(test)]
35mod tests {
36    use core::cell::RefCell;
37
38    use expt::pt::cleanup::{PendingCleanup, TlbFlush, TlbFlusher};
39    use memory_addr::{VirtAddr, VirtAddrRange, va};

41    use crate::tlb::TlbInvalidation;

43    use super::finish_local_cleanup;

45    std::thread_local! {
46        /// The local invalidations completed by the calling test thread.
47        ///
48        /// Each test starts with an independent empty recording.
49        static FLUSHES: RefCell<Vec<Option<VirtAddr>>> = const { RefCell::new(Vec::new()) };
50    }
51
52    /// A local invalidation recorder.
53    ///
54    /// It models the captured strategy without executing architecture instructions.
55    struct RecordingFlusher;
56
57    /// The host recording strategy implementation.
58    ///
59    /// Cleanup completes by storing observations in thread-local test state.
60    impl TlbFlusher<VirtAddr> for RecordingFlusher {
61        /// Records one completed local invalidation.
62        ///
63        /// The returned remote request must be inspected only after this record exists.
64        fn flush(address: Option<VirtAddr>) {
65            FLUSHES.with_borrow_mut(|flushes| flushes.push(address));
66        }
67    }
68
69    /// Completes local pages before returning the requested remote range.
70    ///
71    /// Batching must retain every pending address without widening a sufficient range.
72    #[test]
73    fn finishes_local_pages_before_remote_completion() {
74        let mut cleanup = PendingCleanup::new_with::<RecordingFlusher>();
75        cleanup.push_flush(TlbFlush::Page(va!(0x1000)));
76        cleanup.push_flush(TlbFlush::Page(va!(0x2000)));
77        let request = TlbInvalidation::Range(VirtAddrRange::new(va!(0x1000), va!(0x3000)));
78        let remote = finish_local_cleanup(cleanup, request);
79        FLUSHES.with_borrow(|flushes| {
80            assert_eq!(flushes.as_slice(), &[Some(va!(0x1000)), Some(va!(0x2000))]);
81        });
82        assert_eq!(remote, request);
83    }
84
85    /// Widens remote invalidation when a page-table mutation requires a full flush.
86    ///
87    /// Huge-leaf splits and accumulated flush overflow cannot be completed by one remote page.
88    #[test]
89    fn full_cleanup_requires_full_remote_invalidation() {
90        let mut cleanup = PendingCleanup::new_with::<RecordingFlusher>();
91        cleanup.push_flush(TlbFlush::Page(va!(0x1000)));
92        cleanup.push_flush(TlbFlush::Full);
93        let remote = finish_local_cleanup(cleanup, TlbInvalidation::Page(va!(0x1000)));
94        FLUSHES.with_borrow(|flushes| assert_eq!(flushes.as_slice(), &[None]));
95        assert_eq!(remote, TlbInvalidation::All);
96    }

99/// The serialization lock for kernel mapping changes and delayed reuse.
100///
101/// The request payload has one writer throughout mutation, remote completion and finalization.
102static MAPPING_TRANSACTION: SpinNoPreempt<()> = SpinNoPreempt::new(());
103
104// Lock order:
105// Allocate the MP online/IPI-ready snapshot before acquiring the mapping transaction.
106// mapping transaction (preemption disabled, interrupts enabled)
107//   -> vmalloc metadata lock, only for prepare/finalize phases
108//   -> page-table SpinNoIrq lock, only for PTE mutation and error-path local cleanup
109//   -> release every IRQ-disabling and allocator lock
110//   -> finish pending local cleanup, then publish/send/wait for remote acknowledgements
111//   -> reacquire vmalloc metadata briefly to finalize or quarantine
112//   -> release physical frames outside vmalloc and page-table locks.
113// No scheduler queue lock crosses this boundary. The IPI handler takes none of these locks.
114// A future lock-free mapping/request structure may replace this serialized design after a complete
115// memory-ordering proof. Early not-ready CPUs may add fresh mappings locally, but cannot remove or
116// replace mappings without completing the normal remote protocol.
117
118/// An owned kernel mapping transaction and its stable remote target set.
119///
120/// Only this preemption guard survives the remote completion phase.
121pub(crate) struct MappingTransaction {
122    /// The MP-owned snapshot excluding the pinned caller.
123    ///
124    /// Allocation precedes acquisition of the serialization lock.
125    targets: Vec<LogicalCpuId>,
126    /// The serialization and CPU-pinning guard.
127    ///
128    /// It leaves interrupts enabled throughout remote completion.
129    _guard: SpinNoPreemptGuard<'static, ()>,
130}
131
132impl MappingTransaction {
133    /// Captures ready CPUs and serializes mutation without disabling interrupts.
134    ///
135    /// A ready CPU cannot spin on this lock with interrupts disabled while another owner awaits it.
136    pub(crate) fn begin() -> Result<Self, TlbShootdownError> {
137        let snapshot = mp::ipi_ready_cpu_ids();
138        let pin = kernel_guard::NoPreempt::new();
139        if mp::is_ipi_ready(mp::current_cpu_id()) && !exarch::trap::local_enabled() {
140            return Err(TlbShootdownError::Ipi(
141                crate::ipi::IpiError::InterruptsDisabled,
142            ));
143        }
144        let targets = tlb::remote_targets(snapshot, mp::current_cpu_id());
145        let guard = MAPPING_TRANSACTION.lock();
146        drop(pin);
147        Ok(Self {
148            targets,
149            _guard: guard,
150        })
151    }
152
153    /// Completes remote invalidation after the local page-table guard has been released.
154    ///
155    /// The same snapshot is used for every mutation in this transaction.
156    pub(crate) fn complete(&self, request: TlbInvalidation) -> Result<(), TlbShootdownError> {
157        tlb::shootdown(&self.targets, request)
158    }
159
160    /// Returns the captured remote logical destinations for diagnostics.
161    ///
162    /// The returned slice does not create a second CPU inventory.
163    pub(crate) fn targets(&self) -> &[LogicalCpuId] {
164        &self.targets
165    }
166}

269/// Translates a direct-map virtual address into a physical address.
270///
271/// Returns [`None`] when the layout is unavailable or the address lies outside the direct map.
272pub fn direct_mapping_virt_to_phys(addr: VirtAddr) -> Option<PhysAddr> {
273    let addr_space = VIRTUAL_ADDRESS_SPACE.get()?;
274    if !addr_space.layout.direct_mapping_range.contains(addr) {
275        return None;
276    }
277
278    Some(PhysAddr::from_usize(
279        addr.as_usize() - addr_space.layout.direct_mapping_range.start.as_usize(),
280    ))
281}

283/// Translates a physical address into an active direct-map virtual address.
284///
285/// Returns [`None`] before the final roots are active or when the offset exceeds the reservation.
286pub fn direct_mapping_phys_to_virt(addr: PhysAddr) -> Option<VirtAddr> {
287    if !PAGE_TABLE_ACTIVE.load(Ordering::Acquire) {
288        return None;
289    }
290
291    let addr_space = VIRTUAL_ADDRESS_SPACE.get()?;
292    let offset = addr.as_usize();
293    let start = addr_space.layout.direct_mapping_range.start.as_usize();
294    let virt = VirtAddr::from_usize(start.checked_add(offset)?);
295    addr_space
296        .layout
297        .direct_mapping_range
298        .contains(virt)
299        .then_some(virt)
300}

302/// Runs a closure with exclusive access to the kernel page-table owner.
303///
304/// Operations use the allocator captured by the owner's descriptor. The page-table slot must
305/// already be initialized and cannot be observed while allocator handoff holds the same lock.
306#[inline]
307pub fn with_page_table<F, T>(f: F) -> T
308where
309    F: FnOnce(&mut OpaquePageTable<VirtAddr>) -> T,
310{
311    let mut pt = VIRTUAL_ADDRESS_SPACE.page_table_mutex.lock();
312    f(&mut pt)
313}

315/// Completes local cleanup and preserves its required remote invalidation scope.
316///
317/// The caller releases the page-table lock before calling this helper and supplies a request
318/// covering every changed base page. A full cleanup request also widens remote invalidation,
319/// including changes to huge mappings or intermediate descriptors. The returned request must
320/// complete on all captured remote CPUs before mappings are published or data frames are reused.
321pub(crate) fn finish_local_cleanup(
322    cleanup: PendingCleanup<VirtAddr>,
323    request: TlbInvalidation,
324) -> TlbInvalidation {
325    // SAFETY: Process every request locally through its captured strategy, then return the
326    // required remote scope to the mapping transaction. No pending invalidation is discarded.
327    let (mut flushes, flusher) = unsafe { cleanup.into_inner() };
328    let request = if matches!(flushes, TlbFlushes::Full) {
329        TlbInvalidation::All
330    } else {
331        request
332    };
333    flushes.flush_with(&flusher);
334    request
335}

352/// Initializes the virtual address space.
353///
354/// Detects the translation mode and calculates the kernel layout without allocating table pages.
355/// The initial opaque page table type captures the early allocator `AL` and [`LocalTlbFlusher`]
356/// for cleanup on the calling processor.
357pub(super) fn init_vmm_layout<AL: PageAllocator>() {
358    kprintln!("Virtual address space:");
359
360    let mode = select_va_mode(exarch::mem::virt_addr_space_modes());
361    exarch::mem::set_virt_addr_space_mode(mode);
362    let layout = calculate_vmm_layout(mode);
363
364    VIRTUAL_ADDRESS_SPACE.init_once(VirtualAddressSpace {
365        layout,
366        page_table_type_mutex: SpinNoIrq::new(exarch::mem::get_page_table_type::<
367            AL,
368            LocalTlbFlusher,
369        >(mode)),
370        page_table_mutex: SpinNoIrq::new(OpaquePageTable::dummy()),
371    });
372}

374/// Selects the virtual address space mode to use.
375///
376/// Currently, we require the virtual address space mode to cover both halves
377/// symmetrically, and we prefer those with the largest upper virtual address
378/// bits, and among which we prefer those with the largest page size.
379fn select_va_mode(va_modes: VirtAddrSpaceModes) -> VirtAddrSpaceMode {
380    kprintln!("  Platform virtual address space modes:");
381    for mode in &va_modes.modes {
382        kprintln!("    {}", *mode);
383    }
384
385    kprintln!(
386        "  Current virtual address space mode:\n    {}",
387        va_modes.modes[va_modes.current_index]
388    );
389
390    // Prefer an address-space mode that covers both halves. AArch64 exposes
391    // these as independent TTBR0 and TTBR1 regimes, while x86-64 and RISC-V
392    // expose a unified root.
393    let mut chosen_index = None::<usize>;
394    for (index, mode) in va_modes.modes.iter().enumerate() {
395        if matches!(
396            mode,
397            VirtAddrSpaceMode::Unified(..) | VirtAddrSpaceMode::Independent { .. }
398        ) && mode.upper_va_bits() == mode.lower_va_bits()
399            && mode.upper_page_shift() == mode.lower_page_shift()
400            && chosen_index
401                .map(|index| {
402                    let old_mode = va_modes.modes[index];
403
404                    old_mode.upper_va_bits() < mode.upper_va_bits()
405                        || (old_mode.upper_va_bits() == mode.upper_va_bits()
406                            && old_mode.upper_page_shift() < mode.upper_page_shift())
407                })
408                .unwrap_or(true)
409        {
410            chosen_index = Some(index);
411        }
412    }
413
414    let chosen_index = chosen_index.expect("No good virtual address space mode found");
415    let chosen_mode = va_modes.modes[chosen_index];
416
417    kprintln!(
418        "  Selected virtual address space mode:\n    {}",
419        chosen_mode
420    );
421
422    chosen_mode
423}

425/// Calculates the layout of the virtual address space based on the given mode.
426///
427/// Reserves the first half of the upper address range for the direct map and the next quarter for
428/// vmalloc, leaving the final quarter unused.
429fn calculate_vmm_layout(mode: VirtAddrSpaceMode) -> VirtualAddressSpaceLayout {
430    let upper_bits = mode.upper_va_bits().unwrap().get();
431    let page_shift = mode.upper_page_shift().unwrap().get();
432    let upper_start = va!((1usize << upper_bits).wrapping_neg());
433
434    kprintln!("  Upper half start    : {:#x}", upper_start);
435
436    // Top level memory areas:
437    // - 1st half: direct mapping area
438    // - 2nd half:
439    //   - 3rd quarter: vmalloc area
440    //   - 4th quarter: not used
441    // This scheme gives us at least the same size of these areas as Linux does.
442    let quarter_size = 1usize << (upper_bits - 2);
443    let direct_mapping_range = VirtAddrRange::from_start_size(upper_start, quarter_size * 2);
444    let vmalloc_range =
445        VirtAddrRange::from_start_size(upper_start + quarter_size * 2, quarter_size);
446    kprintln!(
447        "  Direct mapping area : {:x}, {}",
448        direct_mapping_range,
449        direct_mapping_range.size().size_display_wide()
450    );
451    kprintln!(
452        "  Vmalloc area        : {:x}, {}",
453        vmalloc_range,
454        vmalloc_range.size().size_display_wide()
455    );
456
457    VirtualAddressSpaceLayout {
458        mode,
459        page_shift,
460        direct_mapping_range,
461        vmalloc_range,
462    }
463}

465/// Initializes the page table with identity mappings and the physical-memory direct map.
466pub(super) fn init_vmm_mapping_early(phys_mem_regions: &MemoryRegions) {
467    let virt_phys_offset = direct_mapping_offset();
468    let mut pt = VIRTUAL_ADDRESS_SPACE
469        .page_table_type_mutex
470        .lock()
471        .new_pagetable_alloc()
472        .unwrap();
473
474    // TODO: the physical memory regions are not guaranteed to be page-aligned,
475    // that may cause issues when mapping them. Luckily, it's not a serious or
476    // urgent issue now, because free memory regions and kernel sections are
477    // almost guaranteed to be page-aligned. We should fix this in the future.
478    for region in phys_mem_regions {
479        let mapping_flags = region_flags_to_mapping(region.flags);
480        if mapping_flags.is_empty() {
481            continue;
482        }
483
484        let paddr = region.range.start;
485        let vaddr_low = VirtAddr::from_usize(paddr.as_usize());
486        let vaddr_high = vaddr_low + virt_phys_offset;
487        let size = region.range.size();
488
489        // The high direct map is needed for all physical memory. The low
490        // identity map only needs to cover code/data used before the jump and
491        // pages allocated for the bootstrap page tables.
492        let early_range = early::phys_addr_range();
493        let needs_full_identity = region.flags.intersects(
494            MemoryRegionFlags::KERNEL
495                | MemoryRegionFlags::BOOT_SERVICE
496                | MemoryRegionFlags::EXECUTE,
497        );
498        if needs_full_identity {
499            pt.map(vaddr_low, paddr, size, mapping_flags).unwrap();
500        } else if region.range.start < early_range.end && early_range.start < region.range.end {
501            let identity_start = region.range.start.max(early_range.start);
502            let identity_end = region.range.end.min(early_range.end);
503            let identity_size = identity_end.as_usize() - identity_start.as_usize();
504            pt.map(
505                VirtAddr::from_usize(identity_start.as_usize()),
506                identity_start,
507                identity_size,
508                mapping_flags,
509            )
510            .unwrap();
511        }
512        pt.map(vaddr_high, paddr, size, mapping_flags).unwrap();
513    }
514
515    *VIRTUAL_ADDRESS_SPACE.page_table_mutex.lock() = pt;
516}

518/// Transfers the active page-table owner to the formal allocator.
519///
520/// Refreshes relocated function pointers and transfers the early owner's roots to the formal
521/// allocator without freeing them or calling old method pointers. Before this handoff, the formal
522/// allocator must have registered all early page-table allocations so it can release them.
523pub(super) fn use_formal_page_table_allocator() {
524    // Fix function addresses in the opaque page table type first.
525    let page_table_type = exarch::mem::get_page_table_type::<FormalPageAllocator, LocalTlbFlusher>(
526        VIRTUAL_ADDRESS_SPACE.layout.mode,
527    );
528    *VIRTUAL_ADDRESS_SPACE.page_table_type_mutex.lock() = page_table_type.clone();
529
530    let mut page_table_guard = VIRTUAL_ADDRESS_SPACE.page_table_mutex.lock();
531
532    let root = mem::replace(page_table_guard.deref_mut(), OpaquePageTable::dummy()).leak();
533
534    // SAFETY: leak transfers the existing roots and all their table pages out of the only owner.
535    // The refreshed descriptor has the same geometry and can access and release those pages.
536    *page_table_guard = unsafe { page_table_type.new_pagetable_at(root) };
537}

539/// Removes temporary low-half identity mappings from every ready CPU's active view.
540///
541/// Failure is fatal after local mutation and never restores an inconsistent local mapping.
542pub(super) fn remove_identical_mapping(phys_mem_regions: &MemoryRegions) {
543    info!("Removing identical mappings...");
544    let transaction = MappingTransaction::begin().expect("identity mapping transaction rejected");
545
546    let cleanup = with_page_table(|pt| {
547        pt.apply_actions_with_cleanup(phys_mem_regions.iter().map(|region| {
548            let vaddr_low = va!(region.range.start.as_usize());
549            let size = region.range.size();
550            info!(
551                "identity-map clearing range={:#x}",
552                VirtAddrRange::from_start_size(vaddr_low, size)
553            );
554            PageTableAction::Unmap {
555                vaddr: vaddr_low,
556                size,
557            }
558        }))
559        .expect("identity mapping mutation failed")
560    });
561    // The page-table guard is released before cleanup and remote completion.
562    let request = finish_local_cleanup(cleanup, TlbInvalidation::All);
563    transaction.complete(request).unwrap_or_else(|error| {
564        panic!(
565            "identity-map completion failed ranges={phys_mem_regions:?} targets={:?}: {error:?}",
566            transaction.targets()
567        );
568    });
569    info!(
570        "identity-map completion targets={:?} operation=All",
571        transaction.targets()
572    );
573}

575/// The kernel's formal page allocator for page-table storage.
576///
577/// Uses the initialized buddy allocator and the high physical-memory direct map. The buddy
578/// allocator adopts early allocations before this type becomes the active descriptor's allocator.
579pub struct FormalPageAllocator;
580
581impl PageAllocator for FormalPageAllocator {
582    /// Returns the logarithm of the kernel base-page size in bytes.
583    ///
584    /// Page-table allocations use the same granularity as the physical allocator.
585    fn page_size_shift() -> usize {
586        page_size_shift()
587    }
588
589    /// Allocates one physical frame for page-table storage.
590    ///
591    /// Returns [`None`] when the buddy allocator cannot satisfy the request.
592    fn alloc_frame() -> Option<PhysAddr> {
593        allocs::palloc::alloc_frame().ok()
594    }
595
596    /// Allocates contiguous physical frames for page-table storage.
597    ///
598    /// The allocation is aligned to at least one kernel base page.
599    fn alloc_frames(page_count: usize) -> Option<PhysAddr> {
600        allocs::palloc::alloc_frames(page_count, 1 << page_size_shift()).ok()
601    }
602
603    /// Releases one physical page-table frame to the buddy allocator.
604    ///
605    /// The frame must belong to this allocator, including registered early allocations.
606    fn dealloc_frame(addr: PhysAddr) {
607        allocs::palloc::dealloc_frame(addr).unwrap();
608    }
609
610    /// Releases contiguous physical page-table frames to the buddy allocator.
611    ///
612    /// The address and count must match an allocation adopted or created by this allocator.
613    fn dealloc_frames(addr: PhysAddr, page_count: usize) {
614        allocs::palloc::dealloc_frames(addr, page_count).unwrap();
615    }
616
617    /// Translates a physical page-table address through the kernel direct map.
618    ///
619    /// The final page-table roots must be active before this allocator accesses table storage.
620    fn phys_to_virt(addr: PhysAddr) -> VirtAddr {
621        VirtAddr::from_usize(addr.as_usize() + direct_mapping_offset())
622    }
