37/// The serialization lock for kernel mapping changes and delayed reuse.
38///
39/// The request payload has one writer throughout mutation, remote completion and finalization.
40static MAPPING_TRANSACTION: SpinNoPreempt<()> = SpinNoPreempt::new(());
41
42// Lock order:
43// Allocate the MP online/IPI-ready snapshot before acquiring the mapping transaction.
44// mapping transaction (preemption disabled, interrupts enabled)
45//   -> vmalloc metadata lock, only for prepare/finalize phases
46//   -> page-table SpinNoIrq lock, only for PTE mutation and error-path local cleanup
47//   -> release every IRQ-disabling and allocator lock
48//   -> finish pending local cleanup, then publish/send/wait for remote acknowledgements
49//   -> reacquire vmalloc metadata briefly to finalize or quarantine
50//   -> release physical frames outside vmalloc and page-table locks.
51// No scheduler queue lock crosses this boundary. The IPI handler takes none of these locks.
52// A future lock-free mapping/request structure may replace this serialized design after a complete
53// memory-ordering proof. Early not-ready CPUs may add fresh mappings locally, but cannot remove or
54// replace mappings without completing the normal remote protocol.
55
56/// An owned kernel mapping transaction and its stable remote target set.
57///
58/// Only this preemption guard survives the remote completion phase.
59pub(crate) struct MappingTransaction {
60    /// The MP-owned snapshot excluding the pinned caller.
61    ///
62    /// Allocation precedes acquisition of the serialization lock.
63    targets: Vec<LogicalCpuId>,
64    /// The serialization and CPU-pinning guard.
65    ///
66    /// It leaves interrupts enabled throughout remote completion.
67    _guard: SpinNoPreemptGuard<'static, ()>,
68}
69
70impl MappingTransaction {
71    /// Captures ready CPUs and serializes mutation without disabling interrupts.
72    ///
73    /// A ready CPU cannot spin on this lock with interrupts disabled while another owner awaits it.
74    pub(crate) fn begin() -> Result<Self, TlbShootdownError> {
75        let snapshot = mp::ipi_ready_cpu_ids();
76        let pin = kernel_guard::NoPreempt::new();
77        if mp::is_ipi_ready(mp::current_cpu_id()) && !exarch::trap::local_enabled() {
78            return Err(TlbShootdownError::Ipi(
79                crate::ipi::IpiError::InterruptsDisabled,
80            ));
81        }
82        let targets = tlb::remote_targets(snapshot, mp::current_cpu_id());
83        let guard = MAPPING_TRANSACTION.lock();
84        drop(pin);
85        Ok(Self {
86            targets,
87            _guard: guard,
88        })
89    }
90
91    /// Completes remote invalidation after the local page-table guard has been released.
92    ///
93    /// The same snapshot is used for every mutation in this transaction.
94    pub(crate) fn complete(&self, request: TlbInvalidation) -> Result<(), TlbShootdownError> {
95        tlb::shootdown(&self.targets, request)
96    }
97
98    /// Returns the captured remote logical destinations for diagnostics.
99    ///
100    /// The returned slice does not create a second CPU inventory.
101    pub(crate) fn targets(&self) -> &[LogicalCpuId] {
102        &self.targets
103    }
104}

207/// Computes the physical address corresponding to a direct-map virtual address.
208///
209/// Checks only whether the address lies in the configured direct-map window. It does not verify
210/// that a page-table mapping exists for the address, so callers must only pass mapped addresses.
211///
212/// Returns [`None`] when the layout is unavailable or the address lies outside the direct map.
213pub fn direct_mapping_virt_to_phys(addr: VirtAddr) -> Option<PhysAddr> {
214    let addr_space = VIRTUAL_ADDRESS_SPACE.get()?;
215    if !addr_space.layout.direct_mapping_range.contains(addr) {
216        return None;
217    }
218
219    Some(PhysAddr::from_usize(
220        addr.as_usize() - addr_space.layout.direct_mapping_range.start.as_usize(),
221    ))
222}
223
224/// Computes the direct-map virtual address corresponding to a physical address.
225///
226/// Checks that the final roots are active and that the address fits in the reserved direct-map
227/// window. It does not check whether that physical address is mapped there. NO_MAP regions
228/// intentionally have no direct-map mapping.
229///
230/// Returns [`None`] before the final roots are active or when the offset exceeds the reservation.
231pub fn direct_mapping_phys_to_virt(addr: PhysAddr) -> Option<VirtAddr> {
232    if !PAGE_TABLE_ACTIVE.load(Ordering::Acquire) {
233        return None;
234    }
235
236    let addr_space = VIRTUAL_ADDRESS_SPACE.get()?;
237    let offset = addr.as_usize();
238    let start = addr_space.layout.direct_mapping_range.start.as_usize();
239    let virt = VirtAddr::from_usize(start.checked_add(offset)?);
240    addr_space
241        .layout
242        .direct_mapping_range
243        .contains(virt)
244        .then_some(virt)
245}

247/// Runs a closure with exclusive access to the kernel page-table owner.
248///
249/// Operations use the allocator captured by the owner's descriptor. The page-table slot must
250/// already be initialized and cannot be observed while allocator handoff holds the same lock.
251#[inline]
252pub fn with_page_table<F, T>(f: F) -> T
253where
254    F: FnOnce(&mut OpaquePageTable<VirtAddr>) -> T,
255{
256    let mut pt = VIRTUAL_ADDRESS_SPACE.page_table_mutex.lock();
257    f(&mut pt)
258}

260/// Completes local cleanup and preserves its required remote invalidation scope.
261///
262/// The caller releases the page-table lock before calling this helper and supplies a request
263/// covering every changed base page. A full cleanup request also widens remote invalidation,
264/// including changes to huge mappings or intermediate descriptors. The returned request must
265/// complete on all captured remote CPUs before mappings are published or data frames are reused.
266pub(crate) fn finish_local_cleanup(
267    cleanup: PendingCleanup<VirtAddr>,
268    request: TlbInvalidation,
269) -> TlbInvalidation {
270    // SAFETY: Process every request locally through its captured strategy, then return the
271    // required remote scope to the mapping transaction. No pending invalidation is discarded.
272    let (mut flushes, flusher) = unsafe { cleanup.into_inner() };
273    let request = if matches!(flushes, TlbFlushes::Full) {
274        TlbInvalidation::All
275    } else {
276        request
277    };
278    flushes.flush_with(&flusher);
279    request
280}

297/// Initializes the virtual address space.
298///
299/// Detects the translation mode and calculates the kernel layout without allocating table pages.
300/// The initial opaque page table type captures the early allocator `AL` and [`LocalTlbFlusher`]
301/// for cleanup on the calling processor.
302pub(super) fn init_vmm_layout<AL: PageAllocator>() {
303    kprintln!("Virtual address space:");
304
305    let mode = select_va_mode(exarch::mem::virt_addr_space_modes());
306    exarch::mem::set_virt_addr_space_mode(mode);
307    let layout = calculate_vmm_layout(mode);
308
309    VIRTUAL_ADDRESS_SPACE.init_once(VirtualAddressSpace {
310        layout,
311        page_table_type_mutex: SpinNoIrq::new(exarch::mem::get_page_table_type::<
312            AL,
313            LocalTlbFlusher,
314        >(mode)),
315        page_table_mutex: SpinNoIrq::new(OpaquePageTable::dummy()),
316    });
317}

319/// Selects the virtual address space mode to use.
320///
321/// Currently, we require the virtual address space mode to cover both halves
322/// symmetrically, and we prefer those with the largest upper virtual address
323/// bits, and among which we prefer those with the largest page size.
324fn select_va_mode(va_modes: VirtAddrSpaceModes) -> VirtAddrSpaceMode {
325    kprintln!("  Platform virtual address space modes:");
326    for mode in &va_modes.modes {
327        kprintln!("    {}", *mode);
328    }
329
330    kprintln!(
331        "  Current virtual address space mode:\n    {}",
332        va_modes.modes[va_modes.current_index]
333    );
334
335    // Prefer an address-space mode that covers both halves. AArch64 exposes
336    // these as independent TTBR0 and TTBR1 regimes, while x86-64 and RISC-V
337    // expose a unified root.
338    let mut chosen_index = None::<usize>;
339    for (index, mode) in va_modes.modes.iter().enumerate() {
340        if matches!(
341            mode,
342            VirtAddrSpaceMode::Unified(..) | VirtAddrSpaceMode::Independent { .. }
343        ) && mode.upper_va_bits() == mode.lower_va_bits()
344            && mode.upper_page_shift() == mode.lower_page_shift()
345            && chosen_index
346                .map(|index| {
347                    let old_mode = va_modes.modes[index];
348
349                    old_mode.upper_va_bits() < mode.upper_va_bits()
350                        || (old_mode.upper_va_bits() == mode.upper_va_bits()
351                            && old_mode.upper_page_shift() < mode.upper_page_shift())
352                })
353                .unwrap_or(true)
354        {
355            chosen_index = Some(index);
356        }
357    }
358
359    let chosen_index = chosen_index.expect("No good virtual address space mode found");
360    let chosen_mode = va_modes.modes[chosen_index];
361
362    kprintln!(
363        "  Selected virtual address space mode:\n    {}",
364        chosen_mode
365    );
366
367    chosen_mode
368}

370/// Calculates the layout of the virtual address space based on the given mode.
371///
372/// Reserves the first half of the upper address range for the direct map and the next quarter for
373/// vmalloc, leaving the final quarter unused.
374fn calculate_vmm_layout(mode: VirtAddrSpaceMode) -> VirtualAddressSpaceLayout {
375    let upper_bits = mode.upper_va_bits().unwrap().get();
376    let page_shift = mode.upper_page_shift().unwrap().get();
377    let upper_start = va!((1usize << upper_bits).wrapping_neg());
378
379    kprintln!("  Upper half start    : {:#x}", upper_start);
380
381    // Top level memory areas:
382    // - 1st half: direct mapping area
383    // - 2nd half:
384    //   - 3rd quarter: vmalloc area
385    //   - 4th quarter: not used
386    // This scheme gives us at least the same size of these areas as Linux does.
387    let quarter_size = 1usize << (upper_bits - 2);
388    let direct_mapping_range = VirtAddrRange::from_start_size(upper_start, quarter_size * 2);
389    let vmalloc_range =
390        VirtAddrRange::from_start_size(upper_start + quarter_size * 2, quarter_size);
391    kprintln!(
392        "  Direct mapping area : {:x}, {}",
393        direct_mapping_range,
394        direct_mapping_range.size().size_display_wide()
395    );
396    kprintln!(
397        "  Vmalloc area        : {:x}, {}",
398        vmalloc_range,
399        vmalloc_range.size().size_display_wide()
400    );
401
402    VirtualAddressSpaceLayout {
403        mode,
404        page_shift,
405        direct_mapping_range,
406        vmalloc_range,
407    }
408}

410/// Transfers the active page-table owner to the formal allocator.
411///
412/// Refreshes relocated function pointers and transfers the early owner's roots to the formal
413/// allocator without freeing them or calling old method pointers. Before this handoff, the formal
414/// allocator must have registered all early page-table allocations so it can release them.
415pub(super) fn use_formal_page_table_allocator() {
416    // Fix function addresses in the opaque page table type first.
417    let page_table_type = exarch::mem::get_page_table_type::<FormalPageAllocator, LocalTlbFlusher>(
418        VIRTUAL_ADDRESS_SPACE.layout.mode,
419    );
420    *VIRTUAL_ADDRESS_SPACE.page_table_type_mutex.lock() = page_table_type.clone();
421
422    let mut page_table_guard = VIRTUAL_ADDRESS_SPACE.page_table_mutex.lock();
423
424    let root = mem::replace(page_table_guard.deref_mut(), OpaquePageTable::dummy()).leak();
425
426    // SAFETY: leak transfers the existing roots and all their table pages out of the only owner.
427    // The refreshed descriptor has the same geometry and can access and release those pages.
428    *page_table_guard = unsafe { page_table_type.new_pagetable_at(root) };
429}

431/// The kernel's formal page allocator for page-table storage.
432///
433/// Uses the initialized buddy allocator and the high physical-memory direct map. The buddy
434/// allocator adopts early allocations before this type becomes the active descriptor's allocator.
435pub struct FormalPageAllocator;
436
437impl PageAllocator for FormalPageAllocator {
438    /// Returns the logarithm of the kernel base-page size in bytes.
439    ///
440    /// Page-table allocations use the same granularity as the physical allocator.
441    fn page_size_shift() -> usize {
442        page_size_shift()
443    }
444
445    /// Allocates one physical frame for page-table storage.
446    ///
447    /// Returns [`None`] when the buddy allocator cannot satisfy the request.
448    fn alloc_frame() -> Option<PhysAddr> {
449        allocs::palloc::alloc_frame().ok()
450    }
451
452    /// Allocates contiguous physical frames for page-table storage.
453    ///
454    /// The allocation is aligned to at least one kernel base page.
455    fn alloc_frames(page_count: usize) -> Option<PhysAddr> {
456        allocs::palloc::alloc_frames(page_count, 1 << page_size_shift()).ok()
457    }
458
459    /// Releases one physical page-table frame to the buddy allocator.
460    ///
461    /// The frame must belong to this allocator, including registered early allocations.
462    fn dealloc_frame(addr: PhysAddr) {
463        allocs::palloc::dealloc_frame(addr).unwrap();
464    }
465
466    /// Releases contiguous physical page-table frames to the buddy allocator.
467    ///
468    /// The address and count must match an allocation adopted or created by this allocator.
469    fn dealloc_frames(addr: PhysAddr, page_count: usize) {
470        allocs::palloc::dealloc_frames(addr, page_count).unwrap();
471    }
472
473    /// Translates a physical page-table address through the kernel direct map.
474    ///
475    /// The final page-table roots must be active before this allocator accesses table storage.
476    fn phys_to_virt(addr: PhysAddr) -> VirtAddr {
477        VirtAddr::from_usize(addr.as_usize() + direct_mapping_offset())
478    }
479}

481/// Deferred page-table cleanup and remote invalidation tests.
482///
483/// Recording local invalidations avoids privileged instructions on the host.
484#[cfg(test)]
485mod tests {
486    use core::cell::RefCell;
487
488    use expt::pt::cleanup::{PendingCleanup, TlbFlush, TlbFlusher};
489    use memory_addr::{VirtAddr, VirtAddrRange, va};
490
491    use crate::tlb::TlbInvalidation;
492
493    use super::finish_local_cleanup;
494
495    std::thread_local! {
496        /// The local invalidations completed by the calling test thread.
497        ///
498        /// Each test starts with an independent empty recording.
499        static FLUSHES: RefCell<Vec<Option<VirtAddr>>> = const { RefCell::new(Vec::new()) };
500    }
501
502    /// A local invalidation recorder.
503    ///
504    /// It models the captured strategy without executing architecture instructions.
505    struct RecordingFlusher;
506
507    /// The host recording strategy implementation.
508    ///
509    /// Cleanup completes by storing observations in thread-local test state.
510    impl TlbFlusher<VirtAddr> for RecordingFlusher {
511        /// Records one completed local invalidation.
512        ///
513        /// The returned remote request must be inspected only after this record exists.
514        fn flush(address: Option<VirtAddr>) {
515            FLUSHES.with_borrow_mut(|flushes| flushes.push(address));
516        }
517    }
518
519    /// Completes local pages before returning the requested remote range.
520    ///
521    /// Batching must retain every pending address without widening a sufficient range.
522    #[test]
523    fn finishes_local_pages_before_remote_completion() {
524        let mut cleanup = PendingCleanup::new_with::<RecordingFlusher>();
525        cleanup.push_flush(TlbFlush::Page(va!(0x1000)));
526        cleanup.push_flush(TlbFlush::Page(va!(0x2000)));
527        let request = TlbInvalidation::Range(VirtAddrRange::new(va!(0x1000), va!(0x3000)));
528        let remote = finish_local_cleanup(cleanup, request);
529        FLUSHES.with_borrow(|flushes| {
530            assert_eq!(flushes.as_slice(), &[Some(va!(0x1000)), Some(va!(0x2000))]);
531        });
532        assert_eq!(remote, request);
533    }
534
535    /// Widens remote invalidation when a page-table mutation requires a full flush.
536    ///
537    /// Huge-leaf splits and accumulated flush overflow cannot be completed by one remote page.
538    #[test]
539    fn full_cleanup_requires_full_remote_invalidation() {
540        let mut cleanup = PendingCleanup::new_with::<RecordingFlusher>();
541        cleanup.push_flush(TlbFlush::Page(va!(0x1000)));
542        cleanup.push_flush(TlbFlush::Full);
543        let remote = finish_local_cleanup(cleanup, TlbInvalidation::Page(va!(0x1000)));
544        FLUSHES.with_borrow(|flushes| assert_eq!(flushes.as_slice(), &[None]));
545        assert_eq!(remote, TlbInvalidation::All);
546    }
