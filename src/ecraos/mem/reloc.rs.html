5/// Gets the runtime (and current) address of a symbol, with
6/// architecture-specific PC-relative addressing instructions.
7///
8/// It's safe to use this macro even when relocation is not performed.
9#[macro_export]
10macro_rules! get_symbol_addr {
11    ($symbol:expr) => {{
12        let addr: usize;
13        #[cfg(target_arch = "x86_64")]
14        core::arch::asm!(
15            "lea {result}, [rip + {symbol}]",
16            result = out(reg) addr,
17            symbol = sym $symbol,
18            options(nostack, preserves_flags, readonly)
19        );
20
21        #[cfg(target_arch = "aarch64")]
22        core::arch::asm!(
23            "adrp {result}, {symbol}",
24            "add {result}, {result}, :lo12:{symbol}",
25            result = out(reg) addr,
26            symbol = sym $symbol,
27            options(nostack, preserves_flags, readonly)
28        );
29
30        #[cfg(target_arch = "riscv64")]
31        core::arch::asm!(
32            "lla {result}, {symbol}",
33            result = out(reg) addr,
34            symbol = sym $symbol,
35            options(nostack, preserves_flags, readonly)
36        );
37
38        #[cfg(target_arch = "loongarch64")]
39        core::arch::asm!(
40            "la.pcrel {result}, {symbol}",
41            result = out(reg) addr,
42            symbol = sym $symbol,
43            options(nostack, preserves_flags, readonly)
44        );
45
46        addr
47    }};
48}

50/// A relocation entry in the `.rela.dyn` section in an ELF64 file.
51#[repr(C)]
52struct Elf64Rela {
53    /// The virtual address (VMA, during linking) of the place to be patched.
54    offset: u64,
55    /// The symbol table index and the relocation type.
56    info: u64,
57    /// The constant addend to calculate the value to be patched.
58    addend: i64,
59}
60
61/// The value of R_<current_arch>_RELATIVE relocation type.
62const RELATIVE_TYPE: u32 = {
63    cfg_select! {
64        target_arch = "x86_64" => 8, // R_X86_64_RELATIVE
65        target_arch = "aarch64" => 1027, // R_AARCH64_RELATIVE
66        target_arch = "riscv64" => 3, // R_RISCV_RELATIVE
67        target_arch = "loongarch64" => 3, // R_LARCH_RELATIVE
68        _ => panic!("Unsupported target architecture"),
69    }
70};

72/// Handles the relocation of the kernel, by patching the kernel according to
73/// the `R_<current_arch>_RELATIVE` entries from the `.rela.dyn` section.
74///
75/// # Notes
76///
77/// ## VMA Assumption
78///
79/// This function assumes that during linking, [`kernel_entry`] is placed at the
80/// very beginning of the kernel image, and at VMA 0.
81///
82/// The linker script of this crate should handle this correctly. Take care when
83/// updating it.
84///
85/// ## GOT Prevention
86///
87/// This function must be **naturally place-independent**, i.e. it must not
88/// access the GOT or PLT indirections. Updates to this function must be
89/// carefully scrutinized to ensure that this invariant is maintained.
90///
91/// Even the most trivial calls to core library functions can break this. For
92/// example, the following is a wrong implementation:
93///
94/// ```
95/// pub unsafe fn relocate_me() {
96///     unsafe extern "C" {
97///         fn _srela_dyn();
98///         fn _erela_dyn();
99///     }
100///
101///     // The kernel entry is at VMA 0, so the load bias is the actual address of
102///     // the kernel entry.
103///     let load_bias = unsafe { get_symbol_addr!(crate::kernel_entry) };
104///     let srela_dyn_addr = unsafe { get_symbol_addr!(_srela_dyn) };
105///     let erela_dyn_addr = unsafe { get_symbol_addr!(_erela_dyn) };
106///
107///     let rela_base = srela_dyn_addr as *const Elf64Rela;
108///     let rela_cnt = (erela_dyn_addr - srela_dyn_addr) / core::mem::size_of::<Elf64Rela>();
109///
110///     for i in 0..rela_cnt {
111///         let rela = unsafe { rela_base.add(i).as_ref_unchecked() };
112///         if rela.info as u32 == RELATIVE_TYPE {
113///             let target = (load_bias + rela.addend as usize) as *mut u64;
114///             unsafe {
115///                 target.write((load_bias as u64).wrapping_add_signed(rela.addend));
116///             }
117///         }
118///     }
119/// }
120/// ```
121#[inline(never)]
122pub unsafe fn relocate_me() {
123    unsafe extern "C" {
124        fn _srela_dyn();
125        fn _erela_dyn();
126    }
127
128    unsafe {
129        // `kernel_entry` is linked at VMA 0, the runtime base is where that
130        // address landed in memory.
131        let load_bias = get_symbol_addr!(kernel_entry);
132        let srela = get_symbol_addr!(_srela_dyn);
133        let erela = get_symbol_addr!(_erela_dyn);
134        let rela_stride = core::mem::size_of::<Elf64Rela>();
135
136        let mut p = srela;
137        while p < erela {
138            let rela_ptr = p as *const Elf64Rela;
139            let offset = (*rela_ptr).offset;
140            let info = (*rela_ptr).info;
141            let addend = (*rela_ptr).addend;
142
143            // ELF64 r_type is the low 32 bits of r_info.
144            if info as u32 == RELATIVE_TYPE {
145                let target = (load_bias + offset as usize) as *mut u64;
146                let value = (load_bias as i64).wrapping_add(addend) as u64;
147                target.write(value);
148            }
149
150            p += rela_stride;
151        }
152    }
