15/// The global buddy allocator.
16static BUDDY: SpinNoIrq<BuddyAllocator> = SpinNoIrq::new(BuddyAllocator::new());

18/// Initializes the page allocator.
19pub fn init_palloc() {
20    let phys_regions = pmm::phys_mem_regions();
21    let page_size_shift = vmm::page_size_shift();
22    let page_size = vmm::page_size();
23    let dmo = vmm::direct_mapping_offset();
24
25    info!("Initializing page allocator...");
26
27    let early_alloc_range = early::phys_addr_range();
28    let (early_alloc_bitmap, _, early_allocbase_paddr) = early::destroy_early_page_allocator();
29
30    debug!(
31        "Early page allocator at {:x} destroyed, {} pages allocated",
32        early_alloc_range.start,
33        early_alloc_bitmap.len()
34    );
35
36    let mut buddy = BUDDY.lock();
37    // SAFETY: we believe that we have mapped the physical memory to the virtual memory correctly.
38    unsafe { buddy.init(page_size_shift, dmo) }.expect("failed to init buddy allocator");
39
40    for region in phys_regions {
41        if !region.flags.contains(MemoryRegionFlags::FREE) {
42            continue;
43        }
44
45        let Some(range) = region.range.align_inwards(page_size) else {
46            continue;
47        };
48
49        if early_alloc_range.overlaps(range)
50            && buddy
51                .check_metadata_overlap(range, early_alloc_range)
52                .unwrap()
53        {
54            panic!(
55                "Buddy metadata overlaps early allocator for region {:x} ({})",
56                range, region.desc
57            );
58        }
59
60        debug!(
61            "Adding region {:x} ({}) to buddy allocator",
62            range, region.desc
63        );
64        // SAFETY: we trust exarch to provide a valid physical memory map for
65        // the active architecture.
66        unsafe { buddy.add_section(range) }
67            .expect("failed to add the previous region to buddy allocator");
68    }
69
70    assert!(
71        buddy.section_count() > 0,
72        "no usable memory region found for buddy allocator"
73    );
74
75    debug!("Marking early allocator's in-use pages as allocated in the buddy");
76    for index in &early_alloc_bitmap {
77        buddy
78            .alloc_blocks_at(early_allocbase_paddr + (index * page_size), 1)
79            .expect("failed to mark early allocator page as in-use");
80    }
81    drop(buddy);
82
83    info!(
84        "Page allocator initialized, {} pages allocatable in total",
85        stats().heap_pages()
86    );
87    print_buddy_stats("Initial buddy allocator stats");
88}

90/// Runs a function with the buddy allocator locked.
91pub fn with_allocator<F: FnOnce(&mut BuddyAllocator) -> R, R>(f: F) -> R {
92    let mut buddy = BUDDY.lock();
93    f(&mut buddy)
94}

96/// Allocates a single physical frame.
97pub fn alloc_frame() -> exbuddy::BuddyResult<PhysAddr> {
98    with_allocator(|b| b.alloc_frame())
99}
100
101/// Allocates exactly `frame_count` contiguous physical frames.
102///
103/// The returned frame range is aligned to at least `align` bytes. The same
104/// `frame_count` must be passed back to [`dealloc_frames`].
105pub fn alloc_frames(frame_count: usize, align: usize) -> exbuddy::BuddyResult<PhysAddr> {
106    let page_size = vmm::page_size();
107    with_allocator(|b| b.alloc_frames(frame_count, align.max(page_size)))
108}
109
110/// Allocates exactly `frame_count` physical frames at `addr`.
111///
112/// The same `addr` and `frame_count` must be passed back to
113/// [`dealloc_frames_at`].
114pub fn alloc_frames_at(addr: PhysAddr, frame_count: usize) -> exbuddy::BuddyResult<PhysAddr> {
115    with_allocator(|b| b.alloc_frames_at(addr, frame_count).map(|_| addr))
116}
117
118/// Deallocates a single physical frame allocated with [`alloc_frame`].
119pub fn dealloc_frame(addr: PhysAddr) -> exbuddy::BuddyResult {
120    with_allocator(|b| b.dealloc_frame(addr))
121}

123/// Deallocates an exact frame range allocated with [`alloc_frames`].
124///
125/// The `addr` and `frame_count` arguments must match the corresponding
126/// [`alloc_frames`] call exactly.
127pub fn dealloc_frames(addr: PhysAddr, frame_count: usize) -> exbuddy::BuddyResult {
128    with_allocator(|b| b.dealloc_frames(addr, frame_count))
129}

131/// Deallocates an exact frame range allocated with [`alloc_frames_at`].

139/// Returns whether the page containing `addr` is allocated.

144/// Returns aggregate buddy allocator statistics.
145pub fn stats() -> AllocatorStats {
146    with_allocator(|b| b.stats())
147}

149/// Prints aggregate buddy allocator statistics.
150pub fn print_buddy_stats(heading: &str) {
151    let stats = stats();
152    let page_size_shift = vmm::page_size_shift();
153    let total = stats.total_pages();
154    let meta = stats.meta_pages();
155    let heap = stats.heap_pages();
156    let used = stats.used_pages();
157    let free = stats.free_pages();
158    let unused = stats.unused_pages();
159
160    kprintln!("{heading}:");
161    kprintln!(
162        "  Total pages      : {: <10}({})",
163        total,
164        (total << page_size_shift).size_display_wide()
165    );
166    kprintln!(
167        "    Metadata pages : {: <10}({})",
168        meta,
169        (meta << page_size_shift).size_display_wide()
170    );
171    kprintln!(
172        "    Heap pages     : {: <10}({})",
173        heap,
174        (heap << page_size_shift).size_display_wide()
175    );
176    kprintln!(
177        "      Used pages   : {: <10}({})",
178        used,
179        (used << page_size_shift).size_display_wide()
180    );
181    kprintln!(
182        "      Free pages   : {: <10}({})",
183        free,
184        (free << page_size_shift).size_display_wide()
185    );
186    kprintln!(
187        "    Unused pages   : {: <10}({})",
188        unused,
189        (unused << page_size_shift).size_display_wide()
190    );
191}
