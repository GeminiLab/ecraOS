1//! Kernel heap allocator integration.
2//!
3//! Connects the reusable slab allocator core to ecraOS per-CPU state and the
4//! physical page allocator, then exposes the result through Rust's global
5//! allocator interface.
6//!
7//! Each CPU serializes local slab operations with its interrupt-disabling lock. Slab memory regions
8//! follow exslab's precomputed layouts and are aligned to their complete size. Remote frees
9//! publish to the owning slab directly and are drained by its owner on later allocation or
10//! deallocation. Owner lookup uses the original allocation layout without scanning memory.
11//! Local cache selection and owner routing disable interrupts and preemption together.
12

30/// The current CPU's slab allocator.
31#[def_percpu]
32static CURRENT_SLAB: LazyInit<PerCpuSlab> = LazyInit::new();
33
34/// A CPU's slab allocator state.
35///
36/// Couples the logical owner identifier with the lock serializing local allocation operations.
37struct PerCpuSlab {
38    /// Logical CPU id that owns this slab allocator.
39    cpu_id: u16,
40    /// Locked slab allocator core.
41    inner: SpinNoIrq<SlabAllocator>,
42}
43
44/// Locked allocation and slab registration operations for one CPU.
45impl PerCpuSlab {
46    /// Creates an empty per-CPU slab allocator wrapper.
47    fn new(cpu_id: u16, page_size: usize) -> Self {
48        Self {
49            cpu_id,
50            inner: SpinNoIrq::new(
51                SlabAllocator::new(page_size).expect("unsupported slab hardware page size"),
52            ),
53        }
54    }
55
56    /// Allocates one object from this CPU's slab allocator.
57    fn alloc(&self, layout: Layout) -> AllocResult<SlabAllocResult> {
58        self.inner.lock().alloc(layout)
59    }
60
61    /// Registers mapped slab memory with this CPU's allocator.
62    ///
63    /// # Safety
64    ///
65    /// `base` must identify exclusive mapped writable frames of the requested span, aligned to
66    /// that complete span. Slab memory must remain valid until returned or all slab users finish.
67    unsafe fn add_slab(&self, size_class: exslab::SizeClass, base: VirtAddr) -> AllocResult {
68        // SAFETY: The caller guarantees that the supplied frames are valid slab memory.
69        unsafe { self.inner.lock().add_slab(size_class, base, self.cpu_id) }
70    }
71
72    /// Deallocates a valid object on the owner CPU path.
73    ///
74    /// # Safety
75    ///
76    /// `ptr` must be a live allocation owned by this CPU, and `layout` must match its allocation.
77    unsafe fn dealloc_local(&self, ptr: NonNull<u8>, layout: Layout) -> SlabDeallocResult {
78        // SAFETY: The caller upholds the local deallocation contract.
79        unsafe { self.inner.lock().dealloc(ptr, layout) }
80    }
81}

83/// Runs a callback with the current CPU's slab allocator.
84///
85/// Pins the CPU before selecting its slot and prevents interrupt reentry while it is borrowed.
86fn with_current_slab<R>(f: impl FnOnce(&PerCpuSlab) -> R) -> R {
87    let _guard = kernel_guard::NoPreemptIrqSave::new();
88    CURRENT_SLAB.with_current(|slot| f(&*slot))
89}

91/// Initializes the current CPU's heap allocator state.
92pub fn init_malloc_current_cpu() {
93    let cpu_id = mp::current_cpu_id();
94    let slab_cpu_id = u16::try_from(cpu_id).expect("CPU id exceeds slab owner range");
95
96    CURRENT_SLAB.with_current(|slot| {
97        slot.init_once(PerCpuSlab::new(slab_cpu_id, mem::vmm::page_size()));
98    });
99    info!("Small object allocator initialized for CPU {}", cpu_id);
100}

102/// Allocates a small object from a slab.
103fn alloc_small(layout: Layout) -> *mut u8 {
104    let page_size = mem::vmm::page_size();
105
106    loop {
107        match with_current_slab(|slab| slab.alloc(layout)) {
108            Ok(SlabAllocResult::Allocated(ptr)) => return ptr.as_ptr(),
109            Ok(SlabAllocResult::NeedsSlab { size_class, pages }) => {
110                let span = pages * page_size;
111                let paddr = match palloc::alloc_frames(pages, span) {
112                    Ok(paddr) => paddr,
113                    Err(_) => return core::ptr::null_mut(),
114                };
115                let vaddr = va!(paddr.as_usize() + mem::vmm::direct_mapping_offset());
116                // SAFETY: palloc supplies the exact span. The direct-map offset preserves its
117                // alignment, and registration validates the resulting virtual base.
118                let add_result =
119                    with_current_slab(|slab| unsafe { slab.add_slab(size_class, vaddr) });
120                if add_result.is_err() {
121                    palloc::dealloc_frames(paddr, pages)
122                        .expect("failed to dealloc rejected slab frames");
123                    return core::ptr::null_mut();
124                }
125            }
126            Err(_) => return core::ptr::null_mut(),
127        }
128    }
129}

131/// Allocates a page-backed large object.
132fn alloc_large(layout: Layout) -> *mut u8 {
133    let page_size = mem::vmm::page_size();
134    let bytes = layout.size().max(layout.align());
135    let pages = bytes.div_ceil(page_size);
136    let align = layout.align().max(page_size);
137
138    match palloc::alloc_frames(pages, align) {
139        Ok(paddr) => va!(paddr.as_usize() + mem::vmm::direct_mapping_offset()).as_mut_ptr(),
140        Err(_) => core::ptr::null_mut(),
141    }
142}

144/// Deallocates a small object through its owning slab.
145///
146/// # Safety
147///
148/// `ptr` and `layout` must satisfy the GlobalAlloc deallocation contract for this allocator.
149unsafe fn dealloc_small(ptr: *mut u8, layout: Layout) {
150    let page_size = mem::vmm::page_size();
151    // SAFETY: GlobalAlloc requires a non-null live allocation with its original layout.
152    let nn_ptr = unsafe { NonNull::new_unchecked(ptr) };
153    // SAFETY: The live object and its matching layout keep the immutable slab owner readable.
154    let owner_cpu = unsafe { SlabAllocator::owner_cpu(nn_ptr, layout, page_size) }
155        .expect("invalid slab allocation geometry") as usize;
156
157    let local_result = with_current_slab(|slab| {
158        if owner_cpu == slab.cpu_id as usize {
159            // SAFETY: The guard pins this CPU from owner comparison through the locked release.
160            Some(unsafe { slab.dealloc_local(nn_ptr, layout) })
161        } else {
162            None
163        }
164    });
165    match local_result {
166        Some(SlabDeallocResult::Done) => {}
167        Some(SlabDeallocResult::FreeSlab { base, pages }) => {
168            let paddr = pa!(base.as_usize() - mem::vmm::direct_mapping_offset());
169            palloc::dealloc_frames(paddr, pages).expect("failed to dealloc frames");
170        }
171        None => {
172            // SAFETY: The original layout and page size identify the live slot's exact geometry.
173            unsafe { SlabAllocator::remote_free_object(nn_ptr, layout, page_size) }
174                .expect("invalid remote slab allocation geometry");
175        }
176    }
177}

179/// Deallocates a page-backed large object.
180fn dealloc_large(ptr: *mut u8, layout: Layout) {
181    let page_size = mem::vmm::page_size();
182    let bytes = layout.size().max(layout.align());
183    let pages = bytes.div_ceil(page_size);
184    let vaddr = VirtAddr::from_mut_ptr_of(ptr);
185    let paddr = pa!(vaddr.as_usize() - mem::vmm::direct_mapping_offset());
186    palloc::dealloc_frames(paddr, pages).expect("failed to dealloc frames");
187}

189/// The ecraOS global allocator.
190struct EcraosGlobalAlloc;
191
192/// Kernel global allocation behavior for the slab caches and page allocator.
193unsafe impl GlobalAlloc for EcraosGlobalAlloc {
194    /// Allocates an object from a slab or contiguous physical frames according to its layout.
195    ///
196    /// # Safety
197    ///
198    /// The caller must uphold the [`GlobalAlloc::alloc`] contract for the requested layout.
199    unsafe fn alloc(&self, layout: Layout) -> *mut u8 {
200        let effective_size = layout.size().max(layout.align());
201        if effective_size <= exslab::size_class::SLAB_MAX_SIZE
202            && layout.align() <= exslab::size_class::SLAB_MAX_SIZE
203        {
204            alloc_small(layout)
205        } else {
206            alloc_large(layout)
207        }
208    }
209
210    /// Releases an object through the same layout-selected path used during allocation.
211    ///
212    /// # Safety
213    ///
214    /// The caller must uphold the [`GlobalAlloc::dealloc`] contract for the pointer and layout.
215    unsafe fn dealloc(&self, ptr: *mut u8, layout: Layout) {
216        let effective_size = layout.size().max(layout.align());
217        if effective_size <= exslab::size_class::SLAB_MAX_SIZE
218            && layout.align() <= exslab::size_class::SLAB_MAX_SIZE
219        {
220            // SAFETY: The GlobalAlloc caller must provide a live pointer and matching layout.
221            unsafe { dealloc_small(ptr, layout) }
222        } else {
223            dealloc_large(ptr, layout)
224        }
225    }
226}
