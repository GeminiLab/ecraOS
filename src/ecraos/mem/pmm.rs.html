10/// The maximum number of memory regions finally built.
11pub const MAX_MEMORY_REGIONS: usize = 64;
12/// The final physical memory regions.
13pub type MemoryRegions = HeaplessVec<MemoryRegion, MAX_MEMORY_REGIONS>;
14
15/// Static storage for the physical memory region table.
16///
17/// Written once by [`build_phys_mem_regions`] and read back via
18/// [`phys_mem_regions`].
19static PHYS_MEM_REGIONS: LazyInit<MemoryRegions> = LazyInit::new();
20
21/// Retrieves the physical memory regions.
22///
23/// # Panics
24///
25/// Panics if the physical memory regions have not been initialized yet.
26pub fn phys_mem_regions() -> &'static MemoryRegions {
27    PHYS_MEM_REGIONS.as_ref()
28}
29
30/// Retrieves the physical memory regions if they have been initialized.
31pub fn try_phys_mem_regions() -> Option<&'static MemoryRegions> {
32    PHYS_MEM_REGIONS.get()
33}

35/// Pushes a physical memory region into a [`MemoryRegions`], panicking if the
36/// region overlaps with any existing region.
37///
38/// # Panics
39///
40/// Panics if the region overlaps with any existing region or if the number of
41/// regions exceeds the maximum.
42fn push_phys_mem_region_no_overlap(target: &mut MemoryRegions, region: MemoryRegion) {
43    for other in target.iter() {
44        if region.range.overlaps(other.range) {
45            panic!("physical memory region {region:#x?} overlaps with {other:#x?}");
46        }
47    }
48
49    target.push(region).expect("too many memory regions");
50}

52const MAX_SPLIT_SUB_REGIONS: usize = 8;
53
54fn subtract_overlaps<I>(
55    range: PhysAddrRange,
56    occupied: I,
57) -> HeaplessVec<PhysAddrRange, MAX_SPLIT_SUB_REGIONS>
58where
59    I: Iterator<Item = PhysAddrRange>,
60{
61    let mut remaining = HeaplessVec::new();
62    if !range.is_empty() {
63        remaining
64            .push(range)
65            .expect("failed to initialize physical range subtraction");
66    }
67
68    for other in occupied {
69        let current = core::mem::replace(&mut remaining, HeaplessVec::new());
70        for candidate in current {
71            let (before, after) = candidate.subtract(other);
72            if let Some(before) = before.filter(|range| !range.is_empty()) {
73                remaining
74                    .push(before)
75                    .expect("splitting into too many memory regions");
76            }
77            if let Some(after) = after.filter(|range| !range.is_empty()) {
78                remaining
79                    .push(after)
80                    .expect("splitting into too many memory regions");
81            }
82        }
83    }
84
85    remaining
86}

88/// Pushes a physical memory region into a [`MemoryRegions`], splitting the
89/// region into multiple parts to skip overlapping parts if necessary.
90///
91/// # Panics
92///
93/// Panics if the region would be split into too many parts or the number of
94/// regions exceeds the maximum.
95fn push_phys_mem_region_skip_overlap(target: &mut MemoryRegions, region: MemoryRegion) {
96    sort_phys_mem_regions(target);
97    let occupied = target.iter().map(|region| region.range);
98    let remaining = subtract_overlaps(region.range, occupied);
99    target.extend(remaining.into_iter().map(|range| MemoryRegion {
100        range,
101        flags: region.flags,
102        desc: region.desc,
103    }));
104}

106/// Sorts a [`MemoryRegions`] by address range start.
107fn sort_phys_mem_regions(regions: &mut MemoryRegions) {
108    regions.sort_unstable_by_key(|region| region.range.start);
109}
110
111/// Checks if a [`MemoryRegions`] is valid.
112fn check_phys_mem_regions(regions: &MemoryRegions) {
113    for region in regions.iter() {
114        if !region.flags.role_sanity_check() {
115            panic!("physical memory region {region:#x?} has invalid role flags");
116        }
117    }
118
119    for [a, b] in regions.array_windows() {
120        if a.range.end > b.range.start {
121            panic!("physical memory regions {a:#x?} and {b:#x?} overlap");
122        }
123    }
124}

126/// Converts a virtual address range to an identically-mapped physical address
127/// range.
128fn virt_range_to_id_phys_range(virt_range: VirtAddrRange) -> PhysAddrRange {
129    PhysAddrRange::new(
130        PhysAddr::from_usize(virt_range.start.as_usize()),
131        PhysAddr::from_usize(virt_range.end.as_usize()),
132    )
133}

135/// Builds the final physical memory region table.
136///
137/// The final memory regions contains:
138/// - All kernel sections with the [`MemoryRegionFlags::KERNEL`] flag set.
139/// - The loader range with the [`MemoryRegionFlags::BOOT_SERVICE`] flag set.
140/// - All boot-service regions reported by [`exarch::mem::raw_mem_regions`].
141/// - All reserved regions reported by [`exarch::mem::raw_mem_regions`].
142/// - All free regions reported by [`exarch::mem::raw_mem_regions`], with
143///   portions overlapping with other regions dropped.
144pub fn build_phys_mem_regions(raw_mem_regions: &RawMemoryRegions, loader_range: PhysAddrRange) {
145    let mut final_mem_regions = MemoryRegions::new();
146
147    for section in sections::all_sections() {
148        push_phys_mem_region_no_overlap(
149            &mut final_mem_regions,
150            MemoryRegion {
151                range: virt_range_to_id_phys_range(section.aligned_range),
152                flags: section.flags.union(MemoryRegionFlags::KERNEL),
153                desc: section.name,
154            },
155        );
156    }
157
158    push_phys_mem_region_no_overlap(
159        &mut final_mem_regions,
160        MemoryRegion {
161            range: loader_range,
162            // No EXECUTE flag because the loader should no longer be called.
163            flags: MemoryRegionFlags::BOOT_SERVICE
164                | MemoryRegionFlags::READ
165                | MemoryRegionFlags::WRITE,
166            desc: "loader",
167        },
168    );
169
170    for raw_region in raw_mem_regions {
171        if raw_region.flags.contains(MemoryRegionFlags::RESERVED)
172            || raw_region.flags.contains(MemoryRegionFlags::BOOT_SERVICE)
173        {
174            push_phys_mem_region_no_overlap(&mut final_mem_regions, *raw_region)
175        }
176    }
177
178    for raw_region in raw_mem_regions {
179        if raw_region.flags.contains(MemoryRegionFlags::FREE) {
180            push_phys_mem_region_skip_overlap(&mut final_mem_regions, *raw_region);
181        }
182    }
183
184    sort_phys_mem_regions(&mut final_mem_regions);
185    check_phys_mem_regions(&final_mem_regions);
186
187    PHYS_MEM_REGIONS.init_once(final_mem_regions);
188}

190#[cfg(test)]
191mod tests {
192    use memory_addr::{AddrRangeBounds, PhysAddrRange, pa};
193
194    use super::subtract_overlaps;
195
196    #[test]
197    fn subtract_overlaps_removes_multiple_occupied_segments() {
198        let target = PhysAddrRange::new(pa!(0x1000), pa!(0x9000));
199        let occupied = [
200            PhysAddrRange::new(pa!(0x2000), pa!(0x3000)),
201            PhysAddrRange::new(pa!(0x5000), pa!(0x6000)),
202        ];
203
204        let remaining = subtract_overlaps(target, occupied.iter().copied());
205
206        assert_eq!(
207            remaining.as_slice(),
208            &[
209                PhysAddrRange::new(pa!(0x1000), pa!(0x2000)),
210                PhysAddrRange::new(pa!(0x3000), pa!(0x5000)),
211                PhysAddrRange::new(pa!(0x6000), pa!(0x9000)),
212            ]
213        );
214    }
215
216    #[test]
217    fn subtract_overlaps_drops_fully_covered_and_keeps_adjacent_ranges() {
218        let target = PhysAddrRange::new(pa!(0x2000), pa!(0x6000));
219        let occupied = [
220            PhysAddrRange::new(pa!(0x1000), pa!(0x7000)),
221            PhysAddrRange::new(pa!(0x7000), pa!(0x8000)),
222        ];
223
224        let remaining = subtract_overlaps(target, occupied.iter().copied());
225
226        assert!(remaining.is_empty());
227
228        let target = PhysAddrRange::new(pa!(0x2000), pa!(0x6000));
229        let occupied = [
230            PhysAddrRange::new(pa!(0x1000), pa!(0x2000)),
231            PhysAddrRange::new(pa!(0x6000), pa!(0x7000)),
232        ];
233
234        let remaining = subtract_overlaps(target, occupied.iter().copied());
235
236        assert_eq!(remaining.as_slice(), &[target]);
237        assert!(remaining[0].is_valid());
238    }
