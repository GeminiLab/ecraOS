11//! Exception Syndrome Register - EL3
12//!
13//! Holds syndrome information for an exception taken to EL3.
14
15use tock_registers::{interfaces::Readable, register_bitfields};
16
17register_bitfields! {u64,
18    pub ESR_EL3 [
19
20        /// Reserved
21        RES0 OFFSET(37) NUMBITS(27) [],
22
23        /// Instruction Specific Syndrome 2
24        ///
25        /// When FEAT_LS64 is implemented:
26        ///
27        /// When FEAT_LS64_V is implemented, if a memory access generated by an ST64BV instruction
28        /// generates a Data Abort exception for a Translation fault, Access flag fault, or Permission fault, then
29        /// this field holds register specifier, Xs.
30        ///
31        /// When FEAT_LS64_ACCDATA is implemented, if a memory access generated by an ST64BV0
32        /// instruction generates a Data Abort exception for a Translation fault, Access flag fault, or Permission
33        /// fault, then this field holds register specifier, Xs.
34        ///
35        /// Otherwise, this field is RES 0.
36        ISS2 OFFSET(32) NUMBITS(5) [],
37
38        /// Exception Class. Indicates the reason for the exception that this register holds
39        /// information about.
40        ///
41        /// For each EC value, the table references a subsection that gives information about:
42        ///   - The cause of the exception, for example the configuration required to enable the
43        ///     trap.
44        ///   - The encoding of the associated ISS.
45        ///
46        /// All other EC values are reserved by Arm, and:
47        ///   - Unused values in the range 0b000000 - 0b101100 (0x00 - 0x2C) are reserved for future use for
48        ///   synchronous exceptions.
49        ///
50        ///   - Unused values in the range 0b101101 - 0b111111 (0x2D - 0x3F) are reserved for future use, and
51        ///   might be used for synchronous or asynchronous exceptions.
52        ///
53        /// The effect of programming this field to a reserved value is that behavior is CONSTRAINED UNPREDICTABLE.
54        ///
55        /// The reset behavior of this field is:
56        ///   - On a Warm reset, this field resets to an architecturally UNKNOWN value.
57        EC  OFFSET(26) NUMBITS(6) [
58            Unknown               = 0b00_0000,
59            TrappedWFIorWFE       = 0b00_0001,
60            TrappedMCRorMRC       = 0b00_0011, // A32
61            TrappedMCRRorMRRC     = 0b00_0100, // A32
62            TrappedMCRorMRC2      = 0b00_0101, // A32
63            TrappedLDCorSTC       = 0b00_0110, // A32
64            TrappedFP             = 0b00_0111,
65            TrappedPointerAuth    = 0b00_1001,
66            LD64BorST64B          = 0b00_1010,
67            TrappedMRRC           = 0b00_1100, // A32
68            BranchTarget          = 0b00_1101,
69            IllegalExecutionState = 0b00_1110,
70            SMC32                 = 0b01_0011, // A32
71            SVC64                 = 0b01_0101,
72            HVC64                 = 0b01_0110,
73            SMC64                 = 0b01_0111,
74            TrappedMsrMrs         = 0b01_1000,
75            TrappedSve            = 0b01_1001,
76            TSTART                = 0b01_1011,
77            PointerAuth           = 0b01_1100,
78            TrappedSME            = 0b01_1101,
79            GranuleProtection     = 0b01_1110,
80            ImplDefined           = 0b01_1111,
81            InstrAbortLowerEL     = 0b10_0000,
82            InstrAbortCurrentEL   = 0b10_0001,
83            PCAlignmentFault      = 0b10_0010,
84            DataAbortLowerEL      = 0b10_0100,
85            DataAbortCurrentEL    = 0b10_0101,
86            SPAlignmentFault      = 0b10_0110,
87            MemoryOperation       = 0b10_0111,
88            TrappedFP64           = 0b10_1100,
89            SError                = 0b10_1111,
90            Brk64                 = 0b11_1100  // A64 BRK instruction
91        ],
92
93        /// Instruction Length for synchronous exceptions. Possible values of this bit are:
94        ///     0b0 16-bit instruction trapped.
95        ///
96        ///     0b1 32-bit instruction trapped. This value is also used when the exception is one of the
97        ///     following:
98        ///         - An SError interrupt.
99        ///         - An Instruction Abort exception.
100        ///         - A PC alignment fault exception.
101        ///         - An SP alignment fault exception.
102        ///         - A Data Abort exception for which the value of the ISV bit is 0.
103        ///         - An Illegal Execution state exception.
104        ///         - Any debug exception except for Breakpoint instruction exceptions.
105        ///         - An exception reported using EC value 0b000000.
106        ///
107        /// The reset behavior of this field is:
108        ///     - On a Warm reset, this field resets to an architecturally UNKNOWN value.
109        IL  OFFSET(25) NUMBITS(1) [
110            Trapped16 = 0,
111            Trapped32 = 1
112        ],
113
114        /// Instruction Specific Syndrome. Architecturally, this field can be defined independently
115        /// for each defined Exception class. However, in practice, some ISS encodings are used for
116        /// more than one Exception class.
117        ISS OFFSET(0)  NUMBITS(25) []
118    ]
119}
120
121pub struct Reg;
122
123impl Readable for Reg {
124    type T = u64;
125    type R = ESR_EL3::Register;
126
127    sys_coproc_read_raw!(u64, "ESR_EL3", "x");
128}
129
130pub const ESR_EL3: Reg = Reg {};
