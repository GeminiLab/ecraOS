8//! AArch64 Debug Feature Register 0 - EL1
9//!
10//! Provides top level information about the debug system in AArch64 state.
11
12use tock_registers::{interfaces::Readable, register_bitfields};
13
14register_bitfields! {u64,
15    pub ID_AA64DFR0_EL1 [
16        /// Branch Record Buffer Extension.
17        ///
18        /// - 0000 Branch Record Buffer Extension not implemented.
19        /// - 0001 Branch Record Buffer Extension implemented.
20        /// - 0010 As 0b0001, and adds support for branch recording at EL3.
21        ///
22        /// All other values are reserved.
23        BRBE OFFSET(52) NUMBITS(4) [
24            NotImplemented = 0b0000,
25            Implemented = 0b0001,
26            Implemented_El3Support = 0b0010,
27        ],
28
29        /// Multi-threaded PMU extension.
30        ///
31        /// - 0000 FEAT_MTPMU not implemented. If FEAT_PMUv3 is implemented, it is IMPLEMENTATION
32        ///        DEFINED whether PMEVTYPER<n>_EL0.MT and PMEVTYPER<n>.MT are read/write or RES0.
33        /// - 0001 FEAT_MTPMU and FEAT_PMUv3 implemented. PMEVTYPER<n>_EL0.MT and PMEVTYPER<n>.MT
34        ///        are read/write. When FEAT_MTPMU is disabled, the Effective values of
35        ///        PMEVTYPER<n>_EL0.MT and PMEVTYPER<n>.MT are 0.
36        /// - 1111 FEAT_MTPMU not implemented. If FEAT_PMUv3 is implemented, PMEVTYPER<n>_EL0.MT
37        ///        and PMEVTYPER<n>.MT are RES0.
38        ///
39        /// All other values are reserved.
40        MTPMU OFFSET(48) NUMBITS(4) [
41            NotImplemented = 0b0000,
42            Implemented = 0b0001,
43            NotImplemented_Res0 = 0b1111,
44        ],
45
46        /// Trace Buffer Extension.
47        ///
48        /// - 0000 Trace Buffer Extension not implemented.
49        /// - 0001 Trace Buffer Extension implemented.
50        /// - 0010 As 0b0001, and adds:
51        ///   - If EL2 and FEAT_FGT are implemented, a fine-grained trap on the TSB CSYNC instruction.
52        ///   - If EL2 is implemented, an EL2 control to override TRBLIMITR_EL1.nVM.
53        ///   - The TRBE Profiling exception extension, FEAT_TRBE_EXC.
54        ///
55        /// All other values are reserved.
56        TraceBuffer OFFSET(44) NUMBITS(4) [
57            NotImplemented = 0b0000,
58            Implemented = 0b0001,
59            Implemented_Trbe = 0b0010,
60        ],
61
62        /// Armv8.4 Self-hosted Trace Extension version.
63        TraceFilt OFFSET(40) NUMBITS(4) [
64            NotImplemented = 0b0000,
65            Implemented = 0b0001,
66        ],
67
68        /// Statistical Profiling Extension version.
69        PMSVer OFFSET(32) NUMBITS(4) [],
70
71        /// Number of context-aware breakpoints, minus 1.
72        CTX_CMPs OFFSET(28) NUMBITS(4) [],
73
74        /// Number of watchpoints, minus 1.
75        WRPs OFFSET(20) NUMBITS(4) [],
76
77        /// Number of breakpoints, minus 1.
78        BRPs OFFSET(12) NUMBITS(4) [],
79
80        /// Performance Monitors Extension version.
81        PMUVer OFFSET(8) NUMBITS(4) [],
82
83        /// Trace support. Indicates whether System register interface to a trace unit is
84        /// implemented.
85        TraceVer OFFSET(4) NUMBITS(4) [],
86
87        /// Debug architecture version. Indicates presence of Armv8 debug architecture.
88        DebugVer OFFSET(0) NUMBITS(4) [],
89    ]
90}

99}
