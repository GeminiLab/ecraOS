9//! Counter-timer Kernel Control register - EL1
10//!
11//! When FEAT_VHE is implemented and HCR_EL2.{E2H, TGE} is {1, 1}, this register does not
12//! cause any event stream from the virtual counter to be generated, and does not control access to
13//! the counters and timers. The access to counters and timers at EL0 is controlled by CNTHCTL_EL2.
14//!
15//! When FEAT_VHE is not implemented, or when HCR_EL2.{E2H, TGE} is not {1, 1}, this register
16//! controls the generation of an event stream from the virtual counter, and access from EL0 to the
17//! physical counter, virtual counter, EL1 physical timers, and the virtual timer.
18
19use tock_registers::{
20    interfaces::{Readable, Writeable},
21    register_bitfields,
22};
23
24register_bitfields! {u64,
25    pub CNTKCTL_EL1 [
26        /// Controls the scale of the generation of the event stream.
27        ///
28        ///     0b0 The CNTKCTL_EL1.EVNTI field applies to CNTVCT_EL0[15:0].
29        ///
30        ///     0b1 The CNTKCTL_EL1.EVNTI field applies to CNTVCT_EL0[23:8].
31        ///
32        /// This control applies regardless of the value of the CNTHCTL_EL2.ECV bit.
33        ///
34        /// The reset behavior of this field is:
35        ///     - On a Warm reset, this field resets to an architecturally UNKNOWN value.
36        EVNTIS OFFSET(17) NUMBITS(1) [
37            CntVct0_15 = 0,
38            CntVct8_23 = 1
39        ],
40
41        /// Traps EL0 accesses to the physical timer registers to EL1, or to EL2 when it is implemented and
42        /// enabled for the current Security state and HCR_EL2.TGE is 1, as follows:
43        ///
44        /// - In AArch64 state, the following registers are trapped, reported using EC syndrome value
45        ///   0x18:
46        ///
47        ///     — CNTP_CTL_EL0, CNTP_CVAL_EL0, and CNTP_TVAL_EL0.
48        ///
49        /// - In AArch32 state, MRC and MCR accesses to the following registers are trapped, reported
50        /// using EC syndrome value 0x03, MRRC and MCRR accesses are trapped, reported using EC
51        /// syndrome value 0x04:
52        ///     — CNTP_CTL, CNTP_CVAL, CNTP_TVAL.
53        ///
54        ///     0b0 EL0 accesses to the physical timer registers are trapped to EL1.
55        ///
56        ///     0b1 This control does not cause any instructions to be trapped.
57        ///
58        /// When FEAT_VHE is implemented and HCR_EL2.{E2H, TGE} is {1, 1}, this control does not
59        /// cause any instructions to be trapped.
60        ///
61        /// The reset behavior of this field is:
62        ///     - On a Warm reset, this field resets to an architecturally UNKNOWN value.
63        EL0PTEN OFFSET(9) NUMBITS(1) [
64            TrappedPhysical = 0,
65            TrappedNone = 1
66        ],
67
68        /// Traps EL0 accesses to the virtual timer registers to EL1, or to EL2 when it is implemented and
69        /// enabled for the current Security state and HCR_EL2.TGE is 1, as follows:
70        ///
71        /// • In AArch64 state, accesses to the following registers are trapped, reported using EC
72        /// syndrome value 0x18:
73        ///
74        ///     — CNTV_CTL_EL0, CNTV_CVAL_EL0, and CNTV_TVAL_EL0.
75        ///
76        /// • In AArch32 state, MRC and MCR accesses to the following registers are trapped and
77        /// reported using EC syndrome value 0x03, MRRC and MCRR accesses are trapped using EC
78        /// syndrome value 0x04:
79        ///
80        ///     — CNTV_CTL, CNTV_CVAL, and CNTV_TVAL.
81        ///
82        ///     0b0 EL0 accesses to the virtual timer registers are trapped.
83        ///
84        ///     0b1 This control does not cause any instructions to be trapped.
85        ///
86        /// When FEAT_VHE is implemented and HCR_EL2.{E2H, TGE} is {1, 1}, this control does not
87        /// cause any instructions to be trapped.
88        ///
89        /// The reset behavior of this field is:
90        ///     - On a Warm reset, this field resets to an architecturally UNKNOWN value.
91        EL0VTEN OFFSET(8) NUMBITS(1) [
92            TrappedVirtual = 0,
93            TrappedNone = 1
94        ],
95
96        /// Selects which bit of CNTVCT_EL0, as seen from EL1, is the trigger for the event stream generated
97        /// from that counter when that stream is enabled.
98        ///
99        /// If FEAT_ECV is implemented, and CNTKCTL_EL1.EVNTIS is 1, this field selects a trigger bit in
100        /// the range 8 to 23 of CNTVCT_EL0.
101        ///
102        /// Otherwise, this field selects a trigger bit in the range 0 to 15 of CNTVCT_EL0.
103        ///
104        /// The reset behavior of this field is:
105        ///     - On a Warm reset, this field resets to an architecturally UNKNOWN value.
106        EVNTI OFFSET(4) NUMBITS(4) [],
107
108        /// Controls which transition of the CNTVCT_EL0 trigger bit, as seen from EL1 and defined by
109        /// EVNTI, generates an event when the event stream is enabled.
110        ///
111        ///     0b0 A 0 to 1 transition of the trigger bit triggers an event.
112        ///
113        ///     0b1 A 1 to 0 transition of the trigger bit triggers an event.
114        ///
115        /// The reset behavior of this field is:
116        ///     - On a Warm reset, this field resets to an architecturally UNKNOWN value.
117        EVNTDIR OFFSET(3) NUMBITS(1) [
118            ZeroToOne = 0,
119            OneToZero = 1
120        ],
121
122        /// When FEAT_VHE is not implemented, or when HCR_EL2.{E2H, TGE} is not {1, 1}, enables the
123        /// generation of an event stream from CNTVCT_EL0 as seen from EL1.
124        ///
125        ///     0b0 Disables the event stream.
126        ///
127        ///     0b1 Enables the event stream.
128        ///
129        /// When FEAT_VHE is implemented and HCR_EL2.{E2H, TGE} is {1, 1}, this control does not
130        /// enable the event stream.
131        ///
132        /// The reset behavior of this field is:
133        ///     - On a Warm reset, this field resets to an architecturally UNKNOWN value.
134        EVNTEN OFFSET(2) NUMBITS(1) [
135            Disable = 0,
136            Enable = 1
137        ],
138
139        /// Traps EL0 accesses to the frequency register and virtual counter register to EL1, or to EL2 when it
140        /// is implemented and enabled for the current Security state and HCR_EL2.TGE is 1, as follows:
141        ///
142        /// - In AArch64 state, accesses to the following registers are trapped and reported using EC
143        /// syndrome value 0x18:
144        ///
145        ///     — CNTVCT_EL0 and if CNTKCTL_EL1.EL0PCTEN is 0, CNTFRQ_EL0.
146        ///
147        /// - In AArch32 state, MRC and MCR accesses to the following registers are trapped and
148        /// reported using EC syndrome value 0x03, MRRC and MCRR accesses are trapped and
149        /// reported using EC syndrome value 0x04:
150        ///
151        ///     — CNTVCT and if CNTKCTL_EL1.EL0PCTEN is 0, CNTFRQ.
152        ///
153        ///     0b0 EL0 accesses to the frequency register and virtual counter registers are trapped.
154        ///
155        ///     0b1 This control does not cause any instructions to be trapped.
156        ///
157        /// When FEAT_VHE is implemented and HCR_EL2.{E2H, TGE} is {1, 1}, this control does not
158        /// cause any instructions to be trapped.
159        ///
160        /// The reset behavior of this field is:
161        ///     - On a Warm reset, this field resets to an architecturally UNKNOWN value.
162        EL0VCTEN OFFSET(1) NUMBITS(1) [
163            TrappedFreqVct = 0,
164            TrappedNone = 1
165        ],
166
167        /// Traps EL0 accesses to the frequency register and physical counter register to EL1, or to EL2 when
168        /// it is implemented and enabled for the current Security state and HCR_EL2.TGE is 1, as follows:
169        ///
170        /// - In AArch64 state, the following registers are trapped, reported using EC syndrome value
171        /// 0x18:
172        ///
173        ///     — CNTPCT_EL0 and if CNTKCTL_EL1.EL0VCTEN is 0, CNTFRQ_EL0.
174        ///
175        /// - In AArch32 state, MCR or MRC accesses the following registers are trapped, reported using
176        /// EC syndrome value 0x03, MCRR or MRRC accesses are trapped and reported using EC
177        /// syndrome value 0x04:
178        ///
179        ///     — CNTPCT and if CNTKCTL_EL1.EL0VCTEN is 0, CNTFRQ.
180        ///
181        ///     0b0 EL0 accesses to the frequency register and physical counter register are trapped.
182        ///
183        ///     0b1 This control does not cause any instructions to be trapped.
184        ///
185        /// When FEAT_VHE is implemented and HCR_EL2.{E2H, TGE} is {1, 1}, this control does not
186        /// cause any instructions to be trapped.
187        ///
188        /// The reset behavior of this field is:
189        ///     - On a Warm reset, this field resets to an architecturally UNKNOWN value.
190        EL0PCTEN OFFSET(0) NUMBITS(1) [
191            TrappedFreqPct = 0,
192            TrappedNone = 1
193        ],
194    ]
195}
