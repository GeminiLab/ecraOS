8//! Interrupt Controller List Registers - EL2
9//!
10//! Provides interrupt context information for the virtual CPU interface.
11
12use tock_registers::{
13    interfaces::{Readable, Writeable},
14    register_bitfields,
15};
16
17register_bitfields! {u64,
18    pub ICH_LR0_EL2 [
19        /// The state of the interrupt.
20        State OFFSET(62) NUMBITS(2) [
21            Invalid = 0b00,
22            Pending = 0b01,
23            Active = 0b10,
24            PendingAndActive = 0b11,
25        ],
26
27        /// Indicates whether this virtual interrupt maps directly to a hardware interrupt, meaning
28        /// that it corresponds to a physical interrupt. Deactivation of the virtual interrupt also
29        /// causes the deactivation of the physical interrupt with the ID that the pINTID field
30        /// indicates.
31        HW OFFSET(61) NUMBITS(1) [],
32
33        /// Indicates the group for this virtual interrupt.
34        Group OFFSET(60) NUMBITS(1) [],
35
36        /// When FEAT_GICv3_NMI is implemented:
37        /// Indicates whether the virtual priority has the non-maskable property.
38        NMI OFFSET(59) NUMBITS(1) [],
39
40        /// The priority of this interrupt.
41        Priority OFFSET(48) NUMBITS(8) [],
42
43        /// Physical INTID, for hardware interrupts.
44        pINTID OFFSET(32) NUMBITS(13) [],
45
46        /// If this bit is 1, then when the interrupt identified by vINTID is deactivated,
47        /// a maintenance interrupt is asserted.
48        EOI OFFSET(41) NUMBITS(1) [],
49
50        /// Virtual INTID of the interrupt.
51        vINTID OFFSET(0) NUMBITS(32) [],
52    ]
53}
54
55pub struct Reg;
56
57impl Readable for Reg {
58    type T = u64;
59    type R = ();
60
61    sys_coproc_read_raw!(u64, "ICH_LR0_EL2", "x");
62}
63
64impl Writeable for Reg {
65    type T = u64;
66    type R = ();
67
68    sys_coproc_write_raw!(u64, "ICH_LR0_EL2", "x");
69}
70
71pub const ICH_LR0_EL2: Reg = Reg {};
