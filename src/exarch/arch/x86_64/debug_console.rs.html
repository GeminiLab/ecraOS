1//! COM1 (`0x3F8`) UART 16550 debug console with a spinlock for exclusive access.
2
3use kspin::SpinNoIrq;
4use uart_16550::SerialPort;
5
6/// Standard PC COM1 I/O base port.
7const COM1_BASE: u16 = 0x3f8;
8
9/// Shared UART instance guarded by a no-IRQ spinlock.
10static COM1: SpinNoIrq<SerialPort> = unsafe { SpinNoIrq::new(SerialPort::new(COM1_BASE)) };
11
12/// Initializes COM1 for polled transmit/receive.
13pub fn init() {
14    COM1.lock().init();
15}
16
17pub fn write_bytes(bytes: &[u8]) {
18    let mut com = COM1.lock();
19    for &b in bytes {
20        com.send(b);
21    }
22}
23
24pub fn read_bytes(bytes: &mut [u8]) -> usize {
25    let mut com = COM1.lock();
26
27    for (index, byte) in bytes.iter_mut().enumerate() {
28        let Some(n) = com.try_receive().ok() else {
29            return index;
30        };
31
32        *byte = n;
33    }
34
35    bytes.len()
36}
