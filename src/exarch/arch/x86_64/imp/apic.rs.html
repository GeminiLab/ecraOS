13/// Reserved local APIC interrupt vectors.
14///
15/// IOAPIC routing must never allocate any vector in this set.
16pub(crate) mod vectors {
17    /// The local timer vector.
18    ///
19    /// This is independent of controller-routed global interrupt sources.
20    pub const APIC_TIMER_VECTOR: u8 = 0xf0;
21    /// The local spurious vector.
22    ///
23    /// Spurious delivery requires no end-of-interrupt write.
24    pub const APIC_SPURIOUS_VECTOR: u8 = 0xf1;
25    /// The local APIC error vector.
26    ///
27    /// Errors remain in the LAPIC-local namespace.
28    pub const APIC_ERROR_VECTOR: u8 = 0xf2;
29    /// The common software-notification vector.
30    ///
31    /// Mailbox reasons are decoded by the kernel after this vector arrives.
32    pub const APIC_IPI_VECTOR: u8 = 0xf3;
33
34    /// Returns whether a vector belongs to a reserved local source.
35    ///
36    /// The external allocator uses this predicate for every candidate.
37    pub(crate) const fn is_reserved(vector: u8) -> bool {
38        matches!(
39            vector,
40            APIC_TIMER_VECTOR | APIC_SPURIOUS_VECTOR | APIC_ERROR_VECTOR | APIC_IPI_VECTOR
41        )
42    }
43}

73/// Invokes a callback with the current CPU's initialized local APIC.
74///
75/// This panics when local APIC initialization has not installed the current CPU's pointer.
76pub fn with_local_apic<R>(f: impl FnOnce(&mut LocalApic) -> R) -> R {
77    let local = LOCAL_APIC.read_current();
78    assert_ne!(local, 0, "current CPU local APIC is not initialized");
79    // SAFETY: The pointer is installed once by the current CPU and remains valid forever.
80    f(unsafe { &mut *(local as *mut LocalApic) })
81}
82
83/// Sends an end-of-interrupt notification for the current local APIC.
84pub fn end_of_interrupt() {
85    // SAFETY: The local APIC is initialized before external vectors are enabled.
86    with_local_apic(|lapic| unsafe { lapic.end_of_interrupt() });
87}

101pub fn init_bsp() {
102    info!("Initialize Local APIC...");
103
104    unsafe {
105        // Disable 8259A interrupt controllers
106        Port::<u8>::new(0x21).write(0xff);
107        Port::<u8>::new(0xA1).write(0xff);
108    }
109
110    init_local_apic();
111}

114pub fn init_ap() {

118fn init_local_apic() {
119    let has_tsc_deadline = raw_cpuid::CpuId::new()
120        .get_feature_info()
121        .is_some_and(|info| info.has_tsc_deadline());
122    TSC_DEADLINE_SUPPORTED.store(has_tsc_deadline, Ordering::Release);
123    let mut builder = LocalApicBuilder::new();
124    builder
125        .timer_vector(APIC_TIMER_VECTOR as _)
126        .timer_divide(TimerDivide::Div1)
127        .timer_mode(if has_tsc_deadline {
128            TimerMode::TscDeadline
129        } else {
130            TimerMode::OneShot
131        })
132        .error_vector(APIC_ERROR_VECTOR as _)
133        .spurious_vector(APIC_SPURIOUS_VECTOR as _);
134
135    if cpu_has_x2apic() {
136        // x2APIC support was checked before entering this path.
137    } else {
138        // info!("Using xAPIC.");
139        // let base_vaddr = phys_to_virt(pa!(unsafe { xapic_base() } as usize));
140        // builder.set_xapic_base(base_vaddr.as_usize() as u64);
141        panic!("CPU does not support x2APIC.")
142    }
143
144    let mut lapic = builder.build().unwrap();
145    unsafe {
146        if has_tsc_deadline {
147            x86::msr::wrmsr(x86::msr::IA32_TSC_DEADLINE, 0);
148        }
149        lapic.enable();
150        lapic.disable_timer();
151        lapic.set_timer_initial(0);
152    }
153    let lapic = Box::leak(Box::new(lapic)) as *mut LocalApic as usize;
154    LOCAL_APIC.write_current(lapic);
155    if !has_tsc_deadline && LAPIC_TIMER_FREQ_KHZ.load(Ordering::Acquire) == 0 {
156        let frequency = crate::arch::x86_64::time::calibrate_lapic_timer()
157            .expect("failed to calibrate LAPIC timer");
158        LAPIC_TIMER_FREQ_KHZ.store(frequency, Ordering::Release);
159        info!("Calibrated LAPIC timer: {frequency} kHz");
160    }
161}

163/// Returns whether the current CPU supports TSC-Deadline mode.
164pub fn tsc_deadline_supported() -> bool {
165    TSC_DEADLINE_SUPPORTED.load(Ordering::Acquire)
166}
167
168/// Returns the calibrated LAPIC timer frequency in kHz.
169///
170/// This rejects use of the one-shot timer before calibration has completed.
171pub fn timer_frequency_khz() -> u64 {
172    let frequency = LAPIC_TIMER_FREQ_KHZ.load(Ordering::Acquire);
173    assert_ne!(frequency, 0, "LAPIC timer frequency is not calibrated");
174    frequency
175}

177/// Programs the current LAPIC one-shot initial count.
178pub fn program_timer_initial(count: u32) {
179    with_local_apic(|lapic| unsafe {
180        lapic.set_timer_initial(count);
181        lapic.enable_timer();
182    });
183}

185/// Programs an absolute TSC-Deadline value on the current CPU.
186pub fn program_tsc_deadline(deadline: u64) {
187    assert!(tsc_deadline_supported(), "TSC-Deadline mode is unavailable");
188    with_local_apic(|lapic| unsafe {
189        lapic.set_timer_initial(0);
190        lapic.enable_timer();
191    });
192    unsafe { x86::msr::wrmsr(x86::msr::IA32_TSC_DEADLINE, deadline) };
193}
