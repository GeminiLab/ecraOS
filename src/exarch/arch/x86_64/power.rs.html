1//! Power-off via the QEMU ACPI PM control port (`0x604`).
2
3use core::time::Duration;
4
5use expt::opaque::OpaquePageTableRoot;
6use memory_addr::VirtAddr;

8use crate::power::{APEntry, CpuStartError, PhysicalCpuId, ShutdownReason};

10/// Starts an x86-64 application processor.
11pub fn cpu_up(
12    cpu_id: PhysicalCpuId,
13    page_table_root: OpaquePageTableRoot,
14    boot_stack_top: VirtAddr,
15    entry: APEntry,
16) -> Result<(), CpuStartError> {
17    let OpaquePageTableRoot::Single(page_table_root) = page_table_root;
18    let apic_id = u32::try_from(cpu_id).map_err(|_| CpuStartError::InvalidCpu)?;
19    super::imp::ap::setup_ap_start_page(
20        cpu_id,
21        page_table_root,
22        boot_stack_top,
23        VirtAddr::from_usize(entry as usize),
24    );
25
26    unsafe { super::imp::apic::send_init_ipi(apic_id) };
27    crate::time::spin_wait_for(Duration::from_millis(10)); // 10ms
28    unsafe { super::imp::apic::send_sipi(super::imp::ap::AP_START_PAGE_INDEX, apic_id) };
29    crate::time::spin_wait_for(Duration::from_micros(200)); // 200us
30    unsafe { super::imp::apic::send_sipi(super::imp::ap::AP_START_PAGE_INDEX, apic_id) };
31    Ok(())
32}

34/// Shuts down the x86-64 platform.
35pub fn shutdown(_reason: ShutdownReason) -> ! {
36    unsafe {
37        const POWEROFF_PORT: u16 = 0x604;
38        const POWEROFF_VALUE: u16 = 0x2000;
39        core::arch::asm!(
40            "outw %ax, (%dx)",
41            in("ax") POWEROFF_VALUE,
42            in("dx") POWEROFF_PORT,
43            options(att_syntax, nomem, nostack, preserves_flags),
44        );
45
46        loop {
47            core::arch::asm!("hlt", options(nomem, nostack));
48        }
49    }
50}
