7/// The SGI reserved for the common kernel software notification.
8///
9/// Its INTID never enters the global SPI registry.
10pub(crate) const IPI_SGI: u8 = 1;
11
12/// A malformed or unmatched redistributor region.
13///
14/// Selection rejects ambiguous topology before any local interrupt source is enabled.
15#[derive(Clone, Copy, Debug, Eq, PartialEq)]
16pub(crate) enum TopologyError {
17    /// An out-of-bounds or incomplete redistributor frame.
18    ///
19    /// Every frame includes all pages implied by its VLPIS bit.
20    InvalidRegion,
21    /// No frame with the requested physical affinity.
22    ///
23    /// Selection reached Last without finding the current PE.
24    MissingAffinity,
25    /// More than one frame with the requested affinity.
26    ///
27    /// The controller region does not provide an unambiguous physical target.
28    DuplicateAffinity,
29    /// A frame chain without a Last terminator.
30    ///
31    /// The region limit is reached without a valid end marker.
32    MissingLast,
33}
34
35/// Returns the compact affinity value represented by GICR_TYPER.
36///
37/// MPIDR control bits are excluded while Aff3 is packed above Aff2, Aff1 and Aff0.
38pub(crate) const fn mpidr_affinity(mpidr: u64) -> u32 {
39    (((mpidr >> 32) & 0xff) << 24 | (mpidr & 0x00ff_ffff)) as u32
40}
41
42/// Returns the complete frame stride implied by GICR_TYPER.VLPIS.
43///
44/// Virtual LPI frames add two 64-KiB pages to the base redistributor frame pair.
45pub(crate) const fn frame_stride(typer: u64) -> usize {
46    if typer & 2 != 0 { 0x40000 } else { 0x20000 }
47}
48
49/// Selects one redistributor frame by physical MPIDR affinity within the firmware region.
50///
51/// The scan honors Last, validates full strides and rejects duplicate matching frames.
52pub(crate) fn select_redistributor(
53    region_size: usize,
54    mpidr: u64,
55    mut read_typer: impl FnMut(usize) -> Option<u64>,
56) -> Result<usize, TopologyError> {
57    let affinity = mpidr_affinity(mpidr);
58    let mut offset = 0usize;
59    let mut found = None;
60    loop {
61        if offset
62            .checked_add(0x20000)
63            .is_none_or(|end| end > region_size)
64        {
65            return Err(if offset == region_size {
66                TopologyError::MissingLast
67            } else {
68                TopologyError::InvalidRegion
69            });
70        }
71        let typer = read_typer(offset).ok_or(TopologyError::InvalidRegion)?;
72        let end = offset
73            .checked_add(frame_stride(typer))
74            .ok_or(TopologyError::InvalidRegion)?;
75        if end > region_size {
76            return Err(TopologyError::InvalidRegion);
77        }
78        if (typer >> 32) as u32 == affinity {
79            if found.is_some() {
80                return Err(TopologyError::DuplicateAffinity);
81            }
82            found = Some(offset);
83        }
84        if typer & (1 << 4) != 0 {
85            return found.ok_or(TopologyError::MissingAffinity);
86        }
87        offset = end;
88    }
89}

91/// Encodes one Group-one SGI for a physical MPIDR affinity.
92///
93/// Aff0 values above fifteen require Range Selector support and are rejected in this backend.
94pub(crate) fn sgi1r(target: PhysicalCpuId, intid: u8) -> Result<u64, IpiError> {
95    let target = target as u64;
96    let aff0 = target & 0xff;
97    if aff0 >= 16 || intid >= 16 {
98        return Err(IpiError::InvalidTarget);
99    }
100    Ok(((target >> 32) & 0xff) << 48
101        | ((target >> 16) & 0xff) << 32
102        | (intid as u64) << 24
103        | ((target >> 8) & 0xff) << 16
104        | (1 << aff0))
105}

107/// A decoded GIC interrupt namespace.
108///
109/// Private sources never enter the kernel's global-source registry.
110#[derive(Clone, Copy, Debug, Eq, PartialEq)]
111pub(crate) enum GicInterrupt {
112    /// The reserved software SGI.
113    ///
114    /// The common mailbox identifies the semantic reason.
115    Software,
116    /// The virtual timer PPI.
117    ///
118    /// The common timer handler rearms the local source.
119    Timer,
120    /// A shared peripheral source.
121    ///
122    /// The contained INTID is a controller source rather than an exception cause.
123    Spi(u32),
124    /// A special non-delivery INTID.
125    ///
126    /// These values require no end-of-interrupt write.
127    Special,
128    /// An otherwise unsupported private source.
129    ///
130    /// Architecture code completes the acknowledged delivery normally.
131    Unknown,
132}
133
134/// Separates private sources, global sources and special non-delivery values.
135///
136/// The reserved SGI and PPI 27 have distinct semantic local-interrupt consumers.
137pub(crate) const fn decode_intid(intid: u32) -> GicInterrupt {
138    match intid {
139        value if value == IPI_SGI as u32 => GicInterrupt::Software,
140        27 => GicInterrupt::Timer,
141        32..=1019 => GicInterrupt::Spi(intid),
142        1020..=1023 => GicInterrupt::Special,
143        _ => GicInterrupt::Unknown,
144    }
145}

147/// Redistributor and SGI contract tests.
148///
149/// Synthetic register values exercise frame boundaries and physical affinity selection.
150#[cfg(test)]
151mod tests {
152    use crate::ipi::IpiError;
153
154    use super::{
155        GicInterrupt, TopologyError, decode_intid, frame_stride, mpidr_affinity,
156        select_redistributor, sgi1r,
157    };
158
159    /// Selects four distinct QEMU-style redistributor frames by affinity.
160    ///
161    /// The boot CPU frame cannot be reused for every AP.
162    #[test]
163    fn selects_four_cpu_frames() {
164        let typers = [0, 1u64 << 32, 2u64 << 32, (3u64 << 32) | 16];
165        for cpu in 0..4 {
166            assert_eq!(
167                select_redistributor(0x80000, cpu, |offset| typers.get(offset / 0x20000).copied()),
168                Ok(cpu as usize * 0x20000)
169            );
170        }
171        assert_eq!(mpidr_affinity(0x12_8034_5607), 0x1234_5607);
172    }
173
174    /// Honors VLPIS frame stride and stops at the Last redistributor.
175    ///
176    /// Reads must remain inside the complete firmware region and must not pass Last.
177    #[test]
178    fn honors_vlpis_stride_and_last() {
179        assert_eq!(frame_stride(0), 0x20000);
180        assert_eq!(frame_stride(2), 0x40000);
181        let mut reads = 0;
182        assert_eq!(
183            select_redistributor(0x80000, 1, |offset| {
184                reads += 1;
185                match offset {
186                    0 => Some(2),
187                    0x40000 => Some((1u64 << 32) | 16),
188                    _ => panic!("read beyond Last"),
189                }
190            }),
191            Ok(0x40000)
192        );
193        assert_eq!(reads, 2);
194        assert_eq!(
195            select_redistributor(0x20000, 1, |_| Some(16)),
196            Err(TopologyError::MissingAffinity)
197        );
198    }
199
200    /// Rejects duplicate affinities, truncated frames and unterminated frame chains.
201    ///
202    /// Ambiguous or out-of-bounds firmware topology must fail before enabling SGIs.
203    #[test]
204    fn rejects_ambiguous_and_truncated_topology() {
205        assert_eq!(
206            select_redistributor(0x40000, 0, |offset| Some(if offset == 0 { 0 } else { 16 })),
207            Err(TopologyError::DuplicateAffinity)
208        );
209        assert_eq!(
210            select_redistributor(0x10000, 0, |_| Some(16)),
211            Err(TopologyError::InvalidRegion)
212        );
213        assert_eq!(
214            select_redistributor(0x20000, 0, |_| Some(0)),
215            Err(TopologyError::MissingLast)
216        );
217        assert_eq!(
218            select_redistributor(0x20000, 0, |_| Some(2 | 16)),
219            Err(TopologyError::InvalidRegion)
220        );
221    }
222
223    /// Encodes all four affinity fields and the one physical Aff0 target bit.
224    ///
225    /// Unsupported Aff0 values cannot turn into Range Selector or broadcast delivery.
226    #[test]
227    fn encodes_physical_sgi_destination() {
228        assert_eq!(sgi1r(0x12_0034_560f, 1), Ok(0x0012_0034_0156_8000));
229        assert_eq!(sgi1r(3, 1), Ok(0x0100_0008));
230        assert_eq!(sgi1r(16, 1), Err(IpiError::InvalidTarget));
231        assert_eq!(sgi1r(0, 16), Err(IpiError::InvalidTarget));
232    }
233
234    /// Separates the reserved SGI, virtual timer PPI, SPI and special INTID classes.
235    ///
236    /// Software delivery never enters the external-source registry.
237    #[test]
238    fn separates_private_and_global_sources() {
239        assert_eq!(decode_intid(1), GicInterrupt::Software);
240        assert_eq!(decode_intid(27), GicInterrupt::Timer);
241        assert_eq!(decode_intid(32), GicInterrupt::Spi(32));
242        assert_eq!(decode_intid(1019), GicInterrupt::Spi(1019));
243        assert_eq!(decode_intid(1023), GicInterrupt::Special);
244        assert_eq!(decode_intid(0), GicInterrupt::Unknown);
245    }
