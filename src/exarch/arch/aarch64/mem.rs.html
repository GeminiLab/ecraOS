46/// The description attached to the device tree memory range.

48/// Identifies the reserved device-tree blob rather than MMIO described by its nodes.

51/// The alignment used for the device tree memory reservation.
52///
53/// Covers whole pages for all supported AArch64 granules, including 64 KiB, so RAM and
54/// device mappings never share a page at either boundary of the blob.
55const DEVICE_TREE_ALIGN: usize = 64 * 1024;

57/// The TCR bit selecting the 52-bit address-size format used by FEAT_LPA2.
58// `aarch64-cpu` 11.2.0 does not expose TCR_EL1.DS as a bitfield.
59const TCR_DS: u64 = 1 << 59;

61macro_rules! props_tcr_conv_fn {
62    (
63        $props_from_tcr_doc:literal,
64        $props_from_tcr_name:ident,
65        $tcr_from_props_doc:literal,
66        $tcr_from_props_name:ident,
67        $epdx:ident,
68        $enable_ttbrx_walks:ident,
69        $disable_ttbrx_walks:ident,
70        $tgx:ident,
71        $txsz:ident,
72    ) => {
73        #[doc = $props_from_tcr_doc]
74        fn $props_from_tcr_name(
75            tcr: LocalRegisterCopy<u64, TCR_EL1::Register>,
76        ) -> Option<VirtAddrSpaceProps> {
77            use TCR_EL1::{$epdx, $tgx, $txsz};
78
79            if !tcr.matches_all($epdx::$enable_ttbrx_walks) {
80                return None;
81            }
82
83            let page_shift = match tcr.read_as_enum::<$tgx::Value>($tgx) {
84                Some($tgx::Value::KiB_4) => 12,
85                Some($tgx::Value::KiB_16) => 14,
86                Some($tgx::Value::KiB_64) => 16,
87                None => panic!("invalid TCR_EL1.{}", stringify!($tgx)),
88            };
89
90            let va_bits = match tcr.read($txsz) {
91                12 => 52,
92                16 => 48,
93                _ => panic!(
94                    "unsupported TCR_EL1.{}: {}",
95                    stringify!($txsz),
96                    tcr.read($txsz)
97                ),
98            };
99
100            Some(VirtAddrSpaceProps {
101                page_shift,
102                va_bits,
103            })
104        }
105
106        #[doc = $tcr_from_props_doc]
107        fn $tcr_from_props_name(
108            tcr: &mut LocalRegisterCopy<u64, TCR_EL1::Register>,
109            props: Option<VirtAddrSpaceProps>,
110        ) {
111            use TCR_EL1::{$epdx, $tgx, $txsz};
112
113            let Some(VirtAddrSpaceProps {
114                page_shift,
115                va_bits,
116            }) = props
117            else {
118                tcr.modify($epdx::$disable_ttbrx_walks);
119                return;
120            };
121
122            tcr.modify($epdx::$enable_ttbrx_walks + $txsz.val(64 - (va_bits as u64)));
123            tcr.modify(match page_shift {
124                12 => $tgx::KiB_4,
125                14 => $tgx::KiB_16,
126                16 => $tgx::KiB_64,
127                _ => panic!("unsupported page shift: {}", page_shift),
128            });
129
130            if va_bits > 48 && page_shift < 16 {
131                tcr.set(tcr.get() | TCR_DS);
132            }
133        }
134    };
135}

161/// Decodes a virtual-address-space mode from a TCR value.
162fn mode_from_tcr(tcr: LocalRegisterCopy<u64, TCR_EL1::Register>) -> Option<VirtAddrSpaceMode> {
163    let lower = lower_props_from_tcr(tcr);
164    let upper = upper_props_from_tcr(tcr);
165
166    VirtAddrSpaceMode::from_dual_props(lower, upper)
167}

169/// Returns the currently active AArch64 virtual-address-space mode.
170fn current_mode() -> VirtAddrSpaceMode {
171    mode_from_tcr(TCR_EL1.extract())
172        .expect("AArch64 has no valid active virtual address space mode")
173}

175/// Returns a TCR value for a requested AArch64 virtual-address-space mode.
176fn tcr_for_mode(mode: VirtAddrSpaceMode) -> LocalRegisterCopy<u64, TCR_EL1::Register> {
177    let (lower, upper) = VirtAddrSpaceMode::into_dual_props(mode)
178        .expect("AArch64 does not support unified virtual address spaces");
179
180    let mut tcr = LocalRegisterCopy::<u64, TCR_EL1::Register>::new(0);
181    tcr.modify(
182        TCR_EL1::IPS::Bits_48
183            + TCR_EL1::SH0::Inner
184            + TCR_EL1::ORGN0::WriteBack_ReadAlloc_WriteAlloc_Cacheable
185            + TCR_EL1::IRGN0::WriteBack_ReadAlloc_WriteAlloc_Cacheable
186            + TCR_EL1::SH1::Inner
187            + TCR_EL1::ORGN1::WriteBack_ReadAlloc_WriteAlloc_Cacheable
188            + TCR_EL1::IRGN1::WriteBack_ReadAlloc_WriteAlloc_Cacheable,
189    );
190
191    // Set the lower and upper virtual address space properties in the TCR.
192    lower_tcr_from_props(&mut tcr, lower);
193    upper_tcr_from_props(&mut tcr, upper);
194
195    tcr
196}

198/// Collects RAM and device regions reported by the device tree.
199///
200/// Subtracts the page-aligned device-tree blob from each RAM range and reports it once as
201/// reserved device memory, preserving usable RAM on both sides.
202pub fn raw_mem_regions(arg: PlatformBootArg) -> RawMemoryRegions {
203    let mut regions = RawMemoryRegions::new();
204    if let PlatformBootArg::DeviceTree(dtb_addr) = arg {
205        let dtb = unsafe {
206            DevTree::from_raw_pointer(VirtAddr::from_usize(dtb_addr.as_usize()).as_ptr())
207                .expect("failed to parse device tree")
208        };
209        let dtb_range = PhysAddrRange::from_start_size(dtb_addr, dtb.totalsize())
210            .align_outwards(DEVICE_TREE_ALIGN)
211            .expect("device tree range alignment overflow");
212        for node in dtb.nodes().iterator() {
213            let node = node.expect("failed to inspect device tree node");
214            let mut is_memory = false;
215            let mut is_supported = false;
216            let mut is_gic = false;
217            let mut reg_values = [0u64; 4];
218            let mut reg_count = 0;
219            for prop in node.props().iterator() {
220                let prop = prop.expect("failed to inspect device-tree property");
221                match prop.name().expect("invalid device-tree property name") {
222                    "device_type" => {
223                        is_memory = prop.str().is_ok_and(|value| value == "memory");
224                    }
225                    "compatible" => {
226                        for value in prop.iter_str().iterator() {
227                            let value = value.expect("invalid compatible string");
228                            is_gic |= value == "arm,gic-v3";
229                            is_supported |=
230                                matches!(value, "arm,gic-v3" | "arm,pl011" | "arm,armv8-timer");
231                        }
232                    }
233                    "reg" => {
234                        while reg_count < reg_values.len() {
235                            let Some(value) = prop.u64(reg_count).ok() else {
236                                break;
237                            };
238                            reg_values[reg_count] = value;
239                            reg_count += 1;
240                        }
241                    }
242                    _ => {}
243                }
244            }
245
246            if is_memory && reg_count >= 2 {
247                let start = reg_values[0];
248                let end = start
249                    .checked_add(reg_values[1])
250                    .expect("memory range overflow");
251                let memory_range = PhysAddrRange::new(pa!(start as usize), pa!(end as usize));
252                let (before_dtb, after_dtb) = memory_range.subtract(dtb_range);
253                for range in [before_dtb, after_dtb]
254                    .into_iter()
255                    .flatten()
256                    .filter(|range| !range.is_empty())
257                {
258                    regions
259                        .push(MemoryRegion {
260                            range,
261                            flags: DEFAULT_RAM_FLAGS,
262                            desc: DEFAULT_RAM_DESC,
263                        })
264                        .expect("too many platform memory regions");
265                }
266            }
267
268            if is_supported && reg_count >= 2 {
269                let _ = regions.push(MemoryRegion {
270                    range: PhysAddrRange::from_start_size(
271                        pa!(reg_values[0] as usize),
272                        reg_values[1] as usize,
273                    ),
274                    flags: MemoryRegionFlags::READ
275                        | MemoryRegionFlags::WRITE
276                        | MemoryRegionFlags::DEVICE
277                        | MemoryRegionFlags::RESERVED,
278                    desc: "device-tree MMIO",
279                });
280                if is_gic && reg_count >= 4 {
281                    let _ = regions.push(MemoryRegion {
282                        range: PhysAddrRange::from_start_size(
283                            pa!(reg_values[2] as usize),
284                            reg_values[3] as usize,
285                        ),
286                        flags: MemoryRegionFlags::READ
287                            | MemoryRegionFlags::WRITE
288                            | MemoryRegionFlags::DEVICE
289                            | MemoryRegionFlags::RESERVED,
290                        desc: "device-tree MMIO",
291                    });
292                }
293            }
294        }
295
296        regions
297            .push(MemoryRegion {
298                range: dtb_range,
299                flags: MemoryRegionFlags::READ
300                    | MemoryRegionFlags::WRITE
301                    | MemoryRegionFlags::RESERVED,
302                desc: DEVICE_TREE_DESC,
303            })
304            .expect("too many platform memory regions");
305    }
306    regions
307}

309/// Returns whether a [`VirtAddrSpaceProps`] is supported by the CPU.
310///
311/// Returns `None` if the properties are not supported by the CPU. Returns
312/// `Some(true)` if the properties are supported. Returns `Some(false)` if the
313/// properties are supported by the crate but not by the CPU.
314fn is_props_supported_inner(props: VirtAddrSpaceProps, mmfr0: u64, mmfr2: u64) -> Option<bool> {
315    // FEAT_LPA2 uses the previously reserved granule encodings 1 and 2 for
316    // 4 KiB and 16 KiB respectively.  A 64 KiB granule uses VARange instead.
317    /// ID_AA64MMFR0_EL1.TGran4 for "4 KiB granule is supported".
318    const TGRAN4_OK: u64 = 0b0000;
319    /// ID_AA64MMFR0_EL1.TGran4 for "4 KiB granule supports 52-bit input addresses".
320    const TGRAN4_LPA2: u64 = 0b0001;
321    /// ID_AA64MMFR0_EL1.TGran16 for "16 KiB granule is supported".
322    const TGRAN16_OK: u64 = 0b0001;
323    /// ID_AA64MMFR0_EL1.TGran16 for "16 KiB granule supports 52-bit input addresses".
324    const TGRAN16_LPA2: u64 = 0b0010;
325    /// ID_AA64MMFR0_EL1.TGran64 for "64 KiB granule is supported".
326    const TGRAN64_OK: u64 = 0b0000;
327    /// ID_AA64MMFR2_EL1.VARange for "52-bit input addresses are supported".
328    const VARANGE_52BIT: u64 = 0b0001;
329
330    let VirtAddrSpaceProps {
331        page_shift,
332        va_bits,
333    } = props;
334
335    let is_52_bit = match va_bits {
336        48 => false,
337        52 => true,
338        _ => return None,
339    };
340
341    Some(match page_shift {
342        12 => {
343            let tgran4 = ID_AA64MMFR0_EL1::TGran4.read(mmfr0);
344
345            (tgran4 == TGRAN4_LPA2) || (!is_52_bit && tgran4 == TGRAN4_OK)
346        }
347        14 => {
348            let tgran16 = ID_AA64MMFR0_EL1::TGran16.read(mmfr0);
349
350            (tgran16 == TGRAN16_LPA2) || (!is_52_bit && tgran16 == TGRAN16_OK)
351        }
352        16 => {
353            let tgran64 = ID_AA64MMFR0_EL1::TGran64.read(mmfr0);
354            let va_range = ID_AA64MMFR2_EL1::VARange.read(mmfr2);
355
356            (tgran64 == TGRAN64_OK) && (!is_52_bit || va_range == VARANGE_52BIT)
357        }
358        _ => return None,
359    })
360}

362/// Returns whether a [`VirtAddrSpaceProps`] is supported by this crate and the
363/// CPU.
364fn is_props_supported(props: VirtAddrSpaceProps, mmfr0: u64, mmfr2: u64) -> bool {
365    is_props_supported_inner(props, mmfr0, mmfr2).is_some_and(|b| b)
366}

370/// Returns the AArch64 granules and address widths supported by the CPU.
371fn supported_props() -> HeaplessVec<VirtAddrSpaceProps, MAX_PROPS_FOR_HALF> {
372    let mmfr0 = ID_AA64MMFR0_EL1.get();
373    let mmfr2 = ID_AA64MMFR2_EL1.get();
374
375    let mut props_set = HeaplessVec::new();
376
377    for page_shift in [12, 14, 16] {
378        for va_bits in [48, 52] {
379            let props = VirtAddrSpaceProps {
380                page_shift,
381                va_bits,
382            };
383
384            if is_props_supported_inner(props, mmfr0, mmfr2)
385                .expect("invalid AArch64 virtual address-space properties: {props:?}")
386            {
387                props_set
388                    .push(props)
389                    .expect("too many AArch64 virtual address-space properties");
390            }
391        }
392    }
393
394    props_set
395}

397/// Returns whether a [`VirtAddrSpaceMode`] is supported by this crate and the CPU.
398fn is_mode_supported(mode: VirtAddrSpaceMode) -> bool {
399    let mmfr0 = ID_AA64MMFR0_EL1.get();
400    let mmfr2 = ID_AA64MMFR2_EL1.get();
401
402    match mode {
403        VirtAddrSpaceMode::LowerOnly(prop) | VirtAddrSpaceMode::UpperOnly(prop) => {
404            is_props_supported(prop, mmfr0, mmfr2)
405        }
406        VirtAddrSpaceMode::Independent { lower, upper } => {
407            is_props_supported(lower, mmfr0, mmfr2) && is_props_supported(upper, mmfr0, mmfr2)
408        }
409        VirtAddrSpaceMode::Unified(_) => false,
410    }
411}

413/// Returns the virtual-address modes supported by AArch64.
414///
415/// AArch64 uses separate lower and upper translation tables. For each table
416/// there are three possible page granules (4 KiB, 16 KiB, and 64 KiB) and two
417/// virtual address bits (48 and 52, there are actually more and in some cases
418/// neither 48 nor 52 are supported, but we don't consider those cases here).
419pub fn virt_addr_space_modes() -> VirtAddrSpaceModes {
420    let props_for_half = supported_props();
421    if props_for_half.is_empty() {
422        panic!("AArch64 reports no supported translation granule");
423    }
424    let mut modes = VirtAddrSpaceModes::new();
425
426    for p in props_for_half.iter().copied() {
427        modes
428            .modes
429            .push(VirtAddrSpaceMode::LowerOnly(p))
430            .expect("too many virtual address space modes");
431        modes
432            .modes
433            .push(VirtAddrSpaceMode::UpperOnly(p))
434            .expect("too many virtual address space modes");
435    }
436
437    for lower in props_for_half.iter().copied() {
438        for upper in props_for_half.iter().copied() {
439            modes
440                .modes
441                .push(VirtAddrSpaceMode::Independent { lower, upper })
442                .expect("too many virtual address space modes")
443        }
444    }
445
446    let current = current_mode();
447    modes.current_index = modes
448        .modes
449        .iter()
450        .position(|candidate| *candidate == current)
451        .expect("active AArch64 virtual address-space mode is not supported");
452
453    dbcn_println!("  Current mode: {}", current);
454    for prop in props_for_half {
455        dbcn_println!(
456            "  Supported: {} KiB pages, {}-bit virtual addresses",
457            1usize << (prop.page_shift - 10),
458            prop.va_bits
459        );
460    }
461
462    modes
463}

465/// Selects an AArch64 virtual-address-space mode.
466///
467/// Installs identity-mapped temporary roots before changing the translation
468/// geometry. [`set_page_table_root`] replaces those roots with the final roots.
469pub fn set_virt_addr_space_mode(mode: VirtAddrSpaceMode) {
470    if !is_mode_supported(mode) {
471        panic!("Unsupported AArch64 virtual address-space mode: {mode:?}");
472    }
473
474    temp_pt::switch_to_temporary_page_tables(mode, tcr_for_mode(mode));
475}

477/// Returns the page-table descriptor for an AArch64 mode and caller-selected strategies.
478///
479/// The selected single-root or dual-root geometry captures `AL` for all operations that access,
480/// allocate, or deallocate table frames, and `F` for pending cleanup on both sides. See
481/// [`crate::mem::get_page_table_type`] for caller requirements.
482#[cfg_or_dummy(target_arch = "aarch64")]
483pub fn get_page_table_type<AL: PageAllocator, F: TlbFlusher<VirtAddr>>(
484    mode: VirtAddrSpaceMode,
485) -> OpaquePageTableType<VirtAddr> {
486    if !is_mode_supported(mode) {
487        panic!("Unsupported AArch64 virtual address-space mode: {mode:?}");
488    }
489
490    use cartesian_product::product;
491    use expt::{
492        arch::aarch64::AArch64PageTableMeta as A,
493        meta::{LowerCoverage as L, UpperCoverage as U},
494        pte::aarch64::A64PTE,
495    };
496
497    use crate::mem::VirtAddrSpaceProps as P;
498
499    macro_rules! match_half_type {
500        ([$(($page_shift:literal, $va_bits:literal)),+], $page_shift_var:ident, $va_bits_var:ident, $half:ident, $mode:ident) => {
501            match ($page_shift_var, $va_bits_var) {
502                $(($page_shift, $va_bits) => OpaquePageTableType::new::<A<$va_bits, $page_shift, $half<VirtAddr>>, A64PTE, AL, F>(),)+
503                _ => panic!("Unsupported AArch64 virtual address-space mode: {:?}", $mode),
504            }
505        };
506    }
507
508    macro_rules! match_dual_type {
509        ([$(($lower_page_shift:literal, $lower_va_bits:literal, $upper_page_shift:literal, $upper_va_bits:literal)),+], $lower_page_shift_var:ident, $lower_va_bits_var:ident, $upper_page_shift_var:ident, $upper_va_bits_var:ident, $mode:ident) => {
510            match ($lower_page_shift_var, $lower_va_bits_var, $upper_page_shift_var, $upper_va_bits_var) {
511                $(
512                    ($lower_page_shift, $lower_va_bits, $upper_page_shift, $upper_va_bits) =>
513                        OpaquePageTableType::new_dual::<
514                            A<$lower_va_bits, $lower_page_shift, L<VirtAddr>>,
515                            A64PTE,
516                            A<$upper_va_bits, $upper_page_shift, U<VirtAddr>>,
517                            A64PTE,
518                            AL,
519                            F,
520                        >(),
521                )+
522                _ => panic!("Unsupported AArch64 virtual address-space mode: {:?}", $mode),
523            }
524        };
525    }
526
527    match mode {
528        VirtAddrSpaceMode::LowerOnly(P {
529            page_shift,
530            va_bits,
531        }) => product!([12, 14, 16], [48, 52] => match_half_type!(@, page_shift, va_bits, L, mode)),
532        VirtAddrSpaceMode::UpperOnly(P {
533            page_shift,
534            va_bits,
535        }) => product!([12, 14, 16], [48, 52] => match_half_type!(@, page_shift, va_bits, U, mode)),
536        VirtAddrSpaceMode::Independent {
537            lower:
538                P {
539                    page_shift: lower_page_shift,
540                    va_bits: lower_va_bits,
541                },
542            upper:
543                P {
544                    page_shift: upper_page_shift,
545                    va_bits: upper_va_bits,
546                },
547        } => {
548            product!([12, 14, 16], [48, 52], [12, 14, 16], [48, 52] => match_dual_type!(@, lower_page_shift, lower_va_bits, upper_page_shift, upper_va_bits, mode))
549        }
550        VirtAddrSpaceMode::Unified(_) => {
551            panic!("AArch64 does not support unified virtual address spaces")
552        }
553    }
554}

556/// Installs a selected AArch64 translation-table root.
557///
558/// Dual roots are installed in TTBR0 and TTBR1. A single root is installed in
559/// the enabled translation regime for a lower-only or upper-only mode. The MMU
560/// pipeline is synchronized before returning.
561#[cfg_or_dummy(target_arch = "aarch64")]
562pub fn set_page_table_root(root: OpaquePageTableRoot) {
563    let current_tcr = TCR_EL1.get();
564    match root {
565        OpaquePageTableRoot::Single(root) if TCR_EL1::EPD1.is_set(current_tcr) => {
566            TTBR0_EL1.set_baddr(root.as_usize() as u64);
567        }
568        OpaquePageTableRoot::Single(root) if TCR_EL1::EPD0.is_set(current_tcr) => {
569            TTBR1_EL1.set_baddr(root.as_usize() as u64);
570        }
571        OpaquePageTableRoot::Single(_) => {
572            panic!("AArch64 single-root page table requires a disabled opposite regime");
573        }
574        OpaquePageTableRoot::Dual(root) => {
575            TTBR0_EL1.set_baddr(root.lower.as_usize() as u64);
576            TTBR1_EL1.set_baddr(root.upper.as_usize() as u64);
577        }
578    }
579    dsb(ISH);
580    // `aarch64-cpu` exposes architectural barriers and system registers, but
581    // does not expose the EL1 TLB invalidation instruction.
582    unsafe { asm!("tlbi vmalle1is", options(nostack)) };
583    dsb(ISH);
584    isb(SY);
585}
