11/// Discovers the AArch64 controller description from the firmware device tree.
12pub fn probe_platform(arg: PlatformBootArg) {
13    let PlatformBootArg::DeviceTree(dtb_addr) = arg else {
14        return;
15    };
16    let dtb = unsafe {
17        fdt_rs::base::DevTree::from_raw_pointer(
18            memory_addr::VirtAddr::from_usize(dtb_addr.as_usize()).as_ptr(),
19        )
20        .expect("failed to parse device tree")
21    };
22    use fdt_rs::prelude::{FallibleIterator, PropReader};
23    for node in dtb.nodes().iterator() {
24        let node = node.expect("failed to inspect device tree node");
25        if node.name().is_ok_and(|name| name.starts_with("psci"))
26            && let Some(method) = node
27                .props()
28                .find(|prop| Ok(prop.name()? == "method"))
29                .expect("failed to inspect PSCI method")
30        {
31            super::power::set_psci_method(method.str().is_ok_and(|value| value == "hvc"));
32        }
33        let gic = node
34            .props()
35            .any(|prop| Ok(prop.name()? == "compatible" && prop.str()? == "arm,gic-v3"))
36            .expect("failed to inspect GIC compatibility");
37        if !gic {
38            continue;
39        }
40        let Some(reg) = node
41            .props()
42            .find(|prop| Ok(prop.name()? == "reg"))
43            .expect("failed to inspect GIC reg")
44        else {
45            continue;
46        };
47        let gicd = reg.u64(0).expect("invalid GIC distributor base");
48        let gicd_size = reg.u64(1).expect("invalid GIC distributor size");
49        let gicr = reg.u64(2).expect("invalid GIC redistributor base");
50        let gicr_size = reg.u64(3).expect("invalid GIC redistributor size");
51        GIC_CONFIG.init_once(Aarch64ExternalIrqConfig {
52            gicd_base: memory_addr::pa!(gicd as usize),
53            gicd_size: gicd_size as usize,
54            gicr_base: memory_addr::pa!(gicr as usize),
55            gicr_size: gicr_size as usize,
56            spi_count: 988,
57        });
58        break;
59    }
60}

62/// Runs bootstrap-CPU early initialization.
63pub fn init_early(arg: PlatformBootArg) {
64    super::init_early_bsp(arg);
65}
66
67/// Runs late bootstrap-CPU initialization.
68#[cfg_or_dummy(target_arch = "aarch64")]
69pub fn init_later() {
70    if let Some(config) = GIC_CONFIG.get() {
71        crate::trap::irq::init_external_controller(config.clone());
72        super::trap::init_percpu();
73    }
74}
