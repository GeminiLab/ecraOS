22/// Discovers the AArch64 CPU topology, PSCI conduit, and GICv3 configuration.
23#[cfg_or_dummy(target_arch = "aarch64")]
24pub fn discover_platform(arg: PlatformBootArg) -> PlatformDescription {
25    let PlatformBootArg::DeviceTree(dtb_addr) = arg else {
26        panic!("AArch64 platform discovery requires a Device Tree")
27    };
28    // SAFETY: the loader supplies a valid DTB that remains mapped during platform discovery.
29    let dtb = unsafe {
30        DevTree::from_raw_pointer(crate::kernel_if::phys_to_virt(dtb_addr).as_ptr())
31            .expect("failed to parse device tree")
32    };
33    with_index(dtb, |index| {
34        let cpus = index
35            .find_path("/cpus")?
36            .expect("Device Tree has no /cpus node");
37        let cpu_address_cells = cpus.u32_property("#address-cells", 0)?.unwrap_or(2);
38        assert!(
39            matches!(cpu_address_cells, 1 | 2),
40            "unsupported CPU address-cell count: {cpu_address_cells}"
41        );
42        let mut cpu_ids = Vec::new();
43        for node in index.nodes() {
44            if node.is_node_kind("psci")?
45                && let Some(method) = node.string_property("method")?
46            {
47                super::power::set_psci_method(method == "hvc");
48            }
49        }
50        for node in cpus.children() {
51            if node.is_device_type("cpu")? && node.is_enabled()? {
52                let reg = node.property("reg")?.expect("CPU node has no reg property");
53                let affinity = match cpu_address_cells {
54                    1 => u64::from(reg.u32(0).expect("invalid CPU affinity")),
55                    2 => {
56                        let high = u64::from(reg.u32(0).expect("invalid CPU affinity"));
57                        let low = u64::from(reg.u32(1).expect("invalid CPU affinity"));
58                        (high << 32) | low
59                    }
60                    _ => unreachable!("CPU address-cell count validated above"),
61                };
62                cpu_ids.push(PhysicalCpuId::new(affinity as _));
63            }
64        }
65        let bsp_id = crate::kernel_if::current_cpu_phys_id();
66        let cpu_topology = CpuTopology::new(cpu_ids.into_boxed_slice(), bsp_id);
67
68        let gic_node = index
69            .find_compatible("arm,gic-v3")?
70            .expect("Device Tree has no GICv3 controller");
71        assert!(
72            gic_node.is_enabled()?,
73            "Device Tree GICv3 controller is disabled"
74        );
75        let reg = gic_node
76            .property("reg")?
77            .expect("GIC node has no reg property");
78        let gic = Aarch64ExternalIrqConfig {
79            gicd_base: PhysAddr::from_usize(
80                reg.u64(0).expect("invalid GIC distributor base") as usize
81            ),
82            gicd_size: reg.u64(1).expect("invalid GIC distributor size") as usize,
83            gicr_base: PhysAddr::from_usize(
84                reg.u64(2).expect("invalid GIC redistributor base") as usize
85            ),
86            gicr_size: reg.u64(3).expect("invalid GIC redistributor size") as usize,
87            spi_count: 988,
88        };
89        let mut mmio_regions = alloc::vec![
90            MemoryRegion {
91                range: memory_addr::PhysAddrRange::from_start_size(gic.gicd_base, gic.gicd_size),
92                flags: DEFAULT_MMIO_FLAGS,
93                desc: "GIC distributor registers",
94            },
95            MemoryRegion {
96                range: memory_addr::PhysAddrRange::from_start_size(gic.gicr_base, gic.gicr_size),
97                flags: DEFAULT_MMIO_FLAGS,
98                desc: "GIC redistributor registers",
99            },
100        ];
101        if let Some(console_region) = super::debug_console::mmio_region() {
102            mmio_regions.push(console_region);
103        }
104        Ok(PlatformDescription::new(
105            cpu_topology,
106            PlatformInterruptController(gic),
107            mmio_regions,
108        ))
109    })
110    .expect("failed to discover AArch64 platform from Device Tree")
111}
