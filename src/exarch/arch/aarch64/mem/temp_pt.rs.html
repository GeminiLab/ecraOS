1//! Temporary identity-mapped AArch64 translation tables.

21/// The MAIR value used for device and normal memory mappings.
22///
23/// Attribute 0 is Device-nGnRE and attribute 1 is inner-shareable normal
24/// write-back memory.
25const AARCH64_MAIR_EL1: u64 = MAIR_EL1::Attr0_Device::nonGathering_nonReordering_EarlyWriteAck
26    .value
27    | MAIR_EL1::Attr1_Normal_Outer::WriteBack_NonTransient_ReadWriteAlloc.value
28    | MAIR_EL1::Attr1_Normal_Inner::WriteBack_NonTransient_ReadWriteAlloc.value;
29
30/// The number of entries reserved for each temporary page-table level.
31///
32/// This is large enough for the largest supported AArch64 translation granule.
33const TEMPORARY_TABLE_ENTRIES: usize = 8192;
34/// The number of temporary page-table levels reserved per translation regime.
35///
36/// The storage covers the levels below the root and alignment slack separately.
37const TEMPORARY_TABLE_LEVELS: usize = 3;
38/// The maximum supported translation granule size.
39///
40/// Temporary roots are aligned to this value because it is the strictest
41/// alignment required by the supported granules.
42const MAX_PAGE_SIZE: usize = 1 << 16;
43/// The size of the temporary identity-mapped physical window.
44///
45/// The window covers the low physical addresses used during early boot.
46const TEMPORARY_IDENTITY_SIZE: usize = 1 << 39;
47/// The size of the device-memory prefix in the temporary identity mapping.
48///
49/// The prefix preserves the loader's device-memory attributes for MMIO.
50const TEMPORARY_DEVICE_SIZE: usize = 1 << 30;
51/// The number of entries reserved to include runtime-alignment slack.
52///
53/// The extra table accommodates a root aligned within the statically allocated
54/// storage after PIE relocation.
55const TEMPORARY_STORAGE_ENTRIES: usize = TEMPORARY_TABLE_ENTRIES * (TEMPORARY_TABLE_LEVELS + 1);
56
57/// The statically allocated temporary translation tables.
58///
59/// The backing storage is over-allocated so the runtime-selected root can be
60/// aligned even when the PIE load offset changes the link-time address.
61#[repr(C, align(65536))]
62struct TemporaryPageTables {
63    entries: [A64PTE; TEMPORARY_STORAGE_ENTRIES],
64}
65
66impl TemporaryPageTables {
67    /// Creates empty temporary translation tables.
68    ///
69    /// The entries are populated immediately before the roots are installed.
70    const fn new() -> Self {
71        Self {
72            entries: [A64PTE::empty(); TEMPORARY_STORAGE_ENTRIES],
73        }
74    }
75}
76
77/// The temporary lower-regime translation tables.
78///
79/// This storage is used when TTBR0 participates in the selected mode.
80static mut TEMPORARY_LOWER_TABLES: TemporaryPageTables = TemporaryPageTables::new();
81/// The temporary upper-regime translation tables.
82///
83/// This storage is used when TTBR1 participates in the selected mode.
84static mut TEMPORARY_UPPER_TABLES: TemporaryPageTables = TemporaryPageTables::new();

86/// Builds a temporary identity-mapped root for one translation regime.
87///
88/// The root and all subordinate tables are constructed while the old identity
89/// mapping is still active.
90fn temporary_page_table_root(
91    storage: *mut TemporaryPageTables,
92    props: VirtAddrSpaceProps,
93) -> PhysAddr {
94    let page_shift = usize::from(props.page_shift);
95    let level_bits = page_shift - 3;
96    let entries_per_table = 1usize << level_bits;
97    let levels = (usize::from(props.va_bits) - page_shift).div_ceil(level_bits);
98    let normal_flags = MappingFlags::READ | MappingFlags::WRITE | MappingFlags::EXECUTE;
99    let device_flags = MappingFlags::READ | MappingFlags::WRITE | MappingFlags::DEVICE;
100
101    assert!(entries_per_table <= TEMPORARY_TABLE_ENTRIES);
102    assert!((3..=5).contains(&levels));
103
104    // Build the tables while the old identity mapping is still active. A PIE
105    // load offset does not preserve the static's link-time 64 KiB alignment,
106    // so align the root inside the over-allocated storage at runtime.
107    unsafe {
108        let storage_entries = addr_of_mut!((*storage).entries).cast::<A64PTE>();
109        let root = (storage_entries as usize).next_multiple_of(MAX_PAGE_SIZE) as *mut A64PTE;
110        for index in 0..TEMPORARY_TABLE_ENTRIES * TEMPORARY_TABLE_LEVELS {
111            root.add(index).write(A64PTE::empty());
112        }
113
114        let mut table = root;
115        let mut table_slot = 0;
116        let mut level = levels - 1;
117
118        while level > 2 {
119            let next_table = root.add((table_slot + 1) * TEMPORARY_TABLE_ENTRIES);
120            table
121                .add(0)
122                .write(A64PTE::new_table(PhysAddr::from_usize(next_table as usize)));
123            table = next_table;
124            table_slot += 1;
125            level -= 1;
126        }
127
128        let level_two_shift = page_shift + level_bits * 2;
129        let level_two_size = 1usize << level_two_shift;
130
131        if props.page_shift == 12 {
132            // A 4 KiB level-two block is exactly 1 GiB, matching the loader's
133            // initial device/normal memory split.
134            for index in 0..TEMPORARY_IDENTITY_SIZE / level_two_size {
135                let paddr = PhysAddr::from_usize(index * level_two_size);
136                let flags = if index == 0 {
137                    device_flags
138                } else {
139                    normal_flags
140                };
141                table.add(index).write(A64PTE::new_page(paddr, flags, true));
142            }
143        } else {
144            // A 16 KiB or 64 KiB level-two block is larger than the loader's
145            // first 1 GiB device window. Split its first block at level one so
146            // kernel code remains normal memory while MMIO remains Device.
147            let next_table = root.add((table_slot + 1) * TEMPORARY_TABLE_ENTRIES);
148            table
149                .add(0)
150                .write(A64PTE::new_table(PhysAddr::from_usize(next_table as usize)));
151
152            let level_one_shift = page_shift + level_bits;
153            let level_one_size = 1usize << level_one_shift;
154            let device_entries = TEMPORARY_DEVICE_SIZE / level_one_size;
155            for index in 0..entries_per_table {
156                let paddr = PhysAddr::from_usize(index * level_one_size);
157                let flags = if index < device_entries {
158                    device_flags
159                } else {
160                    normal_flags
161                };
162                next_table
163                    .add(index)
164                    .write(A64PTE::new_page(paddr, flags, true));
165            }
166
167            let level_two_entries = TEMPORARY_IDENTITY_SIZE / level_two_size;
168            for index in 1..level_two_entries {
169                let paddr = PhysAddr::from_usize(index * level_two_size);
170                table
171                    .add(index)
172                    .write(A64PTE::new_page(paddr, normal_flags, true));
173            }
174        }
175
176        PhysAddr::from_usize(root as usize)
177    }
178}

180/// Installs temporary roots and switches to a requested TCR geometry.
181///
182/// The function disables stage-one translation only while the old and new
183/// translation geometries are disconnected, then restores the saved SCTLR.
184pub(super) fn switch_to_temporary_page_tables(
185    mode: VirtAddrSpaceMode,
186    tcr: LocalRegisterCopy<u64, TCR_EL1::Register>,
187) {
188    let (lower, upper) = VirtAddrSpaceMode::into_dual_props(mode)
189        .expect("AArch64 does not support unified virtual address spaces");
190
191    let lower_root =
192        lower.map(|props| temporary_page_table_root(addr_of_mut!(TEMPORARY_LOWER_TABLES), props));
193    let upper_root =
194        upper.map(|props| temporary_page_table_root(addr_of_mut!(TEMPORARY_UPPER_TABLES), props));
195
196    MAIR_EL1.set(AARCH64_MAIR_EL1);
197
198    // Changing a translation granule while stage-one translation is enabled
199    // is not a valid live transition. The caller guarantees that the current
200    // mapping is an identity mapping, so execution remains at the same
201    // physical addresses during this short disabled interval.
202    let saved_sctlr = SCTLR_EL1.extract();
203    dsb(SY);
204    SCTLR_EL1.modify(SCTLR_EL1::M::Disable);
205    isb(SY);
206
207    if let Some(root) = lower_root {
208        TTBR0_EL1.set_baddr(root.as_usize() as u64);
209    }
210    if let Some(root) = upper_root {
211        TTBR1_EL1.set_baddr(root.as_usize() as u64);
212    }
213
214    dsb(SY);
215    TCR_EL1.set(tcr.get());
216    dsb(SY);
217    isb(SY);
218    unsafe { asm!("tlbi vmalle1is", options(nostack)) };
219    dsb(aarch64_cpu::asm::barrier::ISH);
220    SCTLR_EL1.set(saved_sctlr.get());
221    isb(SY);
222}

224/// Tests temporary AArch64 translation-table configuration.
225///
226/// These tests keep the memory-attribute contract next to the temporary-table
227/// implementation.
228#[cfg(test)]
229mod tests {
230    use super::AARCH64_MAIR_EL1;
231
232    /// Checks the memory attributes used by temporary mappings.
233    ///
234    /// The values must remain compatible with the loader's device and normal
235    /// memory attributes.
236    #[test]
237    fn configures_temporary_mapping_attributes() {
238        assert_eq!(AARCH64_MAIR_EL1 & 0xff, 0x04);
239        assert_eq!((AARCH64_MAIR_EL1 >> 8) & 0xff, 0xff);
240    }
