14/// The PL011 data-register offset.
15const UARTDR: usize = 0x00;
16/// The PL011 flag-register offset.
17const UARTFR: usize = 0x18;
18/// Indicates that the transmit FIFO is full.
19const FR_TXFF: u32 = 1 << 5;
20/// Indicates that the receive FIFO is empty.
21const FR_RXFE: u32 = 1 << 4;
22
23/// The discovered physical PL011 base address, or zero before initialization.
24static PL011_BASE: AtomicUsize = AtomicUsize::new(0);
25/// The discovered PL011 register span.
26static PL011_SIZE: AtomicUsize = AtomicUsize::new(0);
27
28/// Discovers the enabled PL011 MMIO range without allocating memory.
29///
30/// This early parser supports the 64-bit address and size cells used by the AArch64 QEMU platform.
31/// Invalid or absent required firmware data returns `None` so initialization can halt before any
32/// access to an unverified device address.
33fn probe_pl011(arg: PlatformBootArg) -> Option<(PhysAddr, usize)> {
34    let PlatformBootArg::DeviceTree(dtb_addr) = arg else {
35        return None;
36    };
37    let dtb_ptr = crate::kernel_if::phys_to_virt(dtb_addr).as_ptr();
38    // SAFETY: the loader supplies a valid DTB pointer and the early identity/direct mapping keeps
39    // the blob accessible until the VMM transition completes.
40    let dtb = unsafe { DevTree::from_raw_pointer(dtb_ptr).ok()? };
41    for node in dtb.nodes().iterator() {
42        let node = node.ok()?;
43        if !node.is_compatible("arm,pl011").ok()? || !node.is_enabled().ok()? {
44            continue;
45        }
46        let base = usize::try_from(node.u64_property("reg", 0).ok()??).ok()?;
47        let size = usize::try_from(node.u64_property("reg", 1).ok()??).ok()?;
48        // The DTB supplies the only early device boundary. Reject an empty, null, or overflowing
49        // span here so later register accesses cannot escape it. UARTFR is the highest register
50        // touched by this console, so the range must cover that register in full.
51        let required_size = UARTFR + core::mem::size_of::<u32>();
52        if base == 0 || size < required_size || base.checked_add(size).is_none() {
53            return None;
54        }
55        return Some((PhysAddr::from_usize(base), size));
56    }
57    None
58}

60/// Returns the discovered console register address for volatile access.
61fn register(offset: usize) -> *mut u32 {
62    let base = PL011_BASE.load(Ordering::Acquire);
63    assert_ne!(base, 0, "AArch64 early console is not initialized");
64    // The register offset is a compile-time console constant, but the firmware may describe a
65    // shorter span. Check it before forming the MMIO pointer so malformed DT data cannot make a
66    // console access touch a neighboring device.
67    assert!(
68        offset
69            .checked_add(core::mem::size_of::<u32>())
70            .is_some_and(|end| { end <= PL011_SIZE.load(Ordering::Relaxed) }),
71        "AArch64 early console register exceeds its discovered MMIO range"
72    );
73    // SAFETY: the base and register offset were validated against the firmware-described span.
74    crate::kernel_if::phys_to_virt(PhysAddr::from_usize(base + offset)).as_mut_ptr_of()
75}

77/// Writes bytes through the discovered PL011 transmit FIFO.
78pub fn write_bytes(bytes: &[u8]) {
79    for &byte in bytes {
80        // SAFETY: register validates the firmware-described span before returning this pointer.
81        while unsafe { register(UARTFR).read_volatile() } & FR_TXFF != 0 {
82            core::hint::spin_loop();
83        }
84        // SAFETY: register validates the firmware-described span before returning this pointer.
85        unsafe { register(UARTDR).write_volatile(byte as u32) };
86    }
87}

89/// Reads bytes immediately available in the discovered PL011 receive FIFO.
90pub fn read_bytes(bytes: &mut [u8]) -> usize {
91    let mut count = 0;
92    for byte in bytes {
93        // SAFETY: register validates the firmware-described span before returning this pointer.
94        if unsafe { register(UARTFR).read_volatile() } & FR_RXFE != 0 {
95            break;
96        }
97        // SAFETY: register validates the firmware-described span before returning this pointer.
98        *byte = unsafe { register(UARTDR).read_volatile() as u8 };
99        count += 1;
100    }
101    count
102}

104/// Discovers and stores the firmware-described PL011 console before exceptions are installed.
105///
106/// The supported early environment provides the DTB identity mapping and no allocator. Returns
107/// false when firmware data cannot identify a usable console.
108pub fn init(arg: PlatformBootArg) -> bool {
109    let Some((base, size)) = probe_pl011(arg) else {
110        return false;
111    };
112    PL011_SIZE.store(size, Ordering::Relaxed);
113    PL011_BASE.store(base.as_usize(), Ordering::Release);
114    true
115}

117/// Returns the discovered PL011 range for early and platform memory mapping.
118pub fn mmio_region() -> Option<MemoryRegion> {
119    let base = PL011_BASE.load(Ordering::Acquire);
120    (base != 0).then(|| MemoryRegion {
121        range: PhysAddrRange::from_start_size(
122            PhysAddr::from_usize(base),
123            PL011_SIZE.load(Ordering::Relaxed),
124        ),
125        flags: DEFAULT_MMIO_FLAGS,
126        desc: "PL011 early console",
127    })
