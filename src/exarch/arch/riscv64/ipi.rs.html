19/// Enables supervisor software delivery after installing common mailboxes.
20///
21/// Global interrupt enabling remains part of common per-CPU runtime initialization.
22pub fn init_percpu() {
23    // SAFETY: The common software handler is installed before this source is enabled.
24    unsafe {
25        sip::clear_ssoft();
26        supervisor::enable_interrupt(Interrupt::SupervisorSoft);
27    }
28}
29
30/// Sends one SBI notification to a physical hart identifier.
31///
32/// Rejected calls initiate no delivery and leave retry cleanup to the common publisher.
33pub fn send(target: PhysicalCpuId) -> Result<(), IpiError> {
34    let (mask, base) = hart_mask_parts(target)?;
35    let result = sbi_rt::send_ipi(sbi_rt::HartMask::from_mask_base(mask, base));
36    map_sbi_error(result.error)
37}
