35/// Installs the supervisor trap vector for the current hart.
36///
37/// The caller must invoke [`after_reloc`] after code addresses change through relocation.
38pub(super) fn init_percpu() {
39    let vector = stvec::Stvec::new(
40        trap_vector_base as *const () as usize,
41        stvec::TrapMode::Direct,
42    );
43    // SAFETY: the assembly symbol is aligned and remains valid until the next relocation hook.
44    unsafe { stvec::write(vector) };
45    super::trap::init_percpu();
46}
47
48/// Reloads the supervisor trap vector after relocation.
49///
50/// The current hart retains its interrupt-source mask.
51pub(super) fn after_reloc() {
52    let vector = stvec::Stvec::new(
53        trap_vector_base as *const () as usize,
54        stvec::TrapMode::Direct,
55    );
56    // SAFETY: the relocated assembly symbol is the active kernel trap entry.
57    unsafe { stvec::write(vector) };
58}

60/// Returns whether a trap interrupted user mode.
61///
62/// The saved previous privilege field describes the mode active before trap entry.
63fn is_user(tf: &TrapFrame) -> bool {
64    tf.sstatus.spp() == sstatus::SPP::User
65}
66
67/// Advances the saved program counter past a breakpoint instruction.
68///
69/// The low instruction bits distinguish compressed and ordinary encodings.
70fn handle_breakpoint(sepc: &mut usize) {
71    // SAFETY: `sepc` identifies the mapped instruction that raised the breakpoint exception.
72    let instruction = unsafe { va!(*sepc).as_ptr_of::<u16>().read_volatile() };
73    *sepc += if instruction & 0b11 == 0b11 { 4 } else { 2 };
74}

76/// Dispatches a RISC-V page fault to the common kernel callback.
77///
78/// An unhandled fault is fatal and includes the complete saved context in its diagnostic.
79#[cfg_or_dummy(target_arch = "riscv64")]
80fn handle_page_fault(tf: &mut TrapFrame, mut flags: PageFaultFlags) -> TrapDisposition {
81    let user = is_user(tf);
82    if user {
83        flags |= PageFaultFlags::USER;
84    }
85
86    let fault_addr = va!(stval::read());
87    crate::trap::handle(
88        tf,
89        Trap::Exception(Exception::PageFault {
90            address: fault_addr,
91            flags,
92            is_user: user,
93        }),
94    )
95}

97/// Dispatches one RISC-V supervisor trap.
98///
99/// This symbol is called directly by `trap.S` with a pointer to its stack-resident frame.
100#[cfg_or_dummy(target_arch = "riscv64")]
101#[unsafe(no_mangle)]
102extern "C" fn riscv_trap_handler(tf: &mut TrapFrame) {
103    let scause = scause::read();
104    let is_interrupt = scause.is_interrupt();
105
106    let result = match scause.cause().try_into() {
107        Ok(scause_decoded) => valid_riscv_trap_handler(tf, scause_decoded, scause.bits()),
108        Err(e) => {
109            error!(
110                "Unknown RISC-V scause: {scause:#x} with stval={stval:#x} @ {sepc:#x}, {e:?}, continue anyway...",
111                scause = scause.bits(),
112                stval = stval::read(),
113                sepc = tf.sepc,
114            );
115
116            unknown_riscv_trap_handler(tf, scause)
117        }
118    };
119
120    if result != TrapDisposition::Handled {
121        warn!(
122            "Unhandled RISC-V trap: {scause:#x} with stval={stval:#x} @ {sepc:#x}",
123            scause = scause.bits(),
124            stval = stval::read(),
125            sepc = tf.sepc
126        );
127    }
128    if is_interrupt {
129        // SSIP is cleared, the timer has been rearmed, or all PLIC claims have completed.
130        crate::trap::on_interrupt_exit(tf);
131    }
132}

134/// Handles a RISC-V trap that is known to the kernel.
135#[cfg_or_dummy(target_arch = "riscv64")]
136fn valid_riscv_trap_handler(
137    tf: &mut TrapFrame,
138    scause: RiscVTrap<RiscVInterrupt, RiscVException>,
139    raw_bits: usize,
140) -> TrapDisposition {
141    match scause {
142        RiscVTrap::Interrupt(RiscVInterrupt::SupervisorTimer) => {
143            crate::trap::handle(tf, Trap::Interrupt(Interrupt::Local(LocalInterrupt::Timer)))
144        }
145        RiscVTrap::Interrupt(RiscVInterrupt::SupervisorExternal) => {
146            super::trap::handle_external(tf)
147        }
148        RiscVTrap::Interrupt(RiscVInterrupt::SupervisorSoft) => {
149            // SAFETY: Clear the local source before dispatch so a publication during the
150            // mailbox drain can set SSIP again and cannot be lost by a later clear.
151            unsafe {
152                sip::clear_ssoft();
153                // CSR acknowledgement precedes the common mailbox's memory accesses.
154                core::arch::asm!("fence io, rw", options(nostack));
155            }
156            crate::trap::handle(
157                tf,
158                Trap::Interrupt(Interrupt::Local(LocalInterrupt::Software)),
159            )
160        }
161        RiscVTrap::Exception(RiscVException::InstructionPageFault) => {
162            handle_page_fault(tf, PageFaultFlags::EXECUTE)
163        }
164        RiscVTrap::Exception(RiscVException::LoadPageFault) => {
165            handle_page_fault(tf, PageFaultFlags::READ)
166        }
167        RiscVTrap::Exception(RiscVException::StorePageFault) => {
168            handle_page_fault(tf, PageFaultFlags::WRITE)
169        }
170        RiscVTrap::Exception(RiscVException::Breakpoint) => {
171            handle_breakpoint(&mut tf.sepc);
172            crate::trap::handle(tf, Trap::Exception(Exception::Breakpoint))
173        }
174        RiscVTrap::Exception(RiscVException::IllegalInstruction) => {
175            crate::trap::handle(tf, Trap::Exception(Exception::InvalidInstruction))
176        }
177        RiscVTrap::Exception(other) => {
178            warn!(
179                "Unsupported RISC-V exception: {other:?} with stval={:#x} @ {:#x}",
180                stval::read(),
181                tf.sepc
182            );
183            crate::trap::handle(
184                tf,
185                Trap::Exception(Exception::Unknown(crate::trap::RawTrap(raw_bits))),
186            )
187        }
188    }
189}

191/// Handles a RISC-V trap that is unknown to the kernel.
192#[cfg_or_dummy(target_arch = "riscv64")]
193fn unknown_riscv_trap_handler(tf: &mut TrapFrame, scause: Scause) -> TrapDisposition {
194    let raw_trap = RawTrap(scause.bits());
195
196    crate::trap::handle(
197        tf,
198        if scause.is_exception() {
199            Trap::Exception(Exception::Unknown(raw_trap))
200        } else {
201            Trap::Interrupt(Interrupt::Unknown(raw_trap))
202        },
203    )
