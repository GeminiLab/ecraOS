10use crate::{

12    platform::{
13        CpuTopology, PlatformDescription, PlatformInterruptController,
14        dt::{
15            DeviceTreeNodeExt,
16            index::{DeviceTreeIndexExt, with_index},

22/// Discovers the RISC-V CPU topology and PLIC from Device Tree.
23#[cfg_or_dummy(target_arch = "riscv64")]
24pub fn discover_platform(arg: PlatformBootArg) -> PlatformDescription {
25    let PlatformBootArg::DeviceTree(dtb_addr) = arg else {
26        panic!("RISC-V platform discovery requires a Device Tree")
27    };
28    // SAFETY: the loader supplies a valid DTB that remains mapped during platform discovery.
29    let dtb = unsafe {
30        DevTree::from_raw_pointer(crate::kernel_if::phys_to_virt(dtb_addr).as_ptr())
31            .expect("failed to load device tree")
32    };
33    with_index(dtb, |index| {
34        let dtb_boot_cpu = index.fdt().boot_cpuid_phys() as PhysicalCpuId;
35        let boot_cpu = crate::kernel_if::current_cpu_phys_id();
36        if dtb_boot_cpu != boot_cpu {
37            log::warn!(
38                "Device Tree boot CPU ID {dtb_boot_cpu:#x} differs from runtime hart ID {boot_cpu:#x}; using runtime hart ID as BSP"
39            );
40        }
41        let cpus = index
42            .find_path("/cpus")?
43            .expect("Device Tree has no /cpus node");
44        let mut cpu_ids = Vec::new();
45        for node in cpus.children() {
46            if !node.is_device_type("cpu")? || !node.is_enabled()? {
47                continue;
48            }
49            let cpu_id = node
50                .u32_property("reg", 0)?
51                .expect("CPU node has no reg property") as PhysicalCpuId;
52            cpu_ids.push(cpu_id);
53        }
54        let cpu_topology = CpuTopology::new(cpu_ids.into_boxed_slice(), boot_cpu);
55
56        let plic_node = match index.find_compatible("sifive,plic-1.0.0")? {
57            Some(node) => node,
58            None => index
59                .find_compatible("riscv,plic0")?
60                .expect("Device Tree has no supported PLIC node"),
61        };
62        assert!(plic_node.is_enabled()?, "Device Tree PLIC is disabled");
63        let plic_reg = plic_node
64            .property("reg")?
65            .expect("PLIC node has no reg property");
66        let plic_ndev = plic_node
67            .u32_property("riscv,ndev", 0)?
68            .expect("PLIC node has no riscv,ndev property") as usize;
69        let interrupts = plic_node
70            .property("interrupts-extended")?
71            .expect("PLIC node has no interrupts-extended property");
72
73        let mut interrupt_controllers = BTreeMap::new();
74        for node in index.nodes() {
75            if !node.has_property("interrupt-controller")? {
76                continue;
77            }
78            let Some(phandle) = node.phandle_property("phandle")? else {
79                continue;
80            };
81            let Some(cpu_node) = node.parent() else {
82                continue;
83            };
84            if !cpu_node.is_device_type("cpu")? {
85                continue;
86            }
87            let Some(hart_id) = cpu_node.u32_property("reg", 0)? else {
88                continue;
89            };
90            interrupt_controllers.insert(phandle, hart_id as PhysicalCpuId);
91        }
92
93        let mut contexts = Vec::new();
94        for entry in 0..(interrupts.length() / 8) {
95            let phandle = interrupts
96                .phandle(entry * 2)
97                .expect("invalid PLIC context phandle");
98            let cause = interrupts
99                .u32(entry * 2 + 1)
100                .expect("invalid PLIC context cause");
101            if cause == 9
102                && let Some(&hart_id) = interrupt_controllers.get(&phandle)
103            {
104                contexts.push(RiscvPlicContext {
105                    hart_id,
106                    context: entry,
107                });
108            }
109        }
110        assert!(!contexts.is_empty(), "PLIC has no supervisor contexts");
111
112        let controller = RiscvExternalIrqConfig {
113            physical_base: PhysAddr::from_usize(
114                plic_reg.u64(0).expect("invalid PLIC base") as usize
115            ),
116            size: plic_reg.u64(1).expect("invalid PLIC size") as usize,
117            source_count: plic_ndev,
118            contexts: contexts.into_boxed_slice(),
119        };
120        let mmio_regions = alloc::vec![MemoryRegion {
121            range: memory_addr::PhysAddrRange::from_start_size(
122                controller.physical_base,
123                controller.size,
124            ),
125            flags: DEFAULT_MMIO_FLAGS,
126            desc: "PLIC registers",
127        }];
128        Ok(PlatformDescription::new(
129            cpu_topology,
130            PlatformInterruptController(controller),
131            mmio_regions,
132        ))
133    })
134    .expect("failed to discover RISC-V platform from Device Tree")
135}
