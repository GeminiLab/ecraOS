5fn current_ticks() -> Ticks {
6    Ticks(riscv::register::time::read() as _)
7}
8
9static mut INIT_TICK: Ticks = Ticks(0);
10static mut TIME_FREQ_KHZ: u64 = 0;
11
12/// Time frequency used when calibration fails (10 MHz). It's almost definitely wrong, but it should
13/// be enough for the system to boot and enumerate other time devices.
14const TIME_FREQ_HZ_BLIND: u64 = 10_000_000;
15
16/// Returns monotonic RISC-V time ticks since early initialization.
17pub fn monotonic_ticks() -> Ticks {
18    unsafe { Ticks(current_ticks().0 - INIT_TICK.0) }
19}
20
21/// Converts RISC-V time ticks to nanoseconds.
22pub fn ticks_to_nanos(ticks: Ticks) -> Nanos {
23    unsafe { Nanos(ticks.0 * 1_000_000 / TIME_FREQ_KHZ) }
24}
25
26/// Converts nanoseconds to RISC-V time ticks.
27pub fn nanos_to_ticks(nanos: Nanos) -> Ticks {
28    unsafe { Ticks(nanos.0 * TIME_FREQ_KHZ / 1_000_000) }
29}
30
31/// Programs a RISC-V one-shot timer for an absolute monotonic deadline.
32pub fn set_oneshot_timer(deadline: TimeValue) {
33    let deadline_ns = u64::try_from(deadline.as_nanos())
34        .expect("timer deadline does not fit in the RISC-V time domain");
35    let relative_ticks = nanos_to_ticks(Nanos(deadline_ns));
36    let absolute_ticks = unsafe {
37        INIT_TICK
38            .0
39            .checked_add(relative_ticks.0)
40            .expect("RISC-V timer deadline overflow")
41    };
42    sbi_rt::set_timer(absolute_ticks).expect("SBI TIME set_timer failed");
43}

45pub fn init() {
46    let dtb = super::DTB.get().expect("Device Tree not initialized");
47
48    let cpus = dtb
49        .nodes()
50        .find(|n| n.name().map(|name| name == "cpus"))
51        .ok()
52        .flatten();
53    let freq_prop = cpus.and_then(|cpus| {
54        cpus.props()
55            .find(|p| p.name().map(|name| name == "timebase-frequency"))
56            .ok()
57            .flatten()
58    });
59    let freq_hz = freq_prop
60        .and_then(|p| p.u32(0).ok())
61        .unwrap_or(TIME_FREQ_HZ_BLIND as _);
62
63    unsafe {
64        TIME_FREQ_KHZ = (freq_hz / 1_000) as _;
65        INIT_TICK = current_ticks();
66    }
67}
