32/// The description attached to device-tree MMIO regions.
33///
34/// These ranges are reserved from allocation and mapped with device-memory attributes.
35const DEVICE_MMIO_DESC: &str = "device-tree MMIO";
36
37/// The description attached to the device tree memory range.
38///
39/// Identifies the reserved device-tree blob rather than MMIO described by its nodes.
40const DEVICE_TREE_DESC: &str = "device tree";
41
42/// The alignment used for the device tree memory reservation.
43///
44/// Covers whole base pages so the blob cannot share an allocatable page with RAM.
45const DEVICE_TREE_ALIGN: usize = 4096;
46
47/// Collects physical memory regions reported by the device tree.
48///
49/// Subtracts the page-aligned device-tree blob from each RAM range and reports it once as
50/// reserved device memory, preserving usable RAM on both sides.
51pub fn raw_mem_regions(arg: PlatformBootArg) -> RawMemoryRegions {
52    let mut regions = RawMemoryRegions::new();
53
54    if let PlatformBootArg::DeviceTree(dtb_addr) = arg {
55        let dev_tree = unsafe {
56            DevTree::from_raw_pointer(VirtAddr::from_usize(dtb_addr.as_usize()).as_ptr())
57                .expect("failed to parse device tree")
58        };
59        let dtb_range = PhysAddrRange::from_start_size(dtb_addr, dev_tree.totalsize())
60            .align_outwards(DEVICE_TREE_ALIGN)
61            .expect("device tree range alignment overflow");
62
63        let mems = dev_tree.nodes().filter(|node| {
64            Ok(node
65                .props()
66                .find(|p| Ok(p.name()? == "device_type" && p.str()? == "memory"))
67                .is_ok_and(|p| p.is_some()))
68        });
69
70        for mem in mems.iterator() {
71            let mem = mem.unwrap();
72            let reg = mem
73                .props()
74                .find(|p| Ok(p.name()? == "reg"))
75                .unwrap()
76                .unwrap();
77
78            let mut start = reg.u64(0).unwrap();
79            let mut size = reg.u64(1).unwrap();
80
81            // TODO: remove this magic number
82            if start == 0x8000_0000 {
83                start = 0x8020_0000;
84                size -= 0x20_0000;
85            }
86
87            let memory_range = PhysAddrRange::from_start_size(pa!(start as _), size as _);
88            let (before_dtb, after_dtb) = memory_range.subtract(dtb_range);
89            for range in [before_dtb, after_dtb]
90                .into_iter()
91                .flatten()
92                .filter(|range| !range.is_empty())
93            {
94                regions
95                    .push(MemoryRegion {
96                        range,
97                        flags: DEFAULT_RAM_FLAGS,
98                        desc: DEFAULT_RAM_DESC,
99                    })
100                    .expect("too many platform memory regions");
101            }
102        }
103
104        // TODO: handle other devices
105        for node in dev_tree.nodes().iterator() {
106            let node = node.expect("failed to inspect Device Tree node");
107            let supported = node
108                .props()
109                .any(|prop| {
110                    Ok(prop.name()? == "compatible"
111                        && prop.str().is_ok_and(|compatible| {
112                            matches!(compatible, "sifive,plic-1.0.0" | "riscv,plic0" | "ns16550a")
113                        }))
114                })
115                .expect("failed to inspect Device Tree compatible property");
116            if !supported {
117                continue;
118            }
119            let Some(reg) = node
120                .props()
121                .find(|prop| Ok(prop.name()? == "reg"))
122                .expect("failed to inspect Device Tree reg property")
123            else {
124                continue;
125            };
126            let start = reg.u64(0).expect("invalid device MMIO base");
127            let size = reg.u64(1).expect("invalid device MMIO size");
128            regions
129                .push(MemoryRegion {
130                    range: PhysAddrRange::from_start_size(pa!(start as usize), size as usize),
131                    flags: MemoryRegionFlags::READ
132                        | MemoryRegionFlags::WRITE
133                        | MemoryRegionFlags::DEVICE
134                        | MemoryRegionFlags::RESERVED,
135                    desc: DEVICE_MMIO_DESC,
136                })
137                .expect("too many platform memory regions");
138        }
139
140        regions
141            .push(MemoryRegion {
142                range: dtb_range,
143                flags: MemoryRegionFlags::READ
144                    | MemoryRegionFlags::WRITE
145                    | MemoryRegionFlags::RESERVED,
146                desc: DEVICE_TREE_DESC,
147            })
148            .expect("too many platform memory regions");
149    }
150
151    regions
152}

154/// Returns supported and current RISC-V virtual address-space modes.
155pub fn virt_addr_space_modes() -> VirtAddrSpaceModes {
156    unsafe extern "C" {
157        fn detect_max_va_bits() -> usize;
158        fn detect_current_va_bits() -> usize;
159    }
160
161    let mut modes = VirtAddrSpaceModes::new();
162    modes.current_index = usize::MAX;
163    let max_va_bits = unsafe { detect_max_va_bits() };
164    let current_va_bits = unsafe { detect_current_va_bits() };
165
166    macro_rules! check_va_mode {
167        ($va_bits:expr) => {
168            if max_va_bits >= $va_bits {
169                let _ = modes
170                    .modes
171                    .push(VirtAddrSpaceMode::Unified(VirtAddrSpaceProps {
172                        page_shift: 12,
173                        va_bits: $va_bits,
174                    }));
175
176                if current_va_bits == $va_bits {
177                    modes.current_index = modes.modes.len() - 1;
178                }
179            }
180        };
181    }
182
183    check_va_mode!(57);
184    check_va_mode!(48);
185    check_va_mode!(39);
186
187    if max_va_bits < 39 {
188        panic!(
189            "This machine seems not to support virtual address space, max_va_bits is {}",
190            max_va_bits
191        );
192    }
193    if modes.current_index >= modes.modes.len() {
194        panic!("Unexpected current_va_bits: {}", current_va_bits);
195    }
196
197    modes
198}

200/// Switches to the requested RISC-V virtual address-space mode.
201pub fn set_virt_addr_space_mode(mode: VirtAddrSpaceMode) {
202    if let VirtAddrSpaceMode::Unified(VirtAddrSpaceProps {
203        page_shift: 12,
204        va_bits,
205    }) = mode
206    {
207        unsafe extern "C" {
208            fn set_va_bits(mode: usize) -> bool;
209        }
210
211        if !unsafe { set_va_bits(va_bits as _) } {
212            panic!("Failed to set virtual address space mode: {:?}", mode);
213        }
214    } else {
215        panic!("Unexpected virtual address space mode: {:?}", mode);
216    }
217}

219/// Returns the page-table descriptor for a RISC-V mode and caller-selected strategies.
220///
221/// The selected Sv39, Sv48, or Sv57 format captures `AL` for all operations that access,
222/// allocate, or deallocate table frames, and `F` for pending cleanup. See
223/// [`crate::mem::get_page_table_type`] for caller requirements.
224pub fn get_page_table_type<AL: PageAllocator, F: TlbFlusher<VirtAddr>>(
225    mode: VirtAddrSpaceMode,
226) -> OpaquePageTableType<VirtAddr> {
227    if let VirtAddrSpaceMode::Unified(VirtAddrSpaceProps {
228        page_shift: 12,
229        va_bits,
230    }) = mode
231    {
232        if va_bits == 57 {
233            OpaquePageTableType::new::<Sv57PageTableMeta, Rv64PTE, AL, F>()
234        } else if va_bits == 48 {
235            OpaquePageTableType::new::<Sv48PageTableMeta, Rv64PTE, AL, F>()
236        } else if va_bits == 39 {
237            OpaquePageTableType::new::<Sv39PageTableMeta, Rv64PTE, AL, F>()
238        } else {
239            panic!("Unexpected virtual address space mode: {:?}", mode);
240        }
241    } else {
242        panic!("Unexpected virtual address space mode: {:?}", mode);
243    }
244}

246/// Sets the RISC-V page-table root.
247pub fn set_page_table_root(root: OpaquePageTableRoot) {
248    let OpaquePageTableRoot::Single(root) = root;
249    let root = root.as_usize();
250
251    unsafe {
252        core::arch::asm!(
253            "csrr {satp}, satp",
254            "srli {root}, {root}, 12",      // Get the PPN of the root page table
255            "srli {satp}, {satp}, 44",
256            "slli {satp}, {satp}, 44",
257            "or {satp}, {satp}, {root}",    // Set the new root PPN
258            "csrw satp, {satp}",
259            "sfence.vma",
260            satp = out(reg) _,
261            root = in(reg) root,
262        )
263    }
