35/// The description attached to the device tree memory range.
36///
37/// Identifies the reserved device-tree blob rather than MMIO described by its nodes.
38const DEVICE_TREE_DESC: &str = "device tree";
39
40/// The alignment used for the device tree memory reservation.
41///
42/// Covers whole base pages so the blob cannot share an allocatable page with RAM.
43const DEVICE_TREE_ALIGN: usize = 4096;
44
45/// Collects usable RAM and regions that must remain mapped during early boot.
46///
47/// Reserves the page-aligned device-tree blob as boot data and fills uncovered RAM around it.
48/// Later-discovered devices are mapped separately.
49///
50/// # Panics
51///
52/// Panics if the output table is not empty, memory data is invalid, or capacity is insufficient.
53pub fn raw_mem_regions(arg: PlatformBootArg, regions: &mut RawMemoryRegions) {
54    assert!(
55        regions.is_empty(),
56        "raw memory discovery requires an empty table"
57    );
58
59    if let PlatformBootArg::DeviceTree(dtb_addr) = arg {
60        let dev_tree = unsafe {
61            DevTree::from_raw_pointer(VirtAddr::from_usize(dtb_addr.as_usize()).as_ptr())
62                .expect("failed to parse device tree")
63        };
64        let dtb_range = PhysAddrRange::from_start_size(dtb_addr, dev_tree.totalsize())
65            .align_outwards(DEVICE_TREE_ALIGN)
66            .expect("device tree range alignment overflow");
67
68        regions
69            .insert_disjoint(MemoryRegion {
70                range: dtb_range,
71                flags: MemoryRegionFlags::READ
72                    | MemoryRegionFlags::WRITE
73                    | MemoryRegionFlags::BOOT_SERVICE,
74                desc: DEVICE_TREE_DESC,
75            })
76            .expect("invalid device tree memory region");
77
78        let mems = dev_tree
79            .nodes()
80            .filter(|node| node.is_device_type("memory"));
81
82        for mem in mems.iterator() {
83            let mem = mem.expect("failed to inspect memory device type");
84            let reg = mem
85                .property("reg")
86                .expect("failed to inspect memory reg")
87                .expect("memory node has no reg property");
88
89            let start = reg.u64(0).unwrap();
90            let size = reg.u64(1).unwrap();
91            let memory_range = PhysAddrRange::from_start_size(pa!(start as _), size as _);
92            regions
93                .insert_uncovered(MemoryRegion {
94                    range: memory_range,
95                    flags: DEFAULT_RAM_FLAGS,
96                    desc: DEFAULT_RAM_DESC,
97                })
98                .expect("invalid or excessive platform RAM regions");
99        }
100
101        apply_device_tree_reservations(dev_tree, regions)
102            .expect("failed to inspect device-tree memory reservations");
103    }
104}

106/// Returns supported and current RISC-V virtual address-space modes.
107pub fn virt_addr_space_modes() -> VirtAddrSpaceModes {
108    unsafe extern "C" {
109        fn detect_max_va_bits() -> usize;
110        fn detect_current_va_bits() -> usize;
111    }
112
113    let mut modes = VirtAddrSpaceModes::new();
114    modes.current_index = usize::MAX;
115    let max_va_bits = unsafe { detect_max_va_bits() };
116    let current_va_bits = unsafe { detect_current_va_bits() };
117
118    macro_rules! check_va_mode {
119        ($va_bits:expr) => {
120            if max_va_bits >= $va_bits {
121                let _ = modes
122                    .modes
123                    .push(VirtAddrSpaceMode::Unified(VirtAddrSpaceProps {
124                        page_shift: 12,
125                        va_bits: $va_bits,
126                    }));
127
128                if current_va_bits == $va_bits {
129                    modes.current_index = modes.modes.len() - 1;
130                }
131            }
132        };
133    }
134
135    check_va_mode!(57);
136    check_va_mode!(48);
137    check_va_mode!(39);
138
139    if max_va_bits < 39 {
140        panic!(
141            "This machine seems not to support virtual address space, max_va_bits is {}",
142            max_va_bits
143        );
144    }
145    if modes.current_index >= modes.modes.len() {
146        panic!("Unexpected current_va_bits: {}", current_va_bits);
147    }
148
149    modes
150}

152/// Switches to the requested RISC-V virtual address-space mode.
153pub fn set_virt_addr_space_mode(mode: VirtAddrSpaceMode) {
154    if let VirtAddrSpaceMode::Unified(VirtAddrSpaceProps {
155        page_shift: 12,
156        va_bits,
157    }) = mode
158    {
159        unsafe extern "C" {
160            fn set_va_bits(mode: usize) -> bool;
161        }
162
163        if !unsafe { set_va_bits(va_bits as _) } {
164            panic!("Failed to set virtual address space mode: {:?}", mode);
165        }
166    } else {
167        panic!("Unexpected virtual address space mode: {:?}", mode);
168    }
169}

171/// Returns the page-table descriptor for a RISC-V mode and caller-selected strategies.
172///
173/// The selected Sv39, Sv48, or Sv57 format captures `AL` for all operations that access,
174/// allocate, or deallocate table frames, and `F` for pending cleanup. See
175/// [`crate::mem::get_page_table_type`] for caller requirements.
176pub fn get_page_table_type<AL: PageAllocator, F: TlbFlusher<VirtAddr>>(
177    mode: VirtAddrSpaceMode,
178) -> OpaquePageTableType<VirtAddr> {
179    if let VirtAddrSpaceMode::Unified(VirtAddrSpaceProps {
180        page_shift: 12,
181        va_bits,
182    }) = mode
183    {
184        if va_bits == 57 {
185            OpaquePageTableType::new::<Sv57PageTableMeta, Rv64PTE, AL, F>()
186        } else if va_bits == 48 {
187            OpaquePageTableType::new::<Sv48PageTableMeta, Rv64PTE, AL, F>()
188        } else if va_bits == 39 {
189            OpaquePageTableType::new::<Sv39PageTableMeta, Rv64PTE, AL, F>()
190        } else {
191            panic!("Unexpected virtual address space mode: {:?}", mode);
192        }
193    } else {
194        panic!("Unexpected virtual address space mode: {:?}", mode);
195    }
196}

198/// Sets the RISC-V page-table root.
199pub fn set_page_table_root(root: OpaquePageTableRoot) {
200    let OpaquePageTableRoot::Single(root) = root;
201    let root = root.as_usize();
202
203    unsafe {
204        core::arch::asm!(
205            "csrr {satp}, satp",
206            "srli {root}, {root}, 12",      // Get the PPN of the root page table
207            "srli {satp}, {satp}, 44",
208            "slli {satp}, {satp}, 44",
209            "or {satp}, {satp}, {root}",    // Set the new root PPN
210            "csrw satp, {satp}",
211            "sfence.vma",
212            satp = out(reg) _,
213            root = in(reg) root,
214        )
215    }
