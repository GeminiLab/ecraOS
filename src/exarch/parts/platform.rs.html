15/// The firmware-discovered physical CPU topology.
16///
17/// The architecture-specific platform parser owns the interpretation of CPU identifiers. The
18/// common kernel consumes the identifiers when it builds its logical CPU inventory.
19// TODO: make this struct the real topology record, not merely the CPU list
20pub struct CpuTopology {
21    /// The physical identifiers of all CPUs available to the kernel.
22    cpu_ids: Box<[PhysicalCpuId]>,
23    /// The physical identifier of the bootstrap CPU.
24    bsp_id: PhysicalCpuId,
25}
26
27impl CpuTopology {
28    /// Creates a CPU topology from firmware-discovered identifiers.
29    ///
30    /// The list must be non-empty and must contain the bootstrap identifier. Duplicate identifiers
31    /// are rejected because they cannot be represented by the kernel's physical-to-logical map.
32    pub(crate) fn new(cpu_ids: Box<[PhysicalCpuId]>, bsp_id: PhysicalCpuId) -> Self {
33        assert!(!cpu_ids.is_empty(), "platform discovered no CPUs");
34        assert!(
35            cpu_ids.contains(&bsp_id),
36            "platform CPU list does not contain the bootstrap CPU"
37        );
38        for (index, cpu_id) in cpu_ids.iter().enumerate() {
39            assert!(
40                !cpu_ids[..index].contains(cpu_id),
41                "platform discovered duplicate CPU identifier {cpu_id:#x}"
42            );
43        }
44        Self { cpu_ids, bsp_id }
45    }
46
47    /// Returns the discovered physical CPU identifiers in firmware order.
48    pub fn cpu_ids(&self) -> &[PhysicalCpuId] {
49        &self.cpu_ids
50    }
51
52    /// Returns the physical identifier of the bootstrap CPU.
53    pub const fn bsp_id(&self) -> PhysicalCpuId {
54        self.bsp_id
55    }
56
57    /// Transfers the discovered CPU identifiers and bootstrap identifier to the kernel CPU manager.
58    pub fn into_parts(self) -> (Box<[PhysicalCpuId]>, PhysicalCpuId) {
59        (self.cpu_ids, self.bsp_id)
60    }
61}

63/// Discovered architecture-specific interrupt-controller configuration.
64///
65/// The architecture parser creates this value from firmware data. The value owns the
66/// architecture-specific controller configuration. After mapping its MMIO ranges, the BSP
67/// publishes the configuration for exarch initialization hooks to consume. Architecture code
68/// retains initialized controller state for local-controller setup on the BSP and APs.
69pub struct PlatformInterruptController(pub ArchIrqConfig);
70
71/// The pending discovered controller configuration before BSP controller initialization.
72static INTERRUPT_CONTROLLER_CONFIG: LazyInit<SpinNoIrq<Option<ArchIrqConfig>>> = LazyInit::new();
73
74impl PlatformInterruptController {
75    /// Publishes the discovered configuration for BSP controller initialization.
76    ///
77    /// Call this after the MMIO regions in the containing platform description are mapped. The
78    /// BSP later-initialization hook consumes the configuration once and stores any state required
79    /// by local-controller setup on the BSP and APs.
80    ///
81    /// # Panics
82    ///
83    /// Panics if another controller configuration was already published.
84    pub fn publish(self) {
85        INTERRUPT_CONTROLLER_CONFIG.init_once(SpinNoIrq::new(Some(self.0)));
86    }
87}
88
89/// Takes the discovered controller configuration for one-time global-controller initialization.
90///
91/// The BSP calls this only after platform discovery and MMIO mapping. The architecture controller
92/// then retains the shared state needed by later BSP/AP-local initialization.
93///
94/// # Panics
95///
96/// Panics if no configuration was published or the configuration was already consumed.
97pub(crate) fn take_interrupt_controller_config() -> ArchIrqConfig {
98    INTERRUPT_CONTROLLER_CONFIG
99        .get()
100        .expect("platform interrupt-controller configuration was not published")
101        .lock()
102        .take()
103        .expect("platform interrupt-controller configuration was already consumed")
104}

106/// The platform data needed by the common kernel during early device initialization.
107pub struct PlatformDescription {
108    /// The firmware-discovered CPU topology.
109    cpu_topology: CpuTopology,
110    /// The platform interrupt-controller driver.
111    interrupt_controller: PlatformInterruptController,
112    /// The device MMIO ranges that need direct mappings before controller initialization.
113    mmio_regions: Vec<MemoryRegion>,
114}
115
116impl PlatformDescription {
117    /// Creates a platform description from its discovered components.
118    pub(crate) fn new(
119        cpu_topology: CpuTopology,
120        interrupt_controller: PlatformInterruptController,
121        mmio_regions: Vec<MemoryRegion>,
122    ) -> Self {
123        Self {
124            cpu_topology,
125            interrupt_controller,
126            mmio_regions,
127        }
128    }
129
130    /// Separates the CPU topology, interrupt-controller configuration, and MMIO ranges.
131    pub fn into_parts(self) -> (CpuTopology, PlatformInterruptController, Vec<MemoryRegion>) {
132        (
133            self.cpu_topology,
134            self.interrupt_controller,
135            self.mmio_regions,
136        )
137    }
138}

142    /// Discovers the current platform from the boot argument.
143    ///
144    /// The architecture-specific parser reads ACPI or Device Tree and returns the CPU topology,
145    /// controller configuration, and MMIO regions needed by discovered platform devices. This
146    /// function must run after the VMM and heap are initialized because firmware parsing may
147    /// allocate and access high-half mappings. The caller maps the returned MMIO regions before
148    /// invoking the BSP later-initialization hook.
149    pub fn discover_platform(arg: PlatformBootArg) -> PlatformDescription;
150}

158    /// Preserves the discovered CPU identifiers and bootstrap identifier.
159    #[test]
160    fn cpu_topology_preserves_discovery_result() {
161        let topology = CpuTopology::new(vec![4, 7, 9].into_boxed_slice(), 7);

163        assert_eq!(topology.cpu_ids(), &[4, 7, 9]);
164        assert_eq!(topology.bsp_id(), 7);
