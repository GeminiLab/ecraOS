14/// A failure to validate or update physical memory regions.
15#[derive(Debug, Clone, Copy, PartialEq, Eq)]
16pub enum PhysMemoryRegionsError {
17    /// An address range is empty, reversed, or changed by a transformation.
18    InvalidRange,
19    /// Flags contain unknown bits or do not have exactly one ownership role.
20    InvalidFlags,
21    /// An insertion overlaps an existing region.
22    Overlap,
23    /// An overwrite predicate rejected an existing intersection.
24    OverwriteRejected,
25    /// The result does not fit in fixed-capacity storage.
26    CapacityExceeded,
27    /// Dynamic storage could not allocate the required capacity.
28    AllocationFailed,
29}
30
31/// Storage primitives for a physical region collection.
32///
33/// Default must create empty storage. Clone preserves entries and uses the storage type's normal
34/// allocation-failure behavior. Insertion preserves existing element order. A failed
35/// reservation or insertion must preserve the contents. Storage is never exposed mutably through
36/// the collection, so callers cannot bypass range and flag validation.
37pub trait PhysMemoryRegionsStorage: Default + Clone {
38    /// Returns all stored regions in order.
39    fn as_slice(&self) -> &[MemoryRegion];
40
41    /// Reserves capacity for at least the given total number of elements.
42    ///
43    /// The count includes existing entries, rather than specifying additional entries. A count
44    /// below the current length succeeds without shrinking storage. The contents are unchanged.
45    fn reserve_regions(&mut self, total: usize) -> Result<(), PhysMemoryRegionsError>;
46
47    /// Inserts an element at a valid index without reordering other elements.
48    ///
49    /// The index must not exceed the current length. A successful reservation for the new length
50    /// guarantees that this operation succeeds without allocating.
51    fn insert_region(
52        &mut self,
53        index: usize,
54        region: MemoryRegion,
55    ) -> Result<(), PhysMemoryRegionsError>;
56}

58/// Allocation-free storage operations on an inline-capacity vector.
59impl<const N: usize> PhysMemoryRegionsStorage for HeaplessVec<MemoryRegion, N> {
60    /// Returns the live inline entries.
61    fn as_slice(&self) -> &[MemoryRegion] {
62        self.as_ref()
63    }
64
65    /// Rejects a requested total greater than the inline capacity.
66    fn reserve_regions(&mut self, total: usize) -> Result<(), PhysMemoryRegionsError> {
67        if total <= N {
68            Ok(())
69        } else {
70            Err(PhysMemoryRegionsError::CapacityExceeded)
71        }
72    }
73
74    /// Inserts an inline entry and reports capacity exhaustion.
75    fn insert_region(
76        &mut self,
77        index: usize,
78        region: MemoryRegion,
79    ) -> Result<(), PhysMemoryRegionsError> {
80        self.insert(index, region)
81            .map_err(|_| PhysMemoryRegionsError::CapacityExceeded)
82    }
83}

85/// Fallible storage operations on a dynamically allocated vector.
86impl PhysMemoryRegionsStorage for Vec<MemoryRegion> {
87    /// Returns the allocated entries.
88    fn as_slice(&self) -> &[MemoryRegion] {
89        self
90    }
91
92    /// Grows the allocation to fit the requested total number of entries.
93    fn reserve_regions(&mut self, total: usize) -> Result<(), PhysMemoryRegionsError> {
94        // Vec takes an additional count, while this trait takes the final total count.
95        self.try_reserve(total.saturating_sub(self.len()))
96            .map_err(|_| PhysMemoryRegionsError::AllocationFailed)
97    }
98
99    /// Inserts an entry after fallibly reserving space if necessary.
100    fn insert_region(
101        &mut self,
102        index: usize,
103        region: MemoryRegion,
104    ) -> Result<(), PhysMemoryRegionsError> {
105        self.reserve_regions(self.len() + 1)?;
106        self.insert(index, region);
107        Ok(())
108    }
109}

111/// A sorted collection of nonempty, nonoverlapping regions with one ownership role each.
112///
113/// Ranges retain byte precision. Adjacent regions are not merged. Every fallible mutation keeps
114/// the original contents on error. Callbacks are evaluated at most once per visited intersection
115/// and should not perform external mutations if their effects would need to be rolled back.
116#[derive(Debug, Clone, PartialEq, Eq)]
117pub struct PhysMemoryRegions<S: PhysMemoryRegionsStorage>(S);
118
119/// Empty collection construction for any supported storage.
120impl<S: PhysMemoryRegionsStorage> Default for PhysMemoryRegions<S> {
121    /// Creates an empty collection in the selected storage.
122    fn default() -> Self {
123        Self::new()
124    }
125}

127/// Validated queries and atomic range updates over interchangeable storage.
128impl<S: PhysMemoryRegionsStorage> PhysMemoryRegions<S> {
129    /// Creates an empty region collection.
130    pub fn new() -> Self {
131        Self(S::default())
132    }
133
134    /// Returns the regions without permitting mutation of their ranges or attributes.
135    pub fn as_slice(&self) -> &[MemoryRegion] {
136        self.0.as_slice()
137    }
138
139    /// Returns an iterator in ascending address order.
140    pub fn iter(&self) -> slice::Iter<'_, MemoryRegion> {
141        self.as_slice().iter()
142    }
143
144    /// Returns the number of stored regions.
145    pub fn len(&self) -> usize {
146        self.as_slice().len()
147    }
148
149    /// Returns whether no regions are stored.
150    pub fn is_empty(&self) -> bool {
151        self.as_slice().is_empty()
152    }
153
154    /// Consumes the collection and returns its storage.
155    pub fn into_storage(self) -> S {
156        self.0
157    }
158
159    /// Finds the region containing an address, or returns None for an uncovered address.
160    pub fn find(&self, address: PhysAddr) -> Option<&MemoryRegion> {
161        let index = self
162            .as_slice()
163            .partition_point(|region| region.range.end <= address);
164        self.as_slice()
165            .get(index)
166            .filter(|region| region.range.start <= address)
167    }
168
169    /// Iterates over regions intersecting a range, preserving their full stored ranges.
170    pub fn overlaps(&self, range: PhysAddrRange) -> impl Iterator<Item = &MemoryRegion> {
171        let start = self
172            .as_slice()
173            .partition_point(|region| region.range.end <= range.start);
174        self.as_slice()[start..]
175            .iter()
176            .take_while(move |region| region.range.start < range.end)
177            .filter(move |_| range.start < range.end)
178    }
179
180    /// Iterates over uncovered portions of a byte range in address order.
181    ///
182    /// An empty or reversed input produces no portions. The iterator does not allocate.
183    pub fn uncovered(&self, range: PhysAddrRange) -> impl Iterator<Item = PhysAddrRange> + '_ {
184        let mut cursor = range.start;
185        let mut existing = self.overlaps(range);
186        core::iter::from_fn(move || {
187            while cursor < range.end {
188                let Some(next) = existing.next() else {
189                    let gap = PhysAddrRange::new(cursor, range.end);
190                    cursor = range.end;
191                    return Some(gap);
192                };
193                let start = cursor;
194                cursor = cursor.max(next.range.end);
195                if start < next.range.start {
196                    return Some(PhysAddrRange::new(start, next.range.start));
197                }
198            }
199            None
200        })
201    }
202
203    /// Inserts a complete region and rejects any existing overlap.
204    ///
205    /// # Errors
206    ///
207    /// Returns InvalidRange or InvalidFlags for invalid input, Overlap for an intersection, and
208    /// a storage error if the insertion cannot fit. No error changes the collection contents.
209    pub fn insert_disjoint(&mut self, region: MemoryRegion) -> Result<(), PhysMemoryRegionsError> {
210        validate_region(region)?;
211        let index = self
212            .as_slice()
213            .partition_point(|other| other.range.start < region.range.start);
214        if (index > 0 && self[index - 1].range.end > region.range.start)
215            || (index < self.len() && self[index].range.start < region.range.end)
216        {
217            return Err(PhysMemoryRegionsError::Overlap);
218        }
219        self.0.reserve_regions(self.len() + 1)?;
220        self.0.insert_region(index, region)
221    }
222
223    /// Adds only the uncovered portions of a region, retaining all existing regions.
224    ///
225    /// # Errors
226    ///
227    /// Returns an error for invalid input or insufficient storage. No error changes the
228    /// collection contents, including when the range splits into many uncovered pieces.
229    pub fn insert_uncovered(&mut self, region: MemoryRegion) -> Result<(), PhysMemoryRegionsError> {
230        validate_region(region)?;
231        let mut staged = S::default();
232        let mut cursor = region.range.start;
233        for existing in self.iter().copied() {
234            if cursor < region.range.end && existing.range.start > cursor {
235                let end = existing.range.start.min(region.range.end);
236                append(&mut staged, clipped(region, cursor, end))?;
237            }
238            append(&mut staged, existing)?;
239            if existing.range.end > cursor {
240                cursor = existing.range.end;
241            }
242        }
243        if cursor < region.range.end {
244            append(&mut staged, clipped(region, cursor, region.range.end))?;
245        }
246        self.0 = staged;
247        Ok(())
248    }
249
250    /// Overwrites a complete incoming range if every existing intersection is allowed.
251    ///
252    /// Each predicate argument retains the old flags and description but is clipped to its exact
253    /// intersection with the incoming range. The predicate is called once per intersection until
254    /// one is rejected. Gaps are filled with the incoming region. Old portions outside the range
255    /// retain their attributes.
256    ///
257    /// # Errors
258    ///
259    /// Returns OverwriteRejected when a predicate refuses any overlap, or an input or storage
260    /// error before committing the replacement.
261    pub fn insert_overwrite_if<F>(
262        &mut self,
263        region: MemoryRegion,
264        mut allow: F,
265    ) -> Result<(), PhysMemoryRegionsError>
266    where
267        F: FnMut(MemoryRegion) -> bool,
268    {
269        validate_region(region)?;
270        for old in self.overlaps(region.range).copied() {
271            let overlap = intersection(old, region.range);
272            if !allow(overlap) {
273                return Err(PhysMemoryRegionsError::OverwriteRejected);
274            }
275        }
276        let mut staged = S::default();
277        let mut inserted = false;
278        for old in self.iter().copied() {
279            if old.range.end <= region.range.start {
280                append(&mut staged, old)?;
281                continue;
282            }
283            if old.range.start < region.range.start {
284                append(
285                    &mut staged,
286                    clipped(old, old.range.start, region.range.start),
287                )?;
288            }
289            if !inserted {
290                append(&mut staged, region)?;
291                inserted = true;
292            }
293            if old.range.start >= region.range.end {
294                append(&mut staged, old)?;
295            } else if old.range.end > region.range.end {
296                append(&mut staged, clipped(old, region.range.end, old.range.end))?;
297            }
298        }
299        if !inserted {
300            append(&mut staged, region)?;
301        }
302        self.0 = staged;
303        Ok(())
304    }
305
306    /// Transforms only existing intersections without filling any gaps.
307    ///
308    /// Each callback argument is an old region clipped to the requested range. The returned
309    /// region must retain that exact range and contain valid flags. Portions outside the range
310    /// are preserved.
311    ///
312    /// # Errors
313    ///
314    /// Invalid input, invalid callback results, and storage failures leave the contents unchanged.
315    pub fn transform_overlaps<F>(
316        &mut self,
317        range: PhysAddrRange,
318        transform: F,
319    ) -> Result<(), PhysMemoryRegionsError>
320    where
321        F: FnMut(MemoryRegion) -> MemoryRegion,
322    {
323        self.rewrite(range, None, transform)
324    }
325
326    /// Transforms existing intersections and fills gaps from an incoming region atomically.
327    ///
328    /// The callback has the same contract as transform_overlaps. Uncovered parts receive the
329    /// incoming flags and description. This combines preserving old attributes with recording a
330    /// complete reservation.
331    ///
332    /// # Errors
333    ///
334    /// Invalid input, invalid callback results, and storage failures leave the contents unchanged.
335    pub fn overlay_with<F>(
336        &mut self,
337        region: MemoryRegion,
338        transform: F,
339    ) -> Result<(), PhysMemoryRegionsError>
340    where
341        F: FnMut(MemoryRegion) -> MemoryRegion,
342    {
343        validate_region(region)?;
344        self.rewrite(region.range, Some(region), transform)
345    }
346
347    /// Rebuilds a changed range in separate storage before committing its contents.
348    fn rewrite<F>(
349        &mut self,
350        range: PhysAddrRange,
351        fill: Option<MemoryRegion>,
352        mut transform: F,
353    ) -> Result<(), PhysMemoryRegionsError>
354    where
355        F: FnMut(MemoryRegion) -> MemoryRegion,
356    {
357        validate_range(range)?;
358        let mut staged = S::default();
359        let mut cursor = range.start;
360        for old in self.iter().copied() {
361            if old.range.end <= range.start {
362                append(&mut staged, old)?;
363                continue;
364            }
365            if old.range.start >= range.end {
366                if let Some(fill) = fill
367                    && cursor < range.end
368                {
369                    append(&mut staged, clipped(fill, cursor, range.end))?;
370                }
371                cursor = range.end;
372                append(&mut staged, old)?;
373                continue;
374            }
375            let overlap = intersection(old, range);
376            if old.range.start < overlap.range.start {
377                append(
378                    &mut staged,
379                    clipped(old, old.range.start, overlap.range.start),
380                )?;
381            }
382            if let Some(fill) = fill
383                && cursor < overlap.range.start
384            {
385                append(&mut staged, clipped(fill, cursor, overlap.range.start))?;
386            }
387            let changed = transform(overlap);
388            validate_region(changed)?;
389            if changed.range != overlap.range {
390                return Err(PhysMemoryRegionsError::InvalidRange);
391            }
392            append(&mut staged, changed)?;
393            cursor = overlap.range.end;
394            if overlap.range.end < old.range.end {
395                append(&mut staged, clipped(old, overlap.range.end, old.range.end))?;
396            }
397        }
398        if let Some(fill) = fill
399            && cursor < range.end
400        {
401            append(&mut staged, clipped(fill, cursor, range.end))?;
402        }
403        self.0 = staged;
404        Ok(())
405    }
406}

408/// One-way handoff from inline boot storage to dynamically allocated storage.
409impl<const N: usize> From<PhysMemoryRegions<HeaplessVec<MemoryRegion, N>>>
410    for PhysMemoryRegions<Vec<MemoryRegion>>
411{
412    /// Converts inline entries to dynamic storage after an allocator is available.
413    fn from(regions: PhysMemoryRegions<HeaplessVec<MemoryRegion, N>>) -> Self {
414        // Copy just the live entries instead of moving an inline-capacity owning iterator through
415        // several collection helpers on the small boot stack.
416        Self(regions.as_slice().to_vec())
417    }
418}

456/// Rejects empty or reversed address ranges.
457fn validate_range(range: PhysAddrRange) -> Result<(), PhysMemoryRegionsError> {
458    if range.start >= range.end {
459        Err(PhysMemoryRegionsError::InvalidRange)
460    } else {
461        Ok(())
462    }
463}
464
465/// Checks byte-range validity and architecture-independent flag invariants.
466fn validate_region(region: MemoryRegion) -> Result<(), PhysMemoryRegionsError> {
467    validate_range(region.range)?;
468    if !region.flags.role_sanity_check()
469        || region.flags.bits() & !MemoryRegionFlags::all().bits() != 0
470    {
471        Err(PhysMemoryRegionsError::InvalidFlags)
472    } else {
473        Ok(())
474    }
475}

477/// Copies an existing region's attributes onto a contained byte range.
478fn clipped(region: MemoryRegion, start: PhysAddr, end: PhysAddr) -> MemoryRegion {
479    MemoryRegion {
480        range: PhysAddrRange::new(start, end),
481        ..region
482    }
483}
484
485/// Clips a region to a known nonempty intersection.
486fn intersection(region: MemoryRegion, range: PhysAddrRange) -> MemoryRegion {
487    clipped(
488        region,
489        region.range.start.max(range.start),
490        region.range.end.min(range.end),
491    )
492}
493
494/// Appends a result segment while checking its order against the previous segment.
495fn append<S: PhysMemoryRegionsStorage>(
496    storage: &mut S,
497    region: MemoryRegion,
498) -> Result<(), PhysMemoryRegionsError> {
499    if storage
500        .as_slice()
501        .last()
502        .is_some_and(|last| last.range.end > region.range.start)
503    {
504        return Err(PhysMemoryRegionsError::Overlap);
505    }
506    storage.insert_region(storage.as_slice().len(), region)
507}
