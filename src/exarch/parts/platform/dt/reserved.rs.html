13/// The description attached to regions reserved by a device tree.
14const DEVICE_TREE_RESERVED_DESC: &str = "device-tree reserved memory";
15
16/// A statically located range declared by a device tree reservation mechanism.
17#[derive(Clone, Copy)]
18struct DeviceTreeReservedRegion {
19    /// The physical range that firmware reserves from general use.
20    range: PhysAddrRange,
21    /// Whether the standard direct mapping must omit this range.
22    no_map: bool,
23}
24
25/// A direct child of the device-tree reserved-memory node while it is being parsed.
26struct ReservedMemoryChild<'dt> {
27    /// The structure-token depth of this child node.
28    depth: usize,
29    /// The child reg property, if it describes a static physical range.
30    reg: Option<&'dt [u8]>,
31    /// Whether the child has a no-map property.
32    no_map: bool,
33    /// Whether the child has a reusable property.
34    reusable: bool,
35    /// Whether firmware marks the child enabled.
36    enabled: bool,
37}
38
39/// Applies all static FDT memory reservations to discovered RAM regions.
40///
41/// Reads the FDT reservation map and static reg properties below /reserved-memory. Intersections
42/// with usable RAM become reserved regions. A /reserved-memory child with no-map becomes a NO_MAP
43/// region so the early direct map omits it. Dynamic size requests are not allocated here because
44/// this function runs before the kernel allocator is available.
45///
46/// # Errors
47///
48/// Returns an error when the reservation map, structure block, reserved-memory properties, or
49/// region capacity is invalid.
50pub fn apply_device_tree_reservations(
51    dev_tree: DevTree<'_>,
52    regions: &mut RawMemoryRegions,
53) -> Result<(), DevTreeError> {
54    let mut reserved = HeaplessVec::<DeviceTreeReservedRegion, MAX_RAW_MEM_REGIONS>::new();
55
56    // Parse both reservation sources before changing the discovered memory regions.
57    parse_reservation_map(dev_tree, &mut reserved)?;
58    parse_reserved_memory_nodes(dev_tree, &mut reserved)?;
59
60    // Apply each physical reservation to RAM without allocating temporary storage.
61    for reservation in reserved {
62        apply_reserved_region(regions, reservation)?;
63    }
64
65    Ok(())
66}

68/// Reads entries from the FDT header's memory-reservation block.
69fn parse_reservation_map(
70    dev_tree: DevTree<'_>,
71    reserved: &mut HeaplessVec<DeviceTreeReservedRegion, MAX_RAW_MEM_REGIONS>,
72) -> Result<(), DevTreeError> {
73    let bytes = dev_tree.buf();
74    let mut offset = dev_tree.off_mem_rsvmap();
75    let end = dev_tree.off_dt_struct();
76
77    // The reservation map must be aligned and fit before the structure block begins.
78    if !offset.is_multiple_of(core::mem::align_of::<u64>()) || offset > end {
79        return Err(DevTreeError::InvalidOffset);
80    }
81
82    loop {
83        // Each entry contains two big-endian u64 values; (0, 0) terminates the map.
84        let entry_end = offset
85            .checked_add(2 * core::mem::size_of::<u64>())
86            .ok_or(DevTreeError::InvalidOffset)?;
87        let entry = bytes
88            .get(offset..entry_end)
89            .filter(|_| entry_end <= end)
90            .ok_or(DevTreeError::InvalidOffset)?;
91        let address = u64::from_be_bytes(
92            entry[..8]
93                .try_into()
94                .map_err(|_| DevTreeError::ParseError)?,
95        );
96        let size = u64::from_be_bytes(
97            entry[8..]
98                .try_into()
99                .map_err(|_| DevTreeError::ParseError)?,
100        );
101        if address == 0 && size == 0 {
102            return Ok(());
103        }
104        if size == 0 {
105            return Err(DevTreeError::ParseError);
106        }
107
108        reserved
109            .push(DeviceTreeReservedRegion {
110                range: physical_range(address, size)?,
111                no_map: false,
112            })
113            .map_err(|_| DevTreeError::NotEnoughMemory)?;
114        offset = entry_end;
115    }
116}

118/// Reads static reg ranges from direct children of the /reserved-memory node.
119fn parse_reserved_memory_nodes(
120    dev_tree: DevTree<'_>,
121    reserved: &mut HeaplessVec<DeviceTreeReservedRegion, MAX_RAW_MEM_REGIONS>,
122) -> Result<(), DevTreeError> {
123    let mut tokens = dev_tree.parse_iter();
124    let mut depth = 0usize;
125    let mut reserved_node_depth = None;
126    let mut address_cells = None;
127    let mut size_cells = None;
128    let mut ranges_empty = None;
129    let mut child: Option<ReservedMemoryChild<'_>> = None;
130
131    // Stream the structure once. Depth tracking identifies the reserved-memory node and its
132    // direct children without building an allocator-backed index.
133    while let Some(token) = tokens.next()? {
134        match token {
135            ParsedTok::BeginNode(node) => {
136                if depth == 1 && node.name == b"reserved-memory" {
137                    reserved_node_depth = Some(depth);
138                    address_cells = None;
139                    size_cells = None;
140                    ranges_empty = None;
141                } else if depth
142                    .checked_sub(1)
143                    .is_some_and(|parent_depth| reserved_node_depth == Some(parent_depth))
144                {
145                    child = Some(ReservedMemoryChild {
146                        depth,
147                        reg: None,
148                        no_map: false,
149                        reusable: false,
150                        enabled: true,
151                    });
152                }
153                depth = depth.checked_add(1).ok_or(DevTreeError::ParseError)?;
154            }
155            ParsedTok::Prop(property) => {
156                let owner_depth = depth.checked_sub(1).ok_or(DevTreeError::ParseError)?;
157                let name = property_name(dev_tree, property.name_offset)?;
158
159                // Cell counts and ranges belong to the parent; reg and policy properties belong
160                // only to its direct reservation children.
161                if reserved_node_depth == Some(owner_depth) {
162                    match name {
163                        "#address-cells" => {
164                            address_cells = Some(read_cell_count(property.prop_buf)?);
165                        }
166                        "#size-cells" => {
167                            size_cells = Some(read_cell_count(property.prop_buf)?);
168                        }
169                        "ranges" => ranges_empty = Some(property.prop_buf.is_empty()),
170                        _ => {}
171                    }
172                } else if let Some(child) = child.as_mut()
173                    && child.depth == owner_depth
174                {
175                    match name {
176                        "reg" => child.reg = Some(property.prop_buf),
177                        "no-map" if property.prop_buf.is_empty() => child.no_map = true,
178                        "no-map" => return Err(DevTreeError::ParseError),
179                        "reusable" if property.prop_buf.is_empty() => child.reusable = true,
180                        "reusable" => return Err(DevTreeError::ParseError),
181                        "status" => child.enabled = status_is_enabled(property.prop_buf)?,
182                        _ => {}
183                    }
184                }
185            }
186            ParsedTok::EndNode => {
187                let ended_depth = depth.checked_sub(1).ok_or(DevTreeError::ParseError)?;
188
189                // Defer decoding reg until the child closes so its properties can appear in any
190                // valid order.
191                if child
192                    .as_ref()
193                    .is_some_and(|child| child.depth == ended_depth)
194                {
195                    let child = child.take().ok_or(DevTreeError::ParseError)?;
196                    // The DT specification makes no-map and reusable mutually exclusive.
197                    if child.no_map && child.reusable {
198                        return Err(DevTreeError::ParseError);
199                    }
200                    if child.enabled
201                        && let Some(reg) = child.reg
202                    {
203                        let address_cells = address_cells.ok_or(DevTreeError::InvalidParameter(
204                            "reserved-memory is missing #address-cells",
205                        ))?;
206                        let size_cells = size_cells.ok_or(DevTreeError::InvalidParameter(
207                            "reserved-memory is missing #size-cells",
208                        ))?;
209                        append_reg_ranges(reg, address_cells, size_cells, child.no_map, reserved)?;
210                    }
211                }
212                if reserved_node_depth == Some(ended_depth) {
213                    if ranges_empty != Some(true) {
214                        return Err(DevTreeError::InvalidParameter(
215                            "reserved-memory must have an empty ranges property",
216                        ));
217                    }
218                    reserved_node_depth = None;
219                }
220                depth = ended_depth;
221            }
222            ParsedTok::Nop => {}
223        }
224    }
225
226    Ok(())
227}

229/// Checks whether a null-terminated status property enables its node.
230fn status_is_enabled(bytes: &[u8]) -> Result<bool, DevTreeError> {
231    let Some(value) = bytes.strip_suffix(&[0]) else {
232        return Err(DevTreeError::ParseError);
233    };
234    let value = core::str::from_utf8(value).map_err(DevTreeError::from)?;
235    Ok(matches!(value, "okay" | "ok"))
236}

238/// Returns a property name from the FDT strings block.
239fn property_name<'dt>(dev_tree: DevTree<'dt>, offset: usize) -> Result<&'dt str, DevTreeError> {
240    let strings_start = dev_tree.off_dt_strings();
241    let strings_end = strings_start
242        .checked_add(dev_tree.size_dt_strings() as usize)
243        .ok_or(DevTreeError::InvalidOffset)?;
244    let strings = dev_tree
245        .buf()
246        .get(strings_start..strings_end)
247        .ok_or(DevTreeError::InvalidOffset)?;
248    let suffix = strings.get(offset..).ok_or(DevTreeError::InvalidOffset)?;
249    let length = suffix
250        .iter()
251        .position(|byte| *byte == 0)
252        .ok_or(DevTreeError::ParseError)?;
253    core::str::from_utf8(&suffix[..length]).map_err(DevTreeError::from)
254}

256/// Decodes a one- or two-cell count from a reserved-memory property.
257fn read_cell_count(bytes: &[u8]) -> Result<usize, DevTreeError> {
258    let [a, b, c, d] = bytes else {
259        return Err(DevTreeError::ParseError);
260    };
261    let count = u32::from_be_bytes([*a, *b, *c, *d]) as usize;
262    if !(1..=2).contains(&count) {
263        return Err(DevTreeError::InvalidParameter(
264            "reserved-memory cell counts must be one or two",
265        ));
266    }
267    Ok(count)
268}

270/// Decodes static physical ranges from a reserved-memory reg property.
271fn append_reg_ranges(
272    reg: &[u8],
273    address_cells: usize,
274    size_cells: usize,
275    no_map: bool,
276    reserved: &mut HeaplessVec<DeviceTreeReservedRegion, MAX_RAW_MEM_REGIONS>,
277) -> Result<(), DevTreeError> {
278    // A reg value is a sequence of address/size tuples sized by the parent cell counts.
279    let tuple_size = (address_cells + size_cells)
280        .checked_mul(core::mem::size_of::<u32>())
281        .ok_or(DevTreeError::InvalidOffset)?;
282    if reg.is_empty() || !reg.len().is_multiple_of(tuple_size) {
283        return Err(DevTreeError::ParseError);
284    }
285
286    for tuple in reg.chunks_exact(tuple_size) {
287        let address_size = address_cells * core::mem::size_of::<u32>();
288        let address = decode_cells(&tuple[..address_size], address_cells)?;
289        let size = decode_cells(&tuple[address_size..], size_cells)?;
290        if size == 0 {
291            return Err(DevTreeError::ParseError);
292        }
293        reserved
294            .push(DeviceTreeReservedRegion {
295                range: physical_range(address, size)?,
296                no_map,
297            })
298            .map_err(|_| DevTreeError::NotEnoughMemory)?;
299    }
300
301    Ok(())
302}

304/// Decodes a one- or two-cell big-endian integer.
305fn decode_cells(bytes: &[u8], count: usize) -> Result<u64, DevTreeError> {
306    match count {
307        1 => Ok(u32::from_be_bytes(bytes.try_into().map_err(|_| DevTreeError::ParseError)?) as u64),
308        2 => Ok(u64::from_be_bytes(
309            bytes.try_into().map_err(|_| DevTreeError::ParseError)?,
310        )),
311        _ => Err(DevTreeError::InvalidParameter(
312            "reserved-memory cell counts must be one or two",
313        )),
314    }
315}

317/// Creates a semantic physical range and rejects address-space overflow.
318fn physical_range(address: u64, size: u64) -> Result<PhysAddrRange, DevTreeError> {
319    let end = address
320        .checked_add(size)
321        .ok_or(DevTreeError::InvalidOffset)?;
322    let start = usize::try_from(address).map_err(|_| DevTreeError::InvalidOffset)?;
323    let end = usize::try_from(end).map_err(|_| DevTreeError::InvalidOffset)?;
324    Ok(PhysAddrRange::new(pa!(start), pa!(end)))
325}

327/// Splits discovered regions where a device-tree reservation overlaps usable RAM.
328fn apply_reserved_region(
329    regions: &mut RawMemoryRegions,
330    reservation: DeviceTreeReservedRegion,
331) -> Result<(), DevTreeError> {
332    let mut adjusted = RawMemoryRegions::new();
333    for region in regions.iter().copied() {
334        // Reservations outside this discovered region do not affect it.
335        let start = core::cmp::max(region.range.start, reservation.range.start);
336        let end = core::cmp::min(region.range.end, reservation.range.end);
337        if start >= end {
338            adjusted
339                .push(region)
340                .map_err(|_| DevTreeError::NotEnoughMemory)?;
341            continue;
342        }
343
344        let role = region.flags
345            & (MemoryRegionFlags::FREE | MemoryRegionFlags::RESERVED | MemoryRegionFlags::NO_MAP);
346        // A regular reservation changes only free RAM. no-map must also override a prior
347        // reservation so the final mapping policy is unambiguous.
348        let should_reserve = role.contains(MemoryRegionFlags::FREE);
349        let should_unmap = reservation.no_map && !role.contains(MemoryRegionFlags::NO_MAP);
350        if !should_reserve && !should_unmap {
351            adjusted
352                .push(region)
353                .map_err(|_| DevTreeError::NotEnoughMemory)?;
354            continue;
355        }
356
357        // Preserve any RAM before the intersection, replace the intersection's role, then keep
358        // any RAM after it. Existing access and device attributes are retained.
359        if region.range.start < start {
360            adjusted
361                .push(MemoryRegion {
362                    range: PhysAddrRange::new(region.range.start, start),
363                    ..region
364                })
365                .map_err(|_| DevTreeError::NotEnoughMemory)?;
366        }
367
368        let flags = (region.flags
369            & !(MemoryRegionFlags::FREE
370                | MemoryRegionFlags::RESERVED
371                | MemoryRegionFlags::NO_MAP
372                | MemoryRegionFlags::BOOT_SERVICE
373                | MemoryRegionFlags::KERNEL))
374            | if reservation.no_map {
375                MemoryRegionFlags::NO_MAP
376            } else {
377                MemoryRegionFlags::RESERVED
378            };
379        adjusted
380            .push(MemoryRegion {
381                range: PhysAddrRange::new(start, end),
382                flags,
383                desc: DEVICE_TREE_RESERVED_DESC,
384            })
385            .map_err(|_| DevTreeError::NotEnoughMemory)?;
386
387        if end < region.range.end {
388            adjusted
389                .push(MemoryRegion {
390                    range: PhysAddrRange::new(end, region.range.end),
391                    ..region
392                })
393                .map_err(|_| DevTreeError::NotEnoughMemory)?;
394        }
395    }
396    *regions = adjusted;
397    Ok(())
