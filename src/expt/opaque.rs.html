1//! Runtime type-erased multi-level page table types.
2//!
3//! [`OpaquePageTableType`] captures operations for concrete metadata, entry, and allocator types.
4//! [`OpaquePageTable`] combines those operations with an erased root to represent different
5//! multi-level page table formats in a single type. This permits runtime page table hierarchy
6//! format selection.
7

24/// The actions that mutate a page table hierarchy.
25///
26/// Each action maps or unmaps one range through an opaque multi-level page table.
27pub enum PageTableAction<A: VirtAddr> {
28    /// A mapping of a virtual address range to a physical address range.
29    ///
30    /// The range is rounded outward to base-page boundaries and replaces existing mappings.
31    Map {
32        /// The start of the virtual address range.
33        ///
34        /// This address also selects the lower or upper hierarchy for a dual-root value.
35        vaddr: A,
36        /// The start of the physical address range.
37        ///
38        /// The address is rounded down as described by [`crate::pt::PageTableCursor::map`].
39        paddr: PhysAddr,
40        /// The size of the ranges in bytes.
41        ///
42        /// The virtual range begins at `vaddr` and is expanded to base-page boundaries.
43        size: usize,
44        /// The flags to install in each mapping.
45        ///
46        /// The same flags are used for every leaf installed by this action.
47        flags: MappingFlags,
48    },
49    /// The removal of mappings from a virtual address range.
50    ///
51    /// The range is rounded outward to base-page boundaries. Mapped data frames are retained.
52    Unmap {
53        /// The start of the virtual address range.
54        ///
55        /// This address also selects the lower or upper hierarchy for a dual-root value.
56        vaddr: A,
57        /// The size of the range in bytes.
58        ///
59        /// The virtual range begins at `vaddr` and is expanded to base-page boundaries.
60        size: usize,
61    },
62}

64/// The physical root addresses of an opaque multi-level page table.
65///
66/// `Single` identifies the root of one page table hierarchy. `Dual` identifies independent
67/// lower-address and upper-address hierarchies and is only available on AArch64.
68#[derive(Debug, Clone, Copy, PartialEq, Eq)]
69pub enum OpaquePageTableRoot {
70    /// A single physical root.
71    ///
72    /// This address identifies the root page table of one hierarchy.
73    Single(PhysAddr),
74    /// Independent lower and upper physical roots used by AArch64.
75    ///
76    /// Each address identifies the root page table of a separate hierarchy.
77    #[cfg(any(test, doc, target_arch = "aarch64"))]
78    #[cfg_attr(doc, doc(cfg(target_arch = "aarch64")))]
79    Dual(crate::pt::DualPageTableRoot),
80}

82/// The function type for applying a sequence of page table actions.
83type ApplyActionsFn<A> = for<'a> fn(
84    root: OpaquePageTableRoot,
85    actions: &'a mut dyn Iterator<Item = PageTableAction<A>>,
86) -> PagingResult<A, PendingCleanup<A>>;

88/// The operations for one concrete multi-level page table type.
89///
90/// Mutation functions reconstruct a temporary concrete value while preserving its table frames.
91/// Destruction deallocates table frames through the concrete value's destructor.
92#[derive(Clone)]
93struct PageTableMethods<A: VirtAddr> {
94    /// Creates a new, empty and zeroed page table hierarchy.
95    pub new_alloc: fn() -> PagingResult<A, OpaquePageTableRoot>,
96    /// Applies a sequence of updates to a page table hierarchy and returns the pending cleanup.
97    pub apply_actions: ApplyActionsFn<A>,
98    /// Destroys and deallocates a page table hierarchy.
99    pub destroy: fn(root: OpaquePageTableRoot),
100}
101
102/// Construction of statically specialized opaque operations.
103///
104/// Each method table retains the concrete format and allocator in its function pointers.
105impl<A: VirtAddr> PageTableMethods<A> {
106    /// Creates a method table that rejects allocation and mutation by reporting bugs.
107    ///
108    /// Destruction is a no-op so a constant-initialization sentinel can be dropped.
109    pub const fn dummy() -> Self {
110        /// Reports a bug when an operation is attempted through a dummy method table.
111        ///
112        /// Allocation and mutation both call this helper.
113        fn bug_im_dummy() -> ! {
114            bug!("dummy page table type does not support any actual operations")
115        }
116
117        Self {
118            new_alloc: || bug_im_dummy(),
119            apply_actions: |_, _| bug_im_dummy(),
120            destroy: |_| {},
121        }
122    }
123
124    /// Creates a method table for a concrete single-root multi-level page table format.
125    ///
126    /// `M` supplies the page table hierarchy geometry and must use `A` for virtual addresses.
127    /// `PTE` supplies the concrete entry representation, and `AL` supplies table frame allocation,
128    /// address translation, and destruction.
129    pub const fn new<M, PTE, AL, F>() -> Self
130    where
131        M: PageTableMeta<VirtAddr = A>,
132        PTE: GenericPTE,
133        AL: PageAllocator,
134        F: TlbFlusher<A>,
135        [(); M::LEVELS - 1]: Sized,
136    {
137        Self {
138            new_alloc: || {
139                PageTable::<M, PTE, AL, F>::new_alloc()
140                    .map(|pt| OpaquePageTableRoot::Single(pt.leak()))
141            },
142            apply_actions: |root, actions| {
143                #[cfg(not(any(test, doc, target_arch = "aarch64")))]
144                let OpaquePageTableRoot::Single(root) = root;
145                #[cfg(any(test, doc, target_arch = "aarch64"))]
146                let OpaquePageTableRoot::Single(root) = root else {
147                    bug!("passing dual-root page table to single-root page table operation")
148                };
149                // Use `ManuallyDrop` to avoid dropping the hierarchy on error.
150                let mut pt = ManuallyDrop::new(unsafe { PageTable::<M, PTE, AL, F>::new_at(root) });
151                let mut cursor = pt.cursor();
152
153                for action in actions {
154                    match action {
155                        PageTableAction::Map {
156                            vaddr,
157                            paddr,
158                            size,
159                            flags,
160                        } => {
161                            cursor.map(vaddr, paddr, size, flags)?;
162                        }
163                        PageTableAction::Unmap { vaddr, size } => {
164                            cursor.unmap(vaddr, size)?;
165                        }
166                    }
167                }
168
169                let result = unsafe { cursor.take_cleanup() };
170                drop(cursor);
171
172                let new_root = ManuallyDrop::into_inner(pt).leak();
173                debug_assert_eq!(root, new_root);
174
175                Ok(result)
176            },
177            destroy: |root| match root {
178                OpaquePageTableRoot::Single(root) => {
179                    drop(unsafe { PageTable::<M, PTE, AL, F>::new_at(root) });
180                }
181                #[cfg(any(test, doc, target_arch = "aarch64"))]
182                OpaquePageTableRoot::Dual(_) => {
183                    bug!("passing dual-root page table to single-root page table destruction")
184                }
185            },
186        }
187    }
188
189    /// Creates a method table for a dual-root multi-level page table type.
190    ///
191    /// Both roots use the statically selected allocator for table frames and destruction.
192    #[cfg(any(test, doc, target_arch = "aarch64"))]
193    #[cfg_attr(doc, doc(cfg(target_arch = "aarch64")))]
194    pub const fn new_dual<ML, PTEL, MU, PTEU, AL, F>() -> Self
195    where
196        ML: PageTableMeta<VirtAddr = A, Coverage = crate::meta::LowerCoverage<A>>,
197        PTEL: GenericPTE,
198        MU: PageTableMeta<VirtAddr = A, Coverage = crate::meta::UpperCoverage<A>>,
199        PTEU: GenericPTE,
200        AL: PageAllocator,
201        F: TlbFlusher<A>,
202        [(); ML::LEVELS - 1]: Sized,
203        [(); MU::LEVELS - 1]: Sized,
204    {
205        Self {
206            new_alloc: || {
207                crate::pt::DualPageTable::<A, ML, PTEL, MU, PTEU, AL, F>::new_alloc()
208                    .map(|pt| OpaquePageTableRoot::Dual(pt.leak()))
209            },
210            apply_actions: |root, actions| {
211                let OpaquePageTableRoot::Dual(root) = root else {
212                    bug!("passing single-root page table to dual-root page table operation")
213                };
214
215                // Suppress hierarchy destruction on errors or panics while transferring cleanup.
216                let mut pt = ManuallyDrop::new(unsafe {
217                    crate::pt::DualPageTable::<A, ML, PTEL, MU, PTEU, AL, F>::new_at(root)
218                });
219                let mut cursor = pt.cursor();
220                for action in actions {
221                    match action {
222                        PageTableAction::Map {
223                            vaddr,
224                            paddr,
225                            size,
226                            flags,
227                        } => {
228                            cursor.map(vaddr, paddr, size, flags)?;
229                        }
230                        PageTableAction::Unmap { vaddr, size } => {
231                            cursor.unmap(vaddr, size)?;
232                        }
233                    }
234                }
235
236                let result = unsafe { cursor.take_cleanup() };
237                drop(cursor);
238
239                let new_root = ManuallyDrop::into_inner(pt).leak();
240                debug_assert_eq!(root, new_root);
241
242                Ok(result)
243            },
244            destroy: |root| {
245                let OpaquePageTableRoot::Dual(root) = root else {
246                    bug!("passing single-root page table to dual-root page table destruction")
247                };
248                drop(unsafe {
249                    crate::pt::DualPageTable::<A, ML, PTEL, MU, PTEU, AL, F>::new_at(root)
250                });
251            },
252        }
253    }
254}

256/// A type-erased descriptor for a concrete multi-level page table type.
257///
258/// The descriptor stores operations specialized for one concrete multi-level page table
259/// format. It does not store a root page table and can create any number of
260/// [`OpaquePageTable`] values using that format.
261///
262/// The descriptor captures the allocator used for all table frame allocation and
263/// deallocation performed through those values.
264#[derive(Clone)]
265pub struct OpaquePageTableType<A: VirtAddr> {
266    /// The operations specialized for the selected concrete type.
267    ///
268    /// The function pointers retain the concrete metadata, entry, and allocator types.
269    methods: PageTableMethods<A>,
270}
271
272/// Construction of descriptors and opaque multi-level page tables.
273///
274/// Descriptors select the concrete operations used by every value they construct.
275impl<A: VirtAddr> OpaquePageTableType<A> {
276    /// Creates a descriptor for a concrete single-root multi-level page table format.
277    ///
278    /// `M` supplies the page table hierarchy geometry and must use `A` for virtual addresses.
279    /// `PTE` supplies the concrete page table entry representation and `AL`
280    /// supplies the allocator for table frames.
281    // NOTE: This direct bound currently compiles on nightly-2026-06-01, but
282    // newer nightly trait solvers may report E0284 when downstream crates
283    // instantiate it with architecture metadata.
284    pub const fn new<M, PTE, AL, F>() -> Self
285    where
286        M: PageTableMeta<VirtAddr = A>,
287        PTE: GenericPTE,
288        AL: PageAllocator,
289        F: TlbFlusher<A>,
290        [(); M::LEVELS]: Sized,
291        [(); M::LEVELS - 1]: Sized,
292    {
293        Self {
294            methods: PageTableMethods::<A>::new::<M, PTE, AL, F>(),
295        }
296    }
297
298    /// Creates a descriptor for an AArch64 dual-root multi-level page table type.
299    ///
300    /// `ML` and `PTEL` describe the lower-address root, while `MU` and `PTEU`
301    /// describe the upper-address root. The two metadata types must use the
302    /// same semantic virtual-address type and the corresponding lower and
303    /// upper coverage policies.
304    // NOTE: As with `new`, newer nightly trait solvers may fail to infer these
305    // anonymous const bounds when a downstream crate instantiates this method.
306    #[cfg(any(test, doc, target_arch = "aarch64"))]
307    #[cfg_attr(doc, doc(cfg(target_arch = "aarch64")))]
308    pub const fn new_dual<ML, PTEL, MU, PTEU, AL, F>() -> Self
309    where
310        ML: PageTableMeta<VirtAddr = A, Coverage = crate::meta::LowerCoverage<A>>,
311        PTEL: GenericPTE,
312        MU: PageTableMeta<VirtAddr = A, Coverage = crate::meta::UpperCoverage<A>>,
313        PTEU: GenericPTE,
314        AL: PageAllocator,
315        F: TlbFlusher<A>,
316        [(); ML::LEVELS - 1]: Sized,
317        [(); MU::LEVELS - 1]: Sized,
318    {
319        Self {
320            methods: PageTableMethods::<A>::new_dual::<ML, PTEL, MU, PTEU, AL, F>(),
321        }
322    }
323
324    /// Creates a descriptor whose allocation and mutation operations report bugs.
325    ///
326    /// This sentinel is suitable for constant initialization but must be
327    /// replaced before use. Its destruction operation is a no-op.
328    pub const fn dummy() -> Self {
329        Self {
330            methods: PageTableMethods::dummy(),
331        }
332    }
333
334    /// Constructs an opaque multi-level page table from an existing hierarchy.
335    ///
336    /// The value uses the concrete format captured by this descriptor and does
337    /// not inspect or initialize the referenced root page tables. The returned value
338    /// assumes responsibility for their hierarchies' table-frame deallocation.
339    ///
340    /// # Safety
341    ///
342    /// The root variant must match the descriptor's single-root or dual-root format. The
343    /// hierarchies must match its concrete metadata and entry types, and its allocator must
344    /// be able to access and deallocate their table frames. For a single root, the remaining
345    /// requirements are those of [`PageTable::new_at`]. For dual roots, they are those of
346    /// [`crate::pt::DualPageTable::new_at`].
347    pub unsafe fn new_pagetable_at(&self, root: OpaquePageTableRoot) -> OpaquePageTable<A> {
348        OpaquePageTable {
349            root,
350            methods: self.methods.clone(),
351        }
352    }
353
354    /// Allocates and initializes a runtime type-erased multi-level page table.
355    ///
356    /// The returned value uses this descriptor's concrete format and stores the
357    /// root addresses allocated by the descriptor's captured allocator.
358    pub fn new_pagetable_alloc(&self) -> PagingResult<A, OpaquePageTable<A>> {
359        unsafe { Ok(self.new_pagetable_at((self.methods.new_alloc)()?)) }
360    }
361}

363/// A runtime type-erased multi-level page table.
364///
365/// The value stores root addresses and operations specialized for its concrete format.
366/// See the crate-level documentation for table-frame deallocation and hardware-use requirements.
367pub struct OpaquePageTable<A: VirtAddr> {
368    /// The physical root addresses of the managed hierarchies.
369    ///
370    /// The variant matches the format selected by the originating descriptor.
371    root: OpaquePageTableRoot,
372    /// The operations specialized for the managed hierarchies.
373    ///
374    /// The stored functions use the originating descriptor's concrete format and allocator.
375    methods: PageTableMethods<A>,
376}
377
378/// Operations on an opaque multi-level page table.
379///
380/// Mutation uses a temporary concrete cursor. Cleanup uses the captured [`TlbFlusher`] or is
381/// transferred explicitly to the caller.
382impl<A: VirtAddr> OpaquePageTable<A> {
383    /// Creates a sentinel multi-level page table.
384    ///
385    /// Allocation and mutation report bugs, while destruction is a no-op. The zero-root sentinel
386    /// is suitable for constant initialization before replacement with a concrete value.
387    pub const fn dummy() -> Self {
388        Self {
389            root: OpaquePageTableRoot::Single(pa!(0)),
390            methods: PageTableMethods::dummy(),
391        }
392    }
393
394    /// Returns the erased physical root addresses of the managed hierarchies.
395    ///
396    /// The returned root is not translated or dereferenced by this accessor. Its
397    /// table frames remain associated with this value, and copying the addresses
398    /// does not extend their lifetime.
399    pub fn root(&self) -> OpaquePageTableRoot {
400        self.root
401    }
402
403    /// Consumes the multi-level page table without deallocating the table frames.
404    ///
405    /// This operation reads only the stored root and calls none of the stored method pointers,
406    /// including destruction. Another compatible allocator type can use the hierarchy, and the
407    /// caller is otherwise responsible for arranging its deallocation.
408    pub fn leak(self) -> OpaquePageTableRoot {
409        let manually_dropped = ManuallyDrop::new(self);
410        manually_dropped.root
411    }
412
413    /// Maps a virtual address range to a physical address range.
414    ///
415    /// The concrete page table hierarchy format comes from the originating descriptor. The
416    /// range is rounded to base-page boundaries and existing mappings are
417    /// replaced according to [`crate::pt::PageTableCursorLike::map`].
418    /// See [`Self::apply_actions`] for cleanup and partial-failure behavior.
419    pub fn map(
420        &mut self,
421        vaddr: A,
422        paddr: PhysAddr,
423        size: usize,
424        flags: MappingFlags,
425    ) -> PagingResult<A> {
426        self.apply_actions(core::iter::once(PageTableAction::Map {
427            vaddr,
428            paddr,
429            size,
430            flags,
431        }))
432    }
433
434    /// Removes mappings from a virtual address range.
435    ///
436    /// The range is rounded to base-page boundaries and mapped data frames remain caller-managed.
437    /// See [`Self::apply_actions`] for cleanup and partial-failure behavior.
438    pub fn unmap(&mut self, vaddr: A, size: usize) -> PagingResult<A> {
439        self.apply_actions(core::iter::once(PageTableAction::Unmap { vaddr, size }))
440    }
441
442    /// Applies a sequence of page table hierarchy actions.
443    ///
444    /// The concrete page table hierarchy format comes from the originating descriptor. Each
445    /// action is applied in order, including any immediate table-frame deallocation. Pending TLB
446    /// invalidations are processed through the captured [`TlbFlusher`] before this method returns.
447    ///
448    /// If an action fails, earlier actions remain applied and their pending TLB
449    /// invalidations are flushed before the error is returned. An iterator panic also flushes
450    /// completed mutations during unwinding while retaining the table frames in the remaining
451    /// hierarchy. The operation is not transactional. See [`PendingCleanup`] for synchronization
452    /// requirements and [`Self::apply_actions_with_cleanup`] for transferring cleanup on success.
453    pub fn apply_actions<I: IntoIterator<Item = PageTableAction<A>>>(
454        &mut self,
455        actions: I,
456    ) -> PagingResult<A> {
457        let cleanup = self.apply_actions_with_cleanup(actions)?;
458        cleanup.clean_up();
459        Ok(())
460    }
461
462    /// Applies actions and returns pending cleanup on success.
463    ///
464    /// Same mutation and partial-failure behavior as [`Self::apply_actions`]. Successful calls
465    /// transfer cleanup without processing it. An action error or iterator panic instead causes
466    /// the temporary cursor to process pending cleanup before returning the error or unwinding.
467    /// The selected strategy and hardware-use preconditions must therefore permit automatic
468    /// cleanup on these paths. See [`PendingCleanup`] for synchronization requirements.
469    pub fn apply_actions_with_cleanup<I: IntoIterator<Item = PageTableAction<A>>>(
470        &mut self,
471        actions: I,
472    ) -> PagingResult<A, PendingCleanup<A>> {
473        let mut actions = actions.into_iter();
474        (self.methods.apply_actions)(self.root, &mut actions)
475    }
476}

478/// Opaque multi-level page table destruction.
479///
480/// See the crate-level documentation for table-frame deallocation and hardware-use requirements.
481impl<A: VirtAddr> Drop for OpaquePageTable<A> {
482    /// Deallocates table frames through the stored destruction operation.
483    ///
484    /// The operation uses the concrete type and allocator selected by the originating descriptor.
485    fn drop(&mut self) {
486        (self.methods.destroy)(self.root);
487    }
488}

490/// Lifecycle regressions that require access to erased method pointers.
491///
492/// These tests verify that leaking a multi-level page table works after its stored operations
493/// become unusable.
494#[cfg(test)]
495mod tests {
496    use memory_addr::{VirtAddr, pa};
497    use std::{panic::catch_unwind, string::String};
498
499    use super::{OpaquePageTable, OpaquePageTableRoot, PageTableMethods};
500
501    /// Verifies that leaking an opaque multi-level page table calls none of its stored methods.
502    ///
503    /// Kernel relocation can invalidate these method pointers before another compatible value uses
504    /// the returned root, so even the destruction operation must remain untouched.
505    #[test]
506    fn opaque_leak_returns_root_without_calling_stored_methods() {
507        let root = OpaquePageTableRoot::Single(pa!(0x123000));
508        let page_table = OpaquePageTable::<VirtAddr> {
509            root,
510            methods: PageTableMethods {
511                new_alloc: || panic!("leak called the stored allocation method"),
512                apply_actions: |_, _| panic!("leak called the stored mutation method"),
513                destroy: |_| panic!("leak called the stored destruction method"),
514            },
515        };
516
517        assert_eq!(page_table.leak(), root);
518    }
519
520    /// Verifies that dummy operations report an internal bug before terminating.
521    ///
522    /// Dummy descriptors are initialization sentinels and must never be used as live tables.
523    #[test]
524    fn dummy_operation_reports_bug() {
525        let payload = catch_unwind(|| {
526            let methods = PageTableMethods::<VirtAddr>::dummy();
527            let _ = (methods.new_alloc)();
528        })
529        .expect_err("dummy operation must terminate");
530        let message = payload
531            .downcast_ref::<String>()
532            .map(String::as_str)
533            .or_else(|| payload.downcast_ref::<&str>().copied())
534            .expect("default termination must carry a string message");
535        assert!(
536            message.contains("Bug at"),
537            "unexpected bug report: {message}"
538        );
539    }
