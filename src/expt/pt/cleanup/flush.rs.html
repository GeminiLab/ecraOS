1//! TLB flush management for page table hierarchy mutations.
2//!
3//! Cursor operations accumulate requests for the selected [`TlbFlusher`] to process.
4
5use core::ops::{Add, AddAssign};
6
7use dyn_static_traits::dyn_static_traits;
8use heapless::Vec as HeaplessVec;
9
10use crate::meta::VirtAddr;
11
12/// A TLB flushing strategy.
13///
14/// Implementations select the architecture and invalidation scope. A local strategy affects only
15/// the calling processor. A strategy covering remote processors must complete the required remote
16/// invalidations before returning, including the required translation and page-walk synchronization
17/// within that scope.
18///
19/// Callers must select a strategy suitable for the affected page table hierarchy and its hardware
20/// users. An incorrect strategy can leave stale translations or page-walk state when accesses
21/// resume under the crate-level access coordination contract.
22/// A strategy that waits for remote acknowledgement must not be invoked while the caller holds an
23/// IRQ-disabling guard, page-table lock, or allocator lock.
24#[dyn_static_traits(DynTlbFlusher)]
25pub trait TlbFlusher<A: VirtAddr> {
26    /// Flushes the TLB with a specified address.
27    ///
28    /// `Some(addr)` invalidates the mapping containing `addr`. `None` requests invalidation of all
29    /// addresses within this strategy's scope. Required synchronization completes before return.
30    fn flush(addr: Option<A>);
31}
32
33/// The capacity for individual mapping invalidations before a full flush.
34///
35/// Recording one more page than this capacity promotes the pending operation to
36/// a full TLB flush within the selected strategy's scope.
37pub const SMALL_FLUSH_THRESHOLD: usize = 32;
38
39/// A TLB invalidation requested by one page table hierarchy mutation.
40///
41/// Requests can leave the TLB unchanged, invalidate one virtual mapping, or
42/// invalidate all addresses within the selected strategy's scope.
43#[derive(Default)]
44#[must_use = "the TLB flush must be performed or aggregated into a `TlbFlushes` value"]
45pub enum TlbFlush<A: VirtAddr> {
46    /// No TLB invalidation.
47    ///
48    /// This is used when the operation did not replace a live leaf mapping.
49    #[default]
50    None,
51    /// Invalidation of the mapping containing one virtual address.
52    ///
53    /// The address is passed to the selected [`TlbFlusher`].
54    Page(A),
55    /// Invalidation of all addresses within the selected strategy's scope.
56    ///
57    /// This conservative request is used for changes involving huge mappings or newly installed
58    /// child page table entries. Accumulating too many page requests also requires a full flush.
59    Full,
60}
61
62/// TLB invalidation requests accumulated.
63///
64/// Individual pages are retained up to [`SMALL_FLUSH_THRESHOLD`], after which
65/// the state is promoted to a full flush.
66///
67/// This type is **just a list** and does **NOT** automatically perform TLB flushing on its items.
68/// It's caller's ability and responsibility to interpret the list and perform TLB flushing on its
69/// items.
70#[derive(Default)]
71pub enum TlbFlushes<A: VirtAddr> {
72    /// No pending invalidation.
73    ///
74    /// Flushing this state performs no architecture operation.
75    #[default]
76    None,
77    /// A bounded collection of virtual mappings to invalidate.
78    ///
79    /// Addresses are flushed individually when the collection is drained.
80    Pages(HeaplessVec<A, SMALL_FLUSH_THRESHOLD>),
81    /// A pending invalidation of all addresses within the selected strategy's scope.
82    ///
83    /// Further invalidation requests cannot weaken this state.
84    Full,
85}
86
87impl<A: VirtAddr> AddAssign<TlbFlush<A>> for TlbFlushes<A> {
88    /// Merges one invalidation request into the pending state.
89    ///
90    /// Page requests accumulate until the bounded collection is full. A full
91    /// request, or an overflowing page collection, promotes the state to
92    /// [`TlbFlushes::Full`] until the records are consumed.
93    fn add_assign(&mut self, rhs: TlbFlush<A>) {
94        match rhs {
95            TlbFlush::None => {}
96            TlbFlush::Page(vaddr) => match self {
97                TlbFlushes::None => {
98                    let mut pages = HeaplessVec::new();
99                    // The collection is initially empty, so this push cannot fail.
100                    let _ = pages.push(vaddr);
101                    *self = TlbFlushes::Pages(pages);
102                }
103                TlbFlushes::Pages(pages) => {
104                    if pages.push(vaddr).is_err() {
105                        *self = TlbFlushes::Full;
106                    }
107                }
108                TlbFlushes::Full => {}
109            },
110            TlbFlush::Full => *self = TlbFlushes::Full,
111        }
112    }
113}

115impl<A: VirtAddr> Add<TlbFlush<A>> for TlbFlushes<A> {
116    /// The pending-flush state produced by the addition.
117    ///
118    /// Addition consumes and returns the same state type after merging a request.
119    type Output = Self;
120
121    /// Merges an invalidation request and returns the updated state.
122    ///
123    /// This is the consuming counterpart to [`AddAssign::add_assign`].
124    fn add(mut self, rhs: TlbFlush<A>) -> Self::Output {
125        self += rhs;
126        self
127    }
128}
129
130impl<A: VirtAddr> TlbFlushes<A> {
131    /// Executes the recorded invalidations and clears the list.
132    ///
133    /// See [`TlbFlusher`] for strategy selection and synchronization requirements.
134    pub fn flush_with(&mut self, flusher: &DynTlbFlusher<A>) {
135        match self {
136            TlbFlushes::None => {}
137            TlbFlushes::Pages(pages) => {
138                for vaddr in pages.iter().copied() {
139                    (flusher.flush)(Some(vaddr));
140                }
141            }
142            TlbFlushes::Full => (flusher.flush)(None),
143        }
144
145        self.clear();
146    }
147
148    /// Clears the list and discards all its items.
149    pub fn clear(&mut self) {
150        *self = TlbFlushes::None;
151    }
152
153    /// Merges another collection without performing invalidation.
154    ///
155    /// Full dominates both collections and individual addresses retain their insertion order.
156    pub fn append(&mut self, other: Self) {
157        match &other {
158            Self::None => {}
159            Self::Pages(pages) => {
160                for &address in pages {
161                    *self += TlbFlush::Page(address);
162                }
163            }
164            Self::Full => *self += TlbFlush::Full,
165        }
166    }
167}
