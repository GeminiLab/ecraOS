13/// Metadata for the RISC-V Sv39 page table hierarchy format.
14///
15/// Sv39 uses three nine-bit levels and supports leaf mappings at every level.
16pub struct Sv39PageTableMeta;
17
18impl PageTableMeta for Sv39PageTableMeta {
19    /// The standard virtual-address type used by Sv39.
20    type VirtAddr = VirtAddr;
21
22    /// The three translation levels in Sv39.
23    const LEVELS: usize = 3;
24    /// The twelve offset bits in an Sv39 4 KiB page.
25    const PAGE_OFFSET_BITS: usize = 12;
26    /// The nine virtual-address bits consumed by each Sv39 level.
27    const LEVEL_BITS: [usize; Self::LEVELS] = [9, 9, 9];
28    /// Sv39 uses sign-extended lower and upper address ranges.
29    type Coverage = SymmetricCoverage<Self::VirtAddr>;
30
31    /// The highest level that supports an Sv39 leaf mapping.
32    ///
33    /// Every level in Sv39 may contain a leaf.
34    const MAX_PAGE_LEVEL: usize = 2;
35}

85/// The local TLB flushing strategy for RISC-V 64.
86///
87/// `Some(addr)` uses address-specific `SFENCE.VMA` with ASID zero in a general-purpose register
88/// as `rs2`. It invalidates non-global leaf translations for that address and ASID zero.
89/// `None` uses `SFENCE.VMA x0, x0`, covering all addresses and ASIDs, including global
90/// translations.
91///
92/// Use this strategy only when address-specific invalidations affect ASID-zero, non-global
93/// translations. A global ancestor descriptor also makes a translation global. See
94/// [`TlbFlusher`] for synchronization requirements.
95pub struct Riscv64LocalTlbFlusher;
96
97impl TlbFlusher<VirtAddr> for Riscv64LocalTlbFlusher {
98    fn flush(addr: Option<VirtAddr>) {
99        flush_local_tlb(addr);
100    }
101}
102
103/// Flushes one mapping or the entire local RISC-V TLB.
104///
105/// See [`Riscv64LocalTlbFlusher`] for the ASID and global-translation scope of each operation.
106#[inline]
107fn flush_local_tlb(vaddr: Option<VirtAddr>) {
108    if let Some(vaddr) = vaddr {
109        riscv::asm::sfence_vma(0, vaddr.as_usize())
110    } else {
111        riscv::asm::sfence_vma_all();
112    }
113}
