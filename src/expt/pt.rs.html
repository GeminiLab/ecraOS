20/// The common interface for concrete multi-level page table types.
21///
22/// This trait is primarily used to constrain the behavior of [`single::PageTable`]
23/// and [`dual::DualPageTable`]. Most callers should use one of those concrete types,
24/// or [`crate::opaque::OpaquePageTable`] when runtime type erasure is required.
25pub trait PageTableLike: Sized {
26    /// The semantic virtual-address type accepted by this multi-level page table.
27    ///
28    /// The type determines the addresses accepted by mapping operations.
29    type VirtAddr: VirtAddr;
30    /// The value identifying one or more page table roots.
31    ///
32    /// Copying this value does not change the table frames' lifetime. Therefore,
33    /// the user must not use this value after dropping the page table hierarchy.
34    type Root: Copy;
35    /// The allocator for table frames.
36    ///
37    /// Allocations, physical-address translation, and deallocation use this type.
38    type Allocator: PageAllocator;
39    /// The TLB flusher for this page table hierarchy.
40    ///
41    /// TLB invalidations are performed using this type.
42    type TlbFlusher: TlbFlusher<Self::VirtAddr>;
43    /// The cursor used to mutate this page table hierarchy.
44    ///
45    /// The cursor borrows the multi-level page table mutably during mutation.
46    type Cursor<'a>: PageTableCursorLike<
47            VirtAddr = Self::VirtAddr,
48            Allocator = Self::Allocator,
49            PageTable = Self,
50        >
51    where
52        Self: 'a;
53
54    /// Constructs a multi-level page table from an existing page table hierarchy.
55    ///
56    /// # Safety
57    ///
58    /// The root must satisfy the requirements of [`PageTable::new_at`] for a single-root
59    /// implementation, or [`DualPageTable::new_at`] for a dual-root implementation.
60    unsafe fn new_at(root: Self::Root) -> Self;
61
62    /// Allocates and initializes a multi-level page table through the selected allocator.
63    ///
64    /// Only the root page tables are allocated initially. Child page tables are allocated
65    /// as needed by cursor operations.
66    fn new_alloc() -> PagingResult<Self::VirtAddr, Self>;
67
68    /// Returns the root identifier without changing the table frames' lifetime.
69    ///
70    /// The returned physical address (or pair of physical addresses for dual-root
71    /// multi-level page tables) must not be used after dropping this value.
72    fn root(&self) -> Self::Root;
73
74    /// Consumes the multi-level page table without deallocating its table frames.
75    ///
76    /// Another multi-level page table can use the returned root if its allocator can
77    /// access and deallocate every existing table frame. Otherwise, the caller must
78    /// arrange their deallocation.
79    fn leak(self) -> Self::Root;
80
81    /// Creates a cursor for batched mutations.
82    ///
83    /// See [PageTableCursorLike] for batching and cleanup behavior.
84    fn cursor(&mut self) -> Self::Cursor<'_>;
85}

87/// The common interface for cursors over page table hierarchies.
88///
89/// This trait is the common interface for [`single::PageTableCursor`] and
90/// [`dual::DualPageTableCursor`]. A cursor's mutable borrow excludes other Rust
91/// mutations through the borrowed multi-level page table but does not synchronize
92/// hardware page walks.
93///
94/// Map and unmap round ranges outward to base-page boundaries. A successful return means the PTE
95/// mutations and any table-frame deallocations are complete, with TLB invalidations still pending.
96/// Errors retain earlier mutations, including any intermediate tables already installed and any
97/// table frames already deallocated. Operations are not transactional. See the crate-level
98/// documentation for access coordination requirements.
99///
100/// Cleanup remains pending until [`Self::clean_up`] processes it or [`Self::take_cleanup`]
101/// transfers it to the caller. Implementations also process any remaining cleanup automatically
102/// when the cursor leaves scope.
103pub trait PageTableCursorLike {
104    /// The semantic virtual-address type accepted by this cursor.
105    ///
106    /// It matches the virtual-address type of the borrowed multi-level page table.
107    type VirtAddr: VirtAddr;
108    /// The allocator used by this cursor.
109    ///
110    /// Intermediate page tables are allocated through the borrowed multi-level page table's
111    /// allocator.
112    type Allocator: PageAllocator;
113    /// The multi-level page table type being mutated by this cursor.
114    ///
115    /// The cursor borrows the multi-level page table mutably during mutation.
116    type PageTable: PageTableLike;
117
118    /// Maps a virtual range to physical memory.
119    ///
120    /// This operation changes page table entries while mapped data frames remain
121    /// caller-managed. Replacing a child-table descriptor immediately deallocates its subtree's
122    /// table frames. See the trait-level contract for partial failure and cleanup.
123    fn map(
124        &mut self,
125        vaddr: Self::VirtAddr,
126        paddr: PhysAddr,
127        size: usize,
128        flags: MappingFlags,
129    ) -> PagingResult<Self::VirtAddr>;
130
131    /// Removes mappings from a virtual address range.
132    ///
133    /// This operation clears mappings but does not deallocate their data frames.
134    /// Intermediate tables remain allocated. See the trait-level contract for partial failure
135    /// and cleanup.
136    fn unmap(&mut self, vaddr: Self::VirtAddr, size: usize) -> PagingResult<Self::VirtAddr>;
137
138    /// Processes pending invalidations using the selected TLB flushing strategy.
139    ///
140    /// See [`PendingCleanup`] for synchronization requirements.
141    fn clean_up(&mut self);
142
143    /// Transfers all pending cleanup out of the cursor without processing it.
144    ///
145    /// # Safety
146    ///
147    /// The caller must handle the returned cleanup properly after calling this
148    /// function, and before any other operations on the cursor.
149    ///
150    /// See [`PendingCleanup`] for synchronization requirements.
151    unsafe fn take_cleanup(&mut self) -> PendingCleanup<Self::VirtAddr>;
152}
