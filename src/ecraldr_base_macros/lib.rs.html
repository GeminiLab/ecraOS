1//! Proc macros for marking and calling the ecraldr_base kernel entry point from platform code.
2
3use proc_macro::TokenStream;
4use proc_macro2::TokenStream as TokenStream2;
5use quote::{format_ident, quote};
6use syn::{Ident, ItemFn, parse_macro_input};
7
8/// Exported symbol name for the kernel entry.
9const fn kernel_entry_export_name() -> &'static str {
10    "__ecraos_kernel_entry"
11}
12
13/// Builds the `syn` identifier for the exported kernel entry symbol (`__ecraos_kernel_entry`).
14fn kernel_entry_export_ident() -> Ident {
15    format_ident!("{}", kernel_entry_export_name())
16}
17
18/// Marks a function as the kernel entry function, which will be called by boot
19/// code through [`call_kernel_entry`] once the very early initialization is
20/// complete.
21///
22/// The kernel entry function should then further initialize the platform by
23/// methods defined in `InitIf` trait.
24#[proc_macro_attribute]
25pub fn kernel_entry(attr: TokenStream, input: TokenStream) -> TokenStream {
26    if !attr.is_empty() {
27        return quote! {
28            compile_error!("#[kernel_entry] does not accept attributes");
29        }
30        .into();
31    }
32
33    let kernel_entry = parse_macro_input!(input as ItemFn);
34    let kernel_entry_name = kernel_entry.sig.ident.clone();
35    let kernel_entry_export_name = kernel_entry_export_name();
36
37    quote! {
38        #[unsafe(export_name = #kernel_entry_export_name)]
39        #[unsafe(link_section = ".text.kernel_entry")]
40        #kernel_entry
41
42        #[doc(hidden)]
43        #[allow(clippy::unused_unit)]
44        const KERNEL_ENTRY_SIGNATURE_GUARD: () = {
45            let _kernel_entry_must_match_signature: ::ecraldr_base::KernelEntryType = #kernel_entry_name;
46            ()
47        };
48    }
49    .into()
50}
51
52/// Calls the kernel entry function marked by [`kernel_entry`]. Two arguments
53/// should be passed to the function: `hart_id` (Hardware Thread ID of the
54/// bootstrap processor) and `arg` (the architecture-and-platform-specific
55/// argument that will later be passed to the `InitIf::init_early` method).
56///
57/// The kernel entry function should be called by boot code with the
58/// following conditions met:
59/// - A pagetable providing identical mapping and covering the whole kernel
60///   image must be enabled on the bootstrap processor.
61/// - Interrupts must be globally disabled on the bootstrap processor.
62///
63/// Boot code must call the kernel entry function as early as possible before
64/// any other initialization code is executed.
65#[proc_macro]
66pub fn call_kernel_entry(input: TokenStream) -> TokenStream {
67    let input = TokenStream2::from(input);
68    let kernel_entry_ident = kernel_entry_export_ident();
69
70    quote! {
71        {
72            unsafe extern "Rust" {
73                fn #kernel_entry_ident(hart_id: usize, arg: *const ::ecraldr_base::BootArg) -> !;
74            }
75
76            unsafe {
77                #kernel_entry_ident(#input)
78            }
79        }
80    }
81    .into()
82}
