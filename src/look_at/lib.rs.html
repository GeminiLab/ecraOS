18/// Collects identifiers declared by a function parameter pattern.
19///
20/// The identifiers form a readable forwarding name when a pattern has multiple bindings.
21fn collect_pattern_identifiers(pattern: &Pat, identifiers: &mut Vec<Ident>) {
22    match pattern {
23        Pat::Ident(pattern) => {
24            identifiers.push(pattern.ident.clone());
25            if let Some((_, pattern)) = &pattern.subpat {
26                collect_pattern_identifiers(pattern, identifiers);
27            }
28        }
29        Pat::Or(pattern) => {
30            if let Some(case) = pattern.cases.first() {
31                collect_pattern_identifiers(case, identifiers);
32            }
33        }
34        Pat::Paren(pattern) => collect_pattern_identifiers(&pattern.pat, identifiers),
35        Pat::Reference(pattern) => collect_pattern_identifiers(&pattern.pat, identifiers),
36        Pat::Slice(pattern) => {
37            for element in &pattern.elems {
38                collect_pattern_identifiers(element, identifiers);
39            }
40        }
41        Pat::Struct(pattern) => {
42            for field in &pattern.fields {
43                collect_pattern_identifiers(&field.pat, identifiers);
44            }
45        }
46        Pat::Tuple(pattern) => {
47            for element in &pattern.elems {
48                collect_pattern_identifiers(element, identifiers);
49            }
50        }
51        Pat::TupleStruct(pattern) => {
52            for element in &pattern.elems {
53                collect_pattern_identifiers(element, identifiers);
54            }
55        }
56        Pat::Type(pattern) => collect_pattern_identifiers(&pattern.pat, identifiers),
57        _ => {}
58    }
59}

61/// Chooses a readable forwarding name for a function parameter pattern.
62///
63/// Existing bindings are preserved or combined, with an ordinal fallback for nameless patterns.
64fn forwarding_parameter_name(pattern: &Pat, index: usize, used_names: &mut Vec<String>) -> Ident {
65    let mut identifiers = Vec::new();
66    collect_pattern_identifiers(pattern, &mut identifiers);
67    let candidate = match identifiers.as_slice() {
68        [identifier] => identifier.clone(),
69        [] => format_ident!("argument_{}", index + 1, span = pattern.span()),
70        identifiers => {
71            let name = identifiers
72                .iter()
73                .map(|identifier| identifier.to_string().trim_start_matches("r#").to_owned())
74                .collect::<Vec<_>>()
75                .join("_and_");
76            format_ident!("{name}", span = pattern.span())
77        }
78    };
79    let canonical_name = candidate.to_string().trim_start_matches("r#").to_owned();
80    let candidate = if used_names.contains(&canonical_name) {
81        format_ident!(
82            "{}_argument_{}",
83            canonical_name,
84            index + 1,
85            span = pattern.span()
86        )
87    } else {
88        candidate
89    };
90    used_names.push(candidate.to_string().trim_start_matches("r#").to_owned());
91    candidate
92}

94/// Expands one forwarding function.
95///
96/// The wrapper preserves the declared parameter names and checks the target function signature.
97fn expand_function(
98    target_path: &Path,
99    attrs: Vec<Attribute>,
100    vis: Visibility,
101    mut sig: Signature,
102) -> Result<TokenStream2> {
103    if let Some(asyncness) = &sig.asyncness {
104        return Err(Error::new(
105            asyncness.span(),
106            "`look_at` does not support async functions",
107        ));
108    }
109    if let Some(variadic) = &sig.variadic {
110        return Err(Error::new(
111            variadic.span(),
112            "`look_at` does not support variadic functions",
113        ));
114    }
115
116    let mut call_args = Vec::new();
117    let mut input_types = Vec::new();
118    let mut used_names = Vec::new();
119    for (index, argument) in sig.inputs.iter_mut().enumerate() {
120        let FnArg::Typed(argument) = argument else {
121            return Err(Error::new(
122                argument.span(),
123                "`look_at` only supports free functions",
124            ));
125        };
126        let ident = forwarding_parameter_name(&argument.pat, index, &mut used_names);
127        call_args.push(ident.clone());
128        input_types.push(argument.ty.clone());
129        *argument.pat = parse_quote!(#ident);
130    }
131
132    let name = &sig.ident;
133    let generic_args = sig
134        .generics
135        .params
136        .iter()
137        .filter_map(|parameter| match parameter {
138            GenericParam::Type(parameter) => {
139                let ident = &parameter.ident;
140                Some(quote!(#ident))
141            }
142            GenericParam::Const(parameter) => {
143                let ident = &parameter.ident;
144                Some(quote!(#ident))
145            }
146            GenericParam::Lifetime(_) => None,
147        })
148        .collect::<Vec<_>>();
149    let target = if generic_args.is_empty() {
150        quote!(#target_path::#name)
151    } else {
152        quote!(#target_path::#name::<#(#generic_args),*>)
153    };
154
155    let contains_impl_trait = input_types
156        .iter()
157        .any(|ty| matches!(ty.as_ref(), Type::ImplTrait(_)))
158        || matches!(
159            &sig.output,
160            syn::ReturnType::Type(_, ty) if matches!(ty.as_ref(), Type::ImplTrait(_))
161        );
162    let type_check = if sig.constness.is_none() && !contains_impl_trait {
163        let unsafety = &sig.unsafety;
164        let abi = &sig.abi;
165        let output = &sig.output;
166        quote! { let _: #unsafety #abi fn(#(#input_types),*) #output = #target; }
167    } else {
168        TokenStream2::new()
169    };
170    let call = quote!(#target(#(#call_args),*));
171    let call = if sig.unsafety.is_some() {
172        quote!(unsafe { #call })
173    } else {
174        call
175    };
176    let sig = &sig;
177    Ok(quote! { #(#attrs)* #[inline(always)] #vis #sig { #type_check #call } })
178}

180/// Expands one constant declaration.
181///
182/// The generated constant has the declared type and takes its value from the target constant, so
183/// the compiler checks the target constant against the declared type.
184fn expand_const(
185    target_path: &Path,
186    attrs: Vec<Attribute>,
187    vis: Visibility,
188    name: Ident,
189    declared_type: Type,
190) -> Result<TokenStream2> {
191    Ok(quote! { #(#attrs)* #vis const #name: #declared_type = #target_path::#name; })
192}

194/// Expands one use tree declaration.
195///
196/// The tree is prefixed with the target path, so nested groups and renames resolve inside the
197/// selected module.
198fn expand_use(
199    target_path: &Path,
200    attrs: Vec<Attribute>,
201    vis: Visibility,
202    tree: syn::UseTree,
203) -> Result<TokenStream2> {
204    Ok(quote! { #(#attrs)* #vis use #target_path::#tree; })
205}

207/// Expands one type declaration with optional trait bounds.
208///
209/// The type is re-exported from the target module. Bounds produce an anonymous constant that
210/// checks the target type, and that check keeps only the conditional attributes of the declaration.
211fn expand_type(
212    target_path: &Path,
213    attrs: Vec<Attribute>,
214    vis: Visibility,
215    name: Ident,
216    bounds: Option<TypeParamBounds>,
217) -> Result<TokenStream2> {
218    let reexport = quote! { #(#attrs)* #vis use #target_path::#name; };
219    let conditional_attrs = attrs
220        .iter()
221        .filter(|attribute| {
222            attribute.path().is_ident("cfg") || attribute.path().is_ident("cfg_attr")
223        })
224        .collect::<Vec<_>>();
225    let assertion = bounds.map(|bounds| {
226        quote! {
227            #(#conditional_attrs)*
228            const _: () = {
229                let _: fn() = || {
230                    fn assert_impl<T: #bounds>() {}
231                    let _ = assert_impl::<#target_path::#name>;
232                };
233            };
234        }
235    });
236    Ok(quote! { #reexport #assertion })
237}

239/// Expands one placeholder item.
240///
241/// The item's attributes and visibility are passed to the expansion for its declaration kind.
242fn expand_item(item: LookAtItem, target_path: &Path) -> Result<TokenStream2> {
243    let attrs = item.attrs;
244    let vis = item.vis;
245
246    match item.kind {
247        LookAtItemKind::Function(function) => expand_function(target_path, attrs, vis, function),
248        LookAtItemKind::Const {
249            name,
250            declared_type,
251        } => expand_const(target_path, attrs, vis, name, declared_type),
252        LookAtItemKind::Use(tree) => expand_use(target_path, attrs, vis, tree),
253        LookAtItemKind::Type { name, bounds } => expand_type(target_path, attrs, vis, name, bounds),
254    }
255}

257/// Expands all declarations in one macro invocation.
258///
259/// Every declaration resolves relative to the same selected implementation module.
260fn expand(input: LookAtInput) -> Result<TokenStream2> {
261    let target_path = input.target_path;
262    input
263        .items
264        .into_iter()
265        .map(|item| expand_item(item, &target_path))
266        .collect()
267}

269/// Generates wrappers and re-exports for declarations in an implementation module.
270///
271/// `look_at!` collects the public interface shared by several architecture-specific modules. The
272/// invocation names one implementation module, then lists declarations that should be available at
273/// the invocation site. The declarations are intentionally only signatures or placeholders: the
274/// implementation remains in the selected module.
275///
276/// # Overview
277///
278/// The target path is resolved once and is used for every item in the invocation. Functions are
279/// emitted as wrappers that call the target function. Types and `use` trees are re-exported, while
280/// constants are emitted as typed forwarding constants. The macro does not select the
281/// implementation module. A conditional import, such as the `#[cfg]` imports of `current` below,
282/// still selects it, and `look_at!` states and checks the interface that the selected module must
283/// provide.
284///
285/// ```
286/// # use look_at::look_at;
287/// mod x86_64 {
288///     pub fn halt() {}
289///     pub const PAGE_SIZE: usize = 4096;
290/// }
291///
292/// mod riscv64 {
293///     pub fn halt() {}
294///     pub const PAGE_SIZE: usize = 4096;
295/// }
296///
297/// #[cfg(target_arch = "x86_64")]
298/// use x86_64 as current;
299/// #[cfg(target_arch = "riscv64")]
300/// use riscv64 as current;
301///
302/// look_at! {
303///     @current:
304///     pub fn halt();
305///     pub const PAGE_SIZE: usize;
306/// }
307/// ```
308///
309/// The generated `halt` function calls `x86_64::halt` and `PAGE_SIZE` has the value from
310/// `x86_64::PAGE_SIZE` when building for x86_64, and their `riscv64` counterparts when building for
311/// riscv64.
312///
313/// # Syntax
314///
315/// An invocation starts with `@`, followed by a Rust path and `:`. Every following declaration
316/// must end with `;`. Attributes and visibility precede the declaration just as they do on ordinary
317/// Rust items.
318///
319/// ```ignore
320/// look_at! {
321///     @crate::arch::current:
322///     #[cfg(feature = "debug")]
323///     pub fn initialize();
324///     pub type Context: Send;
325///     pub type Handle;
326///     pub const PAGE_SIZE: usize;
327///     pub use Error as InitError;
328/// }
329/// ```
330///
331/// The invocation must contain at least one item. A function declaration has no body, and all
332/// other declarations are placeholders without a definition or value.
333///
334/// ## Functions
335///
336/// Function declarations keep their name, visibility, generics, `where` clause, ABI, and `const`
337/// and `unsafe` qualifiers. Their parameter patterns are replaced with readable local names so the
338/// generated wrapper can forward each argument positionally. A single binding keeps its name,
339/// several bindings are joined with `_and_`, and a pattern without bindings becomes `argument_N`
340/// for its one-based position. A repeated name receives an `_argument_N` suffix. Wrappers are
341/// marked `#[inline(always)]`.
342///
343/// ```ignore
344/// look_at! {
345///     @crate::arch::current:
346///     pub fn copy_bytes<'a>(source: &'a [u8], destination: &mut [u8]);
347///     pub unsafe extern "C" fn enter(stack: *mut u8) -> !;
348///     pub fn combine<T>(left: T, right: T) -> T where T: Add<Output = T>;
349/// }
350/// ```
351///
352/// Ordinary functions receive a generated function-pointer assignment that checks the declared
353/// signature against the selected target. Functions containing `impl Trait` in an input or output
354/// remain callable wrappers, but skip that generated function-pointer check because opaque types
355/// cannot appear in it. `const` functions also skip it, and their wrappers can only call `const`
356/// targets. The forwarding call still checks the arguments and the return value of these
357/// functions, but it permits coercions.
358///
359/// An `unsafe` wrapper calls its target in an `unsafe` block, so callers take on the safety
360/// requirements of the declaration. `async` functions, variadic functions, functions with a `self`
361/// receiver, and declarations with a body are rejected.
362///
363/// ## Type declarations
364///
365/// A `type` declaration is a placeholder for a target type alias, struct, enum, or union. It emits
366/// a `use` re-export, so the target's complete type definition and generic interface remain intact.
367/// Generic parameters are accepted to describe the intended interface, but are not independently
368/// checked or redeclared by the macro.
369///
370/// ```ignore
371/// look_at! {
372///     @crate::arch::current:
373///     pub type Device;
374///     pub type Buffer<T>;
375///     pub type Context: Send + Sync;
376/// }
377/// ```
378///
379/// A bound is allowed only on a non-generic placeholder. The macro checks it inside an anonymous
380/// constant, for example `pub type Context: Send;` verifies that the selected `Context` implements
381/// `Send` at compile time. Without a bound, the declaration only checks that the target type
382/// exists.
383///
384/// ## Constants and use trees
385///
386/// A constant placeholder contains the name and expected type, but no value. The expansion reads
387/// the value from the selected module and assigns it to the declared type. A target constant of
388/// another type is rejected unless its value coerces to the declared type.
389///
390/// ```ignore
391/// look_at! {
392///     @crate::arch::current:
393///     pub const MAX_CPUS: usize;
394///     pub use scheduler::Task as ArchTask;
395/// }
396/// ```
397///
398/// A `use` declaration is resolved below the target path, so aliases and nested use trees can be
399/// forwarded without repeating the architecture path at each use site. It only checks that the
400/// named items exist.
401///
402/// # Compared with conditional re-exports
403///
404/// The conditional import in the overview example already keeps target selection in one place, so
405/// the selected items could also be re-exported directly:
406///
407/// ```
408/// # mod x86_64 {
409/// #     pub fn halt() {}
410/// # }
411/// # mod riscv64 {
412/// #     pub fn halt() {}
413/// # }
414/// #[cfg(target_arch = "x86_64")]
415/// use x86_64 as current;
416/// #[cfg(target_arch = "riscv64")]
417/// use riscv64 as current;
418/// # #[cfg(not(any(target_arch = "x86_64", target_arch = "riscv64")))]
419/// # use x86_64 as current;
420///
421/// pub use current::halt;
422/// # fn main() {}
423/// ```
424///
425/// Both forms give callers one set of names without their own `#[cfg]` attributes, and both
426/// forward only the listed items. `look_at!` differs in how the interface is checked, which
427/// visibility its items can have, and what it generates. The remaining examples in this section
428/// define `current` directly and omit the `#[cfg]` selection.
429///
430/// ## Interface checks
431///
432/// A re-export exposes whatever the selected module defines. When one implementation diverges from
433/// the intended interface, the crate that owns the interface still compiles. The mismatch surfaces
434/// only where a caller uses the item on that target, or not at all when nothing uses it. If the
435/// interface requires `page_shift` to return `u32`, this re-export of an implementation that
436/// returns `u64` still compiles:
437///
438/// ```
439/// mod current {
440///     pub fn page_shift() -> u64 {
441///         12
442///     }
443/// }
444///
445/// pub use current::page_shift;
446/// # fn main() {}
447/// ```
448///
449/// `look_at!` checks each declaration against the selected module at the invocation site.
450///
451/// - An ordinary function is assigned to a function pointer with the declared signature, so its
452///   parameter types, return type, and ABI must match exactly.
453/// - A constant is assigned to the declared type.
454/// - A type placeholder with bounds asserts that the target type implements them.
455///
456/// Declaring the same interface with `look_at!` rejects that implementation with a type mismatch:
457///
458/// ```compile_fail,E0308
459/// # use look_at::look_at;
460/// mod current {
461///     pub fn page_shift() -> u64 {
462///         12
463///     }
464/// }
465///
466/// look_at! {
467///     @current:
468///     pub fn page_shift() -> u32;
469/// }
470/// # fn main() {}
471/// ```
472///
473/// As with a re-export, a declaration also fails to compile when the selected module lacks the
474/// item. The checks cover only the module selected for the current build, so each implementation
475/// is checked when building for its own target.
476///
477/// ## Visibility
478///
479/// A re-export cannot be more visible than its target, so `pub use` rejects a crate-private
480/// function:
481///
482/// ```compile_fail,E0364
483/// mod current {
484///     pub(crate) fn halt() {}
485/// }
486///
487/// pub use current::halt;
488/// # fn main() {}
489/// ```
490///
491/// Function wrappers and forwarding constants are new items, and their targets only need to be
492/// accessible from the invocation site. An implementation module can therefore keep its items
493/// crate-private while the declarations form the public interface:
494///
495/// ```
496/// # use look_at::look_at;
497/// mod current {
498///     pub(crate) fn halt() {}
499///     pub(crate) const PAGE_SIZE: usize = 4096;
500/// }
501///
502/// look_at! {
503///     @current:
504///     pub fn halt();
505///     pub const PAGE_SIZE: usize;
506/// }
507/// # fn main() {}
508/// ```
509///
510/// Type and `use` declarations expand to re-exports and follow the ordinary re-export rules.
511///
512/// ## Costs
513///
514/// - A function wrapper is a distinct function. Its address differs from the target's, and
515///   attributes on the target, such as `#[must_use]` or `#[track_caller]`, do not apply to the
516///   wrapper unless the declaration repeats them.
517/// - Each signature is written once more in the declaration and changes together with the
518///   implementations.
519/// - Some declarations are checked only partially. Type placeholders without bounds and `use`
520///   declarations only check that the item exists, and generic parameters on type placeholders
521///   are not checked. `const` functions and functions with `impl Trait` skip the function-pointer
522///   check and rely on the forwarding call, which permits coercions.
523/// - Some items cannot be declared, as listed in [Unsupported items](#unsupported-items).
524///
525/// An attribute that callers of the wrapper should observe is therefore written on the declaration
526/// as well as on the implementation:
527///
528/// ```
529/// # use look_at::look_at;
530/// mod current {
531///     #[must_use]
532///     pub fn read_counter() -> u64 {
533///         0
534///     }
535/// }
536///
537/// look_at! {
538///     @current:
539///     #[must_use]
540///     pub fn read_counter() -> u64;
541/// }
542/// # fn main() {}
543/// ```
544///
545/// A conditional re-export remains the simpler choice when callers need the target item itself,
546/// such as its function address or its attributes, when the items cannot be declared here, or when
547/// the interface needs no checks beyond those its callers already perform.
548///
549/// # Attributes and visibility
550///
551/// Outer attributes attached to an item, including doc comments, are copied to its generated
552/// wrapper, re-export, or constant. In particular, `#[cfg(...)]` can remove an item for
553/// architectures where it is unavailable. A bound check copies only the `cfg` and `cfg_attr`
554/// attributes of its declaration, so it is removed together with its re-export.
555///
556/// The visibility written in the invocation is used for the generated item and does not increase
557/// the visibility of the target definition itself. Function wrappers and constants may be more
558/// visible than their targets, while type and `use` re-exports may not, as described in
559/// [Visibility](#visibility).
560///
561/// # Documentation
562///
563/// Function wrappers and forwarding constants are new items, so their documentation consists only
564/// of the doc comments on the declaration. Type and `use` declarations expand to re-exports. When
565/// rustdoc inlines such a re-export, for example because the target module is private, it shows
566/// the doc comments on the declaration followed by those on the target item. That part of the page
567/// therefore depends on the target selected for the documentation build.
568///
569/// Doc comments on a declaration are shared by every implementation. They describe one interface
570/// regardless of the target, but they must hold for every architecture and cannot document the
571/// details of one implementation. A practical split is to describe the architecture-neutral
572/// contract on the declaration and to keep architecture-specific details on each implementation:
573///
574/// ```
575/// # use look_at::look_at;
576/// mod x86_64 {
577///     /// Returns 12 because x86-64 uses 4 KiB base pages.
578///     pub fn page_shift() -> u32 {
579///         12
580///     }
581/// }
582///
583/// use x86_64 as current;
584///
585/// look_at! {
586///     @current:
587///     /// Returns the base-2 logarithm of the base page size.
588///     pub fn page_shift() -> u32;
589/// }
590/// # fn main() {}
591/// ```
592///
593/// The documentation of the generated `page_shift` contains only the declaration's sentence, while
594/// the x86-64 detail stays with the implementation.
595///
596/// # Unsupported items
597///
598/// The current syntax supports functions, `type` placeholders, `const` placeholders, and `use`
599/// trees. Trait declarations, structs with inline fields, enum definitions, union definitions, and
600/// constants with values must remain in the selected implementation module and are not accepted as
601/// declarations here. Their definitions can still be forwarded with a `type`, `const`, or `use`
602/// declaration. Among functions, `async` functions, variadic functions, and functions with a
603/// `self` receiver are rejected, as described in [Functions](#functions).
604#[proc_macro]
605pub fn look_at(input: TokenStream) -> TokenStream {
606    let input = parse_macro_input!(input as LookAtInput);
607    expand(input)
608        .unwrap_or_else(Error::into_compile_error)
609        .into()
610}

677    fn expands_readable_parameter_names() {

679            "@crate::arch: fn tuple((left, right): (u32, u32)); \
680             fn borrow(ref value: String); \
