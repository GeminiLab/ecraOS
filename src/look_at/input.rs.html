11/// The input accepted by the [`look_at`](crate::look_at) macro.
12///
13/// The input contains one target module followed by one or more forwarding declarations.
14#[derive(Clone)]
15pub struct LookAtInput {
16    /// The path to the target module containing the forwarded items.
17    ///
18    /// Every declaration in the invocation resolves below this path.
19    pub target_path: Path,
20    /// The forwarding declarations accepted by `look_at`.
21    ///
22    /// The declarations are expanded in input order and the list is never empty.
23    pub items: Vec<LookAtItem>,
24}
25
26/// The parsing implementation for a complete invocation.
27///
28/// It reads the `@path:` header followed by forwarding declarations until the input ends.
29impl Parse for LookAtInput {
30    /// Parses the target path and every forwarding declaration.
31    ///
32    /// An invocation without declarations is rejected.
33    fn parse(input: ParseStream<'_>) -> Result<Self> {
34        input.parse::<Token![@]>()?;
35        let target_path = input.parse()?;
36        input.parse::<Token![:]>()?;
37
38        let mut items = Vec::new();
39        while !input.is_empty() {
40            items.push(input.parse()?);
41        }
42        if items.is_empty() {
43            return Err(input.error("`look_at` requires at least one item"));
44        }
45        Ok(Self { target_path, items })
46    }
47}

49/// An item supported by `look_at`.
50///
51/// Functions receive forwarding bodies while other declarations become re-exports.
52#[derive(Clone)]
53pub struct LookAtItem {
54    /// The attributes attached to the forwarded item.
55    ///
56    /// They are copied to the generated wrapper, re-export, or constant.
57    pub attrs: Vec<Attribute>,
58    /// The visibility of the forwarded item.
59    ///
60    /// It applies to the generated item and does not affect the target definition.
61    pub vis: Visibility,
62    /// The kind of the forwarded item.
63    ///
64    /// The kind determines how the declaration is expanded.
65    pub kind: LookAtItemKind,
66}
67
68/// The parsing implementation for one forwarding declaration.
69///
70/// It reads outer attributes and visibility before the declaration itself.
71impl Parse for LookAtItem {
72    /// Parses the attributes, visibility, and kind of one declaration.
73    ///
74    /// The declaration kind consumes the terminating semicolon.
75    fn parse(input: ParseStream<'_>) -> Result<Self> {
76        let attrs = Attribute::parse_outer(input)?;
77        let vis = input.parse()?;
78        let kind = input.parse()?;
79
80        Ok(Self { attrs, vis, kind })
81    }
82}

84/// The kinds of items supported by `look_at`.
85///
86/// Each kind describes a declaration without an implementation, which is resolved in the target
87/// module.
88#[derive(Clone)]
89pub enum LookAtItemKind {
90    /// A function declaration.
91    ///
92    /// Forwarded functions contain a signature followed by a semicolon and no body.
93    Function(Signature),
94    /// A const declaration.
95    ///
96    /// Forwarded consts contain a name and a type followed by a semicolon and no value.
97    Const {
98        /// The name of the forwarded constant.
99        ///
100        /// The same name is used for the generated constant and the target constant.
101        name: Ident,
102        /// The type declared for the forwarded constant.
103        ///
104        /// The target constant must have a value of this type.
105        declared_type: Type,
106    },
107    /// A use statement.
108    ///
109    /// The use tree is resolved relative to the selected target module.
110    Use(UseTree),
111    /// A type declaration.
112    ///
113    /// Forwarded types contain a name and optional trait bounds followed by a semicolon and no
114    /// definition. It can point to a type alias, struct, enum, or union. Generic parameters are
115    /// accepted for re-exports, but cannot be combined with trait bounds.
116    Type {
117        /// The name of the forwarded type.
118        ///
119        /// The same name is used for the re-export and the target type.
120        name: Ident,
121        /// The trait bounds that the target type must satisfy.
122        ///
123        /// Bounds are checked at compile time and are absent for generic placeholders.
124        bounds: Option<TypeParamBounds>,
125    },
126}
127
128/// The parsing implementation for a declaration kind.
129///
130/// It selects the kind from the leading keywords without consuming them during lookahead.
131impl Parse for LookAtItemKind {
132    /// Parses one function, const, use, or type declaration.
133    ///
134    /// Function bodies, const values, and bounds on generic type placeholders are rejected.
135    fn parse(input: ParseStream<'_>) -> Result<Self> {
136        let lookahead = input.fork();
137
138        if lookahead.peek(Token![fn])
139            || lookahead.peek(Token![async])
140            || lookahead.peek(Token![unsafe])
141            || lookahead.peek(Token![extern])
142            || (lookahead.peek(Token![const]) && lookahead.peek2(Token![fn]))
143        {
144            // It's Self::Function
145            let result = input.parse().map(Self::Function);
146            if !input.peek(Token![;]) {
147                return Err(input.error("functions in `look_at` must not have a body"));
148            }
149            input.parse::<Token![;]>()?;
150            result
151        } else if lookahead.peek(Token![const]) {
152            // It's Self::Const
153            input.parse::<Token![const]>()?;
154            let name = input.parse()?;
155            input.parse::<Token![:]>()?;
156            let declared_type = input.parse()?;
157            input.parse::<Token![;]>()?;
158            Ok(Self::Const {
159                name,
160                declared_type,
161            })
162        } else if lookahead.peek(Token![use]) {
163            // It's Self::Use
164            input.parse::<Token![use]>()?;
165            let tree = input.parse()?;
166            input.parse::<Token![;]>()?;
167            Ok(Self::Use(tree))
168        } else if lookahead.peek(Token![type]) {
169            // It's Self::Type
170            input.parse::<Token![type]>()?;
171            let name = input.parse()?;
172            let has_generics = input.peek(Token![<]);
173            if has_generics {
174                let _: syn::Generics = input.parse()?;
175            }
176            let bounds = if input.peek(Token![:]) {
177                if has_generics {
178                    return Err(input.error("trait bounds are not supported on generic items"));
179                }
180                input.parse::<Token![:]>()?;
181                Some(TypeParamBounds::parse_separated_nonempty(input)?)
182            } else {
183                None
184            };
185            input.parse::<Token![;]>()?;
186            Ok(Self::Type { name, bounds })
187        } else {
188            Err(input.error("expected a function, type, const, or use"))
189        }
190    }
191}
