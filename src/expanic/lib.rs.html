1//! Kernel panics and bug reports.
2//!
3//! Use [`kpanic!`] to report a fatal kernel error, or [`bug!`] to report an unexpected internal
4//! error. Both call [`terminate`] with a formatted message and the module path, file, line, and
5//! column of the macro call. [`UnwrapOrBug`] adds methods to [`Option`] and [`Result`] that report
6//! missing values and errors as bugs.
7//!
8//! By default, [`terminate`] calls [`core::panic!`] with the supplied information. An application
9//! can replace this function by applying `#[expanic::terminate_handler]` to a function with the
10//! same signature. A kernel can use its replacement to log the error and shut down.
11//!
12//! This crate is `no_std` and requires nightly Rust for `extern_item_impls`. It does not define a
13//! panic handler or provide platform shutdown code.
14
15#![no_std]
16#![feature(extern_item_impls)]
17
18use core::{fmt, panic::Location};
19
20mod bug;

22pub use bug::UnwrapOrBug;

24/// Reports a fatal kernel error and terminates execution.
25///
26/// Accepts a format string and optional arguments. Passes the formatted message and the macro
27/// call's module path, file, line, and column to [`terminate`] with [`TerminateReason::KPanic`].
28#[macro_export]
29macro_rules! kpanic {
30    ($($arg:tt)*) => {{
31        ::core::hint::cold_path();
32        $crate::terminate(
33            $crate::TerminateReason::KPanic,
34            $crate::TerminateLocation::from_raw_parts(
35                Some(module_path!()),
36                file!(),
37                line!(),
38                column!(),
39            ),
40            format_args!($($arg)*),
41        )
42    }};
43}

45/// The reason for terminating execution.
46///
47/// Identifies a Rust panic, a fatal kernel error, or an unexpected internal error.
48#[derive(Debug, Clone, Copy, PartialEq, Eq)]
49pub enum TerminateReason {
50    /// A Rust panic received by the consumer's panic handler.
51    ///
52    /// Used when a panic handler forwards a Rust panic to the application's termination function.
53    Panic,
54    /// A fatal kernel error reported with [`kpanic!`].
55    ///
56    /// Indicates that execution cannot continue because of a kernel error.
57    KPanic,
58    /// An unexpected internal error.
59    ///
60    /// Used by [`bug!`] and the methods in [`UnwrapOrBug`].
61    Bug,
62}
63
64impl TerminateReason {
65    /// Returns the short label for the termination reason.
66    ///
67    /// Returns `"Panic"`, `"Kernel panic"`, or `"Bug"` for display in an error message.
68    pub fn name(self) -> &'static str {
69        match self {
70            TerminateReason::Panic => "Panic",
71            TerminateReason::KPanic => "Kernel panic",
72            TerminateReason::Bug => "Bug",
73        }
74    }
75
76    /// Returns an explanation of the termination reason.
77    ///
78    /// Provides additional text for an error report, including a request to report internal bugs.
79    pub fn desc(self) -> &'static str {
80        match self {
81            TerminateReason::Panic => {
82                "\"Panic\" means some code called the `panic!` macro, usually from an external crate."
83            }
84            TerminateReason::KPanic => {
85                "\"Kernel panic\" means some code called the `kpanic!` macro, usually from the kernel itself or its components."
86            }
87            TerminateReason::Bug => {
88                "\"Bug\" means an unexpected internal error happened in the kernel or its components. Please report this issue to the maintainers."
89            }
90        }
91    }
92}

94/// The source location of a panic or bug.
95///
96/// Stores a file, line, column, and optional module path for an error report. Macros can supply
97/// all four fields. Rust's [`Location`] provides the file, line, and column, but no module path.
98pub struct TerminateLocation<'a> {
99    /// The caller's module path, when available.
100    ///
101    /// Contains `None` when constructed from a Rust panic location.
102    module_path: Option<&'a str>,
103    /// The caller's source file.
104    ///
105    /// Contains `<unknown>` when constructed without a Rust panic location.
106    file: &'a str,
107    /// The caller's source line.
108    ///
109    /// Contains zero when constructed without a Rust panic location.
110    line: u32,
111    /// The caller's source column.
112    ///
113    /// Contains zero when constructed without a Rust panic location.
114    column: u32,
115}
116
117impl<'a> TerminateLocation<'a> {
118    /// Creates a source location from an optional Rust panic location.
119    ///
120    /// Copies the file, line, and column from [`Location`] and leaves the module path as `None`.
121    /// If no location is supplied, uses `<unknown>` for the file and zero for the line and column.
122    pub fn from_panic_location(location: Option<&Location<'a>>) -> Self {
123        if let Some(location) = location {
124            Self::from_raw_parts(None, location.file(), location.line(), location.column())
125        } else {
126            Self::from_raw_parts(None, "<unknown>", 0, 0)
127        }
128    }
129
130    /// Creates a source location from its individual fields.
131    ///
132    /// Stores the supplied module path, file, line, and column without changing them.
133    pub fn from_raw_parts(
134        module_path: Option<&'a str>,
135        file: &'a str,
136        line: u32,
137        column: u32,
138    ) -> Self {
139        Self {
140            module_path,
141            file,
142            line,
143            column,
144        }
145    }
146
147    /// Returns the caller's module path when available.
148    ///
149    /// Returns `None` when constructed from a Rust panic location.
150    pub fn module_path(&self) -> Option<&'a str> {
151        self.module_path
152    }
153
154    /// Returns the caller's source file.
155    ///
156    /// Returns `<unknown>` when constructed without a Rust panic location.
157    pub fn file(&self) -> &'a str {
158        self.file
159    }
160
161    /// Returns the caller's source line.
162    ///
163    /// Returns zero when constructed without a Rust panic location.
164    pub fn line(&self) -> u32 {
165        self.line
166    }
167
168    /// Returns the caller's source column.
169    ///
170    /// Returns zero when constructed without a Rust panic location.
171    pub fn column(&self) -> u32 {
172        self.column
173    }
174}

176/// Terminates execution with a reason, source location, and message.
177///
178/// The default implementation calls [`core::panic!`] with the supplied information in its message.
179/// Whether this unwinds or aborts depends on the application's panic strategy and panic handler.
180///
181/// To change this behavior, apply `#[expanic::terminate_handler]` to a function with the same
182/// signature. A kernel can forward ordinary Rust panics to that function from its panic handler.
183/// Calling the default implementation from a panic handler would cause another panic.
184#[eii(terminate_handler)]
185pub fn terminate(
186    reason: TerminateReason,
187    location: TerminateLocation<'_>,
188    args: fmt::Arguments<'_>,
189) -> ! {
190    terminate_fallback(reason, location, args)
191}
192
193/// Keeps the default termination symbol available to optimized standalone consumers.
194///
195/// EII normally emits the default implementation only when the defining crate references its
196/// public symbol. This reference preserves that weak symbol for consumers that use the fallback.
197#[used]
198static DEFAULT_TERMINATE: fn(
199    TerminateReason,
200    TerminateLocation<'static>,
201    fmt::Arguments<'static>,
202) -> ! = terminate;
203
204/// Panics with the supplied reason, source location, and message.
205///
206/// Includes the original location in the panic message so it remains available to a panic handler.
207#[inline(never)]
208fn terminate_fallback(
209    reason: TerminateReason,
210    location: TerminateLocation<'_>,
211    args: fmt::Arguments<'_>,
212) -> ! {
213    core::panic!(
214        "[{}:{}] {} at {}:{}:{}:\n{}\n\n{}",
215        location.module_path().unwrap_or("<unknown module>"),
216        location.line(),
217        reason.name(),
218        location.file(),
219        location.line(),
220        location.column(),
221        args,
222        reason.desc(),
223    )
224}
