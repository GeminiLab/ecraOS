14/// Tracks one CPU's active IRQ-safe guards and owns the original interrupt state.
15///
16/// Only the outermost guard owns a saved-state token. Inner guards return their temporary token
17/// immediately, while the saved outer token stays in this per-CPU value until the last guard exits.
18pub(crate) struct NestedIrqState<T> {
19    /// The number of IRQ-safe guards currently active on this CPU.
20    depth: usize,
21    /// The token that restores the interrupt state from before the outermost guard entered.
22    outer: Option<T>,
23}
24
25/// Empty per-CPU IRQ nesting state.
26impl<T> NestedIrqState<T> {
27    /// Creates an inactive IRQ nesting state.
28    pub const fn new() -> Self {
29        Self {
30            depth: 0,
31            outer: None,
32        }
33    }
34
35    /// Adds one guard and retains its token only for the outermost guard.
36    ///
37    /// Returns an inner token for the caller to drop after releasing the per-CPU state borrow.
38    /// On error, returns the token so its caller can restore execution state after ending that
39    /// borrow.
40    pub fn enter(&mut self, token: T) -> Result<Option<T>, (NestedIrqError, T)> {
41        if self.depth == 0 {
42            if self.outer.is_some() {
43                return Err((NestedIrqError::UnexpectedOuterToken, token));
44            }
45            self.outer = Some(token);
46            self.depth = 1;
47            return Ok(None);
48        }
49
50        if self.outer.is_none() {
51            return Err((NestedIrqError::MissingOuterToken, token));
52        }
53        let Some(depth) = self.depth.checked_add(1) else {
54            return Err((NestedIrqError::DepthOverflow, token));
55        };
56        self.depth = depth;
57        Ok(Some(token))
58    }
59
60    /// Removes one guard and returns the saved token only after the final guard exits.
61    pub fn exit(&mut self) -> Result<Option<T>, NestedIrqError> {
62        if self.depth == 0 {
63            return Err(NestedIrqError::UnbalancedExit);
64        }
65        if self.outer.is_none() {
66            return Err(NestedIrqError::MissingOuterToken);
67        }
68
69        self.depth -= 1;
70        if self.depth == 0 {
71            Ok(self.outer.take())
72        } else {
73            Ok(None)
74        }
75    }
76
77    /// Returns the current number of active guards.
78    #[cfg(test)]
79    pub fn depth(&self) -> usize {
80        self.depth
81    }
82}

84/// The BSP and each AP keep independent IRQ nesting state in their per-CPU area.
85#[expercpu::def_percpu]
86pub(super) static IRQ_NESTING: NestedIrqState<KernelIrqSave> = NestedIrqState::new();

88/// The execution policy shared by all IRQ-safe spin and reader-writer locks.
89///
90/// This type is public only because it appears in the public lock aliases.
91#[doc(hidden)]
92pub struct NoIrqGuardPolicy;
93
94/// Per-CPU interrupt nesting for IRQ-safe lock acquisitions.
95impl BaseGuard for NoIrqGuardPolicy {
96    /// The saved outer token lives in per-CPU storage rather than in each lock guard.
97    type State = ();
98
99    /// Enters the shared IRQ-safe critical section on the current CPU.
100    fn acquire() -> Self::State {
101        // Prevent migration while resolving and mutating the current CPU's nesting state.
102        NoPreempt::acquire();
103        let irq_state = KernelIrqSave::new();
104        // SAFETY: The temporary token disables local IRQs before the per-CPU state is accessed.
105        // Preemption is already disabled, so this access remains on the same CPU.
106        let entered = unsafe { IRQ_NESTING.current_ref_mut_raw().enter(irq_state) };
107        match entered {
108            Ok(inner_state) => drop(inner_state),
109            Err((error, irq_state)) => {
110                // Restore the temporary token after the mutable per-CPU borrow has ended.
111                drop(irq_state);
112                NoPreempt::release(());
113                panic!("failed to enter the per-CPU IRQ guard: {error:?}");
114            }
115        }
116    }
117
118    /// Leaves the shared IRQ-safe critical section after releasing its data lock.
119    fn release(_state: Self::State) {
120        // SAFETY: A live outer token keeps local IRQs disabled. The matching NoPreempt guard
121        // keeps this state on the CPU that entered it.
122        let exited = unsafe { IRQ_NESTING.current_ref_mut_raw().exit() };
123        match exited {
124            Ok(outer_state) => drop(outer_state),
125            Err(error) => {
126                NoPreempt::release(());
127                panic!("failed to leave the per-CPU IRQ guard: {error:?}");
128            }
129        }
130        NoPreempt::release(());
131    }
132}

134/// A nestable IRQ-save guard shared with the IRQ-safe lock aliases in this crate.
135///
136/// Guards on one CPU may be dropped in any order. IRQs remain disabled until the last related
137/// guard exits. The guard cannot move between CPUs because its state is stored per CPU.
138#[must_use = "dropping the guard releases IRQ and preemption protection"]
139pub struct NoPreemptIrqSave {
140    /// Prevents moving this CPU-local guard to another thread or CPU.
141    _local: PhantomData<*mut ()>,
142}
143
144/// Construction of a shared IRQ-save guard.
145impl NoPreemptIrqSave {
146    /// Enters the current CPU's shared IRQ-safe critical section.
147    pub fn new() -> Self {
148        NoIrqGuardPolicy::acquire();
149        Self {
150            _local: PhantomData,
151        }
152    }
153}
154
155/// Default construction of a nested IRQ-save guard.
156impl Default for NoPreemptIrqSave {
157    /// Enters the current CPU's shared IRQ-safe critical section.
158    fn default() -> Self {
159        Self::new()
160    }
161}
162
163/// Releases one share of the current CPU's IRQ-safe critical section.
164impl Drop for NoPreemptIrqSave {
165    /// Restores IRQ state only when this is the final active guard on the CPU.
166    fn drop(&mut self) {
167        NoIrqGuardPolicy::release(());
168    }
169}

171/// Returns the current CPU's IRQ nesting depth for integration tests.
172#[cfg(test)]
173pub(crate) fn current_depth() -> usize {
174    // SAFETY: Test access runs on the current host or kernel CPU. The test owns this per-CPU
175    // state and does not concurrently mutate it from another execution context.
176    unsafe { IRQ_NESTING.current_ref_raw().depth() }
177}
