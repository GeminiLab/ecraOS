1//! Kernel per-CPU storage.
2//!
3//! [`expercpu`] provides macros for defining per-CPU variables and generating accessors. This
4//! module allocates and initializes their storage and supplies methods for looking up per-CPU area
5//! base addresses for remote accesses.
6//!
7//! The BSP uses the linker-reserved early slot as its per-CPU area, keeping its contents when the
8//! kernel relocates. Once the allocators are ready, the BSP allocates separate vmalloc areas for
9//! APs. Each AP copies the initial values from the linked `.percpu` section into its own area
10//! during startup. All areas remain allocated for the kernel's lifetime.
11
12use alloc::{boxed::Box, vec::Vec};
13
14use expt::pte::MappingFlags;
15use lazyinit::LazyInit;
16use memory_addr::{AddrRangeBounds, VirtAddr};
17
18use crate::{mem, mp::LogicalCpuId};
19
20/// Per-CPU area base addresses indexed by logical CPU ID.
21static PERCPU_AREAS: LazyInit<Box<[VirtAddr]>> = LazyInit::new();
22
23/// The kernel implementation of the `expercpu` area lookup interface.
24struct PerCpuAreaIfImpl;
25
26/// Area lookup for `expercpu` remote accessors.
27#[crate_interface::impl_interface]
28impl expercpu::PerCPUAreaIf for PerCpuAreaIfImpl {
29    /// Returns the area base for a logical CPU index.
30    ///
31    /// # Panics
32    ///
33    /// Panics before area allocation or if `cpu_id` is outside the area table.
34    fn percpu_area_base_for(cpu_id: usize) -> VirtAddr {
35        area_base_for(LogicalCpuId::new(cpu_id))
36    }
37}
38
39/// Returns a CPU's per-CPU area base without checking whether its contents are initialized.
40///
41/// # Panics
42///
43/// Panics before area allocation or if `cpu_id` is outside the area table.
44fn area_base_for(cpu_id: LogicalCpuId) -> VirtAddr {
45    PERCPU_AREAS
46        .get()
47        .expect("percpu areas must be initialized")
48        .get(cpu_id.into_repr())
49        .copied()
50        .expect("invalid cpu id")
51}
52
53/// Returns the per-CPU section size in bytes, including trailing alignment padding.
54fn section_size() -> usize {
55    mem::sections::percpu_aligned().size()
56}
57
58/// Copies initial per-CPU values into the early slot and installs the BSP's per-CPU register.
59///
60/// # Panics
61///
62/// Panics if the linker-reserved slot is too small or incorrectly aligned.
63///
64/// # Safety
65///
66/// The BSP must call this once after BSS clearing, with interrupts disabled and before using
67/// per-CPU values. No other CPU may access the early slot.
68pub(crate) unsafe fn init_bsp_early() {
69    // SAFETY: The linker reserves a writable, aligned slot separate from the initial image.
70    // The caller guarantees exclusive access during early BSP initialization.
71    unsafe { expercpu::init_in_early_slot() };
72}
73
74/// Updates the BSP's per-CPU register after relocation without resetting stored values.
75///
76/// # Safety
77///
78/// The BSP must call this after early-slot initialization and relocation, with interrupts disabled
79/// and no per-CPU references live. The relocated slot must map the same writable storage.
80pub(crate) unsafe fn init_bsp_after_reloc() {
81    // SAFETY: The caller guarantees that the relocated early slot is valid for this BSP.
82    unsafe { expercpu::write_percpu_reg(expercpu::early_slot_start()) };
83}
84
85/// Allocates per-CPU areas for `ap_count` APs (with logical CPU IDs from 1 to `ap_count`) and
86/// records all per-CPU areas' base addresses.
87///
88/// The BSP calls this once before starting APs, after heap, page allocator, and vmalloc
89/// initialization. Each AP receives a read-write mapping with a guard page at each end. The BSP
90/// keeps its early slot. APs initialize their areas later through [`init_ap`].
91///
92/// # Panics
93///
94/// Panics if an area cannot be allocated or the area table has already been initialized.
95pub(crate) fn alloc_percpu_area_for_ap(ap_count: usize) {
96    let mut percpu_area_bases = Vec::with_capacity(ap_count + 1);
97
98    // BSP uses the early slot for percpu.
99    percpu_area_bases.push(expercpu::early_slot_start());
100
101    // APs use the percpu area allocated by the BSP.
102    let percpu_page_count = mem::vmm::page_count_for_bytes(section_size());
103
104    for _ in 0..ap_count {
105        let percpu_area = mem::allocs::vmalloc::alloc_range_and_map_alloc(
106            percpu_page_count,
107            1,
108            MappingFlags::READ | MappingFlags::WRITE,
109        )
110        .expect("failed to allocate percpu area");
111
112        percpu_area_bases.push(percpu_area.start);
113    }
114
115    PERCPU_AREAS.init_once(percpu_area_bases.into_boxed_slice());
116}

118/// Copies initial per-CPU values into the current AP's area and installs its per-CPU register.
119///
120/// # Panics
121///
122/// Panics before area allocation or if `cpu_id` is outside the area table.
123///
124/// # Safety
125///
126/// `cpu_id` must identify the calling AP. It must call this once, with interrupts disabled and
127/// before using per-CPU values. No other CPU may access its area during initialization.
128pub(crate) unsafe fn init_ap(cpu_id: LogicalCpuId) {
129    let percpu_area_base = area_base_for(cpu_id);
130
131    // SAFETY: The allocated area is writable, page-aligned, large enough for the initial image,
132    // and separate from it. The caller guarantees exclusive initialization on the matching AP.
133    unsafe {
134        expercpu::init(percpu_area_base);
135    }
136}
