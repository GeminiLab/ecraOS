101/// Allocates owned shared pages initialized with one value per page.
102///
103/// The guarded reservation is visible to readers only after normal mapping completion.
104pub fn allocate_shared(values: &[u64]) -> Result<VirtAddrRange, VMAllocError> {
105    let range = vmalloc::alloc_range_and_map_alloc(
106        values.len(),
107        1,
108        MappingFlags::READ.union(MappingFlags::WRITE),
109    )?;
110    for (index, &value) in values.iter().enumerate() {
111        // SAFETY: The new owned reservation is mapped and has not yet been published to readers.
112        unsafe {
113            (range.start + index * page_size())
114                .as_mut_ptr_of::<u64>()
115                .write_volatile(value);
116        }
117    }
118    Ok(range)
119}

134/// Verifies that every selected CPU acknowledged exactly one generation.
135///
136/// The observations read the production common TLB slots directly.
137pub fn assert_tlb_completed(targets: &[LogicalCpuId], generation: u64) {
138    assert_ne!(generation, 0);
139    for &cpu in targets {
140        assert_eq!(ipi::tlb_generation(cpu), generation, "TLB ack cpu={cpu}");
141    }
142}

144/// Invalidates all local translations and completes every ready remote CPU.
145///
146/// This uses the same transaction, request payload and common acknowledgement as mapping changes.
147pub fn flush_all() -> Result<u64, VMAllocError> {
148    let mut transaction = vmm::MappingTransaction::begin()?;
149    LocalTlbFlusher::flush(None);
150    transaction.append_remote_flushes(TlbFlushes::Full);
151    transaction.perform_remote_flushes()?;
152    Ok(tlb::published_generation())
153}

155/// Verifies nonwaiting transaction acquisition with local interrupts disabled.
156///
157/// Call from an IPI-ready CPU with interrupts enabled and no active mapping transaction.
158///
159/// # Panics
160///
161/// Panics if waiting acquisition accepts disabled interrupts, nonwaiting acquisition fails on an
162/// available VMM, contention is not reported, or ending a transaction prevents the next attempt.
163pub fn assert_try_mapping_transaction() {
164    assert!(exarch::trap::local_enabled());
165    {
166        let _interrupt_guard = exspin::NoPreemptIrqSave::new();
167        assert!(matches!(
168            vmm::MappingTransaction::begin(),
169            Err(TlbShootdownError::Ipi(IpiError::InterruptsDisabled))
170        ));
171        let transaction = vmm::MappingTransaction::try_begin()
172            .unwrap()
173            .expect("nonwaiting transaction rejected an available VMM");
174        assert!(!exarch::trap::local_enabled());
175        assert!(vmm::MappingTransaction::try_begin().unwrap().is_none());
176        drop(transaction);
177        drop(
178            vmm::MappingTransaction::try_begin()
179                .unwrap()
180                .expect("ending a transaction left the VMM busy"),
181        );
182        assert!(!exarch::trap::local_enabled());
183    }
184    assert!(exarch::trap::local_enabled());
185}

187/// Verifies request retention after a timeout and clearing after successful retry.
188///
189/// `addresses` must contain two distinct page-aligned addresses. `target` must be a ready remote
190/// CPU whose next TLB acknowledgement can be suppressed. No mappings are changed.
191///
192/// # Panics
193///
194/// Panics if the inputs are invalid, the injected timeout is not reported, the retry loses either
195/// request, or a subsequent flush publishes another request without new work.
196pub fn assert_tlb_transaction_retry(addresses: [VirtAddr; 2], target: LogicalCpuId) {
197    let mut transaction = vmm::MappingTransaction::begin().unwrap();
198    assert!(transaction.targets().contains(&target));
199    assert_ne!(addresses[0], addresses[1]);
200    for address in addresses {
201        assert!(address.is_aligned(vmm::page_size()));
202        LocalTlbFlusher::flush(Some(address));
203        transaction.append_remote_flushes(TlbFlushes::None + TlbFlush::Page(address));
204    }
205    suppress_tlb_ack(target);
206    let Err(TlbShootdownError::Ipi(IpiError::Timeout(details))) =
207        transaction.perform_remote_flushes()
208    else {
209        panic!("TLB transaction did not report the injected timeout");
210    };
211    assert_eq!(details.logical_cpu, target);
212    clear_tlb_injection(target);
213
214    transaction.perform_remote_flushes().unwrap();
215    let generation = tlb::published_generation();
216    assert!(generation > details.generation.unwrap());
217    let Some(TlbFlushes::Pages(pages)) = tlb::published_requests(generation) else {
218        panic!("TLB retry did not publish the accumulated page requests");
219    };
220    assert_eq!(pages.as_slice(), addresses.as_slice());
221    assert_tlb_completed(transaction.targets(), generation);
222
223    transaction.perform_remote_flushes().unwrap();
224    assert_eq!(tlb::published_generation(), generation);
225}

227/// Collects the currently live shared backing frame addresses.
228///
229/// Pending and quarantined mappings cannot be reported as live resources.
230pub fn backing_frames(range: VirtAddrRange) -> Vec<PhysAddr> {
231    range
232        .iter(page_size())
233        .expect("aligned shared range")
234        .map(|address| vmalloc::virt_to_phys(address).expect("live shared backing"))
235        .collect()
236}

245/// Extracts common IPI diagnostics from a mapping completion failure.
246///
247/// Non-completion failures remain distinguishable and cannot masquerade as an injected timeout.
248pub fn completion_ipi_error(error: &VMAllocError) -> Option<IpiError> {
249    match error {
250        VMAllocError::Completion(vmalloc::CompletionDiagnostic(TlbShootdownError::Ipi(error))) => {
251            Some(*error)
252        }
253        _ => None,
254    }
255}

285/// Attempts exact physical reuse and releases the probe allocation on success.
286///
287/// A quarantined frame must reject this allocator operation.
288pub fn claim_released_frame(address: PhysAddr) -> bool {
289    if palloc::alloc_frames_at(address, 1).is_err() {
290        return false;
291    }
292    palloc::dealloc_frames_at(address, 1).unwrap();
293    true
294}
