1//! Debug Triggers Extension (EID #0x44425452 "DBTR")
2//!
3//! The RISC-V Sdtrig extension allows machine-mode software to directly
4//! configure debug triggers which in-turn allows native (or hosted) debugging in machine-mode
5//! without any external debugger. Unfortunately, the debug triggers are only accessible to
6//! machine-mode.
7//!
8//! The SBI debug trigger extension defines a SBI based abstraction to provide native debugging
9//! for supervisor-mode software such that it is:
10//! 1. Suitable for the rich operating systems and hypervisors running in supervisor-mode.
11//! 2. Allows Guest (VS-mode) and Hypervisor (HS-mode) to share debug triggers on a hart.
12//!
13//! Each hart on a RISC-V platform has a fixed number of debug triggers which is referred
14//! to as `trig_max` in this SBI extension. Each debug trigger is assigned a logical index
15//! called `trig_idx` by the SBI implementation where `-1 < trig_idx < trig_max`.
16
17use crate::binary::{sbi_call_1, sbi_call_2, sbi_call_3};
18use sbi_spec::binary::{SbiRet, SharedPtr, TriggerMask};

21/// Get the number of debug triggers on the calling hart which can support the trigger
22/// configuration specified by `trig_tdata1` parameter.
23///
24/// This function always returns `SbiRet::success()` in `SbiRet.error`. It will return `trig_max`
25/// in `SbiRet.value` when `trig_tdata1 == 0` otherwise it will return the number of matching
26/// debug triggers in `SbiRet.value`.
27#[doc(alias = "sbi_debug_num_triggers")]
28#[inline]
29pub fn debug_num_triggers(trig_tdata1: usize) -> usize {
30    sbi_call_1(EID_DBTR, NUM_TRIGGERS, trig_tdata1).value
31}

33/// Set and enable the shared memory for debug trigger configuration on the calling hart.
34///
35/// If `shmem` is not all-ones bitwise then `shmem` specifies the bits of the shared memory physical base address.
36/// The `shmem` MUST be `(XLEN / 8)` bytes aligned and the size of shared
37/// memory is assumed to be `trig_max * (XLEN / 2)` bytes.
38///
39/// If `shmem` is all-ones bitwise then shared memory for debug trigger configuration is disabled
40///
41/// The `flags` parameter is reserved for future use and MUST be zero.
42///
43/// # Return value
44///
45/// | Error code                    | Description
46/// |:------------------------------|:---------------------------------
47/// | `SbiRet::success()`           | Shared memory was set or cleared successfully.
48/// | `SbiRet::invalid_param()`     | The `flags` parameter is not zero or the `shmem` parameter is not `(XLEN / 8)` bytes aligned.
49/// | `SbiRet::invalid_address()`   | The shared memory pointed to by the `shmem` parameter does not satisfy the requirements.
50/// | `SbiRet::failed()`            | The request failed for unspecified or unknown other reasons.
51#[doc(alias = "sbi_debug_set_shmem")]
52#[inline]
53pub fn debug_set_shmem(shmem: SharedPtr<u8>, flags: usize) -> SbiRet {
54    sbi_call_3(
55        EID_DBTR,
56        SET_SHMEM,
57        shmem.phys_addr_lo(),
58        shmem.phys_addr_hi(),
59        flags,
60    )
61}

63/// Read the debug trigger state and configuration into shared memory for a range of
64/// debug triggers specified by the `trig_idx_base` and `trig_count` parameters on the calling hart.
65///
66/// For each debug trigger with index `trig_idx_base + i` where `-1 < i < trig_count`, the
67/// debug trigger state and configuration consisting of four XLEN-bit words are written in
68/// little-endian format at `offset = i * (XLEN / 2)` of the shared memory as follows:
69///
70/// ```text
71/// word[0] = `trig_state` written by the SBI implementation
72/// word[1] = `trig_tdata1` written by the SBI implementation
73/// word[2] = `trig_tdata2` written by the SBI implementation
74/// word[3] = `trig_tdata3` written by the SBI implementation
75/// ```
76/// # Return value
77///
78/// | Error code              | Description
79/// |:------------------------|:---------------------------------
80/// | `SbiRet::success()`     | State and configuration of triggers read successfully.
81/// | `SbiRet::no_shmem()`    | Shared memory for debug triggers is disabled.
82/// | `SbiRet::bad_range()`   | Either `trig_idx_base >= trig_max` or `trig_idx_base + trig_count >= trig_max`.
83#[doc(alias = "sbi_debug_read_triggers")]
84#[inline]
85pub fn debug_read_triggers(trig_idx_base: usize, trig_count: usize) -> SbiRet {
86    sbi_call_2(EID_DBTR, READ_TRIGGERS, trig_idx_base, trig_count)
87}

89/// Install debug triggers based on an array of trigger configurations in the shared memory
90/// of the calling hart. The `trig_idx` assigned to each installed trigger configuration is
91/// written back in the shared memory.
92///
93/// The `trig_count` parameter represents the number of trigger configuration entries in
94/// the shared memory at offset `0x0`.
95///
96/// The i'th trigger configuration at `offset = i * (XLEN / 2)` in the shared memory
97/// consists of four consecutive XLEN-bit words in little-endian format which are
98/// organized as follows:
99///
100/// ```text
101/// word[0] = `trig_idx` written back by the SBI implementation
102/// word[1] = `trig_tdata1` read by the SBI implementation
103/// word[2] = `trig_tdata2` read by the SBI implementation
104/// word[3] = `trig_tdata3` read by the SBI implementation
105/// ```
106///
107/// Upon success, `SbiRet.value` is set to zero. Upon failure, `SbiRet.value` is set to the
108/// array index of the failing trigger configuration.
109///
110/// # Return value
111///
112/// | Error code                  | Description
113/// |:----------------------------|:---------------------------------
114/// | `SbiRet::success()`         | Triggers installed successfully.
115/// | `SbiRet::no_shmem()`        | Shared memory for debug triggers is disabled.
116/// | `SbiRet::bad_range()`       | `trig_count >= trig_max`.
117/// | `SbiRet::invalid_param()`   | One of the trigger configuration words `trig_tdata1`, `trig_tdata2`, or `trig_tdata3` has an invalid value.
118/// | `SbiRet::failed()`          | Failed to assign `trig_idx` or HW debug trigger for one of the trigger configurations.
119/// | `SbiRet::not_supported()`   | One of the trigger configuration can't be programmed due to unimplemented optional bits in `tdata1`, `tdata2`, or `tdata3` CSRs.
120#[doc(alias = "sbi_debug_install_triggers")]
121#[inline]
122pub fn debug_install_triggers(trig_count: usize) -> SbiRet {
123    sbi_call_1(EID_DBTR, INSTALL_TRIGGERS, trig_count)
124}

126/// Update already installed debug triggers based on a trigger configuration array in the
127/// shared memory of the calling hart.
128///
129/// The `trig_count` parameter represents the number of trigger configuration entries in
130/// the shared memory at offset `0x0`.
131///
132/// The i'th trigger configuration at `offset = i * (XLEN / 2)` in the shared memory
133/// consists of four consecutive XLEN-bit words in little-endian format as follows:
134///
135/// ```text
136/// word[0] = `trig_idx` read by the SBI implementation
137/// word[1] = `trig_tdata1` read by the SBI implementation
138/// word[2] = `trig_tdata2` read by the SBI implementation
139/// word[3] = `trig_tdata3` read by the SBI implementation
140/// ```
141/// The SBI implementation MUST consider trigger configurations in the increasing order of
142/// the array index and starting with array index `0`. To install a debug trigger for the
143/// trigger configuration at array index `i` in the shared memory, the SBI implementation
144/// MUST do the following:
145///
146/// - Map an unused HW debug trigger which matches the trigger configuration to an
147///   an unused `trig_idx`.
148/// - Save a copy of the `trig_tdata1.vs`, `trig_tdata1.vu`, `trig_tdata1.s`, and
149///   `trig_tdata.u` bits in `trig_state`.
150/// - Update the `tdata1`, `tdata2`, and `tdata3` CSRs of the HW debug trigger.
151/// - Write `trig_idx` at `offset = i * (XLEN / 2)` in the shared memory.
152///
153/// Additionally for each trigger configuration chain in the shared memory, the SBI
154/// implementation MUST assign contiguous `trig_idx` values and contiguous HW debug
155/// triggers when installing the trigger configuration chain.
156///
157/// The last trigger configuration in the shared memory MUST not have `trig_tdata1.chain == 1`
158/// for `trig_tdata1.type = 2 or 6` to prevent incomplete trigger configuration chain
159/// in the shared memory.
160///
161/// The `SbiRet.value` is set to zero upon success or if shared memory is disabled whereas
162/// `SbiRet.value` is set to the array index `i` of the failing trigger configuration upon
163/// other failures.
164///
165/// # Return value
166///
167/// | Error code                  | Description
168/// |:----------------------------|:---------------------------------
169/// | `SbiRet::success()`         | Triggers updated successfully.
170/// | `SbiRet::no_shmem()`        | Shared memory for debug triggers is disabled.
171/// | `SbiRet::bad_range()`       | `trig_count >= trig_max`.
172/// | `SbiRet::invalid_param()`   | One of the trigger configuration in the shared memory has an invalid of `trig_idx` (i.e. `trig_idx >= trig_max`), `trig_tdata1`, `trig_tdata2`, or `trig_tdata3`.
173/// | `SbiRet::failed()`          | One of the trigger configurations has valid `trig_idx` but the corresponding debug trigger is not mapped to any HW debug trigger.
174/// | `SbiRet::not_supported()`   | One of the trigger configuration can't be programmed due to unimplemented optional bits in `tdata1`, `tdata2`, or `tdata3` CSRs.
175#[doc(alias = "sbi_debug_update_triggers")]
176#[inline]
177pub fn debug_update_triggers(trig_count: usize) -> SbiRet {
178    sbi_call_1(EID_DBTR, UPDATE_TRIGGERS, trig_count)
179}

181/// Uninstall a set of debug triggers specified by the `triggers` mask parameter on the calling hart.
182///
183/// The `triggers` specifies which triggers are to be uninstalled.
184/// Each bit in the mask corresponds to a specific trigger, allowing for batch operations
185/// on multiple triggers simultaneously.
186///
187/// For each debug trigger in the specified set of debug triggers, the SBI implementation MUST:
188/// 1. Clear the `tdata1`, `tdata2`, and `tdata3` CSRs of the mapped HW debug trigger.
189/// 2. Clear the `trig_state` of the debug trigger.
190/// 3. Unmap and free the HW debug trigger and corresponding `trig_idx` for re-use in
191///    the future trigger installations.
192///
193/// # Return value
194///
195/// | Error code                  | Description
196/// |:----------------------------|:---------------------------------
197/// | `SbiRet::success()`         | Triggers uninstalled successfully.
198/// | `SbiRet::invalid_param()`   | One of the debug triggers with index `trig_idx` in the specified set of debug triggers either not mapped to any HW debug trigger OR has `trig_idx >= trig_max`.
199#[doc(alias = "sbi_debug_uninstall_triggers")]
200#[inline]
201pub fn debug_uninstall_triggers(triggers: TriggerMask) -> SbiRet {
202    let (trig_idx_mask, trig_idx_base) = triggers.into_inner();
203    sbi_call_2(EID_DBTR, UNINSTALL_TRIGGERS, trig_idx_base, trig_idx_mask)
204}

206/// Enable a set of debug triggers specified by the `triggers` mask parameter on the calling hart.
207///
208/// The `triggers` specifies which triggers are to be enabled.
209/// Each bit in the mask corresponds to a specific trigger, allowing for batch operations
210/// on multiple triggers simultaneously.
211///
212/// To enable a debug trigger in the specified set of debug triggers, the SBI implementation
213/// MUST restore the `vs`, `vu`, `s`, and `u` bits of the mapped HW debug trigger from their
214/// saved copy in `trig_state`.
215///
216/// # Return value
217///
218/// | Error code                  | Description
219/// |:----------------------------|:---------------------------------
220/// | `SbiRet::success()`         | Triggers enabled successfully.
221/// | `SbiRet::invalid_param()`   | One of the debug triggers with index `trig_idx` in the specified set of debug triggers either not mapped to any HW debug trigger OR has `trig_idx >= trig_max`.
222#[doc(alias = "sbi_debug_enable_triggers")]
223#[inline]
224pub fn debug_enable_triggers(triggers: TriggerMask) -> SbiRet {
225    let (trig_idx_mask, trig_idx_base) = triggers.into_inner();
226    sbi_call_2(EID_DBTR, ENABLE_TRIGGERS, trig_idx_base, trig_idx_mask)
227}

238/// # Return value

240/// | Error code                  | Description

244#[doc(alias = "sbi_debug_disable_triggers")]
245#[inline]
246pub fn debug_disable_triggers(triggers: TriggerMask) -> SbiRet {
247    let (trig_idx_mask, trig_idx_base) = triggers.into_inner();
248    sbi_call_2(EID_DBTR, DISABLE_TRIGGERS, trig_idx_base, trig_idx_mask)
