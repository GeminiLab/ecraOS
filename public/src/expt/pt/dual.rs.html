1//! Dual-root multi-level page table implementation.
2//!
3//! This module provides an architecture-independent Rust abstraction over two independent
4//! page table hierarchies. AArch64 uses one hierarchy for the lower address range and
5//! a second hierarchy for the upper address range.
6

21/// The physical root addresses of two page table hierarchies.
22///
23/// Copying these addresses does not change either hierarchy's table-frame lifetime.
24#[derive(Clone, Copy, Debug, Eq, PartialEq)]
25pub struct DualPageTableRoot {
26    /// The root used for lower virtual addresses.
27    ///
28    /// This address alone does not keep the lower page table hierarchy alive.
29    pub lower: PhysAddr,
30    /// The root used for upper virtual addresses.
31    ///
32    /// This address alone does not keep the upper page table hierarchy alive.
33    pub upper: PhysAddr,
34}
35
36/// A dual-root multi-level page table for lower and upper virtual-address ranges.
37///
38/// The type-level constraints prevent combining symmetric roots or roots with
39/// different virtual-address types. The hierarchies may use different geometries
40/// and entry types. Both must form valid trees with disjoint table-frame
41/// allocations. Both roots use `AL` for their table frames, and `F` handles cleanup for both. See
42/// [table-frame lifetime](crate#table-frame-lifetime) and the
43/// [safety requirements for page table updates](crate#safely-updating-page-tables).
44///
45/// Each `map` and `unmap` operation is routed to the root selected by its
46/// starting virtual address. An individual range must remain within one
47/// canonical side and cannot cross the gap between the lower and upper ranges.
48pub struct DualPageTable<A, ML, PTEL, MU, PTEU, AL, F>
49where
50    A: VirtAddr,
51    ML: PageTableMeta<VirtAddr = A, Coverage = LowerCoverage<A>>,
52    MU: PageTableMeta<VirtAddr = A, Coverage = UpperCoverage<A>>,
53    PTEL: GenericPTE,
54    PTEU: GenericPTE,
55    AL: PageAllocator,
56    F: TlbFlusher<A>,
57    [(); ML::LEVELS]: Sized,
58    [(); ML::LEVELS - 1]: Sized,
59    [(); MU::LEVELS]: Sized,
60    [(); MU::LEVELS - 1]: Sized,
61{
62    /// The lower-address multi-level page table.
63    ///
64    /// It manages the lower page table hierarchy through the allocator selected by `AL`.
65    lower: PageTable<ML, PTEL, AL, F>,
66    /// The upper-address multi-level page table.
67    ///
68    /// It manages the upper page table hierarchy through the allocator selected by `AL`.
69    upper: PageTable<MU, PTEU, AL, F>,
70}
71
72/// A mutating cursor over lower and upper page table hierarchies.
73///
74/// The cursor batches TLB invalidations for both roots using `F`. Its mutable borrow excludes
75/// other Rust mutations through this value. See [`PageTableCursor`] for synchronization
76/// requirements.
77pub struct DualPageTableCursor<'a, A, ML, PTEL, MU, PTEU, AL, F>
78where
79    A: VirtAddr,
80    ML: PageTableMeta<VirtAddr = A, Coverage = LowerCoverage<A>>,
81    MU: PageTableMeta<VirtAddr = A, Coverage = UpperCoverage<A>>,
82    PTEL: GenericPTE,
83    PTEU: GenericPTE,
84    AL: PageAllocator,
85    F: TlbFlusher<A>,
86    [(); ML::LEVELS]: Sized,
87    [(); ML::LEVELS - 1]: Sized,
88    [(); MU::LEVELS]: Sized,
89    [(); MU::LEVELS - 1]: Sized,
90{
91    /// The cursor for the lower-address root.
92    ///
93    /// Its destruction processes pending cleanup for the lower page table hierarchy.
94    lower: PageTableCursor<'a, ML, PTEL, AL, F>,
95    /// The cursor for the upper-address root.
96    ///
97    /// Its destruction processes pending cleanup for the upper page table hierarchy.
98    upper: PageTableCursor<'a, MU, PTEU, AL, F>,
99}

101/// Dual-root construction, storage access, and table-frame lifecycle.
102///
103/// Both roots use the same static allocator type.
104impl<A, ML, PTEL, MU, PTEU, AL, F> DualPageTable<A, ML, PTEL, MU, PTEU, AL, F>
105where
106    A: VirtAddr,
107    ML: PageTableMeta<VirtAddr = A, Coverage = LowerCoverage<A>>,
108    MU: PageTableMeta<VirtAddr = A, Coverage = UpperCoverage<A>>,
109    PTEL: GenericPTE,
110    PTEU: GenericPTE,
111    AL: PageAllocator,
112    F: TlbFlusher<A>,
113    [(); ML::LEVELS]: Sized,
114    [(); ML::LEVELS - 1]: Sized,
115    [(); MU::LEVELS]: Sized,
116    [(); MU::LEVELS - 1]: Sized,
117{
118    /// Constructs a dual-root multi-level page table from existing lower and upper hierarchies.
119    ///
120    /// This function does not inspect either root. The caller must ensure that
121    /// both physical addresses identify valid page table roots for their
122    /// respective metadata types.
123    ///
124    /// # Safety
125    ///
126    /// Each root must satisfy [`PageTable::new_at`]. The two page table hierarchies
127    /// must not share table frames.
128    pub unsafe fn new_at(root: DualPageTableRoot) -> Self {
129        Self {
130            lower: unsafe { PageTable::new_at(root.lower) },
131            upper: unsafe { PageTable::new_at(root.upper) },
132        }
133    }
134
135    /// Constructs a dual-root multi-level page table from separate root addresses.
136    ///
137    /// This is a convenience wrapper around [`Self::new_at`].
138    ///
139    /// # Safety
140    ///
141    /// Same as [`Self::new_at`].
142    pub unsafe fn new_at_roots(lower: PhysAddr, upper: PhysAddr) -> Self {
143        unsafe { Self::new_at(DualPageTableRoot { lower, upper }) }
144    }
145
146    /// Allocates and initializes both root page tables through `AL`.
147    ///
148    /// The roots have the size required by the highest level in `ML` and `MU`
149    /// and are zero-initialized before the value is returned.
150    /// If the upper-root allocation fails, the lower root is deallocated.
151    pub fn new_alloc() -> PagingResult<A, Self>
152    where
153        [(); ML::LEVELS - 1]: Sized,
154        [(); MU::LEVELS - 1]: Sized,
155    {
156        let lower = PageTable::<ML, PTEL, AL, F>::new_alloc()?;
157        let upper = PageTable::<MU, PTEU, AL, F>::new_alloc()?;
158        Ok(Self { lower, upper })
159    }
160
161    /// Consumes the value without deallocating table frames and returns both roots.
162    ///
163    /// Another dual-root multi-level page table can use the roots if its allocator can
164    /// access and deallocate every table frame. Otherwise, the caller must arrange
165    /// their deallocation. The allocator type may differ from `AL` if those
166    /// requirements hold.
167    pub fn leak(self) -> DualPageTableRoot {
168        let Self { lower, upper } = self;
169        DualPageTableRoot {
170            lower: lower.leak(),
171            upper: upper.leak(),
172        }
173    }
174
175    /// Returns both physical root addresses without changing table-frame lifetimes.
176    ///
177    /// This value must remain alive while its table frames are in use.
178    pub const fn root(&self) -> DualPageTableRoot {
179        DualPageTableRoot {
180            lower: self.lower.root(),
181            upper: self.upper.root(),
182        }
183    }
184
185    /// Creates a cursor for batched mutations and cleanup of both roots.
186    ///
187    /// See [`DualPageTableCursor`] for synchronization requirements.
188    pub fn cursor(&mut self) -> DualPageTableCursor<'_, A, ML, PTEL, MU, PTEU, AL, F> {
189        let Self { lower, upper } = self;
190        DualPageTableCursor {
191            lower: lower.cursor(),
192            upper: upper.cursor(),
193        }
194    }
195
196    /// Returns the lower-address multi-level page table.
197    ///
198    /// The returned reference borrows the single-root value that manages the lower hierarchy.
199    pub const fn lower(&self) -> &PageTable<ML, PTEL, AL, F> {
200        &self.lower
201    }
202
203    /// Returns the upper-address multi-level page table.
204    ///
205    /// The returned reference borrows the single-root value that manages the upper hierarchy.
206    pub const fn upper(&self) -> &PageTable<MU, PTEU, AL, F> {
207        &self.upper
208    }
209
210    /// Returns a mutable borrow of the lower-address multi-level page table.
211    ///
212    /// This borrow does not synchronize hardware page walks or remote TLBs.
213    pub fn lower_mut(&mut self) -> &mut PageTable<ML, PTEL, AL, F> {
214        &mut self.lower
215    }
216
217    /// Returns a mutable borrow of the upper-address multi-level page table.
218    ///
219    /// This borrow does not synchronize hardware page walks or remote TLBs.
220    pub fn upper_mut(&mut self) -> &mut PageTable<MU, PTEU, AL, F> {
221        &mut self.upper
222    }
223
224    /// Replaces the lower-address multi-level page table and returns the previous value.
225    ///
226    /// The returned value contains the previous page table hierarchy.
227    pub fn swap_lower(&mut self, other: PageTable<ML, PTEL, AL, F>) -> PageTable<ML, PTEL, AL, F> {
228        core::mem::replace(&mut self.lower, other)
229    }
230
231    /// Replaces the upper-address multi-level page table and returns the previous value.
232    ///
233    /// The returned value contains the previous page table hierarchy.
234    pub fn swap_upper(&mut self, other: PageTable<MU, PTEU, AL, F>) -> PageTable<MU, PTEU, AL, F> {
235        core::mem::replace(&mut self.upper, other)
236    }
237}

239/// The common interface for dual-root multi-level page tables.
240///
241/// Operations delegate to the concrete methods with the same lifecycle contract.
242impl<A, ML, PTEL, MU, PTEU, AL, F> PageTableLike for DualPageTable<A, ML, PTEL, MU, PTEU, AL, F>
243where
244    A: VirtAddr,
245    ML: PageTableMeta<VirtAddr = A, Coverage = LowerCoverage<A>>,
246    MU: PageTableMeta<VirtAddr = A, Coverage = UpperCoverage<A>>,
247    PTEL: GenericPTE,
248    PTEU: GenericPTE,
249    AL: PageAllocator,
250    F: TlbFlusher<A>,
251    [(); ML::LEVELS]: Sized,
252    [(); MU::LEVELS]: Sized,
253    [(); ML::LEVELS - 1]: Sized,
254    [(); MU::LEVELS - 1]: Sized,
255{
256    /// The virtual-address type shared by both roots.
257    ///
258    /// Both metadata types require the same semantic address representation.
259    type VirtAddr = A;
260    /// The pair of physical root addresses.
261    ///
262    /// Copying these addresses does not change either hierarchy's table-frame lifetime.
263    type Root = DualPageTableRoot;
264    /// The allocator for both hierarchies' table frames.
265    ///
266    /// Table access, allocation, and deallocation use `AL`.
267    type Allocator = AL;
268    /// The TLB flusher for both hierarchies.
269    ///
270    /// TLB invalidations are performed using `F`.
271    type TlbFlusher = F;
272    /// The cursor borrowing both multi-level page tables.
273    ///
274    /// It uses the metadata and entry types of the corresponding roots.
275    type Cursor<'a>
276        = DualPageTableCursor<'a, A, ML, PTEL, MU, PTEU, AL, F>
277    where
278        Self: 'a;
279
280    /// Constructs a dual-root multi-level page table from existing lower and upper hierarchies.
281    ///
282    /// # Safety
283    ///
284    /// The caller must meet the safety requirements of [`DualPageTable::new_at`].
285    unsafe fn new_at(root: Self::Root) -> Self {
286        unsafe { Self::new_at(root) }
287    }
288
289    /// Allocates and initializes both root page tables.
290    ///
291    /// This delegates to [`DualPageTable::new_alloc`].
292    fn new_alloc() -> PagingResult<Self::VirtAddr, Self> {
293        Self::new_alloc()
294    }
295
296    /// Consumes the value while retaining both hierarchies' table frames.
297    ///
298    /// This delegates to [`DualPageTable::leak`] and returns the roots for transfer.
299    fn leak(self) -> Self::Root {
300        self.leak()
301    }
302
303    /// Returns both root addresses without changing table-frame lifetimes.
304    ///
305    /// This delegates to [`DualPageTable::root`].
306    fn root(&self) -> Self::Root {
307        self.root()
308    }
309
310    /// Creates a cursor borrowing both roots.
311    ///
312    /// This delegates to [`DualPageTable::cursor`].
313    fn cursor(&mut self) -> Self::Cursor<'_> {
314        self.cursor()
315    }
316}

318/// The side of dual page table hierarchies that a virtual address belongs to.
319///
320/// Selection is based on the coverage policies of the two roots.
321#[derive(Debug, Clone, Copy, PartialEq, Eq)]
322enum DualPageTableSide {
323    /// The lower-address page table hierarchy.
324    ///
325    /// Operations use the lower root for this side.
326    Lower,
327    /// The upper-address page table hierarchy.
328    ///
329    /// Operations use the upper root for this side.
330    Upper,
331}
332
333/// Selection of a root from a virtual address.
334///
335/// Canonical coverage determines which root receives each operation.
336impl<A, ML, PTEL, MU, PTEU, AL, F> DualPageTableCursor<'_, A, ML, PTEL, MU, PTEU, AL, F>
337where
338    A: VirtAddr,
339    ML: PageTableMeta<VirtAddr = A, Coverage = LowerCoverage<A>>,
340    MU: PageTableMeta<VirtAddr = A, Coverage = UpperCoverage<A>>,
341    PTEL: GenericPTE,
342    PTEU: GenericPTE,
343    AL: PageAllocator,
344    F: TlbFlusher<A>,
345    [(); ML::LEVELS]: Sized,
346    [(); ML::LEVELS - 1]: Sized,
347    [(); MU::LEVELS]: Sized,
348    [(); MU::LEVELS - 1]: Sized,
349{
350    /// Selects the lower or upper page table hierarchy for a canonical virtual address.
351    ///
352    /// Addresses accepted by neither root, or by both roots, are rejected.
353    fn side(&self, vaddr: A) -> PagingResult<A, DualPageTableSide> {
354        let lower =
355            <ML::Coverage as PageTableCoverage>::validate_and_truncate_vaddr(vaddr, ML::VA_BITS)
356                .is_ok();
357        let upper =
358            <MU::Coverage as PageTableCoverage>::validate_and_truncate_vaddr(vaddr, MU::VA_BITS)
359                .is_ok();
360
361        match (lower, upper) {
362            (true, false) => Ok(DualPageTableSide::Lower),
363            (false, true) => Ok(DualPageTableSide::Upper),
364            _ => Err(PagingError::NonCanonical { vaddr }),
365        }
366    }
367
368    /// Maps a virtual range through the root selected by its starting address.
369    ///
370    /// Same as [`PageTableCursor::map`] for the selected hierarchy. See [`DualPageTable`] for
371    /// range coverage requirements. The starting address must select exactly one root, even
372    /// for a zero-size request.
373    pub fn map(
374        &mut self,
375        vaddr: A,
376        paddr: PhysAddr,
377        size: usize,
378        flags: MappingFlags,
379    ) -> PagingResult<A> {
380        match self.side(vaddr)? {
381            DualPageTableSide::Lower => self.lower.map(vaddr, paddr, size, flags),
382            DualPageTableSide::Upper => self.upper.map(vaddr, paddr, size, flags),
383        }
384    }
385
386    /// Removes mappings through the root selected by the starting address.
387    ///
388    /// Same as [`PageTableCursor::unmap`] for the selected hierarchy. See [`DualPageTable`] for
389    /// range coverage requirements. The starting address must select exactly one root, even
390    /// for a zero-size request.
391    pub fn unmap(&mut self, vaddr: A, size: usize) -> PagingResult<A> {
392        match self.side(vaddr)? {
393            DualPageTableSide::Lower => self.lower.unmap(vaddr, size),
394            DualPageTableSide::Upper => self.upper.unmap(vaddr, size),
395        }
396    }
397
398    /// Transfers both roots' pending cleanup into one batch without processing it.
399    ///
400    /// # Safety
401    ///
402    /// The caller must meet the cleanup requirements of [`PageTableCursorLike::take_cleanup`].
403    pub unsafe fn take_cleanup(&mut self) -> PendingCleanup<A> {
404        // SAFETY: the user promised it.
405        let mut cleanup = unsafe { self.lower.take_cleanup() };
406        // SAFETY: they use the same TLB flusher.
407        unsafe { cleanup.append(self.upper.take_cleanup()) };
408        cleanup
409    }
410}

412/// The common mutation interface for a dual-root cursor.
413///
414/// Each mapping operation selects a root from its starting virtual address.
415impl<A, ML, PTEL, MU, PTEU, AL, F> PageTableCursorLike
416    for DualPageTableCursor<'_, A, ML, PTEL, MU, PTEU, AL, F>
417where
418    A: VirtAddr,
419    ML: PageTableMeta<VirtAddr = A, Coverage = LowerCoverage<A>>,
420    MU: PageTableMeta<VirtAddr = A, Coverage = UpperCoverage<A>>,
421    PTEL: GenericPTE,
422    PTEU: GenericPTE,
423    AL: PageAllocator,
424    F: TlbFlusher<A>,
425    [(); ML::LEVELS]: Sized,
426    [(); ML::LEVELS - 1]: Sized,
427    [(); MU::LEVELS]: Sized,
428    [(); MU::LEVELS - 1]: Sized,
429{
430    /// The virtual-address type shared by both roots.
431    ///
432    /// It matches the address type of the borrowed dual-root multi-level page table.
433    type VirtAddr = A;
434    /// The allocator for both borrowed hierarchies' table frames.
435    ///
436    /// Intermediate page table frame allocations use `AL`.
437    type Allocator = AL;
438    /// The multi-level page table type being mutated by this cursor.
439    ///
440    /// It matches the type of the borrowed dual-root multi-level page table.
441    type PageTable = DualPageTable<A, ML, PTEL, MU, PTEU, AL, F>;
442
443    /// Maps a virtual range through the root selected by its starting address.
444    ///
445    /// Mapped data frames remain caller-managed.
446    fn map(
447        &mut self,
448        vaddr: A,
449        paddr: PhysAddr,
450        size: usize,
451        flags: MappingFlags,
452    ) -> PagingResult<A> {
453        self.map(vaddr, paddr, size, flags)
454    }
455
456    /// Removes mappings through the root selected by the starting address.
457    ///
458    /// This does not deallocate the mapped data frames.
459    fn unmap(&mut self, vaddr: A, size: usize) -> PagingResult<A> {
460        self.unmap(vaddr, size)
461    }
462
463    /// Processes both cursors' accumulated cleanup.
464    ///
465    /// See [`PendingCleanup`] for the selected strategy's synchronization requirements.
466    fn clean_up(&mut self) {
467        // SAFETY: the user promised it.
468        unsafe { self.take_cleanup() }.clean_up();
469    }
470
471    /// Transfers both roots' pending cleanup without processing it.
472    ///
473    /// # Safety
474    ///
475    /// The caller must meet the safety requirements of [`DualPageTableCursor::take_cleanup`].
476    unsafe fn take_cleanup(&mut self) -> PendingCleanup<A> {
477        // SAFETY: the user promised it.
478        unsafe { self.take_cleanup() }
479    }
