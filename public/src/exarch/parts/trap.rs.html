21/// The architecture-neutral accessors for a saved trap frame.
22///
23/// Kernel handlers use this trait rather than layout-specific fields so their control flow remains
24/// independent of x86 and RISC-V frame representations.
25pub trait TrapFrameAccess {
26    /// Returns the saved instruction pointer.
27    ///
28    /// The returned address identifies the interrupted instruction or its architecture-adjusted
29    /// continuation point.
30    fn instruction_pointer(&self) -> VirtAddr;
31
32    /// Replaces the saved instruction pointer.
33    ///
34    /// The new address becomes the continuation target when the architecture returns from the
35    /// trap.
36    fn set_instruction_pointer(&mut self, instruction_pointer: VirtAddr);
37
38    /// Returns whether the trap interrupted user mode.
39    ///
40    /// This describes the saved privilege state rather than the kernel's current execution mode.
41    fn is_user(&self) -> bool;
42}

66#[def_interface(gen_caller)]
67pub trait KernelTrapIf {
68    /// Dispatches one decoded trap to the kernel's handlers.
69    ///
70    /// When a trap happens, the architecture-specific code will decode the trap and call this
71    /// function to dispatch it to the kernel, which is usually expected to use [`HandlerSlots`]
72    /// to find the appropriate handler for the trap.
73    ///
74    /// The kernel may also choose to handle it in a different way, but it must return a
75    /// [`TrapDisposition`] to indicate whether the trap was handled or not.
76    fn handle(frame: &mut TrapFrame, trap: Trap) -> TrapDisposition {
77        (_, _) = (frame, trap);
78        TrapDisposition::Unhandled
79    }
80
81    /// Reaches the common scheduling safe point after completing an interrupt source.
82    ///
83    /// Architecture code invokes this only after local-source clearing or controller EOI.
84    fn on_interrupt_exit(frame: &mut TrapFrame) {
85        _ = frame;
86    }
87}

89/// Semantic trap contract tests.
90///
91/// These tests describe the typed boundary before its architecture decoders are implemented.
92#[cfg(test)]
93mod tests {
94    use super::{ArchIrq, Exception, Interrupt, LocalInterrupt, RawTrap, Trap, TrapDisposition};
95
96    /// Verifies that semantic trap variants preserve their distinct domains.
97    #[test]
98    fn semantic_trap_preserves_distinct_domains() {
99        let raw = RawTrap(0xfeed);
100        let global = ArchIrq::new(7);
101
102        assert!(matches!(
103            Trap::Exception(Exception::Unknown(raw)),
104            Trap::Exception(Exception::Unknown(observed)) if observed == raw
105        ));
106        assert!(matches!(
107            Trap::Interrupt(Interrupt::Local(LocalInterrupt::Timer)),
108            Trap::Interrupt(Interrupt::Local(LocalInterrupt::Timer))
109        ));
110        assert!(matches!(
111            Trap::Interrupt(Interrupt::Global(global)),
112            Trap::Interrupt(Interrupt::Global(observed)) if observed == global
113        ));
114        assert!(matches!(
115            Trap::Interrupt(Interrupt::Unknown(raw)),
116            Trap::Interrupt(Interrupt::Unknown(observed)) if observed == raw
117        ));
118    }
119
120    /// Verifies that trap disposition combination preserves an unhandled result.
121    #[test]
122    fn external_dispatch_preserves_unhandled_result_until_a_handler_runs() {
123        assert_eq!(
124            TrapDisposition::Unhandled + TrapDisposition::Unhandled,
125            TrapDisposition::Unhandled
126        );
127        assert_eq!(
128            TrapDisposition::Unhandled + TrapDisposition::Handled,
129            TrapDisposition::Unhandled
130        );
131        assert_eq!(
132            TrapDisposition::Handled + TrapDisposition::Handled,
133            TrapDisposition::Handled
134        );
135    }
