10/// Attribute-like macro that implements the traits of the `riscv-types` crate for a given enum.
11///
12/// As these traits are unsafe, the macro must be called with the `unsafe` keyword followed by the trait name.
13/// In this way, we warn callers that they must comply with the requirements of the trait.
14///
15/// The trait name must be one of `ExceptionNumber`, `CoreInterruptNumber`, `ExternalInterruptNumber`,
16/// `PriorityNumber`, or `HartIdNumber`.
17///
18/// # Note
19///
20/// Crates using this macro must depend on the `riscv` crate, as the generated code references it.
21///
22/// If the `rt` feature is enabled, the generated code may also include the necessary runtime support
23/// for interrupt and exception handling. Thus, the calling crate must also depend on the `riscv-rt` crate.
24///
25/// # Safety
26///
27/// The struct to be implemented must comply with the requirements of the specified trait.
28///
29/// # Example
30///
31/// ```rust,ignore,no_run
32/// use riscv::*;
33///
34/// #[repr(usize)]
35/// #[pac_enum(unsafe ExceptionNumber)]
36/// #[derive(Clone, Copy, Debug, Eq, PartialEq)]
37/// enum Exception {
38///     E1 = 1,
39///     E3 = 3,
40/// }
41///
42/// assert_eq!(Exception::E1.number(), 1);
43/// assert_eq!(Exception::E3.number(), 3);
44///
45/// assert_eq!(Exception::from_number(1), Ok(Exception::E1));
46/// assert_eq!(Exception::from_number(2), Err(2));
47/// assert_eq!(Exception::from_number(3), Ok(Exception::E3));
48///
49/// assert_eq!(Exception::MAX_EXCEPTION_NUMBER, 3);
50///```
51#[proc_macro_attribute]
52pub fn pac_enum(attr: TokenStream, item: TokenStream) -> TokenStream {
53    let input = parse_macro_input!(item as DeriveInput);
54    let pac_enum = riscv::PacEnumItem::new(&input);
55
56    let attr = parse_macro_input!(attr as riscv::PacTrait);
57
58    let trait_impl = pac_enum.impl_trait(&attr);
59    quote! {
60        #input
61        #(#trait_impl)*
62    }
63    .into()
64}

66/// Attribute to mark which function will be called before jumping to the entry point.
67///
68/// You must enable the `post-init` feature in the `riscv-rt` crate to use this macro.
69///
70/// In contrast with `__pre_init`, this function is called after the static variables
71/// are initialized, so it is safe to access them. It is also safe to run Rust code.
72///
73/// The function must have the signature of `[unsafe] fn([usize])`, where the argument
74/// corresponds to the hart ID of the current hart. This is useful for multi-hart systems
75/// to perform hart-specific initialization.
76///
77/// # IMPORTANT
78///
79/// This attribute can appear at most *once* in the dependency graph.
80///
81/// # Examples
82///
83/// ```
84/// #[riscv_macros::post_init]
85/// unsafe fn before_main(hart_id: usize) {
86///     // do something here
87/// }
88/// ```
89#[cfg(feature = "riscv-rt")]
90#[proc_macro_attribute]
91pub fn post_init(args: TokenStream, input: TokenStream) -> TokenStream {
92    riscv_rt::Fn::post_init(args, input)
93}

95/// Attribute to mark which function will set interrupts before jumping to the entry point.
96///
97/// The `riscv-rt` crates provides a default implementation that works for most cases.
98/// If you want to provide your own implementation, you must enable the `custom-setup-interrupts`
99/// feature in the `riscv-rt` crate and use this macro on your function.
100/// The `riscv-rt` crate re-exports this macro if the `custom-setup-interrupts` feature is enabled,
101/// so you can use it as `riscv_rt::setup_interrupts` without depending on `riscv-macros` directly.
102///
103/// The function must have the signature of `[unsafe] fn([usize])`, where the argument
104/// corresponds to the hart ID of the current hart. This is useful for multi-hart systems
105/// to perform hart-specific interrupt setup.
106///
107/// # IMPORTANT
108///
109/// This attribute can appear at most *once* in the dependency graph.
110///
111/// # Examples
112///
113/// ```
114/// #[riscv_macros::setup_interrupts]
115/// unsafe fn setup_interrupts(hart_id: usize) {
116///     // do something here
117/// }
118/// ```
119#[cfg(feature = "riscv-rt")]
120#[proc_macro_attribute]
121pub fn setup_interrupts(args: TokenStream, input: TokenStream) -> TokenStream {
122    riscv_rt::Fn::setup_interrupts(args, input)
123}

125/// Attribute to declare the entry point of the program
126///
127/// The specified function will be called by the reset handler *after* RAM has been initialized.
128/// If present, the FPU will also be enabled before the function is called.
129///
130/// # Signature
131///
132/// ## Regular Usage
133///
134/// The type of the specified function must be `[unsafe] fn([usize[, usize[, usize]]]) -> !` (never ending function).
135/// The optional arguments correspond to the values passed in registers `a0`, `a1`, and `a2`.
136/// The first argument holds the hart ID of the current hart, which is useful for multi-hart systems.
137/// The other two arguments are currently unused and reserved for future use.
138///
139/// ## With U-Boot
140///
141/// This runtime supports being booted by U-Boot. In this case, the entry point function
142/// must have the signature `[unsafe] fn([c_int[, *const *const c_char]]) -> !`, where the first argument
143/// corresponds to the `argc` parameter and the second argument corresponds to the `argv` parameter passed by U-Boot.
144///
145/// Remember to enable the `u-boot` feature in the `riscv-rt` crate to use this functionality.
146///
147/// # IMPORTANT
148///
149/// This attribute can appear at most *once* in the dependency graph.
150///
151/// The entry point will be called by the reset handler. The program can't reference to the entry
152/// point, much less invoke it.
153///
154/// # Examples
155///
156/// ``` no_run
157/// #[riscv_macros::entry]
158/// fn main() -> ! {
159///     loop {
160///         /* .. */
161///     }
162/// }
163/// ```
164#[cfg(feature = "riscv-rt")]
165#[proc_macro_attribute]
166pub fn entry(args: TokenStream, input: TokenStream) -> TokenStream {
167    riscv_rt::Fn::entry(args, input)
168}

170/// Attribute to declare an exception handler.
171///
172/// The function must have the signature `[unsafe] fn([&[mut] riscv_rt::TrapFrame]) [-> !]`.
173///
174/// The argument of the macro must be a path to a variant of an enum that implements the `riscv_rt::ExceptionNumber` trait.
175///
176/// # Example
177///
178/// ``` ignore,no_run
179/// #[riscv_rt::exception(riscv::interrupt::Exception::LoadMisaligned)]
180/// fn load_misaligned(trap_frame: &mut riscv_rt::TrapFrame) -> ! {
181///     loop{};
182/// }
183/// ```
184#[cfg(feature = "riscv-rt")]
185#[proc_macro_attribute]
186pub fn exception(args: TokenStream, input: TokenStream) -> TokenStream {
187    riscv_rt::Fn::trap(args, input, riscv_rt::TrapType::Exception)
188}

190/// Attribute to declare a core interrupt handler.
191///
192/// The function must have the signature `[unsafe] fn() [-> !]`.
193///
194/// The argument of the macro must be a path to a variant of an enum that implements the `riscv_rt::CoreInterruptNumber` trait.
195///
196/// If the `v-trap` feature is enabled, this macro generates the corresponding interrupt trap handler in assembly.
197/// This feature relies on the `RISCV_RT_BASE_ISA` environment variable being set to one of
198/// `rv32i`, `rv32e`, `rv64i`, or `rv64e`. Otherwise, this will **panic**.
199///
200/// # Example
201///
202/// ``` ignore,no_run
203/// #[riscv_rt::core_interrupt(riscv::interrupt::Interrupt::SupervisorSoft)]
204/// fn supervisor_soft() -> ! {
205///     loop{};
206/// }
207/// ```
208#[cfg(feature = "riscv-rt")]
209#[proc_macro_attribute]
210pub fn core_interrupt(args: TokenStream, input: TokenStream) -> TokenStream {
211    riscv_rt::Fn::trap(args, input, riscv_rt::TrapType::CoreInterrupt)
212}

234/// Temporary patch macro to deal with LLVM bug.
235///
236/// # Note
237///
238/// This macro is intended to be used internally by the `riscv-rt` crate. Do not use it directly in your code.
239#[cfg(feature = "riscv-rt")]
240#[proc_macro]
241pub fn rvrt_llvm_arch_patch(_input: TokenStream) -> TokenStream {
242    let q = if let Ok(arch) = std::env::var("RISCV_RT_LLVM_ARCH_PATCH") {
243        let patch = format!(".attribute arch,\"{arch}\"");
244        quote! { core::arch::global_asm!{#patch} }
245    } else {
246        quote!(compile_error!("RISCV_RT_LLVM_ARCH_PATCH is not set"))
247    };
248    q.into()
249}

251/// Generates assembly code required for the default handling of traps.
252///
253/// The main routine generated is `_default_start_trap`. If no `_start_trap` function
254/// is defined, the linker will use this function as the default trap entry point.
255///
256/// If the `pre-default-start-trap` feature is enabled, the generated code will also
257/// include a call to a user-defined function `_pre_default_start_trap` at the beginning
258/// of the `_default_start_trap` routine.
259///
260/// If the `rt-v-trap` feature is enabled, the macro will also include the assembly code
261/// for the `_start_DefaultInterrupt_trap` and `_continue_interrupt_trap` routines, which
262/// are required for handling core interrupts in vectored trap mode.
263///
264/// # Note
265///
266/// This macro is intended to be used internally by the `riscv-rt` crate. Do not use it directly in your code.
267#[cfg(feature = "riscv-rt")]
268#[proc_macro]
269pub fn rvrt_default_start_trap(_input: TokenStream) -> TokenStream {
270    match riscv_rt::asm::RiscvArch::try_from_env() {
271        Some(arch) => arch.default_start_trap().into(),
272        None => quote! {
273            compile_error!("RISCV_RT_BASE_ISA environment variable is not set or is invalid");
274        }
275        .into(),
276    }
