8/// Hart states.
9///
10/// Declared in Table 1 at §9.
11pub mod hart_state {
12    /// The hart is physically powered-up and executing normally.
13    #[doc(alias = "SBI_HSM_STATE_STARTED")]
14    pub const STARTED: usize = 0;
15    /// The hart is not executing in supervisor-mode or any lower privilege mode.
16    ///
17    /// It is probably powered-down by the SBI implementation if the underlying platform
18    /// has a mechanism to physically power-down harts.
19    #[doc(alias = "SBI_HSM_STATE_STOPPED")]
20    pub const STOPPED: usize = 1;
21    /// The hart is pending before being started
22    ///
23    /// Some other hart has requested to start (or power-up) the hart from the STOPPED state,
24    /// and the SBI implementation is still working to get the hart in the STARTED state.
25    #[doc(alias = "SBI_HSM_STATE_START_PENDING")]
26    pub const START_PENDING: usize = 2;
27    /// The hart is pending before being stopped.
28    ///
29    /// The hart has requested to stop (or power-down) itself from the STARTED state,
30    /// and the SBI implementation is still working to get the hart in the STOPPED state.
31    #[doc(alias = "SBI_HSM_STATE_STOP_PENDING")]
32    pub const STOP_PENDING: usize = 3;
33    /// The hart is in a platform-specific suspend (or low-power) state.
34    #[doc(alias = "SBI_HSM_STATE_SUSPENDED")]
35    pub const SUSPENDED: usize = 4;
36    /// The hart is pending before being suspended.
37    ///
38    /// The hart has requested to put itself in a platform-specific low-power state
39    /// from the STARTED state, and the SBI implementation is still working to get
40    /// the hart in the platform-specific SUSPENDED state.
41    #[doc(alias = "SBI_HSM_STATE_SUSPEND_PENDING")]
42    pub const SUSPEND_PENDING: usize = 5;
43    /// The hart is pending before being resumed.
44    ///
45    /// An interrupt or platform specific hardware event has caused the hart to resume
46    /// normal execution from the SUSPENDED state, and the SBI implementation is still
47    /// working to get the hart in the STARTED state.
48    #[doc(alias = "SBI_HSM_STATE_RESUME_PENDING")]
49    pub const RESUME_PENDING: usize = 6;
50}
