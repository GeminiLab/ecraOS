1//! Per-page metadata stored in the metadata region of each section.
2//!
3//! Each page frame in the heap has a corresponding [`PageMeta`] entry.
4//! Free pages are linked together via intrusive doubly-linked lists using PFN indices.
5
6use crate::pfn::{OptionSectionFrameNumber, SectionFrameNumber};
7
8/// Flags describing a page's block state.
9///
10/// It can be considered to have the following bitfield layout:
11/// - Bit 0: **Allocated**. Whether the block is allocated.
12/// - Bit 1: **Head**. Whether the page is the head of a buddy block.
13///
14/// The representation makes zero-initialization safe for pages that are not block heads.
15#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
16#[repr(u8)]
17pub enum PageFlags {
18    /// Page is inside a larger buddy block.
19    #[default]
20    InBlock = 0,
21    /// Page is the head of a free block.
22    Free = 2,
23    /// Page is the head of an allocated block.
24    Allocated = 3,
25}
26
27/// Metadata for a single page frame.
28///
29/// This struct should be 12 bytes in size and 4-byte aligned.
30///
31/// For free or allocated blocks, only the head page carries the `flags` and
32/// `order` of the whole block, while other pages are marked as `InBlock`.
33///
34/// The representation makes zero-initialization safe for pages that are not
35/// block heads.
36#[derive(Debug, Clone, Copy)]
37#[repr(C)]
38pub struct PageMeta {
39    /// Current state of this page.
40    pub flags: PageFlags,
41    /// Order of the block.
42    ///
43    /// This field is valid only if `flags` does not equal `PageFlags::InBlock`.
44    pub order: u8,
45    /// Reserved padding.
46    pub _pad: u16,
47    /// Previous PFN in the same-order free list.
48    ///
49    /// This field is valid only if `flags` equals `PageFlags::Free`.
50    pub prev: OptionSectionFrameNumber,
51    /// Next PFN in the same-order free list.
52    ///
53    /// This field is valid only if `flags` equals `PageFlags::Free`.
54    pub next: OptionSectionFrameNumber,
55}
56
57/// Compile-time assertions to ensure `PageMeta` is exactly 12 bytes.
58const _: () = assert!(core::mem::size_of::<PageMeta>() == 12);
59/// Compile-time assertions to ensure `PageMeta` is 4-byte aligned.
60const _: () = assert!(core::mem::align_of::<PageMeta>() == 4);
61
62/// Provides a free order-0 page-meta entry.
63impl Default for PageMeta {
64    /// Creates a free order-0 page-meta entry.
65    fn default() -> Self {
66        Self::new()
67    }
68}
69
70/// Page metadata operations.
71impl PageMeta {
72    /// Creates a free order-0 page-meta entry.
73    pub const fn new() -> Self {
74        Self {
75            flags: PageFlags::Free,
76            order: 0,
77            _pad: 0,
78            prev: OptionSectionFrameNumber::NONE,
79            next: OptionSectionFrameNumber::NONE,
80        }
81    }
82
83    /// Reads the [`SectionFrameNumber`] of the previous page in the same-order
84    /// free list.
85    ///
86    /// The value is valid only if `flags` equals `PageFlags::Free`.
87    #[inline]
88    pub const fn prev(&self) -> Option<SectionFrameNumber> {
89        self.prev.into_option_sfn()
90    }
91
92    /// Reads the [`SectionFrameNumber`] of the next page in the same-order
93    /// free list.
94    ///
95    /// The value is valid only if `flags` equals `PageFlags::Free`.
96    #[inline]
97    pub const fn next(&self) -> Option<SectionFrameNumber> {
98        self.next.into_option_sfn()
99    }
100
101    /// Sets the previous page in the same-order free list to the given
102    /// [`SectionFrameNumber`] or `None`.
103    #[inline]
104    pub const fn set_prev(&mut self, sfn: Option<SectionFrameNumber>) {
105        self.prev = OptionSectionFrameNumber::from_option_sfn(sfn);
106    }
107
108    /// Sets the next page in the same-order free list to the given
109    /// [`SectionFrameNumber`] or `None`.
110    #[inline]
111    pub const fn set_next(&mut self, sfn: Option<SectionFrameNumber>) {
112        self.next = OptionSectionFrameNumber::from_option_sfn(sfn);
113    }
114
115    /// Sets the previous page in the same-order free list to the given
116    /// [`SectionFrameNumber`].
117    #[inline]
118    pub const fn set_prev_to(&mut self, sfn: SectionFrameNumber) {
119        self.set_prev(Some(sfn));
120    }
121
122    /// Sets the next page in the same-order free list to the given
123    /// [`SectionFrameNumber`].
124    #[inline]
125    pub const fn set_next_to(&mut self, sfn: SectionFrameNumber) {
126        self.set_next(Some(sfn));
127    }
128
129    /// Sets the previous page in the same-order free list to `None`.
130    #[inline]
131    pub const fn clear_prev(&mut self) {
132        self.set_prev(None);
133    }
134
135    /// Sets the next page in the same-order free list to `None`.
136    #[inline]
137    pub const fn clear_next(&mut self) {
138        self.set_next(None);
139    }
140}
