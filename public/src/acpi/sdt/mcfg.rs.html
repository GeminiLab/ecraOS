1use crate::{
2    AcpiTable,
3    sdt::{SdtHeader, Signature},
4};
5use core::{fmt, mem, slice};
6
7#[repr(C, packed)]
8pub struct Mcfg {
9    pub header: SdtHeader,
10    _reserved: u64,
11    // Followed by `n` entries with format `McfgEntry`
12}
13
14/// ### Safety: Implementation properly represents a valid MCFG.
15unsafe impl AcpiTable for Mcfg {
16    const SIGNATURE: Signature = Signature::MCFG;
17
18    fn header(&self) -> &SdtHeader {
19        &self.header
20    }
21}
22
23impl Mcfg {
24    /// Returns a slice containing each of the entries in the MCFG table. Where possible, `PlatformInfo.interrupt_model` should
25    /// be enumerated instead.
26    pub fn entries(&self) -> &[McfgEntry] {
27        let length = self.header.length as usize - mem::size_of::<Mcfg>();
28
29        // Intentionally round down in case length isn't an exact multiple of McfgEntry size - this
30        // has been observed on real hardware (see rust-osdev/acpi#58)
31        let num_entries = length / mem::size_of::<McfgEntry>();
32
33        unsafe {
34            let pointer = (self as *const Mcfg as *const u8).add(mem::size_of::<Mcfg>()) as *const McfgEntry;
35            slice::from_raw_parts(pointer, num_entries)
36        }
37    }
38}
39
40impl fmt::Debug for Mcfg {
41    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
42        formatter.debug_struct("Mcfg").field("header", &self.header).field("entries", &self.entries()).finish()
43    }
44}
